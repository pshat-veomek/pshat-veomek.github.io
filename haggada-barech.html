וְאַל יֶחְסַר לָנוּ מָזוֹן לְעוֹלָם וָעֶד - כמובן אין זו קביעה על העתיד, אלא בקשה אודותיו. בַּעֲבוּר שְׁמוֹ הַגָּדוֹל - בעבור שיתגלה שמו, דהיינו: מידותיו - החסד והרחמים, בכך שהוא זן ומפרנס לכול. כִּי הוּא אֵל זָן וּמְפַרְנֵס לַכֹּל וּמֵטִיב לַכֹּל, וּמֵכִין מָזוֹן לְכָל בְּרִיּוֹתָיו אֲשֶׁר בָּרָא - ולכן לא חסר לנו וכו', כאמור לעיל. אתה פּוֹתֵחַ אֶת יָדֶך המשפיעה שפע וברכה לעולם, וע"י כך מַשְׂבִּיעַ לְכָל חַי, באופן של רָצוֹן = הארת פנים, כלומר: בכבוד וברווח.

שֶׁהוֹצֵאתָנוּ יְהֹוָה אֱלֹהֵינוּ מֵאֶרֶץ מִצְרַיִם - באופן פיזי, ופדית את נשמותינו מִבֵּית עֲבָדִים - שבמצרים נשמות ישראל היו משועבדות לתרבות מצרים והשפעתה, ואף שלא אצל כולם הדבר התבטא באופן מעשי, מכל מקום האומה בכללותה חסרה את החרות הנשמתית, שהיא הדבקות האלוהית (השווה לכתוב לעיל בברכת 'גאל ישראל': "על גאולתנו ועל פדות נפשנו": ה'גאולה' - פיזית, וה'פדות' - רוחנית)123. וְעַל בְּרִיתְךָ שֶׁחָתַמְתָּ בְּבְשָׂרֵנוּ - ובזכות מצוה זו זכינו לירושת הארץ (שהיא הנושא של ברכה זו), שהיא כביכול חלקו של הקב"ה בקיום הברית, כמו שאמר לאברהם בציווי על המילה (בראשית יז ב-ח): "וְאֶתְּנָה בְרִיתִי בֵּינִי וּבֵינֶךָ... וְנָתַתִּי לְךָ וּלְזַרְעֲךָ אַחֲרֶיךָ אֵת אֶרֶץ מְגֻרֶיךָ". וְעַל תּוֹרָתְךָ החלק האמוני והמוסרי שבתורה124 שֶׁלִּמַּדְתָּנוּ, וְעַל חֻקֶּיךָ = המצוות שטעמן אינו ידוע125 שֶׁהוֹדַעְתָּנוּ - שהודעת לנו שאנו צריכים לקיימם, אף בלא ש"לימדתנו" אותם, דהיינו הבנת טעמי הדברים.

בסיומה של ברכת הודאה זו אנו מביעים את תקוותנו לכך שכל באי העולם יכירו במלכות ה' ויברכוהו: יִתְבָּרַךְ שִׁמְךָ בְּפִי כָּל חַי תָּמִיד לְעוֹלָם וָעֶד - יאמרו שאתה לבדך הוא מקור הברכה והטוב בעולם.

רְצֵה בנו וְהַחֲלִיצֵנוּ - חזקנו ותן בנו כוח כמו ליוצאים לקרב, מלשון (במדבר לא ה): "חֲלוּצֵי צָבָא", וכן (ישעיהו נח יא): "וְעַצְמֹתֶיךָ יַחֲלִיץ", בְּמִצְוֹתֶיךָ וּבְמִצְוַת יוֹם הַשְּׁבִיעִי-ע"י שנקיים את מצוותיך ואת מצוַת השבת.

כִּי יוֹם זֶה גָּדוֹל וְקָדוֹשׁ הוּא לְפָנֶיךָ – המילה 'כי' מלמדת שמשפט זה הוא נתינת טעם לאמור לעיל, כלומר: בזכות שאנו מקיימים מצוה גדולה וקדושה זו, רצה בנו והחליצנו.

המילה 'זה', מתייחסת לדבר הנמצא לפנינו, שאפשר להצביע עליו ולומר "זה", ואילו המילה 'הוא' מתייחסת לדבר הנסתר מעינינו; לפי זה יתפרש: יום זה – יום השבת המוּכר לנו מהעולם הזה - גדול וקדוש הוא לפניך – יום השבת הגלוי כַּיום לפני ה' בלבד, ומעינינו הוא נסתר, "יום שכולו שבת ומנוחה לחיי העולמים", שהוא מצבו העתידי של העולם כשיגיע לתיקונו השלם, והשבת המוכרת לנו מהעולם הזה היא רק התנוצצות מזערית ממנו - בו מתממשות בשלמות גדולת השבת וקדושתה127.

  1. 128

    הצעה

וּבִרְצוֹנְךָ - ע"י שתהיה איתנו במצב של רצון ופיוס, הָנִיחַ לָנוּ - עשה שתהיה לנו מנוחה, שֶׁלֹּא תְהֵא צָרָה בפועל, וְיָגוֹן בהרגשת הלב (מלבי"ם, ישעיהו ח כב, וירמיהו לא יב).

לִפְלֵיטָה - שתתןלנו מפלט והצלה, וגם לְטוֹבָה - שתיטיב לנו.

זָכְרֵנוּ יְהֹוָה אֱלֹהֵינוּ בּוֹ לְטוֹבָה וּפָקְדֵנוּ בוֹ לִבְרָכָה - משמעות המושג 'פקידה' שונה ממשמעות המושג 'זכירה', בכך ש'פקידה' איננה זכירה גרידא, אלא כזו הגורמת לזוכר לעשות את הדבר המתבקש ע"פ תוכן הזכירה (מלבי"ם, שמות ג טז). אנו מבקשים מה' שכשיזכור אותנו ויבחן את מעשינו, נהיה רצויים וטובים בעיניו, ויפקוד אותנו ואת מעשינו הטובים, ועקב 'פקידה' זו יברך אותנו.

מוצגות כאן שתי סיבות שבעבורן אנו מצפים מה' שיושיענו: א. מצידנו: כִּי אֵלֶיךָ עֵינֵינוּ - בזכות זה שאנחנו בוטחים בו. ב. מצידו: כִּי אֵל מֶלֶךְ חַנּוּן וְרַחוּם אָתָּה.

התואר בּוֹרְאֵנוּ - מתייחס לבריאת החומר עצמו, והתואר יוֹצְרֵנוּ - מתייחס ליצירת הצורה מהחומר (מלבי"ם, ישעיהו מה ז). (ייתכן129 שהתואר 'גואלנו', שמבחינה עניינית מקומו המתאים אינו בין 'בוראנו' ל'יוצרנו', משובץ שם כדי ליצור רצף אלפביתי: אדירנו בוראנו גואלנו, שגם יש בו משמעות ע"פ הנסתר130). קְדוֹשֵׁנוּ - קדוש, ומשפיע עלינו, על ישראל שבדור זה, מקדושתו, קְדוֹשׁ יַעֲקֹב - משפיע מקדושתך על כלל ישראל שבכל הדורות (מקור הביטוי בישעיהו כט כג: "וְהִקְדִּישׁוּ אֶת קְדוֹשׁ יַעֲקֹב"). משמעות התואר 'קדוש', הוא - נבדל, מופרש, ומרומם, מהעולם הזה על חוקיו ומגבלותיו, ואין לשכל אנושי הבנה במהותו (ראה מלבי"ם, ישעיהו ו ג). משמעות הקדושה הנשפעת ממנו לישראל היא, מעלה רוחנית המרוממת אותם מעל למדרגה האנושית הכללית. רוֹעֵנוּ - מנהיג אותנו, ישראל שבדור זה, רוֹעֵה יִשְׂרָאֵל - מנהיג את כלל ישראל בכל הדורות (מקור הביטוי בתהילים פ ב: "רֹעֵה יִשְׂרָאֵל הַאֲזִינָה"). הַמֶּלֶךְ הַטּוֹב מצד עצמו, וְהַמֵּטִיב בפועל לַכֹּל, שֶׁבְּכָל יוֹם וָיוֹם הוּא הֵטִיב... הוּא גְמָלָנוּ... לָעַד - משמעות המושג 'גמול' היא נתינה לאחֵר מתוך רגש, בין רגש חיובי, כפי המשמעות כאן, ובין רגש שלילי, כמו (עובדיה א טו): "גְּמֻלְךָ יָשׁוּב בְּרֹאשֶׁךָ" שנאמר על ענישתה של מלכות אדום ע"י הקב"ה (מלבי"ם, ישעיהו נט יח). זאת, בשונה מההבנה הרגילה לפיה 'גמול' הוא דווקא הבא כתגובה; הבנה זו לא מתיישבת עם מספר מקומות במקרא (ראה יואל ד ד, תהילים קלז ח, ומצודת ציון שם ושם). במשפט זה אנו מוסיפים, שלא זו בלבד שהקב"ה מטיב לנו תמיד, אלא הוא עושה זאת באופן של גמול, דהיינו מתוך רגשי אהבה131. פירוש נוסף (ע"פ ההבנה הרגילה)132: הטובות שאנו מקבלים מהקב"ה מוגדרות גם כהטבה = מתנת חינם, שהרי אינו חייב לנו דבר, וגם כגמול = שכר על פעולה, כיון שהקב"ה מיטיב לאדם בהתאם למספר זכויותיו.

הָרַחֲמָן - שלושת 'הרחמן' הראשונים אינם בקשות, אלא שבח.

הָרַחֲמָן הוּא יִתְבָּרַךְ בַּשָּׁמַיִם - בפי המלאכים, וּבָאָרֶץ - בפי בני האדם.

הָרַחֲמָן הוּא יִשְׁתַּבַּח לְדוֹר דּוֹרִים - בכל דור יכירו בשבחו של הקב"ה, וישבחוהו = יספרו את שבחו, ובכל דור יתווסף שבח יותר מהדור שלפניו (מלשון 'השבחה' = תוספת שבח), כיון שהעולם נבנה ומתפתח, וחכמתו של הקב"ה מתגלה יותר; וְיִתְפָּאַר בָּנוּ - פארו של הקב"ה יתבטא בעולם דרכנו, דרך פארו של עם ישראל, וזה יהיה לָעַד וּלְנֵצַח נְצָחִים - כיון שסגולת עם ישראל והבחירה האלוהית בו, הן נצחיות; וְיִתְהַדַּר בָּנוּ - וכן הדרו של הקב"ה יתבטא בעולם דרך הדרו של עם ישראל לָעַד וּלְעוֹלְמֵי עוֹלָמִים133.

    הצעה

כְּמוֹ שֶׁנִּתְבָּרְכוּ אֲבוֹתֵינוּ אַבְרָהָם יִצְחָק וְיַעֲקֹב: בַּכֹּל מִכֹּל כֹּל - על הברכה שנתברך כל אחד משלושת האבות נאמרה אחת מלשונות אלו: על אברהם (בראשית כד א): "וה' בֵּרַךְ אֶת אַבְרָהָם בַּכֹּל". על יצחק (שם כז לג): "וָאֹכַל מִכֹּל". על יעקב (שם לג יא): "יֶשׁ לִי כֹל".

בַּמָּרוֹם יְלַמְּדוּ עֲלֵיהֶם - על בעלי הבית שהוזכרו קודם, וְעָלֵינוּ, זְכוּת - שיזכירו את זכויותינו וידונו אותנו בהתאם להן, וכן שידונו את מעשינו לכף זכות וינהגו עמנו לפנים משורת הדין, וע"י כך תְּהֵא לנו מִשְׁמֶרֶת שָׁלוֹם - ממרום ישמרו עלינו שתמיד יהיה לנו שלום מבית ומחוץ, וְנִשָּׂא בְרָכָה מֵאֵת יְהֹוָה, וּצְדָקָה מֵאֱלֹהֵי יִשְׁעֵנוּ - על פי לשון הפסוק (תהילים כד ה): "יִשָּׂא בְרָכָה מֵאֵת ה' וּצְדָקָה מֵאֱלֹהֵי יִשְׁעוֹ"; אנו מבקשים שה' יברכנו בהתאם למגיע לנו כביכול על פי זכויותינו, ומלבד זאת, מצד שהוא אלוהי ישענו, שמשגיח עלינו באופן מיוחד להושיענו ולהיטיב לנו, יוסיף לנו עוד טובות בתור צדקה ולא בתור שכר134.

וְנִמְצָא חֵן וכו'. כלשון הפסוק (משלי ג ד): "וּמְצָא חֵן וְשֵׂכֶל טוֹב בְּעֵינֵי אֱלֹהִים וְאָדם". הָרַחֲמָן הוּא יַנְחִילֵנוּ יוֹם שֶכֻּלּוֹ שַׁבָּת וּמְנוּחָה לְחַיֵּי הָעוֹלָמִים - בקשה זו מבוססת על לשון המשנה בסוף מסכת תמיד (הנאמרת בנוסח אשכנז לאחר פיטום הקטורת במוסף של שבת), המפרטת את המזמורים שהיו נאמרים ע"י הלווים במקדש בכל אחד מימי השבוע: "בשבת היו אומרים: מזמור שיר ליום השבת - מזמור שיר לעתיד לבוא, ליום שכולו שבת ומנוחה לחיי העולמים". המשנה מסבירה כי "יום השבת" האמור במזמור הוא המציאות העתידית, שבה העולם יגיע לשלמותו הרוחנית וכולו יהיה קודש לערכים אלוהיים, בדומה ליום השבת, שעניינו הוא שביתה מחיי החולין ועיסוק בענייני הרוח; אז יאבדו כל אויבי ה' והצדיקים יפרחו וישגשגו, כמתואר באותו מזמור (תהילים צב). בשבת, שהיא מעין התגלות בזעיר אנפין של אותה מציאות נעלית, בתוך המציאות ההווית, אנו מוסיפים בקשה זו, בה אנו מבקשים שנהיה זכאים לחיות באותו זמן.

הָרַחֲמָן הוּא יְזַכֵּנוּ לִימוֹת הַמָּשִׁיחַ - הם הימים שנאמר עליהם135:"אין בין העולם הזה לימות המשיח אלא שעבוד מלכויות בלבד" - אז עם ישראל לא יהיה משועבד לאומות העולם, אלא הן תהייה כפופות לו, אבל חוקי הטבע לא ישתנו כלל, וּלְחַיֵּי הָעוֹלָם הַבָּא - הוא הזמן שיבוא לאחר ימות המשיח, עליו נאמר136: "העולם הבא אין בו לא אכילה ולא שתייה ולא פרייה ורבייה, אלא צדיקים יושבים ועטרותיהם בראשיהם ונהנים מזיו השכינה". אפשר גם לפרש שהבקשה מתייחסת לעולם הנשמות אליו מגיע כל אדם לאחר פטירתו, אך מאיזכור "חיי העולם הבא" לאחר "ימות המשיח" נראה כפירוש הראשון. שני פירושים אלו אפשריים גם לגבי 'הרחמן' שמוסיפים ביו"ט: "הָרַחֲמָן הוּא יַנְחִילֵנוּ יוֹם שֶׁכֻּלּוֹ טוֹב" - יתן לנו זכות לחיים בזמן שהמציאות בו היא כולה שלימה. בליל הסדר מוסיפים לו המשך: יוֹם שֶׁכֻּלּוֹ אָרוֹךְ - שהוא עולם נצחי, יוֹם שֶׁצַּדִּיקִים יוֹשְׁבִים וְעַטְרוֹתֵיהֶם בְּרָאשֵׁיהֶם - כלומר: הנשמות זוכות לכבוד רוחני, וְנֶהֱנִים מִזִּיו הַשְּׁכִינָה, וִיהִי חֶלְקֵנוּ עִמָּהֶם.

מִגְדּוֹל יְשׁוּעוֹת מַלְכּוֹ - עושה לדוד, המולך מטעמו, ישועות גדולות137. פירוש נוסף: ה' הוא כמגדל הגנה המושיע את דוד138, וְעֹשֶׂה חֶסֶד לִמְשִׁיחוֹ – לדוד, שנמשח בשמן המשחה; החסד הוא שהמלוכה תהיה שייכת לְדָוִד וּלְזַרְעוֹ עַד עוֹלָם, כפי שניבא לו נתן הנביא (שמואל ב' ז טז): "וְנֶאְמַן בֵּיתְךָ וּמַמְלַכְתְּךָ עַד עוֹלָם לְפָנֶיךָ, כִּסְאֲךָ יִהְיֶה נָכוֹן עַד עוֹלָם". פסוק זה נאמר כאן כהמשך לבקשה שלפניו, שנזכה לימות המשיח, והוא מתאר את תוכנם, שבאותם ימים נזכה לישועות גדולות כמו בימי דוד, ולהקמת מלך מזרע דוד שמלכותו ומלכות זרעו תתקיים לעולם. פסוק זה חותם את שירת דוד (שמואל ב' כב), וייתכן שגם משום כך נתקן לאמרו בחתימת ברכת המזון139. עֹשֶׂה שָׁלוֹם בִּמְרוֹמָיו - הוא לשון הפסוק באיוב (כה ב), ופירושו, שהקב"ה נמצא במרום, כלומר: מעל כל המציאות, ומשם הוא מנהיגהּ בחכמה כך שהעולם יתקיים בשלום ובאיזון המתאים, על אף שיש בו כוחות שונים המנוגדים זה לזה; הוּא יַעֲשֶׂה שָׁלוֹם עָלֵינוּ וְעַל כָּל יִשְׂרָאַל - שלא יהיה עלינו אִיוּם חיצוני, ולא בתוכנו סכסוך פנימי.

יְראוּ אֶת יְהֹוָה - היו יראים ממנו, אתם עם ישראל קְדשָׁיו, כִּי אֵין מַחְסוֹר לִירֵאָיו - ואז, כשתהיו ראויים להיקרא 'יראיו', יספק לכם כל צרכיכם. כְּפִירִים = גורי אריות רָשׁוּ וְרָעֵבוּ - לא מצאו טרף ונהיו רשים = עניים ורעבים, אבל דֹרְשֵׁי יְהֹוָה - מבקשי קרבתו, לֹא יַחְסְרוּ כָל טוֹב.

הוֹדוּ לַיְהֹוָה כִּי הוא טוֹב140, או: כִּי טוֹב להודות לו141, כִּי לְעוֹלָם הוא משפיע את חַסְדּוֹ על בריותיו. אתה פּוֹתֵחַ אֶת יָדֶך המשפיעה שפע לעולם, וע"י כך מַשְׂבִּיעַ לְכָל חַי, באופן של רָצוֹן = הארת פנים, כלומר: בכבוד וברווח. בָּרוּךְ הַגֶּבֶר אֲשֶׁר יִבְטַח בַּיְהֹוָה, ואז יהיה יְהֹוָה מִבְטַחוֹ, שאכן יספק לו כל צרכיו, והוא ימשיך לבטוח בו.

נַעַר הָיִיתִי גַם זָקַנְתִּי - ובכל חיי הארוכים לֹא רָאִיתִי צַדִּיק נֶעֱזָב ע"י הקב"ה, אלא תמיד הוא מספק לצדיק את צרכיו, ואפילו לא ראיתי את זַרְעוֹ של הצדיק מְבַקֶּשׁ לָחֶם, כיון שגם אותם מפרנס הקב"ה בזכות אביהם. יְהֹוָה עֹז לְעַמּוֹ יִתֵּן - לגבור על אויביהם מבחוץ, יְהֹוָה יְבָרֵךְ אֶת עַמּוֹ בַשָּׁלוֹם - שישרור בקרב העם שלום פנימי. חז"ל142 פירשו פסוק זה כך: ה' יברך את עמו בברכותיו, באמצעות השלום, כלומר: השלום אינו תוכן הברכה עצמה אלא הוא הכלי שמכיל אותה; שבזמן שהשלום שורר בקרב העם, הם יכולים ליהנות מברכת ה', אך כשיש מצה ומריבה אין הנאה משפע הברכה, כמאמר הכתוב (משלי יז א): "טוֹב פַּת חֲרֵבָה וְשַׁלְוָה בָהּ, מִבַּיִת מָלֵא זִבְחֵי רִיב".
