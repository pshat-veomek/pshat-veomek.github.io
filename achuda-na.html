דמויותכרך אקל

בַּתְּחִלָּה חָשַׁד וּלְבַסּוֹף בֵּרַךְ

גִּדֵּל הַנָּבִיא מִהְיוֹתוֹ רַךְ

שְׁכַב הוֹרָהוּ פְּעָמִים שָׁלוֹשׁ

וּלְמָחֳרָת בָּא פִּיהוּ לִדְרֹשׁ

בְּמִלְחָמָה מֵתוּ בָּנָיו

אַף נִינוֹ נִרְצָח

בְּזִקְנָתוֹ כָּהוּ עֵינָיו

וּמֵת בְּגִיל צַח

הצגת הפתרון

עלי הכהן

בַּתְּחִלָּה חָשַׁד וּלְבַסּוֹף בֵּרַךְ – בהתחלה חשד בחנה שהיא שיכורה, ולבסוף בירכה: "וְחַנָּה הִיא מְדַבֶּרֶת עַל לִבָּהּ רַק שְׂפָתֶיהָ נָּעוֹת וְקוֹלָהּ לֹא יִשָּׁמֵעַ, וַיַּחְשְׁבֶהָ עֵלִי לְשִׁכֹּרָה. וַיֹּאמֶר אֵלֶיהָ עֵלִי: עַד מָתַי תִּשְׁתַּכָּרִין הָסִירִי אֶת יֵינֵךְ מֵעָלָיִךְ. וַתַּעַן חַנָּה וַתֹּאמֶר: לֹא אֲדֹנִי [...] וַיַּעַן עֵלִי וַיֹּאמֶר: לְכִי לְשָׁלוֹם, וֵאלֹהֵי יִשְׂרָאֵל יִתֵּן אֶת שֵׁלָתֵךְ אֲשֶׁר שָׁאַלְתְּ מֵעִמּוֹ" (שמואל א' א יג-יז).

גִּדֵּל הַנָּבִיא מִהְיוֹתוֹ רַךְ – גידל את שמואל במשכן מילדותו: "וַתַּעֲלֵהוּ עִמָּהּ כַּאֲשֶׁר גְּמָלַתּוּ [...] וַתְּבִאֵהוּ בֵית ה' שִׁלוֹ [...] וַיָּבִיאוּ אֶת הַנַּעַר אֶל עֵלִי" (שם שם כד-כה), "וְהַנַּעַר שְׁמוּאֵל מְשָׁרֵת אֶת ה' לִפְנֵי עֵלִי" (שם ג א).

שְׁכַב הוֹרָהוּ פְּעָמִים שָׁלוֹשׁ – "וַיִּקְרָא ה' אֶל שְׁמוּאֵל וַיֹּאמֶר הִנֵּנִי. וַיָּרָץ אֶל עֵלִי וַיֹּאמֶר: הִנְנִי כִּי קָרָאתָ לִּי, וַיֹּאמֶר: לֹא קָרָאתִי, שׁוּב שְׁכָב, וַיֵּלֶךְ וַיִּשְׁכָּב. וַיֹּסֶף ה' קְרֹא עוֹד שְׁמוּאֵל, וַיָּקָם שְׁמוּאֵל וַיֵּלֶךְ אֶל עֵלִי, וַיֹּאמֶר: הִנְנִי כִּי קָרָאתָ לִי, וַיֹּאמֶר: לֹא קָרָאתִי בְנִי, שׁוּב שְׁכָב. וּשְׁמוּאֵל טֶרֶם יָדַע אֶת ה', וְטֶרֶם יִגָּלֶה אֵלָיו דְּבַר ה'. וַיֹּסֶף ה' קְרֹא שְׁמוּאֵל בַּשְּׁלִשִׁית, וַיָּקָם וַיֵּלֶךְ אֶל עֵלִי, וַיֹּאמֶר: הִנְנִי כִּי קָרָאתָ לִי, וַיָּבֶן עֵלִי כִּי ה' קֹרֵא לַנָּעַר. וַיֹּאמֶר עֵלִי לִשְׁמוּאֵל: לֵךְ שְׁכָב, וְהָיָה אִם יִקְרָא אֵלֶיךָ וְאָמַרְתָּ דַּבֵּר ה' כִּי שֹׁמֵעַ עַבְדֶּךָ" (שם שם ד-ט).

וּלְמָחֳרָת בָּא פִּיהוּ לִדְרֹשׁ – "וַיִּשְׁכַּב שְׁמוּאֵל עַד הַבֹּקֶר [...] וַיִּקְרָא עֵלִי אֶת שְׁמוּאֵל, וַיֹּאמֶר: שְׁמוּאֵל בְּנִי, וַיֹּאמֶר הִנֵּנִי. וַיֹּאמֶר: מָה הַדָּבָר אֲשֶׁר דִּבֶּר אֵלֶיךָ, אַל נָא תְכַחֵד מִמֶּנִּי" (שם שם טו-יז).

בְּמִלְחָמָה מֵתוּ בָּנָיו – "וַיִּלָּחֲמוּ פְלִשְׁתִּים וַיִּנָּגֶף יִשְׂרָאֵל [...] וּשְׁנֵי בְנֵי עֵלִי מֵתוּ חָפְנִי וּפִינְחָס" (שם ד י-יא).

אַף נִינוֹ נִרְצָח – אחימלך בן אחיטוב היה נינו, שכן אחיטוב היה בנו של פנחס, כפי שנאמר: "וַאֲחִיָּה בֶן אֲחִטוּב אֲחִי אִיכָבוֹד בֶּן פִּינְחָס בֶּן עֵלִי כֹּהֵן ה' בְּשִׁלוֹ" (שם יד ג), ונרצח על ידי דואג האדומי בפקודת שאול: "וַיִּשְׁלַח הַמֶּלֶךְ לִקְרֹא אֶת אֲחִימֶלֶךְ בֶּן אֲחִיטוּב הַכֹּהֵן וְאֵת כָּל בֵּית אָבִיו הַכֹּהֲנִים אֲשֶׁר בְּנֹב [...] וַיֹּאמֶר הַמֶּלֶךְ לְדוֹאֵג: סֹב אַתָּה וּפְגַע בַּכֹּהֲנִים, וַיִּסֹּב דּוֹאֵג הָאֲדֹמִי וַיִּפְגַּע הוּא בַּכֹּהֲנִים וַיָּמֶת בַּיּוֹם הַהוּא שְׁמֹנִים וַחֲמִשָּׁה אִישׁ נֹשֵׂא אֵפוֹד בָּד, וְאֵת נֹב עִיר הַכֹּהֲנִים הִכָּה לְפִי חֶרֶב" (שם כב יא-יט).

בְּזִקְנָתוֹ כָּהוּ עֵינָיו, וּמֵת בְּגִיל צַח – צ"ח, תשעים ושמונה: "וְעֵלִי בֶּן תִּשְׁעִים וּשְׁמֹנֶה שָׁנָה, וְעֵינָיו קָמָה וְלֹא יָכוֹל לִרְאוֹת [...] וַיִּפֹּל מֵעַל הַכִּסֵּא אֲחֹרַנִּית בְּעַד יַד הַשַּׁעַר וַתִּשָּׁבֵר מַפְרַקְתּוֹ וַיָּמֹת" (שם ד טו-יח).

מקומותכרך בקל

בְּתֶקַע שׁוֹפָר נָפְלָה

וְהָחֳרַם כָּל שְׁלָלָהּ

נִכְלָמִים הֳפְנוּ לָהּ

וּמֵתוּ בָּנִים בִּגְלָלָהּ

הצגת הפתרון

יריחו

בְּתֶקַע שׁוֹפָר נָפְלָה, וְהָחֳרַם כָּל שְׁלָלָהּ – "וַיְהִי בַּפַּעַם הַשְּׁבִיעִית תָּקְעוּ הַכֹּהֲנִים בַּשּׁוֹפָרוֹת וַיֹּאמֶר יְהוֹשֻׁעַ אֶל הָעָם הָרִיעוּ כִּי נָתַן ה' לָכֶם אֶת הָעִיר. וְהָיְתָה הָעִיר חֵרֶם הִיא וְכָל אֲשֶׁר בָּהּ לַה' [...] וְרַק אַתֶּם שִׁמְרוּ מִן הַחֵרֶם פֶּן תַּחֲרִימוּ וּלְקַחְתֶּם מִן הַחֵרֶם וְשַׂמְתֶּם אֶת מַחֲנֵה יִשְׂרָאֵל לְחֵרֶם וַעֲכַרְתֶּם אוֹתוֹ. וְכֹל כֶּסֶף וְזָהָב וּכְלֵי נְחֹשֶׁת וּבַרְזֶל קֹדֶשׁ הוּא לַה' אוֹצַר ה' יָבוֹא. וַיָּרַע הָעָם וַיִּתְקְעוּ בַּשֹּׁפָרוֹת וַיְהִי כִשְׁמֹעַ הָעָם אֶת קוֹל הַשּׁוֹפָר וַיָּרִיעוּ הָעָם תְּרוּעָה גְדוֹלָה וַתִּפֹּל הַחוֹמָה תַּחְתֶּיהָ וַיַּעַל הָעָם הָעִירָה אִישׁ נֶגְדּוֹ וַיִּלְכְּדוּ אֶת הָעִיר. [...] וְהָעִיר שָׂרְפוּ בָאֵשׁ וְכָל אֲשֶׁר בָּהּ רַק הַכֶּסֶף וְהַזָּהָב וּכְלֵי הַנְּחֹשֶׁת וְהַבַּרְזֶל נָתְנוּ אוֹצַר בֵּית ה'" (יהושע ו טז-כד).

נִכְלָמִים הֳפְנוּ לָהּ – "וַיֹּאמְרוּ שָׂרֵי בְנֵי עַמּוֹן אֶל חָנוּן אֲדֹנֵיהֶם: הַמְכַבֵּד דָּוִד אֶת אָבִיךָ בְּעֵינֶיךָ כִּי שָׁלַח לְךָ מְנַחֲמִים?! הֲלוֹא בַּעֲבוּר חֲקוֹר אֶת הָעִיר וּלְרַגְּלָהּ וּלְהָפְכָהּ שָׁלַח דָּוִד אֶת עֲבָדָיו אֵלֶיךָ! וַיִּקַּח חָנוּן אֶת עַבְדֵי דָוִד וַיְגַלַּח אֶת חֲצִי זְקָנָם וַיִּכְרֹת אֶת מַדְוֵיהֶם בַּחֵצִי עַד שְׁתוֹתֵיהֶם וַיְשַׁלְּחֵם. וַיַּגִּדוּ לְדָוִד וַיִּשְׁלַח לִקְרָאתָם כִּי הָיוּ הָאֲנָשִׁים נִכְלָמִים מְאֹד, וַיֹּאמֶר הַמֶּלֶךְ: שְׁבוּ בִירֵחוֹ עַד יְצַמַּח זְקַנְכֶם וְשַׁבְתֶּם" (שמואל ב' י ג-ה).

וּמֵתוּ בָּנִים בִּגְלָלָהּ – "וַיַּשְׁבַּע יְהוֹשֻׁעַ בָּעֵת הַהִיא לֵאמֹר: אָרוּר הָאִישׁ לִפְנֵי ה' אֲשֶׁר יָקוּם וּבָנָה אֶת הָעִיר הַזֹּאת אֶת יְרִיחוֹ, בִּבְכֹרוֹ יְיַסְּדֶנָּה וּבִצְעִירוֹ יַצִּיב דְּלָתֶיהָ" (יהושע שם כו), "בְּיָמָיו [של אחאב] בָּנָה חִיאֵל בֵּית הָאֱלִי אֶת יְרִיחֹה, בַּאֲבִירָם בְּכֹרוֹ יִסְּדָהּ ובשגיב [וּבִשְׂגוּב] צְעִירוֹ הִצִּיב דְּלָתֶיהָ כִּדְבַר ה' אֲשֶׁר דִּבֶּר בְּיַד יְהוֹשֻׁעַ בִּן נוּן" (מלכים א' טז לד).

דמויותכרך גבינוני

אֵשֶׁת גִּבּוֹר אַחֲרוֹן

לִקּוּחֶיהָ עוֹרְרוּ חֲרוֹן

מֵת בְּנָהּ לְמִנְיַן שְׁמָהּ

וְעַל בְּנָהּ אַחֵר לְמֶלֶךְ נָאֲמָה

הצגת הפתרון

בת שבע

אֵשֶׁת גִּבּוֹר אַחֲרוֹן – "בַּת שֶׁבַע בַּת אֱלִיעָם, אֵשֶׁת אוּרִיָּה הַחִתִּי" (שמואל ב' יא ג), ואוריה החתי נמנה אחרון מבין גיבורי דוד: "אוּרִיָּה הַחִתִּי, כֹּל שְׁלֹשִׁים וְשִׁבְעָה" (שם כג לט),

לִקּוּחֶיהָ עוֹרְרוּ חֲרוֹן – נישואיה על ידי דוד עוררו את חרון ה': "וַיִּשְׁלַח דָּוִד וַיַּאַסְפָהּ אֶל בֵּיתוֹ וַתְּהִי לוֹ לְאִשָּׁה וַתֵּלֶד לוֹ בֵּן, וַיֵּרַע הַדָּבָר אֲשֶׁר עָשָׂה דָוִד בְּעֵינֵי ה'" (שם יא כז);

מֵת בְּנָהּ לְמִנְיַן שְׁמָהּ – בנה הראשון שילדה לדוד מת בעת היותו בן שבעה ימים, כמניין שבשמה 'בת שבע': "וַיְהִי בַּיּוֹם הַשְּׁבִיעִי וַיָּמָת הַיָּלֶד" (שם יב יח),

וְעַל בְּנָהּ אַחֵר לְמֶלֶךְ נָאֲמָה – על אודות שלמה בנה, שאדוניה מבקש לקחת ממנו את המלוכה: "וַתָּבֹא בַת שֶׁבַע אֶל הַמֶּלֶךְ [...] וַתֹּאמֶר לוֹ: אֲדֹנִי אַתָּה נִשְׁבַּעְתָּ בַּה' אֱלֹהֶיךָ לַאֲמָתֶךָ כִּי שְׁלֹמֹה בְנֵךְ יִמְלֹךְ אַחֲרָי וְהוּא יֵשֵׁב עַל כִּסְאִי! וְעַתָּה הִנֵּה אֲדֹנִיָּה מָלָךְ וְעַתָּה אֲדֹנִי הַמֶּלֶךְ לֹא יָדָעְתָּ [...] וְהָיָה כִּשְׁכַב אֲדֹנִי הַמֶּלֶךְ עִם אֲבֹתָיו וְהָיִיתִי אֲנִי וּבְנִי שְׁלֹמֹה חַטָּאִים" (מלכים א' א טו-כא).

מקומותכרך דקשה

בָּהּ מֶלֶךְ בֶּן מֶלֶךְ

נֶאֱסַר בְּיַד מֶלֶךְ

וּבְנֵי אָחִיו הַמֶּלֶךְ

הוּמְתוּ בְּיַד מֶלֶךְ

הצגת הפתרון

רִבְלָה

בָּהּ מֶלֶךְ בֶּן מֶלֶךְ נֶאֱסַר בְּיַד מֶלֶךְ – יהואחז בן יאשיהו, נאסר ברבלה על ידי פרעה מלך מצרים, כמסופר בכתוב: "בֶּן עֶשְׂרִים וְשָׁלֹשׁ שָׁנָה יְהוֹאָחָז בְּמָלְכוֹ [...] וַיַּאַסְרֵהוּ פַרְעֹה נְכֹה בְרִבְלָה בְּאֶרֶץ חֲמָת" (מלכים ב' כג לא-לג);

וּבְנֵי אָחִיו הַמֶּלֶךְ – בני צדקיהו המלך, שהיה אחיו של יהואחז,

הוּמְתוּ בְּיַד מֶלֶךְ – נבוכדנצר מלך בבל, כמסופר בכתוב: "וַיַּעֲלֻהוּ אֶל נְבוּכַדְרֶאצַּר מֶלֶךְ בָּבֶל רִבְלָתָה בְּאֶרֶץ חֲמָת [...], וַיִּשְׁחַט מֶלֶךְ בָּבֶל אֶת בְּנֵי צִדְקִיָּהוּ בְּרִבְלָה לְעֵינָיו" (ירמיהו לט ה-ו).

צמחיםכרך הקל

חָגְרוּ פְקוּחִים בְּעָלֶיהָ

וְנֹצְרָהּ יֹאכַל גִּדּוּלֶיהָ

אַרְבֶּה הִלְבִּין שָׂרִיגֶיהָ

וּבַשִּׁיר חָנְטָה פַגֶּיהָ

הצגת הפתרון

תאנה

חָגְרוּ פְקוּחִים בְּעָלֶיהָ – "וַתִּפָּקַחְנָה עֵינֵי שְׁנֵיהֶם וַיֵּדְעוּ כִּי עֵירֻמִּם הֵם, וַיִּתְפְּרוּ עֲלֵה תְאֵנָה וַיַּעֲשׂוּ לָהֶם חֲגֹרֹת" (בראשית ג ז),

וְנֹצְרָהּ יֹאכַל גִּדּוּלֶיהָ – "נֹצֵר תְּאֵנָה יֹאכַל פִּרְיָהּ" (משלי כז יח).

אַרְבֶּה הִלְבִּין שָׂרִיגֶיהָ – אכל את קליפת הענפים וחשף אותם בלבנותם, כאמור בדברי הנביא: "יֶתֶר הַגָּזָם אָכַל הָאַרְבֶּה וְיֶתֶר הָאַרְבֶּה אָכַל הַיָּלֶק וְיֶתֶר הַיֶּלֶק אָכַל הֶחָסִיל [...] שָׂם גַּפְנִי לְשַׁמָּה וּתְאֵנָתִי לִקְצָפָה, חָשֹׂף חֲשָׂפָהּ וְהִשְׁלִיךְ הִלְבִּינוּ שָׂרִיגֶיהָ" (יואל א ד-ז),

וּבַשִּׁיר חָנְטָה פַגֶּיהָ – בשיר השירים (ב יג) נאמר: "הַתְּאֵנָה חָנְטָה פַגֶּיהָ".

דמויותכרך הבינוני

אִשָּׁה בַּעֲלַת כְּשָׁפִים

נֶחְרְדָה וְקָרְאָה בִּשְׁמוֹ

אַף בֶּן נְבִיאִים

נֶחֱרַד וְהִזְכִּיר אֶת שְׁמוֹ

הצגת הפתרון

שאול

אִשָּׁה בַּעֲלַת כְּשָׁפִים – בעלת האוב בעין דור,

נֶחְרְדָה וְקָרְאָה בִּשְׁמוֹ – של שאול, כאמור: "וַתֵּרֶא הָאִשָּׁה אֶת שְׁמוּאֵל וַתִּזְעַק בְּקוֹל גָּדוֹל, וַתֹּאמֶר הָאִשָּׁה אֶל שָׁאוּל לֵאמֹר: לָמָּה רִמִּיתָנִי, וְאַתָּה שָׁאוּל!" (שמואל א' כח יב),

אַף בֶּן נְבִיאִים, נֶחֱרַד וְהִזְכִּיר אֶת שְׁמוֹ – "וַיֹּאמְרוּ בְנֵי הַנְּבִיאִים אֶל אֱלִישָׁע: הִנֵּה נָא הַמָּקוֹם אֲשֶׁר אֲנַחְנוּ יֹשְׁבִים שָׁם לְפָנֶיךָ צַר מִמֶּנּוּ, נֵלְכָה נָּא עַד הַיַּרְדֵּן וְנִקְחָה מִשָּׁם אִישׁ קוֹרָה אֶחָת וְנַעֲשֶׂה לָּנוּ שָׁם מָקוֹם לָשֶׁבֶת שָׁם [...], וַיֵּלֶךְ אִתָּם וַיָּבֹאוּ הַיַּרְדֵּנָה וַיִּגְזְרוּ הָעֵצִים, וַיְהִי הָאֶחָד מַפִּיל הַקּוֹרָה וְאֶת הַבַּרְזֶל נָפַל אֶל הַמָּיִם, וַיִּצְעַק וַיֹּאמֶר: אֲהָהּ אֲדֹנִי, וְהוּא שָׁאוּל!" (מלכים ב' ו א-ה) ("הִזְכִּיר אֶת שְׁמוֹ" במובן שאול, תרתי משמע).

מקומותכרך גקל

סְמוּכָה לְבֵית אֵל

וּבֵינֵיהֶן הֻצַּב הַרְאֵל

אֵלֶיהָ נִשְׁלְחוּ מְרַגְּלִים

וּשְׁלֹשֶׁת אַלְפֵי חַיָּלִים

בָּרִאשׁוֹן נִצְּחָה

וּבַשֵּׁנִי נֻצְּחָה

בִּמְנוּסַת הַסָּחָה

נֶעֶזְבָה פְּתוּחָה

הצגת הפתרון

העי

סְמוּכָה לְבֵית אֵל – "הָעַי אֲשֶׁר עִם בֵּית אָוֶן, מִקֶּדֶם לְבֵית אֵל" (יהושע ז ב),

וּבֵינֵיהֶן הֻצַּב הַרְאֵל – 'הראל' הוא מזבח, כאמור: "וְהַהַרְאֵל אַרְבַּע אַמּוֹת, וּמֵהָאֲרִיאֵל וּלְמַעְלָה הַקְּרָנוֹת אַרְבַּע" (יחזקאל מג טו); אברהם בנה מזבח בין בית אל ובין העי, כאמור: "וַיַּעְתֵּק מִשָּׁם הָהָרָה מִקֶּדֶם לְבֵית אֵל וַיֵּט אָהֳלֹה, בֵּית אֵל מִיָּם וְהָעַי מִקֶּדֶם, וַיִּבֶן שָׁם מִזְבֵּחַ לַה' (בראשית יב ח);

אֵלֶיהָ נִשְׁלְחוּ מְרַגְּלִים – "וַיִּשְׁלַח יְהוֹשֻׁעַ אֲנָשִׁים מִירִיחוֹ הָעַי אֲשֶׁר עִם בֵּית אָוֶן מִקֶּדֶם לְבֵית אֵל, וַיֹּאמֶר אֲלֵיהֶם לֵאמֹר: עֲלוּ וְרַגְּלוּ אֶת הָאָרֶץ, וַיַּעֲלוּ הָאֲנָשִׁים וַיְרַגְּלוּ אֶת הָעָי" (יהושע שם),

וּשְׁלֹשֶׁת אַלְפֵי חַיָּלִים – כאמור בהמשך: "וַיָּשֻׁבוּ אֶל יְהוֹשֻׁעַ וַיֹּאמְרוּ אֵלָיו: אַל יַעַל כָּל הָעָם, כְּאַלְפַּיִם אִישׁ אוֹ כִּשְׁלֹשֶׁת אֲלָפִים אִישׁ יַעֲלוּ וְיַכּוּ אֶת הָעָי [...] וַיַּעֲלוּ מִן הָעָם שָׁמָּה כִּשְׁלֹשֶׁת אֲלָפִים אִישׁ" (פס' ג-ד).

בָּרִאשׁוֹן נִצְּחָה – בקרב הראשון, ניצחו אנשי העי את ישראל, כמסופר שם בהמשך: "וַיָּנֻסוּ לִפְנֵי אַנְשֵׁי הָעָי, וַיַּכּוּ מֵהֶם אַנְשֵׁי הָעַי כִּשְׁלֹשִׁים וְשִׁשָּׁה אִישׁ, וַיִּרְדְּפוּם לִפְנֵי הַשַּׁעַר עַד הַשְּׁבָרִים וַיַּכּוּם בַּמּוֹרָד, וַיִּמַּס לְבַב הָעָם וַיְהִי לְמָיִם" (פס' ד-ה),

וּבַשֵּׁנִי נֻצְּחָה – בקרב השני נוצחה העי על ידי ישראל, שכן

בִּמְנוּסַת הַסָּחָה נֶעֶזְבָה פְּתוּחָה – ישראל ביימו מנוסה מפניהם כדי להטעות אותם, והם רדפו אחריהם ועזבו את עירם פתוחה, כך שישראל יכלו לחדור אליה ולשרפה, כמתואר בכתוב: "וַיִּנָּגְעוּ יְהוֹשֻׁעַ וְכָל יִשְׂרָאֵל לִפְנֵיהֶם וַיָּנֻסוּ דֶּרֶךְ הַמִּדְבָּר. וַיִּזָּעֲקוּ כָּל הָעָם אֲשֶׁר בָּעַי לִרְדֹּף אַחֲרֵיהֶם, וַיִּרְדְּפוּ אַחֲרֵי יְהוֹשֻׁעַ וַיִּנָּתְקוּ מִן הָעִיר. וְלֹא נִשְׁאַר אִישׁ בָּעַי וּבֵית אֵל אֲשֶׁר לֹא יָצְאוּ אַחֲרֵי יִשְׂרָאֵל וַיַּעַזְבוּ אֶת הָעִיר פְּתוּחָה וַיִּרְדְּפוּ אַחֲרֵי יִשְׂרָאֵל [...] וְהָאוֹרֵב קָם מְהֵרָה מִמְּקוֹמוֹ וַיָּרוּצוּ כִּנְטוֹת יָדוֹ וַיָּבֹאוּ הָעִיר וַיִּלְכְּדוּהָ וַיְמַהֲרוּ וַיַּצִּיתוּ אֶת הָעִיר בָּאֵשׁ. וַיִּפְנוּ אַנְשֵׁי הָעַי אַחֲרֵיהֶם וַיִּרְאוּ וְהִנֵּה עָלָה עֲשַׁן הָעִיר הַשָּׁמַיְמָה וְלֹא הָיָה בָהֶם יָדַיִם לָנוּס הֵנָּה וָהֵנָּה, וְהָעָם הַנָּס הַמִּדְבָּר נֶהְפַּךְ אֶל הָרוֹדֵף. וִיהוֹשֻׁעַ וְכָל יִשְׂרָאֵל רָאוּ כִּי לָכַד הָאֹרֵב אֶת הָעִיר וְכִי עָלָה עֲשַׁן הָעִיר, וַיָּשֻׁבוּ וַיַּכּוּ אֶת אַנְשֵׁי הָעָי. וְאֵלֶּה יָצְאוּ מִן הָעִיר לִקְרָאתָם וַיִּהְיוּ לְיִשְׂרָאֵל בַּתָּוֶךְ אֵלֶּה מִזֶּה וְאֵלֶּה מִזֶּה, וַיַּכּוּ אוֹתָם עַד בִּלְתִּי הִשְׁאִיר לוֹ שָׂרִיד וּפָלִיט" (שם ח טו-כב).
