נוסח הפיוט

אָנָּא אֵל נָא, הוֹשַׁע נָא וְהוֹשִׁיעָה נָּא, אָבִינוּ אָתָּה.

לְמַעַן תָּמִים בְּדֹרֹתָיו, הַנִּמְלָט בְּרוֹב צִדְקוֹתָיו,

מֻצָּל מִשֶּׁטֶף בְּבֹא מַבּוּל מָיִם,

לְאוֹם אֲנִי חוֹמָה, הוֹשַׁע נָא וְהוֹשִׁיעָה נָּא, אָבִינוּ אָתָּה.

לְמַעַן שָׁלֵם בְּכָל מַעֲשִׂים, הַמְנֻסֶּה בַּעֲשָׂרָה נִסִּים,

כְּשָׁר מַלְאָכִים נָם יֻקַּח נָא מְעַט מָיִם,

לְבָרָה כַּחַמָּה, הוֹשַׁע נָא וְהוֹשִׁיעָה נָּא, אָבִינוּ אָתָּה.

לְמַעַן רַךְ וְיָחִיד נֶחֱנַט פְּרִי לְמֵאָה, זָעַק אַיֵּה הַשֶּׂה לְעוֹלָה,

בִּשְּׂרוּהוּ עֲבָדָיו מָצָאנוּ מָיִם,

לְגוֹלָה וְסוּרָה, הוֹשַׁע נָא וְהוֹשִׁיעָה נָּא, אָבִינוּ אָתָּה.

לְמַעַן קִדַּם שְׂאֵת בְּרָכָה, הַנִּשְׂטַם וּלְשִׁמְךָ חִכָּה,

מְיַחֵם בְּמַקְלוֹת בְּשִׁקֲּתוֹת הַמָּיִם,

לְדָמְתָה לְתָמָר, הוֹשַׁע נָא וְהוֹשִׁיעָה נָּא, אָבִינוּ אָתָּה.

לְמַעַן צָדַק הֱיוֹת לְךָ לְכֹהֵן, כֶּחָתָן פְּאֵר יְכַהֵן,

מְנֻסֶּה בְּמַסָּה בְּמֵי מְרִיבַת מָיִם,

לְהָהָר הַטּוֹב, הוֹשַׁע נָא וְהוֹשִׁיעָה נָּא, אָבִינוּ אָתָּה.

לְמַעַן פֹּאַר הֱיוֹת גְּבִיר לְאֶחָיו, יְהוּדָה אֲשֶׁר גָּבַר בְּאֶחָיו,

מִסְפַּר רֹבַע מִדָּלְיָו יִזַּל מָיִם,

לוֹא לָנוּ כִּי אִם לְמַעַנְךָ, הוֹשַׁע נָא וְהוֹשִׁיעָה נָּא, אָבִינוּ אָתָּה.

לְמַעַן עָנָו מִכֹּל וְנֶאֱמָן, אֲשֶׁר בְּצִדְקוֹ כִּלְכֵּל הַמָּן,

מָשׁוּךְ לְגוֹאֵל וּמָשׁוּי מִמָּיִם,

לְזֹאת הַנִּשְׁקָפָה, הוֹשַׁע נָא וְהוֹשִׁיעָה נָּא, אָבִינוּ אָתָּה.

לְמַעַן שַׂמְתּוֹ כְּמַלְאֲכֵי מְרוֹמִים, הַלּוֹבֵשׁ אוּרִים וְתֻמִּים,

מְצֻוֶּה לָבֹא בַּמִּקְדָּשׁ בְּקִדּוּשׁ יָדַיִם וְרַגְלַיִם וּרְחִיצַת מָיִם,

לְחוֹלַת אַהֲבָה, הוֹשַׁע נָא וְהוֹשִׁיעָה נָּא, אָבִינוּ אָתָּה.

לְמַעַן נְבִיאָה מְחֹלַת מַחֲנָיִם, לִכְמֵהֵי לֵב הוּשְׂמָה עֵינָיִם,

לְרַגְלָהּ רָצָה עֲלוֹת וָרֶדֶת בְּאֵר מָיִם,

לְטֹּבוּ אֹהָלָיו, הוֹשַׁע נָא וְהוֹשִׁיעָה נָּא, אָבִינוּ אָתָּה.

לְמַעַן מְשָׁרֵת לֹא מָשׁ מֵאֹהֶל, וְרוּחַ הַקֹּדֶשׁ עָלָיו אִהֵל,

בְּעָבְרוֹ בַיַרְדֵּן נִכְרְתוּ הַמָּיִם,

לְיָפָה וּבָרָה, הוֹשַׁע נָא וְהוֹשִׁיעָה נָּא, אָבִינוּ אָתָּה.

לְמַעַן לָמַד רְאוֹת לְטוֹבָה אוֹת, זָעַק אַיֵּה נִפְלָאוֹת,

מִצָּה טַל מִגִּזָּה מְלֹא הַסֵּפֶל מָיִם,

לְכַלַּת לְבָנוֹן, הוֹשַׁע נָא וְהוֹשִׁיעָה נָּא, אָבִינוּ אָתָּה.

לְמַעַן כְּלוּלֵי עֲשׂוֹת מִלְחַמְתֶּךָ, אֲשֶׁר בְּיָדָם תַּתָּה יְשׁוּעָתֶךָ,

צְרוּפֵי מִגּוֹי בְּלָקְקָם בְּיָדָם מָיִם,

לְלֹא בָּגְדוּ בָךְ, הוֹשַׁע נָא וְהוֹשִׁיעָה נָּא, אָבִינוּ אָתָּה.

לְמַעַן יָחִיד צוֹרְרִים דָּשׁ, אֲשֶׁר מֵרֶחֶם לְנָזִיר הֻקְדָּשׁ,

מִמַּכְתֵּשׁ לֶחִי הִבְקַעְתָּ לוֹ מָיִם,

לְמַעַן שֵׁם קָדְשֶׁךָ, הוֹשַׁע נָא וְהוֹשִׁיעָה נָּא, אָבִינוּ אָתָּה.

לְמַעַן טוֹב הֹלֵךְ וְגָדֵל, אֲשֶׁר מֵעשֶׁק עֵדָה חִדֵּל,

בְּשׁוּב עָם מֵחֲטֹא צָו שְׁאָב מָיִם,

לְנָאוָה כִּירוּשָׁלָיִם, הוֹשַׁע נָא וְהוֹשִׁיעָה נָּא, אָבִינוּ אָתָּה.

לְמַעַן חַיָּךְ מְכַרְכֵּר בְּשִׁיר, הַמְלַמֵּד תּוֹרָה בְּכָל כְּלֵי שִׁיר,

מְנַסֵּךְ לְפָנֶיךָ כְּתָאַב שְׁתוֹת מָיִם,

לְשָׂמוּ בְךָ סִבְרָם, הוֹשַׁע נָא וְהוֹשִׁיעָה נָּא, אָבִינוּ אָתָּה.

לְמַעַן זָךְ עָלָה בִּסְעָרָה, הַמְקַנֵּא וּמֵשִׁיב עֶבְרָה,

לְפִלּוּלוֹ יָרְדָה אֵשׁ וְלִחֲכָה עָפָר וּמָיִם,

לְעֵינֶיהָ בְּרֵכוֹת, הוֹשַׁע נָא וְהוֹשִׁיעָה נָּא, אָבִינוּ אָתָּה.

לְמַעַן וְשֵׁרֵת בֶּאֱמֶת לְרַבּוֹ, פִּי שְׁנַיִם בְּרוּחוֹ נֶאֱצַל בּוֹ,

בְּקַחְתּוֹ מְנַגֵּן נִתְמַלְּאוּ גֵבִים מָיִם,

לְפוֹצְחֵי מִי כָמוֹךָ, הוֹשַׁע נָא וְהוֹשִׁיעָה נָּא, אָבִינוּ אָתָּה.

לְמַעַן הִרְהֵר עֲשׂוֹת רְצוֹנֶךָ, הַמַּכְרִיז תְּשׁוּבָה לְצֹאנֶךָ,

אָז בְּבֹא מְחָרֵף סָתַם עֵינֹת מָיִם,

לְצִיּוֹן מִכְלַל יֹפִי, הוֹשַׁע נָא וְהוֹשִׁיעָה נָּא, אָבִינוּ אָתָּה.

לְמַעַן דְּרָשׁוּךָ בְּתוֹךְ הַגּוֹלָה, וְסוֹדְךָ לָמוֹ נִגְלָה,

בְּלִי לְהִתְגָּאֵל דָּרְשׁוּ זֵרְעֹנִים וּמָיִם,

לְקוֹרְאֶיךָ בַּצָּר, הוֹשַׁע נָא וְהוֹשִׁיעָה נָּא, אָבִינוּ אָתָּה.

לְמַעַן גָּמַר חָכְמָה וּבִינָה, סֹפֵר מָהִיר מְפַלֵּשׂ אֲמָנָה,

מְחַכְּמֵנוּ אֲמָרִים הַמְּשׁוּלִים בְּרַחֲבֵי מָיִם,

לְרַבָּתִי עָם, הוֹשַׁע נָא וְהוֹשִׁיעָה נָּא, אָבִינוּ אָתָּה.

לְמַעַן בָּאֵי לְךָ הַיּוֹם בְּכָל לֵב, שׁוֹפְכִים לְךָ שִׂיחַ בְּלֹא לֵב וָלֵב,

שׁוֹאֲלִים מִמְּךָ עֹז מִטְרוֹת מָיִם,

לְשׁוֹרְרוּךָ בַּיָּם, הוֹשַׁע נָא וְהוֹשִׁיעָה נָּא, אָבִינוּ אָתָּה.

לְמַעַן אוֹמְרֵי יִגְדַּל שְׁמֶךָ, וְהֵם נַחֲלָתְךָ וְעַמֶּךָ,

צְמֵאִים לְיֶשְׁעֲךָ כְּאֶרֶץ עֲיֵפָה לְמָיִם,

לְתַרְתָּ לָמוֹ מְנוּחָה, הוֹשַׁע נָא וְהוֹשִׁיעָה נָּא, אָבִינוּ אָתָּה.

על הפיוט

הפיוט הבא מזכיר את צדיקי הדורות (על פי סדר תשר"ק), ואת זכויותיהם הקשורות לעניין המים, ובו אנו מתפללים לזכות לישועה ולברכת גשמים בזכות צדיקים אלו.244 בסוף כל בית מופיע כינוי של עם ישראל, ירושלים, או בית המקדש (על פי סדר א"ב), הקשור לזכותו של אותו צדיק. הזכרת שמות הצדיקים ומעשיהם, מרוממת את המתפלל ומעוררת אותו לשאוף להידמות להם, ללמוד ממידותיהם, ולדבוק בדרכיהם; כך התפילה מממשת את ייעודה – העלאת מדרגתו של המתפלל וזיכוך רצונותיו.

אָנָּא אֵל נָא, הוֹשַׁע נָא וְהוֹשִׁיעָה נָּא,245אָבִינוּ אָתָּה – כלשון הכתוב: "וְעַתָּה ה' »אƒבינוּ »א»תּה, אֲנַחְנוּ הַחֹמֶר וְאַתָּה יֹצְרֵנוּ, וּמַעֲשֵׂה יָדְךָ כֻּלָּנוּ" (ישעיהו סד ז), ופירש 'מצודת דוד': "'וְעַתָּה ה'' – הלא אתה אבינו, ודרך האב לרחם על הבן; [...] 'אֲנַחְנוּ הַחֹמֶר' – הלוא אתה בראת אותנו וכולנו המה מעשה ידיך, והיאך תשחית מעשה ידיך". הביטוי שאנו מזכירים כאן: 'אָבִינוּ אָתָּה' – כולל את שתי הטענות שבפסוק, שכן הבן הוא יצירתו של האב. אנו פונים לקב"ה ומבקשים: 'הוֹשַׁע נָא וְהוֹשִׁיעָה נָּא' – הן מצד רחמיך עלינו כאב, הן מצד הקשר המיוחד הקיים בינינו כיוצר ויצירתו.246

הַנִּמְלָט בְּרוֹב צִדְקוֹתָיו – שניצל בזכות צדקותיו הרבות, כפי שאמר לו הקב"ה: "בֹּא אַתָּה וְכָל בֵּיתְךָ אֶל הַתֵּבָה, כִּי אֹתְךָ רָאִיתִי צַדִּיק לְפָנַי בַּדּוֹר הַזֶּה" (שם ז א), וכך היה מֻצָּל מִשֶּׁטֶף בְּבֹא מַבּוּל מָיִם.

לְאוֹם אֲנִי חוֹמָה – לאומת ישראל, העומדת כחומה בצורה בשמירתה על אמונתה, תורתה, ומצוותיה248– הוֹשַׁע נָא וְהוֹשִׁיעָה נָּא, אָבִינוּ אָתָּה. כשם שנח שמר על צדקותו ותמימותו על אף שהיה 'בְּדֹרֹתָיו' – מוקף באנשי דורו המושחתים, כך עם ישראל שומר על ייחודו ועל נאמנותו לקב"ה, על אף קשיי הגלות ואומות העולם הסובבות אותו ומנסות להדיחו מדרך התורה והאמונה.249 מתוך הזכרת דמותו של נח, והעמידה שלו ושל עם ישראל כחומה בצורה כנגד כל מתנגדיהם, מתעורר המתפלל לחזק אף בקרבו תכונה זו לצורך עבודת ה', כדברי ה'טור':

"יהודה בן תימא אומר: הוי עז כנמר וקל כנשר, רץ כצבי וגיבור כארי לעשות רצון אביך שבשמיים"250– פֵּרֵט ארבעה דברים בעבודת הבורא יתברך. והתחיל ב'עז כנמר', לפי שהוא כלל גדול בעבודת הבורא יתברך; לפי שפעמים אדם חפץ לעשות מצוה, ונמנע מלעשותה מפני בני אדם שמלעיגים עליו. ועל כן הזהיר שתעיז פניך כנגד המלעיגים ואל תימנע מלעשות המצוה.251

לְמַעַן שָׁלֵם בְּכָל מַעֲשִׂים – בזכות אברהם, עליו נאמר: "וּמָצָאתָ אֶת לְבָבוֹ נֶאֱמָן לְפָנֶיךָ" (נחמיה ט ח), ופירש ה'רוקח' (בפירוש הסידור): "שהיה עובד בלב נאמן לפניך, כאדם שחושב: איך אעבוד לאדוני ביראה ובאהבה ובלב שלם". זהו פירוש הביטוי: 'שָׁלֵם בְּכָל מַעֲשִׂים' – את כל מעשיו הוא קיים בשלימות, התלויה בכוונת הלב בעשייתם. ניתן גם לפרש: 'שָׁלֵם בְּכָל מַעֲשִׂים' – אברהם עצמו נהיה שלם, באמצעות כל המעשים שעשה; כל מעשיו נבעו מהרצון התמידי לקיים את רצון ה', וכך חייו כולם היוו מארג אחדותי ורציף בעל מגמה אחת ויחידה. מי שחי כך הרי הוא 'שָׁלֵם' – ליבו אינו מחולק לרצונות שונים, אלא הוא מסור כולו בשלימות לרצונו האמיתי והיחיד.252 וכך כתב הרמב"ם:

העובד מאהבה – עוסק בתורה ובמצוות והולך בנתיבות החכמה, לא מפני דבר בעולם, ולא מפני יראת הרעה, ולא כדי לירש הטובה, אלא עושה האמת מפני שהוא אמת, וסוף הטובה לבוא בגללה. ומעלה זו היא מעלה גדולה מאוד, ואין כל חכם זוכה לה. והיא מעלת אברהם אבינו, שקראו הקב"ה אוהבו – לפי שלא עבד אלא מאהבה. והיא המעלה שציוונו בה הקב"ה על ידי משה, שנאמר: "וְאָהַבְתָּ אֵת ה' אֱלֹהֶיךָ" (דברים ו ה). ובזמן שיאהב אדם את ה' אהבה הראויה – מיד יעשה כל המצוות מאהבה.253

הַמְנֻסֶּה בַּעֲשָׂרָה נִסִּים – שנתנסה בעשרה נסיונות ועמד בכולם.254 הרמב"ם מונה את עשרה הניסיונות:

הראשון – הגֵּרות, והוא אומרו יתעלה לו: "לֶךְ לְךָ מֵאַרְצְךָ" (בראשית יב א); והשני – הרעב אשר מצא בארץ כנען כאשר השתקע בארץ, וכבר יועד לו: "וְאֶעֶשְׂךָ ... וַאֲבָרֶכְךָ וַאֲגַדְּלָה שְׁמֶךָ" (שם שם ב), וזה ניסיון גדול; [...] השלישי – עשוֹק המצרים אותו בלקיחת שרה לפרעה; והרביעי – הלחמו בארבעה מלכים; והחמישי – לוקחו הגר כשנואש מההולדה משרה; והשישי – המילה, אשר צֻווה בה בשנות הזיקנה; והשביעי – עשוֹק מלך גרר אותו בלקיחת שרה גם כן; והשמיני – הרחקת הגר אחר היבנותו ממנה; והתשיעי – הרחקת בנו ישמעאל, והוא אומרו לו יתעלה: "אַל יֵרַע בְּעֵינֶיךָ עַל הַנַּעַר וְעַל אֲמָתֶךָ" (שם כא יב), וכבר העיד הכתוב כמה היה קשה הדבר עליו: "וַיֵּרַע הַדָּבָר מְאֹד בְּעֵינֵי אַבְרָהָם" (שם שם יא), אבל הוא קיים ציוויו יתעלה, וגרשם; והעשירי – עקידת יצחק.255

ניסיון הוא מלשון 'נֵס', כדברי המדרש: "'וְהָאֱלֹהִים נִסָּה אֶת אַבְרָהָם' (בראשית כב א) – רבי יוסי הגלילי אומר: גידלו כַּנֵּס הזה של ספינה";256 מטרת הניסיון היא לנשא את האדם למדרגה גבוהה יותר, ולהוציא אל הפועל את המידות הטובות הגנוזות בעמקי נפשו.

כְּשָׁר מַלְאָכִים נָם יֻקַּח נָא מְעַט מָיִם – כאשר ראה את המלאכים ('שָׁר' = ראה),257 רץ לקראתם ואמר להם ('נָם' מלשון נאום ודיבור):258"יֻקַּח נָא מְעַט מַיִם וְרַחֲצוּ רַגְלֵיכֶם, וְהִשָּׁעֲנוּ תַּחַת הָעֵץ. וְאֶקְחָה פַת לֶחֶם וְסַעֲדוּ לִבְּכֶם, אַחַר תַּעֲבֹרוּ, כִּי עַל כֵּן עֲבַרְתֶּם עַל עַבְדְּכֶם" (שם יח ד-ה). מעשה זה אירע ביום השלישי למילתו של אברהם,259 ועל אף חולשתו ישב בפתח האוהל כדי למצוא עוברי אורח ולארחם; דבר זה מבטא את תכונתו המיוחדת שהוזכרה לעיל: 'שָׁלֵם בְּכָל מַעֲשִׂים' – מחמת השתוקקותו הפנימית להיטיב, לא הסתפק בביצוע המוטל עליו, אלא חיפש הזדמנויות בהן יוכל לממשה. לְבָרָה כַּחַמָּה – 'בָּרָה' מלשון 'ברור': בעלת אור ברור (=בהיר וזך) כשל השמש; זהו דימוי לאומת ישראל, המאירה לעולם באור האלוהי שבקרבה; כלשון הכתוב: "מִי זֹאת הַנִּשְׁקָפָה כְּמוֹ שָׁחַר, יָפָה כַלְּבָנָה, בָּרָה כַּחַמָּה" (שיר השירים ו י)260– הוֹשַׁע נָא וְהוֹשִׁיעָה נָּא, אָבִינוּ אָתָּה. כשם שאברהם הזריח בעולם את שמש האמונה והאיר לכל האנושות באורה, כך עם ישראל – ממשיך דרכו של אברהם, הוא כחמה המאירה לעולם כולו; תפקידו הוא ללמד את האומות כולן את האמונה בקב"ה ואת דרך עבודתו.

נֶחֱנַט פְּרִי לְמֵאָה – נולד לאביו כשהיה בן מאה, כפרי הנחנט (חנטה = תחילת היווצרות הפרי),262 כמאמר הכתוב: "וְאַבְרָהָם בֶּן מְאַת שָׁנָה בְּהִוָּלֶד לוֹ אֵת יִצְחָק בְּנוֹ" (בראשית כא ה).

זָעַק אַיֵּה הַשֶּׂה לְעֹלָה – כמו שכתוב: "וַיֹּאמֶר: הִנֵּה הָאֵשׁ וְהָעֵצִים, וְאַיֵּה הַשֶּׂה לְעֹלָה" (שם כב ז). כשיצחק ראה את האש והעצים, הבין שמטרת הליכתם היא הקרבת קורבן; כיוון שלא היתה עמהם בהמה, החל להבין מעצמו מיהו השה לעולה, ונפלו עליו מורא וחרדה, ולפיכך – 'זָעַק'.263

בִּשְּׂרוּהוּ עֲבָדָיו מָצָאנוּ מָיִם – כמו שכתוב: "וַיְהִי בַּיּוֹם הַהוּא וַיָּבֹאוּ עַבְדֵי יִצְחָק וַיַּגִּדוּ לוֹ עַל אֹדוֹת הַבְּאֵר אֲשֶׁר חָפָרוּ, וַיֹּאמְרוּ לוֹ: מָצָאנוּ מָיִם" (שם כו לב), וכתב שם רמב"ן: "וקרוב לי, שהוא הבאר שכרה אברהם, ונתן לו [=לאבימלך] שבע כבשות לעדה, כי סתמוהו פלישתים עם האחרים, וישב יצחק ויחפור אותו, ויקרא לו גם הוא שם כשם אשר קרא לו אביו". יצחק שב וחופר את הבארות שחפר אביו וסתמום הפלישתים, ואף קורא להן באותם שמות, כמו שכתוב: "וַיָּשָׁב יִצְחָק וַיַּחְפֹּר אֶת בְּאֵרֹת הַמַּיִם אֲשֶׁר חָפְרוּ בִּימֵי אַבְרָהָם אָבִיו וַיְסַתְּמוּם פְּלִשְׁתִּים אַחֲרֵי מוֹת אַבְרָהָם, וַיִּקְרָא לָהֶן שֵׁמוֹת כַּשֵּׁמֹת אֲשֶׁר קָרָא לָהֶן אָבִיו" (שם שם יח). כוונתו של יצחק היתה לכבד בכך את זכרו של אברהם, כדברי המדרש:

בנוהג שבעולם, אדם בונה בית וקורא לו שם, ועומד בנו ומחדש בו דבר ומחדש לו שם אחר. אבל הצדיקים אינן כן; בוא וראה: כשחפר אברהם בארות אלו – קרא להן שם. קינאו בו פלישתים, וסתמו פלישתים אותם בימי יצחק. ואף על פי שחזר עליהן יצחק וחפרן, לא חידש להן שם, אלא קרא אותם 'כַּשֵּׁמֹת אֲשֶׁר קָרָא לָהֶן אָבִיו'.264

לְגוֹלָה וְסוּרָה – לאומת ישראל, השרויה בגלות ומוּסרת ממקומה הטבעי – ארץ ישראל265– הוֹשַׁע נָא וְהוֹשִׁיעָה נָּא, אָבִינוּ אָתָּה. יצחק הוא היחיד מבין האבות שלא יצא מארץ ישראל, על פי ציווי הקב"ה: "וַיֵּרָא אֵלָיו ה', וַיֹּאמֶר: אַל תֵּרֵד מִצְרָיְמָה, שְׁכֹן בָּאָרֶץ אֲשֶׁר אֹמַר אֵלֶיךָ" (שם כו ב), וכתב שם רש"י: "שאתה עולה תמימה, ואין חוצה לארץ כדאי לך". יצחק הוא 'עולה תמימה' – חייו צריכים להתנהל באופן האידיאלי והשלם, ולפיכך אין לו לצאת מהארץ. זאת, משום שארץ ישראל היא המקום הטבעי של זרע אברהם, ורק בה יכולים בניו לחיות כראוי את חייהם הרוחניים; וכך כתב הראי"ה קוק:

ארץ ישראל איננה דבר חיצוני, קניין חיצוני לאומה, רק בתור אמצעי למטרה של ההתאגדות הכללית, והחזקת קיומה החומרי או אפילו הרוחני. ארץ ישראל היא חטיבה עצמותית, קשורה בקשר חיים עם האומה, חבוקה בסגולות פנימיות עם מציאותה.266

עניין זה, שהתבטא בכך שיצחק נשאר בארץ, מבליט את החיסרון במצב האומה כעת, בו היא 'גּוֹלָה וְסוּרָה'; על כך תפילתנו: 'לְגוֹלָה וְסוּרָה, הוֹשַׁע נָא וְהוֹשִׁיעָה נָּא' – והשיבה לארצה, 'אָבִינוּ אָתָּה'.

אֲשֶׁר בְּצִדְקוֹ כִּלְכֵּל הַמָּן – שבזכותו ירד המן לישראל, שהרי כשמת פסקה ירידתו, כדברי הגמרא: "שלושה פרנסים טובים עמדו לישראל, אלו הן: משה ואהרון ומרים, ושלוש מתנות טובות ניתנו על ידם, ואלו הן: באר וענן ומן. באר בזכות מרים, עמוד ענן בזכות אהרון, מן בזכות משה"281– "בשבעה באדר מת משה, ופסק מן מלירד".282

מָשׁוּךְ לְגוֹאֵל – שלא רצה להתמנות לתפקיד הגואל והמנהיג של ישראל, אלא היה 'משוך' לכך בעל כורחו על ידי הקב"ה, כמתואר בכתוב (שמות פרקים ג-ד).283 פירוש נוסף: שכבר בעת שנולד ונמשך מבטן אימו, היה מיועד להיות הגואל, כדברי הגמרא: "מלמד שהיתה [מרים] מתנבאה כשהיא אחות אהרון, ואומרת: עתידה אימי שתלד בן שמושיע את ישראל. וכיוון שנולד משה – נתמלא כל הבית כולו אור, עמד אביה ונשקה על ראשה, אמר לה: בתי, נתקיימה נבואתך".284

וּמָשׁוּי מִמָּיִם – על ידי בת פרעה, כמו שכתוב: "וַתִּקְרָא שְׁמוֹ מֹשֶׁה, וַתֹּאמֶר: כִּי מִן הַמַּיִם מְשִׁיתִהוּ" (שמות ב י).

לְזֹאת הַנִּשְׁקָפָה – אומת ישראל, כלשון הכתוב: "מִי זֹאת הַנִּשְׁקָפָה כְּמוֹ שָׁחַר, יָפָה כַלְּבָנָה, בָּרָה כַּחַמָּה, אֲיֻמָּה כַּנִּדְגָלוֹת" (שיר השירים ו י) – הוֹשַׁע נָא וְהוֹשִׁיעָה נָּא, אָבִינוּ אָתָּה. במדרש נאמר:

כשהיו ישראל מהלכין במדבר, היו אומות העולם רואים היאך הקב"ה עושה להם נס, ומוריד להן את המן, ומעלה להם את הבאר, ומגיז להן את השליו, והוא בכבודו מאיר להם, שנאמר: "וַה' הֹלֵךְ לִפְנֵיהֶם יוֹמָם בְּעַמּוּד עָנָן לַנְחֹתָם הַדֶּרֶךְ, וְלַיְלָה בְּעַמּוּד אֵשׁ לְהָאִיר לָהֶם" (שמות יג כא), והיו אומות העולם רואים את הענן ביום, ואת אורו של אש בלילה, ואומרים: "מִי זֹאת הַנִּשְׁקָפָה כְּמוֹ שָׁחַר".285

על פי המדרש, פסוק זה מבטא את התפעלות האומות ממעלתם של ישראל כפי שנגלתה בתקופת דור המדבר.286 מכיוון שאז הונהגו על ידי משה, ונס ירידת המן שהוזכר במדרש אירע בזכותו – בחר הפייטן להשתמש כאן בכינוי זה לעם ישראל.

לְמַעַן מְשָׁרֵת לֹא מָשׁ מֵאֹהֶל – בזכות יהושע משרת משה, עליו נאמר: "וּמְשָׁרְתוֹ יְהוֹשֻׁעַ בִּן נוּן נַעַר, לֹא יָמִישׁ מִתּוֹךְ הָאֹהֶל" (שמות לג יא), ובזכות דבר זה – וְרוּחַ הַקֹּדֶשׁ עָלָיו אִהֵל – שרתה עליו רוח הקודש כאוהל,295 כדברי המדרש: "לפי ששימש יהושע [את] אדוניו – זכה לרוח הקודש, שנאמר: "וַיְהִי אַחֲרֵי מוֹת מֹשֶׁה עֶבֶד ה', וַיֹּאמֶר ה' אֶל יְהוֹשֻׁעַ בִּן נוּן מְשָׁרֵת מֹשֶׁה לֵאמֹר" (יהושע א א), שאין תלמוד [=צורך] לומר: 'מְשָׁרֵת מֹשֶׁה', למה נאמר? לומר לך: לפי שהיה 'מְשָׁרֵת מֹשֶׁה' – זכה לנבואה",296 וכן אמרו: "גדול שימושה יותר מלימודה, שהרי כל ישראל למדו תורה ממשה, ולא בחר הקב"ה אלא במשרתו יהושע בן נון".297

בְּעָבְרוֹ בַיַּרְדֵּן נִכְרְתוּ הַמָּיִם – שבהנהגתו עברו ישראל את הירדן, כמסופר בספר יהושע (פרקים ג-ד). מלשון הפיוט משמע שהוא היה הגורם לנס זה; ייתכן שהמקור לכך הוא הפסוק: "וַיֹּאמֶר ה' אֶל יְהוֹשֻׁעַ: הַיּוֹם הַזֶּה אָחֵל גַּדֶּלְךָ בְּעֵינֵי כָּל יִשְׂרָאֵל, אֲשֶׁר יֵדְעוּן כִּי כַּאֲשֶׁר הָיִיתִי עִם מֹשֶׁה אֶהְיֶה עִמָּךְ" (שם ג ז) – הרי שבמידה מסויימת הנס נעשה לצורכו, כדי להגדיל את כבודו ומעלתו בעיני ישראל.

לְיָפָה וּבָרָה – אומת ישראל, על פי לשון הכתוב: "מִי זֹאת הַנִּשְׁקָפָה כְּמוֹ שָׁחַר, י»»פה כַלְּבָנָה, »בּר»ה כַּחַמָּה" (שיר השירים ו י)298– הוֹשַׁע נָא וְהוֹשִׁיעָה נָּא, אָבִינוּ אָתָּה. הלבנה, אין אורה רב, ועיקר ייחודיותה הוא ביופי שלה, בזריחתה בתוך השמיים החשוכים; לפיכך, לדימוי ישראל ללבנה מתאים התואר 'יָפָה'. החמה, לעומת זאת, עיקר עניינה הוא הארתה לעולם; לפיכך, בדימוי ישראל לחמה מתאים התואר 'בָּרָה' = בעלת אור בהיר וזך. שני צדדים אלו קיימים בעם ישראל: הוא דומה ללבנה בכך שהוא לבדו מואר באור ה' בתוך העולם המלא אלילות, כפירה, וחומרנות – כלבנה המוארת באור השמש, בתוך השמיים החשוכים; הוא דומה אף לחמה, בכך שהוא מאיר לעולם כולו באורו, ומלמדו את האמונה בקב"ה ואת דרך עבודתו. בבית זה, העוסק בדמותו של יהושע, בחר הפייטן לכנות את אומת ישראל 'יָפָה וּבָרָה'; ייתכן, שזאת משום שיהושע הנהיג את עם ישראל בתקופת המעבר בין הזמן בו היו 'יָפָה כַלְּבָנָה', לבין הזמן בו היו 'בָּרָה כַּחַמָּה' – בין תקופת דור המדבר בה היו דומים ללבנה, שכן חיו אז בינם לבין עצמם, ללא יחסים עם אומות העולם והשפעה עליהן, לבין תקופת הכניסה לארץ, כיבושה, חלוקתה לנחלות, וביסוס שלטון האומה בארצה – מצב שמתוכו יכולים הם להוות אור לגויים, כחמה המאירה לארץ ולדרים.

'זָעַק' – שהתפלל על ישראל ולימד עליהם זכות בכל כוחו, כדברי המדרש: "'וְאַיֵּה כָל נִפְלְאֹתָיו' – [מוטעם] ב'זקף גדול',301 לפי שהרים קול בתפילתו בכוח גדול".302

מִצָּה (=סחט) טַל מִגִּזָּה מְלֹא הַסֵּפֶל מָיִם – כמסופר בכתוב. מטרת גדעון בניסיון זה לא היתה בחינת יכולתו של הקב"ה, שכן על זה נאמר: "לֹא תְנַסּוּ אֶת ה' אֱלֹהֵיכֶם" (דברים ו טז), אלא לבחון האם הוא עצמו ראוי שיֵעשה על ידו נס.303 המלבי"ם מסביר את משמעותו הסמלית של ניסיון זה:

ואמר שיציג גיזת הצמר בגורן וינסה בטל, כי הטל הוא מליצה [=משל ורמז] על השפע האלוהי היורד מלמעלה, כמו שכתוב: "אֶהְיֶה כַטַּל לְיִשְׂרָאֵל" (הושע יד ו), "שֶֹׁאשִׁי נִמְלָא טָל" (שיר השירים ה ב); ואם טל יהיה על הגיזה לבדה, הגם שעל כל הארץ יהיה חורב – רוצה לומר: הגם שכל הארץ – שהוא כל ישראל – בלתי ראויים אל הטל העליון, ויהיה שם חורב, בכל זאת אמשיך הטל בזכותי [שכן אותי מייצגת גיזת הטל], אז – וידעתי כי תושיע בידי ובזכותי את ישראל.

לְכַלַּת לְבָנוֹן – אומת ישראל, על פי לשון הכתוב: "אִתִּי מִלְּבָנוֹן כַּלָּה" (שיר השירים ד ח), הוֹשַׁע נָא וְהוֹשִׁיעָה נָּא, אָבִינוּ אָתָּה. חז"ל פירשו ש'לְבָנוֹן' הוא בית המקדש, "המלבין עוונותיהם של ישראל";304 על פי זה ניתן לפרש את הכינוי 'כַּלַּת לְבָנוֹן' – שכולה לבנה ונקייה מעוונות, כמו שכתוב בפסוק שלפני הפסוק הנזכר: "כֻּלָּךְ יָפָה רַעְיָתִי וּמוּם אֵין בָּךְ"; חטאיהם של ישראל הם חיצוניים להם, ואינם פוגמים בסגולתם הפנימית.305 זהו תוכן לימוד הזכות שלימד גדעון על ישראל: אם עשית ניסים לאבותינו על אף שלא היו ראויים לכך מצד זכויותיהם, עשֵה אף לנו, שהרי סגולתם העצמית של ישראל, הנותרת תמיד בתמימותה, היא הגורמת לניסים שאתה עושה עבורם.

וַיֹּאמֶר ה' אֶל גִּדְעוֹן: רַב הָעָם אֲשֶׁר אִתָּךְ מִתִּתִּי אֶת מִדְיָן בְּיָדָם, פֶּן יִתְפָּאֵר עָלַי יִשְׂרָאֵל לֵאמֹר: יָדִי הוֹשִׁיעָה לִּי [...]. וַיֹּאמֶר ה' אֶל גִּדְעוֹן: עוֹד הָעָם רָב, הוֹרֵד אוֹתָם אֶל הַמַּיִם וְאֶצְרְפֶנּוּ לְךָ שָׁם, וְהָיָה אֲשֶׁר אֹמַר אֵלֶיךָ: זֶה יֵלֵךְ אִתָּךְ – הוּא יֵלֵךְ אִתָּךְ, וְכֹל אֲשֶׁר אֹמַר אֵלֶיךָ: זֶה לֹא יֵלֵךְ עִמָּךְ – הוּא לֹא יֵלֵךְ. וַיּוֹרֶד אֶת הָעָם אֶל הַמָּיִם, וַיֹּאמֶר ה' אֶל גִּדְעוֹן: כֹּל אֲשֶׁר יָלֹק בִּלְשׁוֹנוֹ מִן הַמַּיִם כַּאֲשֶׁר יָלֹק הַכֶּלֶב – תַּצִּיג אוֹתוֹ לְבָד, וְכֹל אֲשֶׁר יִכְרַע עַל בִּרְכָּיו לִשְׁתּוֹת. וַיְהִי מִסְפַּר הַמְלַקְקִים בְּיָדָם אֶל פִּיהֶם שְׁלֹשׁ מֵאוֹת אִישׁ, וְכֹל יֶתֶר הָעָם כָּרְעוּ עַל בִּרְכֵיהֶם לִשְׁתּוֹת מָיִם. וַיֹּאמֶר ה' אֶל גִּדְעוֹן: בִּשְׁלֹשׁ מֵאוֹת הָאִישׁ הַמְלַקְקִים אוֹשִׁיעַ אֶתְכֶם וְנָתַתִּי אֶת מִדְיָן בְּיָדֶךָ, וְכָל הָעָם יֵלְכוּ אִישׁ לִמְקֹמוֹ.307

הקב"ה מורה לגדעון לצמצם את צבאו, כדי שיהיה ניכר שניצחונם הוא מאת ה' ולא מכוחם. לצמצום זה היתה מטרה נוספת – 'צירוף' הצבא מעובדי העבודה הזרה (כמו צירוף מתכת מהסיגים שבה). צירוף זה נעשה על פי האופן בו האנשים שתו: אלו שכרעו על ברכיהם והרכינו ראשם אל המים, סימן היה הדבר לכך שהם רגילים לכרוע לפני העבודה הזרה, והם הושבו לביתם; רק את אלו שהעלו מים בתוך כפות ידיהם וליקקום מתוכן – לקח עימו גדעון למלחמה, והם היו 'צְרוּפֵי מִגּוֹי' = הנבחרים מתוך העם.

לְלֹא בָּגְדוּ בָךְ – עם ישראל, ששומרים על נאמנותם לתורתך ומצוותיך על אף קשיי הגלות והנסיונות הרבים – הוֹשַׁע נָא וְהוֹשִׁיעָה נָּא, כשם שאז הושעת את ישראל על ידי שלוש מאות האנשים שלא בגדו בך לעבוד עבודה זרה, אָבִינוּ אָתָּה.

לְמַעַן יָחִיד צוֹרְרִים דָּשׁ – בזכות שמשון, שדש והרג את הפלישתים, צוררי ישראל, כשהוא 'יָחִיד' – ללא עזרת צבא. דבר זה רמוז כבר בברכת יעקב לשבט דן, המכוונת בעיקר לשמשון:308"דָּן יָדִין עַמּוֹ כְּאַחַד שִׁבְטֵי יִשְׂרָאֵל. יְהִי דָן נָחָשׁ עֲלֵי דֶרֶךְ, שְׁפִיפֹן עֲלֵי אֹרַח, הַנּשֵׁךְ עִקְּבֵי סוּס וַיִּפֹּל רֹכְבוֹ אָחוֹר" (בראשית מט טז-יז), וכתב שם רד"ק: "דימהו לנחש ולשפיפון שהלך יחידי, כי שאר החיות – ברוב [=בדרך כלל] הולכים חבורות, והנחש לעולם הולך יחידי וממית רבים. וכן שמשון הלך יחיד, והיה ממית רבים משונאי ישראל, רגלִים ופרשים".

אֲשֶׁר מֵרֶחֶם לְנָזִיר הֻקְדָּשׁ – שהוקדש על ידי מלאך ה' להיות נזיר כבר מאז היותו ברחם אימו, כפי שאמר לה: "הִנָּךְ הָרָה וְיֹלַדְתְּ בֵּן, וְעַתָּה אַל תִּשְׁתִּי יַיִן וְשֵׁכָר, וְאַל תֹּאכְלִי כָּל טֻמְאָה, כִּי נְזִיר אֱלֹהִים יִהְיֶה הַנַּעַר מִן הַבֶּטֶן עַד יוֹם מוֹתוֹ" (שופטים יג ז).

מִמַּכְתֵּשׁ לֶחִי הִבְקַעְתָּ לוֹ מָיִם – כמסופר בכתוב:

הוּא בָא עַד לֶחִי [=שם מקום] וּפְלִשְׁתִּים הֵרִיעוּ לִקְרָאתוֹ, וַתִּצְלַח עָלָיו רוּחַ ה' [...] וַיִּמְצָא לְחִי חֲמוֹר טְרִיָּה וַיִּשְׁלַח יָדוֹ וַיִּקָּחֶהָ, וַיַּךְ בָּהּ אֶלֶף אִישׁ. וַיֹּאמֶר שִׁמְשׁוֹן: בִּלְחִי הַחֲמוֹר חֲמוֹר חֲמֹרָתָיִם, בִּלְחִי הַחֲמוֹר הִכֵּיתִי אֶלֶף אִישׁ. וַיְהִי כְּכַלֹּתוֹ לְדַבֵּר וַיַּשְׁלֵךְ הַלְּחִי מִיָּדוֹ, וַיִּקְרָא לַמָּקוֹם הַהוּא רָמַת לֶחִי. וַיִּצְמָא מְאֹד, וַיִּקְרָא אֶל ה' וַיֹּאמַר: אַתָּה נָתַתָּ בְיַד עַבְדְּךָ אֶת הַתְּשׁוּעָה הַגְּדֹלָה הַזֹּאת, וְעַתָּה אָמוּת בַּצָּמָא וְנָפַלְתִּי בְּיַד הָעֲרֵלִים. וַיִּבְקַע אֱלֹהִים אֶת הַמַּכְתֵּשׁ אֲשֶׁר בַּלֶּחִי, וַיֵּצְאוּ מִמֶּנּוּ מַיִם וַיֵּשְׁתְּ וַתָּשָׁב רוּחוֹ וַיֶּחִי, עַל כֵּן קָרָא שְׁמָהּ עֵין הַקּוֹרֵא אֲשֶׁר בַּלֶּחִי עַד הַיּוֹם הַזֶּה.309

רש"י מפרש ש'הַמַּכְתֵּשׁ אֲשֶׁר בַּלֶּחִי' שממנו נבעו המים הוא הגומה של השן שבלחי החמור; רלב"ג מפרש שבאותו מקום, ששמו היה 'לֶחִי', היה מכתש באדמה, וממנו נבעו המים.

לְמַעַן שֵׁם קָדְשֶׁךָ – כדי ששמך יתקדש בעולם על ידי ישועתם של ישראל, הוֹשַׁע נָא וְהוֹשִׁיעָה נָּא, כשם שהתקדש שמך על ידי פעולותיו של שמשון להצלת ישראל מהפלישתים, אָבִינוּ אָתָּה.

לְמַעַן טוֹב הוֹלֵךְ וְגָדֵל – בזכות שמואל הנביא, עליו נאמר: "וְהַנַּעַר שְׁמוּאֵל הֹלֵךְ וְגָדֵל וָטוֹב, גַּם עִם ה' וְגַם עִם אֲנָשִׁים" (שמואל א' ב כו), אֲשֶׁר מֵעשֶׁק עֵדָה חִדֵּל – שגרם לעדת ישראל לחדול מעושק ומקלקולים חברתיים אחרים, כמו שכתוב: "וַיִּשְׁפֹּט שְׁמוּאֵל אֶת יִשְׂרָאֵל כֹּל יְמֵי חַיָּיו. וְהָלַךְ מִדֵּי שָׁנָה בְּשָׁנָה וְסָבַב בֵּית אֵל וְהַגִּלְגָּל וְהַמִּצְפָּה, וְשָׁפַט אֶת יִשְׂרָאֵל אֵת כָּל הַמְּקוֹמוֹת הָאֵלֶּה" (שם ז טו-טז). יש גורסים: מֵעִקְּשׁוּת לֵב (=עקמימות הלב, שאינו ישר בדרך ה') עֵדָה חִדֵּל310– שגרם להם לחדול מהעבירות שבינם לבין המקום, כמו שכתוב: "וַיֹּאמֶר שְׁמוּאֵל אֶל כָּל בֵּית יִשְׂרָאֵל לֵאמֹר: אִם בְּכָל לְבַבְכֶם אַתֶּם שָׁבִים אֶל ה' – הָסִירוּ אֶת אֱלֹהֵי הַנֵּכָר מִתּוֹכְכֶם וְהָעַשְׁתָּרוֹת, וְהָכִינוּ לְבַבְכֶם אֶל ה' וְעִבְדֻהוּ לְבַדּוֹ, וְיַצֵּל אֶתְכֶם מִיַּד פְּלִשְׁתִּים. וַיָּסִירוּ בְּנֵי יִשְׂרָאֵל אֶת הַבְּעָלִים וְאֶת הָעַשְׁתָּרֹת, וַיַּעַבְדוּ אֶת ה' לְבַדּוֹ" (שם שם ג-ד).

בְּשׁוּב עָם מֵחֲטֹא צָו שְׁאָב מָיִם – בעצרת התשובה הכללית שכינס, בה העם שבו מחטאיהם, ציווה לשאוב מים ולשופכם כסימן להכנעה לפני ה', כאומרים: "הרי אנו לפניך כמים הללו הנשפכין",311 כמסופר בכתוב: "וַיֹּאמֶר שְׁמוּאֵל: קִבְצוּ אֶת כָּל יִשְׂרָאֵל הַמִּצְפָּתָה, וְאֶתְפַּלֵּל בַּעַדְכֶם אֶל ה'. וַיִּקָּבְצוּ הַמִּצְפָּתָה, וַיִּשְׁאֲבוּ מַיִם וַיִּשְׁפְּכוּ לִפְנֵי ה', וַיָּצוּמוּ בַּיּוֹם הַהוּא, וַיֹּאמְרוּ שָׁם: חָטָאנוּ לַה', וַיִּשְׁפֹּט שְׁמוּאֵל אֶת בְּנֵי יִשְׂרָאֵל בַּמִּצְפָּה" (שם שם ה-ו).

לְנָאוָה כִּירוּשָׁלָיִם – לאומת ישראל, כלשון הכתוב: "יָפָה אַתְּ רַעְיָתִי כְּתִרְצָה, נָאוָה כִּירוּשָׁלִָם, אֲיֻמָּה כַּנִּדְגָּלוֹת" (שיר השירים ו ד) – הוֹשַׁע נָא וְהוֹשִׁיעָה נָּא, אָבִינוּ אָתָּה. שמואל הוא זה שזיהה – יחד עם דוד – את ירושלים כ'מָקוֹם אֲשֶׁר יִבְחַר ה' אֱלֹהֵיכֶם בּוֹ לְשַׁכֵּן שְׁמוֹ שָׁם' (דברים יא יב), המיועד לבניית בית המקדש;312 משום כך, בבית זה העוסק בדמותו – כונתה אומת ישראל 'נָאוָה כִּירוּשָׁלָיִם',313 כינוי הרומז לעבודת בית המקדש העומד בירושלים, כפי שיבואר להלן. הגר"א מפרש,314 שבפסוק שהובא משיר השירים הקב"ה משבח את ישראל בהיותם בגלות, בה הם חסרים שניים מהעמודים עליהם עומד העולם:315 עמוד התורה – כמו שכתוב: "מַלְכָּהּ וְשָׂרֶיהָ בַגּוֹיִם – אֵין תּוֹרָה" (איכה ב ט), וכן אמרו חז"ל: "כיוון שגלו ישראל ממקומן – אין לך ביטול תורה גדול מזה",316 ועמוד העבודה – עבודת בית המקדש, ולא נותר בידם כי אם עמוד גמילות החסדים. אף על פי כן, אומר הקב"ה שהם חביבים עליו כאילו יש בידם את זכות קיום שלושת העמודים, הרמוזים בשלושת הכינויים שבפסוק: הכינוי 'יָפָה כְּתִרְצָה' רומז לגמילות החסדים, שכן תרצה היא עיר המלוכה של מלכי ישראל,317 שהיו מלכי חסד, כפי שנאמר: "הִנֵּה נָא שָׁמַעְנוּ כִּי מַלְכֵי בֵּית יִשְׂרָאֵל כִּי מַלְכֵי חֶסֶד הֵם" (מלכים א' כ לא); הכינוי 'נָאוָה כִּירוּשָׁלִָם' רומז לעבודת בית המקדש בירושלים, והכינוי 'אֲיֻמָּה כַּנִּדְגָּלוֹת' רומז לתורה, שכן ה'נִדְגָּלוֹת' הן מחנות בני ישראל במדבר שחנו על דגליהם, והם אלו שקיבלו את התורה. ייתכן שמשמעות הדבר היא, שאף על פי שלימוד התורה ועבודת המקדש אינם מופיעים בפועל בשלימות אצל ישראל בעת הגלות – מעלתם העצמית, שמכוחה זכו לדברים אלו – נותרה בעינה. על פי זה ניתן לפרש שכוונת הפייטן בכינוי 'נָאוָה כִּירוּשָׁלִָם' היא, הזכרת מעלתם העצמית של ישראל וחביבותם לפני המקום, שמכוחן אנו מתפללים לישועה ולהשבת עבודת בית המקדש – המממשת בפועל את מעלת 'נָאוָה כִּירוּשָׁלִָם'.318

לְמַעַן וְשֵׁרֵת בֶּאֱמֶת לְרַבּוֹ – בזכות אלישע, ששֵרֵת בנאמנות את אליהו רבו, כמו שכתוב: "פֹּה אֱלִישָׁע בֶּן שָׁפָט אֲשֶׁר יָצַק מַיִם עַל יְדֵי אֵלִיָּהוּ" (מלכים ב' ג יא), פִּי שְׁנַיִם בְּרוּחוֹ נֶאֱצַל בּוֹ – שרוח הנבואה של אליהו נאצלה ממנו לאלישע בכפליים,328 כמתואר בכתוב:

וַיְהִי כְעָבְרָם, וְאֵלִיָּהוּ אָמַר אֶל אֱלִישָׁע: שְׁאַל מָה אֶעֱשֶׂה לָּךְ בְּטֶרֶם אֶלָּקַח מֵעִמָּךְ, וַיֹּאמֶר אֱלִישָׁע: וִיהִי נָא פִּי שְׁנַיִם בְּרוּחֲךָ אֵלָי. וַיֹּאמֶר: הִקְשִׁיתָ לִשְׁאוֹל, אִם תִּרְאֶה אֹתִי לֻקָּח מֵאִתָּךְ יְהִי לְךָ כֵן, וְאִם אַיִן – לֹא יִהְיֶה. וַיְהִי הֵמָּה הֹלְכִים הָלוֹךְ וְדַבֵּר, וְהִנֵּה רֶכֶב אֵשׁ וְסוּסֵי אֵשׁ וַיַּפְרִדוּ בֵּין שְׁנֵיהֶם, וַיַּעַל אֵלִיָּהוּ בַּסְּעָרָה הַשָּׁמָיִם. וֶאֱלִישָׁע רֹאֶה וְהוּא מְצַעֵק: אָבִי אָבִי רֶכֶב יִשְׂרָאֵל וּפָרָשָׁיו, וְלֹא רָאָהוּ עוֹד, וַיַּחֲזֵק בִּבְגָדָיו וַיִּקְרָעֵם לִשְׁנַיִם קְרָעִים. וַיָּרֶם אֶת אַדֶּרֶת אֵלִיָּהוּ אֲשֶׁר נָפְלָה מֵעָלָיו, וַיָּשָׁב וַיַּעֲמֹד עַל שְׂפַת הַיַּרְדֵּן. וַיִּקַּח אֶת אַדֶּרֶת אֵלִיָּהוּ אֲשֶׁר נָפְלָה מֵעָלָיו, וַיַּכֶּה אֶת הַמַּיִם, וַיֹּאמַר: אַיֵּה ה' אֱלֹהֵי אֵלִיָּהוּ; אַף הוּא וַיַּכֶּה אֶת הַמַּיִם וַיֵּחָצוּ הֵנָּה וָהֵנָּה, וַיַּעֲבֹר אֱלִישָׁע. וַיִּרְאֻהוּ בְנֵי הַנְּבִיאִים אֲשֶׁר בִּירִיחוֹ מִנֶּגֶד וַיֹּאמְרוּ: נָחָה רוּחַ אֵלִיָּהוּ עַל אֱלִישָׁע, וַיָּבֹאוּ לִקְרָאתוֹ וַיִּשְׁתַּחֲווּ לוֹ אָרְצָה.329

בְּקַחְתּוֹ מְנַגֵּן נִתְמַלְּאוּ גֵבִים מָיִם – בספר מלכים מסופר שיהורם מלך ישראל, יהושפט מלך יהודה, ומלך אדום, יצאו להילחם עם מואב; בשלב מסויים הגיעו למצב של מחסור במים עד כדי סכנת מוות, והמלכים ביקשו מאלישע, שהיה עימהם, שינבא להם כיצד להחלץ מסכנה זו. אלישע הגיב בכעס כלפי יהורם, שהיה רשע ועובד עבודה זרה: "וַיֹּאמֶר אֱלִישָׁע אֶל מֶלֶךְ יִשְׂרָאֵל: מַה לִּי וָלָךְ, לֵךְ אֶל נְבִיאֵי אָבִיךָ וְאֶל נְבִיאֵי אִמֶּךָ [...]. חַי ה' צְבָאוֹת אֲשֶׁר עָמַדְתִּי לְפָנָיו, כִּי לוּלֵי פְּנֵי יְהוֹשָׁפָט מֶלֶךְ יְהוּדָה אֲנִי נֹשֵׂא, אִם אַבִּיט אֵלֶיךָ וְאִם אֶרְאֶךָּ". ובכל זאת, נעתר לבקשתם: "וְעַתָּה קְחוּ לִי מְנַגֵּן" – מפני שכעס, היה זקוק למנגן כדי להשיב אליו את השמחה, שכן הנבואה אינה שורה אלא מתוך שמחה, ולא מתוך כעס או עצבות;330"וְהָיָה כְּנַגֵּן הַמְנַגֵּן – וַתְּהִי עָלָיו יַד ה'. וַיֹּאמֶר: כֹּה אָמַר ה': »עשׂ…ה הנּחל הזּ∆ה גּ≈בƒים גּ≈בƒים. כִּי כֹה אָמַר ה': לֹא תִרְאוּ רוּחַ וְלֹא תִרְאוּ גֶשֶׁם, ו¿הנּחל ההוּא יƒ»מּל≈א »מיƒם, וּשְׁתִיתֶם אַתֶּם וּמִקְנֵיכֶם וּבְהֶמְתְּכֶם. וְנָקַל זֹאת בְּעֵינֵי ה' וְנָתַן אֶת מוֹאָב בְּיֶדְכֶם". אלישע התנבא שעליהם לחפור באותו עמק 'גֵּבִים גֵּבִים' = בורות רבים, והם יתמלאו מים באופן ניסי, ואכן: "וַיְהִי בַבֹּקֶר כַּעֲלוֹת הַמִּנְחָה, וְהִנֵּה מַיִם בָּאִים מִדֶּרֶךְ אֱדוֹם, וַתִּמָּלֵא הָאָרֶץ אֶת הַמָּיִם"331– הקב"ה הוריד גשמים רבים בארץ אדום, וכך נוצר שיטפון שהגיע עד מחנה ישראל, ומילא את הגבים במים.332

לְפוֹצְחֵי מִי כָמוֹךָ – עם ישראל, שלאחר קריעת ים-סוף פצחו בשירת הים, בה אמרו: "מִי כָמֹכָה בָּאֵלִם ה', מִי כָּמֹכָה נֶאְדָּר בַּקֹּדֶשׁ, נוֹרָא תְהִלֹּת עֹשֵׂה פֶלֶא" (שמות טו יא), הוֹשַׁע נָא וְהוֹשִׁיעָה נָּא, אָבִינוּ אָתָּה. השירה מכונה כאן על שם פסוק זה, כיוון שהוא הפסוק המרכזי בשירה: בין חציה הראשון של השירה, המתאר את התרחשות הנס, לבין חציה השני, המתאר את תוצאותיו – מופיע פסוק זה, המבטא את ההתרשמות מהנס, את אמירת ההתפעלות אליה הוא מוביל. ייתכן שמשום כך נבחר פסוק זה להיות משובץ בברכת 'גאל ישראל' בברכות קריאת שמע, יחד עם הפסוק החותם את השירה ועומד בשיאה: "ה' יִמְלֹךְ לְעֹלָם וָעֶד" (שם שם יח). ניתן לשאול: 'פּוֹצְחֵי מִי כָמוֹךָ' הוא כינוי המתאים לדור המדבר, אך מדוע אותנו, שלא נכחנו באותו מעמד, אנו מכנים כך? ייתכן שההסבר לכך הוא, שכשם שבאמירת השירה – ובפרט בפסוק זה – אומרי השירה תרגמו את הנס שאירע במציאות, להכרזה מפורשת "כִּי יַד ה' עָשְׂתָה זֹּאת וּקְדוֹשׁ יִשְׂרָאֵל בְּרָאָהּ"333– כך ישראל בכל דור הם "מְסַפְּרִים תְּהִלּוֹת ה' וֶעֱזוּזוֹ וְנִפְלְאוֹתָיו אֲשֶׁר עָשָׂה";334 לפיכך – על ידי ישועתנו יתקדש שמו של הקב"ה, בכך שנודה לו על הניסים ונודיע בעמים עלילותיו, ומכוח זאת אנו מבקשים: "הוֹשַׁע נָא וְהוֹשִׁיעָה נָּא, אָבִינוּ אָתָּה". בבית זה, הפייטן מקשר בין שני ניסים שפעלו באופן הפוך זה מזה: בנס קריעת ים-סוף הים הפך ליבשה, ואילו בנס של אלישע – הנחל היבש התמלא מים; שני הניסים יחד מראים את שליטתו המוחלטת של הקב"ה בכל כוחות הטבע. אנו מתפללים: 'הוֹשַׁע נָא' – בהורדת גשמים ובפרנסה ברווח, כשם שאז סיפקת לישראל מים בלב המדבר; ו'הוֹשַׁע נָא' – בגאולתנו השלימה, שבה אף אנו נפצח בשירת 'מִי כָמוֹךָ', "ונודה לך שיר חדש על גאולתנו ועל פדות נפשנו",335 על ניסי קריעת הים שיהיו אז, כדברי הנביא:

וְנָשָׂא נֵס לַגּוֹיִם, וְאָסַף נִדְחֵי יִשְׂרָאֵל, וּנְפֻצוֹת יְהוּדָה יְקַבֵּץ מֵאַרְבַּע כַּנְפוֹת הָאָרֶץ. וְהֶחֱרִים [=יְיַבֵּש] ה' אֵת לְשׁוֹן יָם מִצְרַיִם, וְהֵנִיף יָדוֹ עַל הַנָּהָר בַּעְיָם [=בחוזק] רוּחוֹ, וְהִכָּהוּ לְשִׁבְעָה נְחָלִים, וְהִדְרִיךְ בַּנְּעָלִים. וְהָיְתָה מְסִלָּה לִשְׁאָר עַמּוֹ אֲשֶׁר יִשָּׁאֵר מֵאַשּׁוּר, כַּאֲשֶׁר הָיְתָה לְיִשְׂרָאֵל בְּיוֹם עֲלֹתוֹ מֵאֶרֶץ מִצְרָיִם. וַאֲמַרְתֶּם בַּיּוֹם הַהוּא: הוֹדוּ לַה' קִרְאוּ בִשְׁמוֹ הוֹדִיעוּ בָעַמִּים עֲלִילֹתָיו, הַזְכִּירוּ כִּי נִשְׂגָּב שְׁמוֹ. זַמְּרוּ ה' כִּי גֵאוּת עָשָׂה, מידעת (קרֵי: מוּדַעַת) זֹאת בְּכָל הָאָרֶץ. צַהֲלִי וָרֹנִּי יוֹשֶׁבֶת צִיּוֹן, כִּי גָדוֹל בְּקִרְבֵּךְ קְדוֹשׁ יִשְׂרָאֵל.336

לְמַעַן הִרְהֵר עֲשׂוֹת רְצוֹנֶךָ – בזכות חזקיהו מלך יהודה, שכל ימיו הרהר כיצד לקיים את רצון ה', כפי שנאמר עליו: "וַיַּעַשׂ הַטּוֹב וְהַיָּשָׁר וְהָאֱמֶת לִפְנֵי ה' אֱלֹהָיו. וּבְכָל מַעֲשֶׂה אֲשֶׁר הֵחֵל בַּעֲבוֹדַת בֵּית הָאֱלֹהִים וּבַתּוֹרָה וּבַמִּצְוָה לִדְרֹשׁ לֵאלֹהָיו – בְּכָל לְבָבוֹ עָשָׂה וְהִצְלִיחַ" (דברי הימים ב' לא כ-כא).

הַמַּכְרִיז תְּשׁוּבָה לְצֹאנֶךָ – שעורר את עם ישראל לשוב בתשובה, כמו שכתוב:

וַיֵּלְכוּ הָרָצִים בָּאִגְּרוֹת מִיַּד הַמֶּלֶךְ וְשָׂרָיו בְּכָל יִשְׂרָאֵל וִיהוּדָה וּכְמִצְוַת הַמֶּלֶךְ לֵאמֹר: בְּנֵי יִשְׂרָאֵל, שׁוּבוּ אֶל ה' אֱלֹהֵי אַבְרָהָם יִצְחָק וְיִשְׂרָאֵל, וְיָשֹׁב אֶל הַפְּלֵיטָה הַנִּשְׁאֶרֶת לָכֶם מִכַּף מַלְכֵי אַשּׁוּר. וְאַל תִּהְיוּ כַּאֲבוֹתֵיכֶם וְכַאֲחֵיכֶם אֲשֶׁר מָעֲלוּ בַּה' אֱלֹהֵי אֲבוֹתֵיהֶם, וַיִּתְּנֵם לְשַׁמָּה כַּאֲשֶׁר אַתֶּם רֹאִים. עַתָּה אַל תַּקְשׁוּ עָרְפְּכֶם כַּאֲבוֹתֵיכֶם, תְּנוּ יָד לַה' וּבֹאוּ לְמִקְדָּשׁוֹ אֲשֶׁר הִקְדִּישׁ לְעוֹלָם, וְעִבְדוּ אֶת ה' אֱלֹהֵיכֶם וְיָשֹׁב מִכֶּם חֲרוֹן אַפּוֹ. כִּי בְשׁוּבְכֶם עַל ה' – אֲחֵיכֶם וּבְנֵיכֶם לְרַחֲמִים לִפְנֵי שׁוֹבֵיהֶם וְלָשׁוּב לָאָרֶץ הַזֹּאת, כִּי חַנּוּן וְרַחוּם ה' אֱלֹהֵיכֶם, וְלֹא יָסִיר פָּנִים מִכֶּם אִם תָּשׁוּבוּ אֵלָיו.337

אָז בְּבֹא מְחָרֵף סָתַם עֵינֹת מָיִם – כשבא סנחריב מלך אשור וצר על ירושלים, שלח את רבשקה שר צבאו לומר דברי חירוף כלפי הקב"ה באזני יושבי ירושלים, שמוטב להם להיכנע משום שהקב"ה לא יוכל להצילם: "מִי בְּכָל אֱלֹהֵי הָאֲרָצוֹת אֲשֶׁר הִצִּילוּ אֶת אַרְצָם מִיָּדִי, כִּי יַצִּיל ה' אֶת יְרוּשָׁלִַם מִיָּדִי" (מלכים ב' יח לה). לפני שהגיע צבא אשור לירושלים, סתם חזקיהו את המעיינות שסביבה, כמסופר בכתוב:

אַחֲרֵי הַדְּבָרִים וְהָאֱמֶת הָאֵלֶּה בָּא סַנְחֵרִיב מֶלֶךְ אַשּׁוּר וַיָּבֹא בִיהוּדָה, וַיִּחַן עַל הֶעָרִים הַבְּצֻרוֹת וַיֹּאמֶר לְבִקְעָם אֵלָיו. וַיַּרְא יְחִזְקִיָּהוּ כִּי בָא סַנְחֵרִיב, וּפָנָיו לַמִּלְחָמָה עַל יְרוּשָׁלָםִ. וַיִּוָּעַץ עִם שָׂרָיו וְגִבֹּרָיו לִסְתּוֹם אֶת מֵימֵי הָעֲיָנוֹת אֲשֶׁר מִחוּץ לָעִיר, וַיַּעְזְרוּהוּ. וַיִּקָּבְצוּ עַם רָב וַיִּסְתְּמוּ אֶת כָּל הַמַּעְיָנוֹת וְאֶת הַנַּחַל הַשּׁוֹטֵף בְּתוֹךְ הָאָרֶץ, לֵאמֹר: לָמָּה יָבוֹאוּ מַלְכֵי אַשּׁוּר וּמָצְאוּ מַיִם רַבִּים.338

לְצִיּוֹן מִכְלַל יֹפִי – ירושלים, כלשון הכתוב: "מִצִּיּוֹן מִכְלַל יֹפִי אֱלֹהִים הוֹפִיעַ" (תהילים נ ב)339– הוֹשַׁע נָא וְהוֹשִׁיעָה נָּא, כשם שבימי חזקיהו הושעת את ירושלים מהמצור, בכך שהמתת בין לילה את חיילי אשור שצרו עליה, כמסופר בכתוב: "וַיְהִי בַּלַּיְלָה הַהוּא וַיֵּצֵא מַלְאַךְ ה' וַיַּךְ בְּמַחֲנֵה אַשּׁוּר מֵאָה שְׁמוֹנִים וַחֲמִשָּׁה אָלֶף, וַיַּשְׁכִּימוּ בַבֹּקֶר וְהִנֵּה כֻלָּם פְּגָרִים מֵתִים" (מלכים ב' יט לה); אָבִינוּ אָתָּה.

לְמַעַן בָּאֵי לְךָ הַיּוֹם בְּכָל לֵב – בעבור354 ישראל הבאים לפניך היום בתפילה בכל ליבם, שׁוֹפְכִים לְךָ שִׂיחַ – על פי לשון הכתוב: "תְּפִלָּה לְעָנִי כִי יַעֲטֹף, וְלִפְנֵי ה' יִשְׁפֹּךְ שִׂיחוֹ" (תהילים קב א), בְּלֹא לֵב וָלֵב – בלא פירוד לבבות ביניהם, ובלא 'אחד בפה ואחד בלב',355 אלא באחדות ובכוונת לב שלימה, כדברי המדרש: "בשעה שאתם מתפללים לפני הקב"ה, לא יהיו לכם שני לבבות – אחד לפני הקב"ה ואחד לדבר אחר";356 וכפי שכתב ה'שולחן ערוך':

המתפלל צריך שיכוון בליבו פירוש המילות שמוציא בשפתיו. ויחשוב כאילו שכינה כנגדו, ויסיר כל המחשבות הטורדות אותו, עד שתישאר מחשבתו וכוונתו זכה בתפילתו [...] ואם תבוא לו מחשבה אחרת בתוך התפילה ישתוק עד שתתבטל המחשבה. וצריך שיחשוב בדברים המכניעים הלב ומכוונים אותו לאביו שבשמיים [...] הגהת רמ"א: ויחשוב קודם התפילה מרוממות האל יתעלה ומשפלות האדם, ויסיר כל תענוגי עולם מליבו.357

שׁוֹאֲלִים מִמְּךָ עֹז מִטְרוֹת מָיִם358– שהרי ביום זה נחתם הדין על גשמי השנה, כדברי המשנה: "ובחג נידונין על המים",359"וכל חיי האדם תלויים במים, והכול הולך אחר החיתום".360

לְשׁוֹרְרוּךָ בַּיָּם – לישראל, ששרו לך על נס קריעת ים-סוף ושיבחו את מידת עוזך שהתגלתה אז, כמו שכתוב: "עָזִּי וְזִמְרָת יָהּ וַיְהִי לִי לִישׁוּעָה" (שמות טו ב) – הוֹשַׁע נָא וְהוֹשִׁיעָה נָּא, וגלה אף להם מידה זו בהורדת 'עֹז מִטְרוֹת מַיִם',361אָבִינוּ אָתָּה.

לְמַעַן אוֹמְרֵי יִגְדַּל שְׁמֶךָ – בעבור עם ישראל, המייחלים לכך ששמך יגדל בעולם על ידי הגאולה השלימה, ומתפללים יום יום: 'יִתְגַּדַּל וְיִתְקַדַּשׁ שְׁמֵהּ רַבָּא', וְהֵם נַחֲלָתְךָ וְעַמֶּךָ – על פי לשון הכתוב בתפילת משה לאחר חטא העגל: "אַל תַּשְׁחֵת עַמְּךָ וְנַחֲלָתְךָ אֲשֶׁר פָּדִיתָ בְּגָדְלֶךָ [...] וְהֵם עַמְּךָ וְנַחֲלָתֶךָ" (דברים ט כו-כט). משה, בתפילתו לביטול הגזירה על השמדת עם ישראל, משתמש בטענה שעם ישראל הוא עמו ונחלתו של הקב"ה – הם העם המיועד להפיץ בעולם את האמונה בה', ודבר זה אינו יכול להשתנות אף אם חטאו, שכן ה'נחלה' היא נצחית;362 הסגולה האלוהית עודנה קיימת בהם, ולפיכך גם על חטא חמור זה ראוי לקב"ה לסלוח להם ולבטל את הגזירה. אף אנו משתמשים בטענה זו, ומבקשים מהקב"ה לזכות לגאולה השלימה אף אם איננו ראויים לכך מצד זכויותינו, שהרי אנו עמו ונחלתו, מקדשי שמו בעולם, ועל ידי גאולתנו – 'יִגְדַּל שְׁמֶךָ'.

צְמֵאִים לְיֶשְׁעֲךָ כְּאֶרֶץ עֲיֵפָה לְמָיִם – מתאווים ומצפים לישועתך השלימה, כאדמה הצמאה למים, על פי לשון הכתוב: "פֵּרַשְׂתִּי יָדַי אֵלֶיךָ, נַפְשִׁי כְּאֶרֶץ עֲיֵפָה לְךָ סֶלָה" (תהילים קמג ו). כשם שהאדמה אינה צמאה למים לתועלת עצמה, אלא כדי שתוכל להצמיח את יבולה כראוי, כך ישראל אינם מצפים לישועה בעבור תועלתם האישית, אלא כדי שיוכלו לפעול לתיקון העולם במלכות שדי ולממש את ייעודם בעולם. דבר זה מתאפשר רק בהיותם חיים על אדמתם עם כינון מלכותם, עבודת המקדש, וחזרת הנבואה, כששכינת ה' בתוכם, שכך יקדשו את שם ה' וייצגוהו באופן השלם.363

לְתַרְתָּ לָמוֹ מְנוּחָה – לעם ישראל, שבעת הליכתם במדבר ארון הברית הלך לפניהם והתקין להם מקום נוח לחנות בו, כמו שכתוב: "וַאֲרוֹן בְּרִית ה' נֹסֵעַ לִפְנֵיהֶם דֶרֶךְ שְׁלֹשֶׁת יָמִים לָתוּר לָהֶם מְנוּחָה" (במדבר י לג); וכתב שם ה'כלי יקר':

דרשו רבותינו זכרונם לברכה: אותו היום סרו מאחרי ה', וברחו מהר ה' כתינוק הבורח מבית הספר;364 לכך ברחו להם – מיראה פן יוסיף להם מצוות. ואמר הכתוב: ראֵה כמה הם כפויי טובה, כי אף בזמן המרד הזה – ארון ברית ה' נוסע לפניהם דרך שלושת ימים לתור להם מנוחה.

כשם שאז הקב"ה היטיב עם ישראל אף על פי שמעשיהם לא הצדיקו זאת, כך אנו מתפללים להיגאל אף אם אין לנו זכויות לכך. פירוש נוסף: 'מְנוּחָה' היא ארץ ישראל, כלשון הכתוב: "כִּי לֹא בָאתֶם עַד עָתָּה אֶל הַמְּנוּחָה וְאֶל הַנַּחֲלָה אֲשֶׁר ה' אֱלֹהֶיךָ נֹתֵן לָךְ" (דברים יב ט);365 כשם שלאחר יציאת מצרים הנחית את ישראל אל הארץ המובטחת, בה מצאו מנוחה, כך אף אותנו הוֹשַׁע נָא וְהוֹשִׁיעָה נָּא – קבץ את גלויותינו מארבע כנפות הארץ לארצנו, ו'תְּשִׁיבֵנוּ אֶל הַחֶבֶל וְאֶל הַנַּחֲלָה',366אָבִינוּ אָתָּה.
