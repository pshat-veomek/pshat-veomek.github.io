הפסוקים של השער (בראשית כו, כג – לה)

כג וַיַּעַל מִשָּׁם בְּאֵר שָׁבַע. כד וַיֵּרָא אֵלָיו יְהֹוָה בַּלַּיְלָה הַהוּא וַיֹּאמֶר אָנֹכִי אֱלֹהֵי אַבְרָהָם אָבִיךָ אַל תִּירָא כִּי אִתְּךָ אָנֹכִי וּבֵרַכְתִּיךָ וְהִרְבֵּיתִי אֶת זַרְעֲךָ בַּעֲבוּר אַבְרָהָם עַבְדִּי. כה וַיִּבֶן שָׁם מִזְבֵּחַ וַיִּקְרָא בְּשֵׁם יְהֹוָה וַיֶּט שָׁם אָהֳלוֹ וַיִּכְרוּ שָׁם עַבְדֵי יִצְחָק בְּאֵר. כו וַאֲבִימֶלֶךְ הָלַךְ אֵלָיו מִגְּרָר וַאֲחֻזַּת מֵרֵעֵהוּ וּפִיכֹל שַׂר צְבָאוֹ. כז וַיֹּאמֶר אֲלֵהֶם יִצְחָק מַדּוּעַ בָּאתֶם אֵלָי וְאַתֶּם שְׂנֵאתֶם אֹתִי וַתְּשַׁלְּחוּנִי מֵאִתְּכֶם. כח וַיֹּאמְרוּ רָאוֹ רָאִינוּ כִּי הָיָה יְהֹוָה עִמָּךְ וַנֹּאמֶר תְּהִי נָא אָלָה בֵּינוֹתֵינוּ בֵּינֵינוּ וּבֵינֶךָ וְנִכְרְתָה בְרִית עִמָּךְ. כט אִם תַּעֲשֵׂה עִמָּנוּ רָעָה כַּאֲשֶׁר לֹא נְגַעֲנוּךָ וְכַאֲשֶׁר עָשִׂינוּ עִמְּךָ רַק טוֹב וַנְּשַׁלֵּחֲךָ בְּשָׁלוֹם אַתָּה עַתָּה בְּרוּךְ יְהֹוָה. ל וַיַּעַשׂ לָהֶם מִשְׁתֶּה וַיֹּאכְלוּ וַיִּשְׁתּוּ. לא וַיַּשְׁכִּימוּ בַבֹּקֶר וַיִּשָּׁבְעוּ אִישׁ לְאָחִיו וַיְשַׁלְּחֵם יִצְחָק וַיֵּלְכוּ מֵאִתּוֹ בְּשָׁלוֹם. לב וַיְהִי בַּיּוֹם הַהוּא וַיָּבֹאוּ עַבְדֵי יִצְחָק וַיַּגִּדוּ לוֹ עַל אֹדוֹת הַבְּאֵר אֲשֶׁר חָפָרוּ וַיֹּאמְרוּ לוֹ מָצָאנוּ מָיִם. לג וַיִּקְרָא אֹתָהּ שִׁבְעָה עַל כֵּן שֵׁם הָעִיר בְּאֵר שֶׁבַע עַד הַיּוֹם הַזֶּה. ס לד וַיְהִי עֵשָׂו בֶּן אַרְבָּעִים שָׁנָה וַיִּקַּח אִשָּׁה אֶת יְהוּדִית בַּת בְּאֵרִי הַחִתִּי וְאֶת בָּשְׂמַת בַּת אֵילֹן הַחִתִּי. לה וַתִּהְיֶיןָ מֹרַת רוּחַ לְיִצְחָק וּלְרִבְקָה.

פרק כו · פסוקים כג–כד

וַיַּעַל מִשָּׁם בְּאֵר שָׁבַע. וַיֵּרָא אֵלָיו ה' בַּלַּיְלָה הַהוּא, וַיֹּאמֶר: אָנֹכִי אֱלֹהֵי אַבְרָהָם אָבִיךָ, אַל תִּירָא כִּי אִתְּךָ אָנֹכִי וּבֵרַכְתִּיךָ וְהִרְבֵּיתִי אֶת זַרְעֲךָ בַּעֲבוּר אַבְרָהָם עַבְדִּי.

יצחק עובר ממושבו בסמוך ל'רחובות' ועולה לבאר שבע. רשב"ם כתב כי עשה כך בעקבות יראתו מהפלשתים ורצונו להתרחק מהם: "'וַיַּעַל מִשָּׁם בְּאֵר שָׁבַע' – כי היה מתיירא מהם, שכן מוכיח שאמר לו הקב"ה 'אַל תִּירָא כִּי אִתְּךָ אָנֹכִי', נמצא שהיה מתיירא". מחמת יראתו מהפלשתים הוא התרחק מהם, ומחמת זאת גם נגלה אליו ה' והרגיע אותו לבל יירא מהם, וכפי שכתב 'בכור שור': "לפי שהיו רגילים לעשות עימו מריבה ולהתקוטט עימו, הוצרך לומר 'אַל תִּירָא'", וכן הרמב"ן: "בעבור שהבריחו אבימלך מעליו מקנאתם בו, וגם רועי גרר רבו עימו, היה מתיירא פן יֵאספו עליו והכוהו הוא וביתו, והבטיחו הקב"ה שלא יירא מהם ויברך אותו". באופן דומה פירש 'אור החיים': "להיות שראה השפלה ממלך פלשתים שגרשו מארצו ולא ידע אן יפנה, חש כי ה' עזבו; לזה אמר: אל תירא, לא עזבתיך, איתך אני".

ייתכן גם שנסיעתו לבאר שבע היתה על פי הדיבור, הגם שלא נזכר כך בכתוב; וייתכן גם שכדי לפרסם את שם ה' רצה לעבור למקום אחר שיש בו אנשים רבים, וכדרך שעשה אברהם אביו. ואם 'באר שבע' זו היא אותה 'באר שבע' שבה גר אברהם (ראה בעניין זה להלן פס' לג), ניתן גם להניח שרצה יצחק לדור במקום שבו גר אביו, ואף הוא עצמו בצעירותו, ולהמשיך את מפעלו של אברהם באותו המקום שבו פעל הוא.

"וַיֵּרָא אֵלָיו ה' בַּלַּיְלָה הַהוּא" – כלומר בלילה הראשון להיותו בבאר שבע; וכתב על כך ר' נפתלי הרץ ויזל: "כי אם לא נראה אליו באותה הלילה, ביום שלאחריו היה נוסע משם לתור לו הארץ אשר יצווה ה' שישכון שם"; וכפי שציווהו לעיל "שְׁכֹן בָּאָרֶץ אֲשֶׁר אֹמַר אֵלֶיךָ" (פס' ב), דהיינו כפי שפירש הרמב"ן שם: "ושכון כל ימיך בארץ שאומר אליך בכל פעם ופעם, על פי י"י תיסע ועל פי י"י תחנה".

"אָנֹכִי אֱלֹהֵי אַבְרָהָם אָבִיךָ" – ה' מציג את עצמו כ"אֱלֹהֵי אַבְרָהָם", ובכך מזכיר ליצחק את הסיוע שהעניק לאברהם אביו בשעתו – "רוצה לומר: אני שנתתי לאברהם אביך כוח לנצח את המלכים, ואני הנותן לך כוח לעשות חיל, ולכן לא תירא" (אברבנאל); מלבד מלחמת המלכים, השגחת ה' ליוותה את אברהם כל ימי חייו, וכאן רומז ה' ליצחק – כשם שהייתי עם אברהם אביך, כך אהיה עימך. בנוסף לכך, אמירה זו עשויה לכלול גם את המסר שזכותו של אברהם אביו עומדת לו, וכפי שיאמר ה' בסוף הדברים: "בַּעֲבוּר אַבְרָהָם עַבְדִּי".

"אַל תִּירָא כִּי אִתְּךָ אָנֹכִי" – בתחילת הפרק, בבוא יצחק לגרר, אמר לו ה': "גּוּר בָּאָרֶץ הַזֹּאת וְאֶהְיֶה עִמְּךָ וַאֲבָרֲכֶךָּ", ולא אמר לו גם "אַל תִּירָא", כמו בהבטחה שבפסוקנו; וכך על כך ר' נפתלי הרץ ויזל:

ובבואו לגרר לא אמר לו ה' ["אַל תִּירָא"], כי היה גלוי לפניו שיקנאו אותו פלשתים ושיצטרך להתרחק מהם במצות המלך וישטמוהו גם בהיותו רחוק מהם, ושיצטרך לירוא מהם. אלא שהבטיחו "וְאֶהְיֶה עִמְּךָ", כמו "עִמּוֹ אָנֹכִי בְצָרָה, אֲחַלְּצֵהוּ וַאֲכַבְּדֵהוּ" (תהלים צא טו). עתה הבטיחו שאין לו לירוא כלל, וְשֶׁשָׁלְמוּ ימי עָנְיוֹ.

"כִּי אִתְּךָ אָנֹכִי, וּבֵרַכְתִּיךָ וְהִרְבֵּיתִי אֶת זַרְעֲךָ" – הבטחה זו חוזרת על מה שנאמר ליצחק בבואו לגרר; שם הבטיח לו ה' שיהיה עימו, שיברכנו, ושירבה את זרעו – "וְאֶהְיֶה עִמְּךָ וַאֲבָרֲכֶךָּ [...] וְהִרְבֵּיתִי אֶת זַרְעֲךָ כְּכוֹכְבֵי הַשָּׁמַיִם", ואף כאן כן – "כִּי אִתְּךָ אָנֹכִי, וּבֵרַכְתִּיךָ, וְהִרְבֵּיתִי אֶת זַרְעֲךָ". ר' נפתלי הרץ ויזל מפרש כי בעקבות הגעת יצחק למקום חדש הבטיחו ה' על כך בשנית, ומשמעות הדברים שמעבר לברכות שזכה להן במקומו הקודם, הוא יתברך גם במקום זה:

"וּבֵרַכְתִּיךָ" – כבר אמר לו למעלה "וַאֲבָרֲכֶךָּ", וחוזר על "גּוּר בָּאָרֶץ הַזֹּאת" [כלומר שתתברך בארץ הזאת], ונתקיימה ההבטחה, כמו שאמר בעניין: "וַיְבָרְכֵהוּ ה'" (פס' יב). ועתה שיצא מארץ גרותו ובא לבאר שבע, הוצרך להבטיחו שנית שיברכהו גם בארץ ההיא.120

"בַּעֲבוּר אַבְרָהָם עַבְדִּי" – מתוספת זו משמע שיצחק יזכה לברכות אלו רק "בַּעֲבוּר אַבְרָהָם", ולא בזכות עצמו. האמנם לא היה יצחק ראוי לכך גם מכוח עצמו, ובזכותו שלו?

רלב"ג כתב על כך: "'בַּעֲבוּר אַבְרָהָם עַבְדִּי' ― רוצה לומר: כי מצד אברהם תדבק בך השגחה יותר נפלאה מההשגחה הראויה להיות לך מצד עצמך"; ורד"צ הופמן הסביר זאת באופן אחר: "הברכה מובטחת בזכות אברהם מכיוון שכבר מת, והוכיח את עצמו כצדיק תמים במשך כל ימי חייו".

פרק כו · פסוק כה

וַיִּבֶן שָׁם מִזְבֵּחַ וַיִּקְרָא בְּשֵׁם ה' וַיֶּט שָׁם אָהֳלוֹ, וַיִּכְרוּ שָׁם עַבְדֵי יִצְחָק בְּאֵר.

"וַיִּבֶן שָׁם מִזְבֵּחַ" – כתב רלב"ג: "וכאשר ראה יצחק ששם דבק בו השפע האלוהי בנה שם מזבח, כדי שתהיה הנבואה מתמדת לו שם". ניתן גם להסביר שבניית המזבח היתה דרך של הודאה לה' על הדברים שנאמרו לו בנבואה זו, וכשם שעשה אברהם אביו, כאמור: "וַיִּבֶן שָׁם מִזְבֵּחַ לַה' הַנִּרְאֶה אֵלָיו" (לעיל יב ז) – לשם הודאה, כפי שכתב רש"י שם: "'וַיִּבֶן שָׁם מִזְבֵּחַ' – על בשורת הזרע ועל בשורת ארץ ישראל", וכן הרמב"ן שם: "כי הודה לשם הנכבד וזבח לו זבח תודה על שנראה אליו".

בהיותו בגרר נמנע יצחק מלבנות מזבח בעקבות דברי ה' שנאמרו לו בבואו לשם, כמתואר בתחילת הפרק – שכן חשש מהתנגדות הפלשתים; אך כאן הובטח מאת ה' שאין לו לירא, ומשכך בנה במקום זה מזבח, וייסד בו מרכז רוחני של קריאה בשם ה', כאמור כאן: "וַיִּבֶן שָׁם מִזְבֵּחַ, וַיִּקְרָא בְּשֵׁם ה'"; ומלבד זאת, גם קבע בו את מושבו – בנטיית אוהל וכריית באר, כאמור לאחר מכן: "וַיֶּט שָׁם אָהֳלוֹ, וַיִּכְרוּ שָׁם עַבְדֵי יִצְחָק בְּאֵר". וכך כתב ר' נפתלי הרץ ויזל:

כששמע מפי ה' שישכון בבאר שבע בטח, ושהיא הארץ אשר אמר אליו, בנה שם בית תפילה ובית כנסת לקרוא ברבים, ללמדם דעת ויראת ה', כאמרו "וַיִּקְרָא בְּשֵׁם ה'" כמו שנאמר אצל אביו. ולא עשה כן בארץ פלשתים, שהיו שונאי נפשו ומסתכן ביניהם. גם אברהם לא קרא במקום הזה בשם ה' עד אחר שבאו אליו אבימלך ופיכל וכרתו עמו ברית ושבועה, ויצחק בנבואה שמע שישכון שמה בטח.

לדבריו, כשהיו האבות בקרבת מקום לפלשתים נמנעו מייסוד מרכז רוחני; אברהם נטע אשל בבאר שבע וקרא שם בשם ה' רק לאחר שכרת ברית עם אבימלך, כאמור: "וַיִּכְרְתוּ בְרִית בִּבְאֵר שָׁבַע וַיָּקָם אֲבִימֶלֶךְ וּפִיכֹל שַׂר צְבָאוֹ וַיָּשֻׁבוּ אֶל אֶרֶץ פְּלִשְׁתִּים, וַיִּטַּע אֵשֶׁל בִּבְאֵר שָׁבַע וַיִּקְרָא שָׁם בְּשֵׁם ה' אֵל עוֹלָם" (לעיל כא לב-לג); ויצחק נמנע מכך בהיותו בגרר בין הפלשתים, ורק כעת – כשהבטיחו ה' "אַל תִּירָא" – עשה כן.

"וַיִּקְרָא בְּשֵׁם ה'" – כך נאמר גם על אברהם: "וַיַּעְתֵּק מִשָּׁם הָהָרָה מִקֶּדֶם לְבֵית אֵל וַיֵּט אָהֳלֹה, בֵּית אֵל מִיָּם וְהָעַי מִקֶּדֶם, וַיִּבֶן שָׁם מִזְבֵּחַ לַה' וַיִּקְרָא בְּשֵׁם ה'" (לעיל יב ח), וכתב שם הרמב"ן:

"וַיִּקְרָא בְּשֵׁם ה'" – פירש אונקלוס: שהתפלל שם, כמו "קָרָאתִי שִׁמְךָ ה' מִבּוֹר תַּחְתִּיּוֹת" (איכה ג נה). והנכון – שהיה קורא בקול גדול שם לפני המזבח את שם ה', מודיע אותו ואלוהותו לבני אדם; כי באור כשדים היה מלמדם ולא אבו שמוע, ועתה כשבא בארץ הזאת שהובטח בה "וַאֲבָרְכָה מְבָרֲכֶיךָ", היה למוד ללמד ולפרסם האלוהות.

וכן אמר הכתוב ביצחק, כאשר הלך אל נחל גרר והובטח "אַל תִּירָא כִּי אִתְּךָ אָנֹכִי", שבנה מזבח "וַיִּקְרָא בְּשֵׁם ה'"; כי בא במקום חדש, אשר לא שמעו את שִׁמְעוֹ ולא ראו את כבודו [של ה'], והגיד כבודו בגויים ההם.

סדר הפעולות בכתוב מבליט את העובדה שראשית לכל בנה יצחק מזבח וקרא בשם ה', ורק לאחר מכן נפנה לצרכיו שלו – נטיית האוהל וכריית הבאר: "וַיִּבֶן שָׁם מִזְבֵּחַ, וַיִּקְרָא בְּשֵׁם ה' – וַיֶּט שָׁם אָהֳלוֹ, וַיִּכְרוּ שָׁם עַבְדֵי יִצְחָק בְּאֵר"; וכפי שכתב הנצי"ב ב'העמק דבר': "והקדים לקבוע מקום לקרבן ולתפילה ואחר כך קבע אהלו לדירה, ואחר כך ראה לכרות באר לצורך דירה מרווחת". לעומת זאת, אצל אברהם מופיע סדר פעולות שונה: "וַיַּעְתֵּק מִשָּׁם הָהָרָה מִקֶּדֶם לְבֵית אֵל וַיֵּט אָהֳלֹה, בֵּית אֵל מִיָּם וְהָעַי מִקֶּדֶם, וַיִּבֶן שָׁם מִזְבֵּחַ לַה' וַיִּקְרָא בְּשֵׁם ה'" (לעיל יב ח) – תחילה נטה לו אוהל, ולאחר מכן בנה מזבח וקרא בשם ה'. נראה שטעם הדבר הוא, שבענייננו בנה יצחק את המזבח בעקבות התגלות ה' אליו והבטחותיו לו, ולפיכך עשה זאת מיד, בעוד רושם ההתגלות עליו – על מנת לבטא בכך את הודאתו לה' על ההבטחות הללו; ומכיון שבנה שם מזבח, החליט גם לקבוע שם את מושבו, ונטה את אהלו.121 אך לאברהם לא היתה התגלות באותו מקום,122 ומחמת סיבה אחרת בחר לקבוע שם את מושבו; ולפיכך נהג כסדר הנורמלי והטבעי – בתחילה נטה את אהלו והתמקם שם עם אנשיו ומקנהו, ולאחר שהיתה קיימת התשתית החומרית הנצרכת לחיים באותו מקום, בנה שם מזבח וקרא בשם ה'.123

כתב רש"ר הירש:

כשיצחק בונה שם מיד מזבח, ומניף את הדגל הרוחני של אברהם; ואף נוטה שם את אהלו ללא בדיקה מוקדמת של טיב הקרקע, ומבלי שיתברר תחילה שיכולים להמצא שם מים – הרי שהכל בא אליו מיד, כאילו נעשה מאליו! אנשיו מוצאים מים בניסיון החפירה הראשון, והמלך ששילחו מעל פניו גומל לו בביקור על פי כל כללי הכבוד. כל אלה הם דברים שעד כה טרח עליהם לשווא, מתוך מאבקים והפרעות.

כלומר, ייסוד המרכז הרוחני פתח ליצחק את שערי הברכה גם בתחומים החומריים – עבדיו חופרים ומוצאים מים, ואבימלך בא אליו לכרות עימו ברית, כפי שיסופר בהמשך. וניתן להוסיף שיש בכך גם 'סימן לבנים' – כשעם ישראל יבטא ויבליט את היותו עם ה' ויקרא בשם ה', יזכה לשפע וברכה גשמיים, ואף לביטחון, הערכה וכבוד מצד אומות העולם.

"וַיִּכְרוּ שָׁם עַבְדֵי יִצְחָק בְּאֵר" – בכל יתר הפסוקים בפרקנו משתמש הכתוב בפועל 'חפר' ביחס לבארות, וכאן משתמש בפועל 'כרה'; וכתב על כך המלבי"ם: "ויש הבדל בין 'כרייה' ל'חפירה'; ה'כרייה' מציין התחלת החפירה, ו'החפירה' מציין גמר החפירה, כמו שכתבתי בפירוש תהלים בפסוק 'בּוֹר כָּרָה וַיַּחְפְּרֵהוּ' (תהלים ז טז); ורוצה לומר שהתחילו לכרות, ואחר כך אמר [על באר זו]: 'עַל אֹדוֹת הַבְּאֵר אֲשֶׁר חָפָרוּ' (פס' לב) [-כשגמרו לחפרה]".

פרק כו · פסוק כו

וַאֲבִימֶלֶךְ הָלַךְ אֵלָיו מִגְּרָר וַאֲחֻזַּת מֵרֵעֵהוּ וּפִיכֹל שַׂר צְבָאוֹ.

"וַאֲבִימֶלֶךְ הָלַךְ אֵלָיו מִגְּרָר"

אבימלך בא מגרר עם "אֲחֻזַּת מֵרֵעֵהוּ וּפִיכֹל שַׂר צְבָאוֹ" אל יצחק, על מנת לבקש ממנו שיכרות עימו ברית – כפי שיתואר בפסוקים הבאים. מה הניע את אבימלך לבוא ולבקש כך מיצחק, דווקא כעת, לאחר ששילחו מגרר?

באופן פשוט, ניתן להסביר זאת על פי דברי אבימלך עצמו: "רָאוֹ רָאִינוּ כִּי הָיָה ה' עִמָּךְ, וַנֹּאמֶר תְּהִי נָא אָלָה בֵּינוֹתֵינוּ בֵּינֵינוּ וּבֵינֶךָ וְנִכְרְתָה בְרִית עִמָּךְ" (פס' כח), וכפי שכתב רד"צ הופמן: "משראה אבימלך שיצחק מצליח בדרכו גם מחוץ לארצו, הוא השתכנע שידו המגינה והמברכת של כוח עילאי מסייעת לו ליצחק, ולכן הוא מבקש לכרות עמו ברית".

ניתן גם לומר, שבעקבות כך ששילח את יצחק מגרר, חשש אבימלך שיצחק יראה בכך הפרת הברית שכרתו בעבר בינם לבין אברהם (כמסופר לעיל בפרק כא), או שחשש פן ירצה יצחק להילחם בו כנקמה על כך – ולפיכך בא אליו כעת כדי לכרות עימו ברית ולהבטיח את שלומו ושלום ארצו. וראה בדברי הרמב"ן המובאים להלן בביאור לפסוק כח (עמ' 200), שכתב כן.

בנוסף לכך כתב הרמב"ן (בפס' כד) שכך סובב ה' בעקבות הבטחתו ליצחק: "הבטיחו הקב"ה שלא יירא מהם ויברך אותו; ואז נתן בליבם שהלך המלך לפניו בכבוד גדול יותר מאשר עשה לאביו, כי בא עם פיכל שר צבאו, ועוד הוסיף להביא עימו שרים רבים רעיו" – כריתת הברית עם אבימלך הביאה לידי ביטוי ומימוש את הבטחת ה' ליצחק שיהיה עימו, ושאין לו לירא מהפלשתים. כך פירש גם ר' נפתלי הרץ ויזל: "סיפר הכתוב [ש]מלבד שהבטיחו ה' 'אַל תִּירָא', ראה בעיניו קיום זאת ההבטחה, כי מלך פלשתים עם שריו באו אליו לכרות עימו ברית ולבקש שלומו".124

לשון הכתוב כאן מנוסחת בלשון עבר רגיל – "וַאֲבִימֶלֶךְ הָלַךְ", במקום הלשון הרגילה במקרא, בו' ההיפוך – "וילך אבימלך"; וניתן להסביר זאת בכך שמאחר והכתוב מתאר את ההתרחשות מנקודת מבטו של יצחק, הוא מזכיר את עצם הליכת אבימלך מגרר בלשון עבר רגיל – שכן בשעה שהוא נפגש עם יצחק הליכה זו כבר נעשתה; מה שאין כן כשמתאר הכתוב פעילות של דמות העומדת במרכז הסיפור המקראי, הוא משתמש בו' ההיפוך, שמכך משמע שזהו השלב הנוכחי בסיפור.

"וַאֲחֻזַּת מֵרֵעֵהוּ"

ניתן לפרש מילים אלו בכמה דרכים; אונקלוס תרגם ביטוי זה כ"וְסִיעַת מְרַחֲמוֹהִי" – סיעת מאוהביו; לפי זה, "אֲחֻזַּת" היינו קבוצת אנשים, כפי שכתב רש"י: "לשון קבוצה ואגודה, שנאחזין יחד", ו"מֵרֵעֵהוּ" היינו שאותם אנשים היו רעיו של אבימלך. בדרך זו, ניתן לפרש בשני אופנים:

האחד, כך שמשמעות המילה "אֲחֻזַּת" היא "אחוזה", ולא בצורת סמיכות, ואם כן המ' במילה "מֵרֵעֵהוּ" אינה חלק מהמילה אלא מוספית – אחוזה מ-רעהו, קבוצה מתוך רעיו של אבימלך; לפי זה, הגירסה הנכונה בתרגום אונקלוס היא "וְסִיעַת מְרַחֲמוֹהִי", כמצוטט לעיל. והשני, שהמילה "אֲחֻזַּת" היא בצורת סמיכות, והמ' במילה "מֵרֵעֵהוּ" היא חלק מן המילה – קבוצת מרעיו, ולפי זה הגירסה הנכונה בתרגום אונקלוס היא "וְסִיעַת רַחֲמוֹהִי". שתי הבנות אלו הובאו בדברי רש"י:

"וַאֲחֻזַּת [מֵרֵעֵהוּ]" – "סִיעַת מְרַחֲמוֹהִי", סיעה מאוהביו. ואל תתמה על תי"ו שב"אֲחֻזַּת", ואף על פי שאינה תיבה דבוקה לחברתה, יש דוגמתה במקרא: "הָבָה לָּנוּ עֶזְרָת מִצָּר" (תהלים ס יג), "וּשְׁכֻרַת וְלֹא מִיָּיִן" (ישעיהו נא כא).

ויש פותרים: "מֵרֵעֵהוּ" – המ"ם מיסוד התיבה, כמו: "שְׁלֹשִׁים מֵרֵעִים" (שופטים יד יא) דשמשון, כדי לעשות "וַאֲחֻזַּת" תיבה דבוקה. ומתרגמין: "סִיעַת רַחֲמוֹהִי".125

לפי שתי האפשרויות הללו, המילה "מֵרֵעֵהוּ" היא לשון רבים, כמו "מרעיו", ואף על פי שלרוב צורה זו מהווה לשון יחיד; וכפי שכתב רד"ק בשיטת אונקלוס: "ויהיה 'רעהו' כמו 'רעיו', וכן 'מָגֵן גִּבֹּרֵיהוּ' (נחום ב ד) – כמו 'גִּבּוֹרָיו'".

במדרש בראשית רבה (סד ט) מפרש ר' נחמיה כאונקלוס, אך ר' יהודה מפרש באופן אחר: ר' יהודה אמר: 'אֲחֻזַּת מֵרֵעֵהוּ' היה שמו; ר' נחמיה אמר: 'סִיעַת מְרַחֲמוֹהִי'". לדעת ר' יהודה, "אֲחֻזַּת" הוא שם אדם שהיה מרעהו, דהיינו חבירו, של אבימלך;126 כך פירש גם רס"ג, וכן רד"ק, שכתב:

"וַאֲחֻזַּת מֵרֵעֵהוּ" – שר גדול לאבימלך, היה שמו "אֲחֻזַּת". ואמרה המסורה: "לית שום בר נש".127 ואמר "מֵרֵעֵהוּ" – כמו "רֵעֶה הַמֶּלֶךְ" (מלכים א' ד ה), כי השר הגדול שיושב תמיד עם המלך ייקרא רֵעַ המלך. ומ"ם "מֵרֵעֵהוּ" – מאותיות אמתי"ן הנוספות, וכן "לְמֵרֵעֵהוּ אֲשֶׁר רֵעָה לוֹ" (שופטים יד כ).

"וּפִיכֹל שַׂר צְבָאוֹ"

גם לגבי פיכל נחלקו ר' יהודה ור' נחמיה במדרש (שם): "ר' יהודה אמר: פיכל שמו; ר' נחמיה אמר: פה שכל צבאותיו נושקים לו על פיו"; לדעת ר' יהודה 'פיכל' היה שמו הפרטי של אותו שר הצבא – ואם כן, מסתבר שזהו אותו פיכל שהשתתף בכריתת הברית עם אברהם, כמסופר לעיל בפרק כא: "וַיְהִי בָּעֵת הַהִוא וַיֹּאמֶר אֲבִימֶלֶךְ וּפִיכֹל שַׂר צְבָאוֹ אֶל אַבְרָהָם לֵאמֹר [...] וְעַתָּה הִשָּׁבְעָה לִּי" (פס' כב-כג), ואף על פי שעברו שנים רבות מאז; לעומת זאת, לדעת ר' נחמיה 'פיכל' היה שמו הקבוע של הנושא בתפקיד שר צבא פלשתים, על שם זאת "שכל צבאותיו נושקים לו על פיו" – כלומר שעל פיו יישק כל דבר; ואם כן, מסתבר שאין זה אותו פיכל המוזכר בעניין כריתת הברית עם אברהם, שכן כאמור עברו מאז שנים רבות.

פרק כו · פסוק כז

וַיֹּאמֶר אֲלֵהֶם יִצְחָק: מַדּוּעַ בָּאתֶם אֵלָי, וְאַתֶּם שְׂנֵאתֶם אֹתִי וַתְּשַׁלְּחוּנִי מֵאִתְּכֶם.

יצחק מבין כי אבימלך וסיעתו באו אליו לצורך אינטרס שלהם, ואינו ממתין לשמוע מה בפיהם, אלא מקדים ומביע את תרעומתו כלפיהם. הם ראויים לתוכחה בין על כך ששילחוהו מאיתם ובין על כך שכעת הם באים אליו כשהם צריכים לו; אך את גערתו הוא מביע בצורה עדינה ומרומזת, בתמיהה שניתן לקראה כמעין היתממות – "מַדּוּעַ בָּאתֶם אֵלָי?", אם באתם אלי לצרכי ולטובתי, או כדי לכרות עימי ברית הדדית – הרי "אַתֶּם שְׂנֵאתֶם אֹתִי וַתְּשַׁלְּחוּנִי מֵאִתְּכֶם", ואם כן מוכח שאינכם רוצים בטובתי, ואף אין לכם עניין ביחסי שלום עימי; ואם באתם אלי לצרככם ולטובתכם – וכי ראוי הדבר שלאחר ש"וְאַתֶּם שְׂנֵאתֶם אֹתִי וַתְּשַׁלְּחוּנִי מֵאִתְּכֶם" תבואו אלי לבקש ממני טובה?! וכדרך שאמר יפתח לזקני גלעד: "הֲלֹא אַתֶּם שְׂנֵאתֶם אוֹתִי וַתְּגָרְשׁוּנִי מִבֵּית אָבִי, וּמַדּוּעַ בָּאתֶם אֵלַי עַתָּה כַּאֲשֶׁר צַר לָכֶם?" (שופטים יא ז).128

בכך נוהג יצחק כמנהג אברהם אביו בבוא אבימלך אליו לכרות עימו ברית, שאף הוא הוכיחו על העוול שנעשה כלפיו: "וְהוֹכִחַ אַבְרָהָם אֶת אֲבִימֶלֶךְ עַל אֹדוֹת בְּאֵר הַמַּיִם אֲשֶׁר גָּזְלוּ עַבְדֵי אֲבִימֶלֶךְ" (לעיל כא כה). בכך הוא שומר על כבודו ומעמדו מול אבימלך; לאחר ששֻׁלַּח על ידו מגרר, הוא מעלה את תרעומתו בדבריו אליו, שלא להיות כמי שמקבל את בקשותיו של מלך פלשתים בציות כנוע. יצחק זוכר את שנעשה לו, והוא גם מראה זאת לאבימלך; הוא מבהיר לו בדבריו ובאופן התנהלותו כי הוא זוכר ונוטר את הפגיעה בו ובזכויותיו, ואינו מקבל זאת בהכנעה ובמחילה. וכך כתב 'צרור המור': "והוא בגודל לבבו לא שת ליבו למלך ושרים, ואמר להם: 'מַדּוּעַ בָּאתֶם אֵלַי' עתה, אחר הנבלה הגדולה שעשיתם שלא כדין לשלחני מאתכם בהיותי תושב עִמָּכֶם".

מעבר לכך, בתרעומת זו היה עניין גם מצד גילוי הלב שבה, כפי שכתב בעל מדרש 'לקח טוב': "לא החניפם בדברים, אלא בדברי תוכחות אמר לו [...], שמתוך התוכחות באה האהבה", וכן ר' אברהם בן הרמב"ם: "להראות מה שבלב, ולהינצל מן המרמה ומהסתרת שום דבר בלב".

"וְאַתֶּם שְׂנֵאתֶם אֹתִי וַתְּשַׁלְּחוּנִי מֵאִתְּכֶם" – בדברים אלו פונה יצחק לאבימלך וסיעתו בלשון רבים, ואף על פי שלעיל נאמר כי אבימלך עצמו הוא אשר אמר לו "לֵךְ מֵעִמָּנוּ" (פס' טז) – וטעם הדבר, "כי אבימלך בעצת שריו וגדוליו אמר לו 'לֵךְ מֵעִמָּנוּ'" (רד"ק); ואכן מסתבר שלא היתה זו החלטה של אבימלך על דעת עצמו, אלא גם גדולי המלכות והמקורבים לו היו שותפים בעצה זו.

לעיל שם הוזכרה גם אפשרות להבין כי אבימלך לא גירש את יצחק בתקיפות, אלא במעין עצה-בקשה אמר לו ללכת מעימם, על מנת שלא יציקו לו פלשתים ולא יפגעו בו או בנכסיו; לפי הבנה זו, טענת יצחק "וְאַתֶּם שְׂנֵאתֶם אֹתִי וַתְּשַׁלְּחוּנִי מֵאִתְּכֶם" אינה מתייחסת בעיקרה לעצם אמירת אבימלך "לֵךְ מֵעִמָּנוּ" אלא לכל מה שקדם לה – השנאה וההתנכלות אליו מצד הפלשתים, שהתבטאו בסתימת הבארות שחפר אביו (לפי המפרשים כך, כמבואר לעיל) ושמא גם בדברים נוספים, עד שהגיעו לידי כך שאמר לו אבימלך "לֵךְ מֵעִמָּנוּ"; וכל זאת כולל יצחק במילים "וְאַתֶּם שְׂנֵאתֶם אֹתִי וַתְּשַׁלְּחוּנִי מֵאִתְּכֶם" – המופנות למלך ולגדולי הפלשתים, המייצגים את העם כולו, ששנאוהו – "וְאַתֶּם שְׂנֵאתֶם אֹתִי", ובכך גרמו לשילוחו בפועל על ידי אבימלך – "וַתְּשַׁלְּחוּנִי מֵאִתְּכֶם".

"רָאוֹ רָאִינוּ כִּי הָיָה ה' עִמָּךְ"

למה כוונתם באמרם שראו "כִּי הָיָה ה' עִמָּךְ"? וכיצד דבר זה מהווה סיבה לכך שהם רוצים לכרות עימו ברית?

דברים דומים אמרו אבימלך ופיכל לאברהם: "אֱלֹהִים עִמְּךָ בְּכֹל אֲשֶׁר אַתָּה עֹשֶׂה, וְעַתָּה הִשָּׁבְעָה לִּי בֵאלֹהִים הֵנָּה [...]" (לעיל כא כב-כג); וכתב שם המלבי"ם:

"אֱלֹהִים עִמְּךָ" – באשר כמו זר נחשב דבר זה, איך מלך הארץ ושריו יבקשו מאיש גֵּר מתגורר בארצם שיישבע להם בל ירע להם, ושיבואו עוד לביתו לבקש מאיתו כזאת? על זה הקדימו, שלכן יראים מפניו באשר אלוהים איתו, כמו שראו במלחמת המלכים ולקיחת שרה ולידת בן לזקוניו.

אם כן, אף כאן נפרש כך את דבריהם: מכיון שראינו שהיה ה' עימך ואתה מצליח במעשיך – יש בנו חשש מכך שגם אם תפעל נגדנו יהיה ה' בעזרך, ולפיכך אנו מבקשים לכרות עימך ברית, כדי להבטיח שלא תעשה עימנו רעה.

יש שפירשו באופן אחר, כך שכוונתם במילים אלו היתה לומר: ראינו כי הצלחתך היא מאת ה' בהשגחה מיוחדת, ולא כפי שסברנו בעבר שהצלחתך היא "מִמֶּנּוּ"; כך כתב האלשיך, בהמשך לדבריו שהובאו לעיל בביאור לפסוק טז, שבהם פירש את טענת אבימלך "כִּי עָצַמְתָּ מִמֶּנּוּ מְאֹד" כך שיצחק נוטל מהשפע שמגיע אליהם ממרום ובכך ממעט את חלקם: "'רָאוֹ רָאִינוּ' וכו' – לומר, ראינו מה שגדלת עלינו בהיותך עימנו, ואחר כך ראינו שלא נוסף לנו שפע בלכתך, והכרנו בזה כי לא מעושר שלנו היית מושפע בהיותך עימנו, רק כי היה ה' עימך", וכן המלבי"ם (שאף הוא פירש כך לעיל): "תחילה חשבנו שהצלחתך היא מצד הארץ ואתה ממעט את השפע שלנו, אבל עתה ראינו שהצלחתך היא מצד כי היה ה' עימך, וזה אינו מגרע השפע שלנו". אולם לפי פירוש זה, קשה להבין כיצד אמירה זו מהווה סיבה לכך שהם רוצים לכרות עימו ברית – הרי אינם מבקשים ממנו או מציעים לו לחזור אליהם, ואף אינם מתנצלים על כך ששילחוהו מאיתם, כך שתידרש הקדמה שמסבירה כי הם מבינים כעת במה הם טעו!129

"וַנֹּאמֶר: תְּהִי נָא אָלָה בֵּינוֹתֵינוּ בֵּינֵינוּ וּבֵינֶךָ"

'אלה' היא קללה המלווה ברית או שבועה, שתחול על מי שיפר אותן, ובכך מחזקת את תוקפן. אבימלך רוצה לכרות ברית שכרוכה בה אלה, בינם לבין יצחק – "תְּהִי נָא אָלָה בֵּינוֹתֵינוּ בֵּינֵינוּ וּבֵינֶךָ, וְנִכְרְתָה בְרִית עִמָּךְ".

מה פשר האמירה הכפולה "בֵּינוֹתֵינוּ, בֵּינֵינוּ וּבֵינֶךָ"?

באופן פשוט, אמירה זו מתפרשת כך שהמילים "בֵּינֵינוּ וּבֵינֶךָ" מהוות פירוש למילה "בֵּינוֹתֵינוּ"; המילה "בֵּינוֹתֵינוּ" כוללת את שני הצדדים הכורתים ביניהם את הברית, ומי הם? "בֵּינֵינוּ" – בין צד אחד, שהוא אבימלך ופלשתים, "וּבֵינֶךָ" – לבין הצד השני, שהוא יצחק. וכך פירש רד"צ הופמן: "'בֵּינוֹתֵינוּ' וגו' – נראה שביטוי זה שונה מן 'בֵּינֵינוּ' בזה שהוא בא לציין את שני הגורמים, כאילו נאמר 'בינינו ובינך' או 'וביניכם'; והשווה גם [להלן] מב כג.130 'בֵּינֵינוּ וּבֵינֶךָ' איננו איפוא אלא פירושו של 'בֵּינוֹתֵינוּ'".

הבנה אחרת למילים אלו עולה מתוך תרגום אונקלוס, שתרגמן כך: "תִּתְקַיַּים כְּעַן מוֹמָתָא דַּהֲוָת בֵּין אֲבָהָתַנָא בֵּינַנָא וּבֵינָךְ" – תתקיים כעת השבועה שהיתה בין אבותינו – בינינו ובינך; להבנתו, אבימלך זה היה בנו של אבימלך שכרת ברית עם אברהם, וכעת הוא מעוניין לחדש את הברית שכרתו אבותיהם ביניהם, שהיא האלה שכבר כעת עומדת "בֵּינוֹתֵינוּ" – בינם לבין יצחק, ושיעור הכתוב: "תְּהִי נָא [ה]אָלָה [ש]בֵּינוֹתֵינוּ [מכוח אבותינו, גם] בֵּינֵינוּ וּבֵינֶךָ [מכוח כריתת ברית מחדש שנעשה בינותינו]". וכך פירש רש"י: "האלה אשר 'בֵּינוֹתֵינוּ' מימי אביך, תהי גם עתה 'בֵּינֵינוּ וּבֵינֶךָ'".

ומדוע רצה אבימלך לחדש את הברית שהיתה בין אביו לבין אברהם? מדוע נזקקה ברית זו לחידוש, מאחר והיא נכרתה לשלושה דורות, כדברי אבימלך שם: "אִם תִּשְׁקֹר לִי וּלְנִינִי וּלְנֶכְדִּי"?

החזקוני כתב על כך: "אף על פי שכבר נשבעו אברהם ואבימלך עד שלושה דורות, מכל מקום ביקש אבימלך לכרות ברית חדשה ביניהם, לפי שהוא פשע בברית על דבר הבארות ועל ששילחו מאיתו". וכך כתב גם הרמב"ן, בפירושו הראשון:

ועניין הרעה הזאת אשר פחדו ממנו [כאמרם "אִם תַּעֲשֵׂה עִמָּנוּ רָעָה"], איננו נראה שיירא מלך פלשתים שלא יבוא עליו יצחק למלחמה. אבל הבטיחם אברהם לו ולנינו ולנכדו, ועתה אמרו בליבם: בעבור שהפרנו אנחנו את בריתנו עם יצחק ושילחנוהו מעלינו, גם הוא יפר בריתו עימנו ויגרשו זרעו את זרענו מן הארץ! ולכן עשו עימו ברית חדשה, והתנצלו131 לאמר לו שלא הפרו הברית הראשונה, שלא עשו עימו [אלא] רק טוב.

וזה טעם "תְּהִי נָא אָלָה בֵּינוֹתֵינוּ" – שנבוא עתה עימך באָלָה להחרים על כל מי שיעבור על הברית, כטעם "לְעבְרְךָ בִּבְרִית ה' אֱלֹהֶיךָ וּבְאָלָתוֹ" (דברים כט יא).

לפי פירוש זה, פלשתים חששו שמא יראה יצחק את מה שעשו לו כהפרת הברית, ומשכך גם הוא לא יראה עצמו מחויב לה; ואף כי לא חששו שיצחק עצמו יילחם בהם, חששו מכך שזרעו יילחם בזרעם, ולפיכך רצו לחדש את הברית ביניהם. בדבריהם אליו, אינם מציגים את הדברים כך שהברית הופרה וכעת הם מבקשים ברית חדשה – שכן בכך היו מפלילים את עצמם כמי שנהגו שלא כהוגן, והיו נותנים אישור למחשבה שאכן הברית הופרה, והלוא ייתכן שיצחק לא ירצה לחדש ביניהם ברית. לפיכך, הם מציגים את הדברים כך שהברית הקודמת עודנה בתוקפה והם לא הפרו אותה – שהרי עשו עימו רק טוב ושילחוהו בשלום; ואם כן, מה הם מבקשים? להוסיף לברית זו גם אָלָה, שתחזק את תוקפה ותאיים על מי שירצה לעבור עליה.132

הרמב"ן מוסיף פירוש שני:

וייתכן שהיה אברהם גדול מאוד ורב כוח, שהיו בביתו שלוש מאות איש שולף חרב,133 ולו בעלי ברית רבים, והוא גם בן חיל אשר ליבו כלב האריה,134 ורדף ארבעה מלכים גיבורים מאוד ונצחם; וכאשר ראו הצלחתו כי היא מאת האלוהים, פחד ממנו מלך פלשתים פן יכבוש ממנו מלכותו, ונקל זאת ממלחמת ארבעת המלכים; אולי שמע עוד דבר ה' שנתן לו הארץ – ולכן בא עימו בברית והשביעו פן "תִּשְׁקֹר לִי וּלְנִינִי וּלְנֶכְדִּי" (לעיל כא כג), כי היה שקר אם ימרוד על המלך, וחשב כי אפשר שיחיה אברהם עד מלוך נכדו.

וכמעשה אבות עשו בנים, כי היה יצחק גדול כאביו ופחד המלך פן יילחם בו בגרשו אותו מארצו.

לפי פירוש זה, אבימלך ביקש מאברהם לכרות עימו ברית מפני חשש שהיה לו מאברהם עצמו, שהיה גיבור מלחמה וניצח את ארבעת המלכים – ולא מפני דאגה לעתיד עמו בדורות הבאים, פן יילחמו בהם זרע אברהם. משכך, מלכתחילה ברית זו כללה את אברהם לבדו, ולא את זרעו, ורק בצד של אבימלך היתה הברית לשלושה דורות – "לִי וּלְנִינִי וּלְנֶכְדִּי", מפני ש"חשב כי אפשר שיחיה אברהם עד מלוך נכדו". לפיכך, כעת כשאבימלך חושש מפני יצחק פן יילחם בו – שכן ה' עימו, ואנשים רבים סרים למשמעתו (ובניגוד לפירוש הראשון, שלפיו אבימלך לא חשש מכך) – הוא מבקש ממנו לכרות עימו ברית, שכן הברית שכרת אברהם עימו, או עם אביו, אינה מחייבת את יצחק כלל.

"אִם תַּעֲשֵׂה עִמָּנוּ רָעָה כַּאֲשֶׁר לֹא נְגַעֲנוּךָ"

בפסוק זה קיים קושי תחבירי, שכן הוא פותח במילה "אִם", המציינת לכאורה תנאי שגורר תוצאה, אך לא נאמר בו מה היא התוצאה; "אִם תַּעֲשֵׂה עִמָּנוּ רָעָה כַּאֲשֶׁר לֹא נְגַעֲנוּךָ, וְכַאֲשֶׁר עָשִׂינוּ עִמְּךָ רַק טוֹב וַנְּשַׁלֵּחֲךָ בְּשָׁלוֹם" – מה יהא אז?

באופן פשוט, המילה "אִם" מציינת שבועת אלה, ואכן לשון הכתוב חסרה, כך שפסוקנו מהווה המשך לפסוק הקודם, וכוונת הדברים לומר: "תְּהִי נָא אָלָה בֵּינוֹתֵינוּ בֵּינֵינוּ וּבֵינֶךָ וְנִכְרְתָה בְרִית עִמָּךְ – אִם תַּעֲשֵׂה עִמָּנוּ רָעָה", דהיינו שתוכן האלה הוא: "אִם תַּעֲשֵׂה עִמָּנוּ רָעָה" ובכך תעבור על האלה – תבוא עליך פורענות. צורה לשונית זו מאפיינת שבועות נוספות במקרא, וכפי שפירש הרמב"ן בעניין ברית אברהם ואבימלך, על הפסוק "וְעַתָּה הִשָּׁבְעָה לִּי בֵאלֹהִים הֵנָּה אִם תִּשְׁקֹר לִי וּלְנִינִי וּלְנֶכְדִּי" (לעיל כא כג):

מילת 'אם', בכל מקום – מספקת, אל תחשב בה בלתי זה. והיא באה ברוב מקומות השבועה: "אִם תִּשְׁקֹר לִי", "וְלָכֵן נִשְׁבַּעְתִּי לְבֵית עֵלִי אִם יִתְכַּפֵּר עֲוֹן בֵּית עֵלִי" (שמואל א' ג יד), "אַחַת נִשְׁבַּעְתִּי בְקָדְשִׁי אִם לְדָוִד אֲכַזֵּב" (תהלים פט לו), "[אֲשֶׁר נִשְׁבַּעְתִּי בְאַפִּי] אִם יְבֹאוּן אֶל מְנוּחָתִי" (תהלים צה יא), "וַיִּקְצֹף וַיִּשָּׁבַע לֵאמֹר: אִם יִרְאֶה אִישׁ בָּאֲנָשִׁים הָאֵלֶּה" (דברים א לד-לה). והעניין, כי בעבור היות השבועות בְּאָלָה, יאמר: השבעה לי לאמר כה יעשה לך אלוהים וכה יוסיף אם תשקור לי, וכן אמר: "תְּהִי נָא אָלָה בֵּינוֹתֵינוּ בֵּינֵינוּ וּבֵינֶךָ". ובעניין הגבוה: "נִשְׁבַּעְתִּי בְקָדְשִׁי אִם לְדָוִד אֲכַזֵּב", ו"אִם יִתְכַּפֵּר עֲוֹן בֵּית עֵלִי" – כלומר: אם יהיה כן אין דברי אמת. וכיוצא בזה שלא ירצה לפרש התנאי, והכתוב יכנה ויקצר בהם וכמותם בחסרון התנאי: "וַיִּקְרָא יַעְבֵּץ לֵאלֹהֵי יִשְׂרָאֵל לֵאמֹר: אִם בָּרֵךְ תְּבָרֲכֵנִי וְהִרְבִּיתָ אֶת גְּבוּלִי וְהָיְתָה יָדְךָ עִמִּי וְעָשִׂיתָ מֵּרָעָה לְבִלְתִּי עָצְבִּי, וַיָּבֵא אֱלֹהִים אֵת אֲשֶׁר שָׁאָל" (דברי הימים א' ד י) – יחסר התנאי כולו. וכן: "אִם יִרְאוּ אֶת הָאָרֶץ" (במדבר יד כג), יחזור לפסוק ראשון: "חַי אָנִי וְיִמָּלֵא כְבוֹד ה' אֶת כָּל הָאָרֶץ" (שם פס' כא), ויקצר השבועה.

לדבריו, שבועה כוללת את המילה 'אם' במשמעות של קבלת אלה, קללה, אם השבועה לא תקויים – וכפי שאומר כאן אבימלך: "תְּהִי נָא אָלָה בֵּינוֹתֵינוּ בֵּינֵינוּ וּבֵינֶךָ, וְנִכְרְתָה בְרִית עִמָּךְ", אלא שהלשון מקוצרת. כך הוא כשהשבועה נאמרת על ידי אדם, וכשהיא נאמרת על ידי ה' משמעות התנאי הוא: "אם יהיה כן, אין דברי אמת".

אך בעניינו פירש הרמב"ן באופן אחר, שלפיו פסוקנו אינו מהווה המשך ישיר לקודמו, אלא מתחיל אמירה חדשה; וכך כתב בפירושו כאן:

"אִם תַּעֲשֵׂה עִמָּנוּ רָעָה כַּאֲשֶׁר לֹא נְגַעֲנוּךָ" – יאמר: אם תעשה עימנו רעה כמו שאנחנו לא נגענוך – "אַתָּה עַתָּה בְּרוּךְ ה'", ואין לאל ידינו לעשות עימך רע, אבל ישתנה העת בעבור החמס שאתה עושה, ותצטרך לשוב אל ארצנו ונשיב לך גמול.

לפי פירוש זה, ההשלמה לתנאי "אִם תַּעֲשֵׂה עִמָּנוּ רָעָה" היא סוף הפסוק – "אַתָּה עַתָּה בְּרוּךְ ה'", שבו הם רומזים לו: אמנם עתה אתה מבורך מאת ה' ואינך צריך לנו, אך אם תעשה עימנו רעה, על אף שאנחנו לא נגענוך ועשינו עימך רק טוב, ישתנה מזלך לרעה "ותצטרך לשוב אל ארצנו, ונשיב לך גמול" רע, על החמס שעשית עימנו.

אולם יש לתמוה על כך, מה ראה הרמב"ן לפרש באופן זה – ולא כפי ההבנה הפשוטה, המשווה בין פסוקנו לבין פסוקים אחרים של שבועה שנאמר בהם 'אם' – וכפי שפירש הוא עצמו לעיל? ועוד, שלפי פירושו כאן גם המילה "כַּאֲשֶׁר" אינה מובנת היטב – מה פירוש "אם תעשה עימנו רעה כמו שאנחנו לא נגענוך"? הרי אין כאן כל דימוי – אתה תעשה רעה, ואילו אנחנו לא עשינו רעה! מה שאין כן אם נפרש כאפשרות הקודמת, הרי שכוונת הדברים היא: "וְנִכְרְתָה בְרִית עִמָּךְ, אִם תַּעֲשֵׂה עִמָּנוּ רָעָה" – דהיינו שלא תעשה עימנו רעה, ואם תעשה רעה תתקיים בך האלה, "כַּאֲשֶׁר" גם אנחנו "לֹא נְגַעֲנוּךָ". ר' נפתלי הרץ ויזל הוסיף להקשות על פירוש זה של הרמב"ן: "ואין דבר מדבריו בכתוב! ועוד, בעודם מבקשים אהבתו ידברו אליו קשות, ודברים המשברים הלב?!".

"כַּאֲשֶׁר לֹא נְגַעֲנוּךָ"

"אִם תַּעֲשֵׂה עִמָּנוּ רָעָה כַּאֲשֶׁר לֹא נְגַעֲנוּךָ" – המילה "כַּאֲשֶׁר" אינה מופיעה כאן כציון זמן אלא כציון דימוי, "כמו אשר"; נכרות ברית על כך שלא תעשה עימנו רעה, כמו אשר אנחנו "לֹא נְגַעֲנוּךָ" לרעה, וכמו אשר אנחנו "עָשִׂינוּ עִמְּךָ רַק טוֹב, וַנְּשַׁלֵּחֲךָ בְּשָׁלוֹם".

אימתי "לֹא נְגַעֲנוּךָ"? רש"י פירש: "'לֹא נְגַעֲנוּךָ' – כשאמרנו לך 'לֵךְ מֵעִמָּנוּ'", ואילו הרמב"ן פירש שאמירה זו מתייחסת לזמן שבו ישב יצחק ביניהם: "וטעם 'לֹא נְגַעֲנוּךָ' – שלא נפתה לִבֵּנוּ על אשתך ליגע באחד מכם, כלשון 'הַנֹּגֵעַ בָּאִישׁ הַזֶּה וּבְאִשְׁתּוֹ' (פס' יא)".

ניתן לפרש, שאכן אמירה זו מתייחסת הן לעת שבת יצחק בגרר, כרמב"ן, והן לעת שילוחו משם, כרש"י, שהן בזה והן בזה "לֹא נְגַעֲנוּךָ" – והיא מהווה כלל שלאחריו פרט, ומתפרטת בדברים שהם אומרים מיד לאחר מכן: "וְכַאֲשֶׁר עָשִׂינוּ עִמְּךָ רַק טוֹב" – בשבתך עימנו, ומלבד זאת גם "וַנְּשַׁלֵּחֲךָ בְּשָׁלוֹם".

"וְכַאֲשֶׁר עָשִׂינוּ עִמְּךָ רַק טוֹב"

מה הוא אותו "טוֹב" שעשו עימו? הרמב"ן כתב: "לשמור כל אשר לך, בצוותינו את העם להשמר מכם" – וכאמור לעיל: "וַיְצַו אֲבִימֶלֶךְ אֶת כָּל הָעָם לֵאמֹר: הַנֹּגֵעַ בָּאִישׁ הַזֶּה וּבְאִשְׁתּוֹ מוֹת יוּמָת" (פס' יא). ייתכן שכוונתם גם לעצם הרשות שנתנו לו לגור בארצם, וכפי שפירש רש"י את דברי אבימלך לאברהם "כַּחֶסֶד אֲשֶׁר עָשִׂיתִי עִמְּךָ תַּעֲשֶׂה עִמָּדִי" (לעיל כא כג) – "שאמרתי לך 'הִנֵּה אַרְצִי לְפָנֶיךָ [בַּטּוֹב בְּעֵינֶיךָ שֵׁב]' (שם כ טו)". בנוסף לכך, ייתכן שכוונתם גם לקשרי המסחר שניהלו עימו, שמכוחם היתה לו פרנסה בתקופת הרעב, ולאחר מכן גם עושר רב.

מילת הדימוי "כַּאֲשֶׁר" מופיעה פעמיים בפסוקנו: "אִם תַּעֲשֵׂה עִמָּנוּ רָעָה כַּאֲשֶׁר לֹא נְגַעֲנוּךָ, וְכַאֲשֶׁר עָשִׂינוּ עִמְּךָ רַק טוֹב וַנְּשַׁלֵּחֲךָ בְּשָׁלוֹם". כאמור לעיל, הדימוי "כַּאֲשֶׁר לֹא נְגַעֲנוּךָ" מובן בהקשר זה, שכן הם מבקשים ממנו שלא יעשה עמהם רעה, כמו שהם לא נגעו בו לרעה; אך מהי משמעות הדימוי "וְכַאֲשֶׁר עָשִׂינוּ עִמְּךָ רַק טוֹב"? הרי הם אינם אומרים "אם לא תעשה עימנו טובה כמו שאנו עשינו עימך טובה", ואין האלה מחייבת אותו לעשות עמהם טוב, אלא רק שלא לעשות עמהם רע, כאמרם "אִם תַּעֲשֵׂה עִמָּנוּ רָעָה", ואם כן – מה עניינה של הזכרת הטוב שעשו עימו? מהי הציפייה מיצחק שמקבילה לכך שהם עשו עימו טוב?

כפי שיתבאר להלן, רש"י פירש את המילים שבסוף הפסוק "אַתָּה עַתָּה בְּרוּךְ ה'" כרומזות לרצונם לקבל מאיתו טובה, וכך כתב: "'אַתָּה' – גם אתה עשה עימנו כמו כן", דהיינו עתה אתה מבורך מאת ה', ולפיכך אנו מצפים ממך לעשות עימנו טובה. ולפי דבריו, מובן כי זוהי גם משמעות המילים "וְכַאֲשֶׁר עָשִׂינוּ עִמְּךָ רַק טוֹב", המקדימות את הסיומת "אַתָּה עַתָּה בְּרוּךְ ה'"; ושיעור הכתוב: כשם שעשינו עימך טוב, "גם אתה עשה עימנו כמו כן", שהרי אתה "בְּרוּךְ ה'".

עם זאת, אבימלך ואנשיו אינם כוללים בפירוש בתוכן האלה אמירה כגון "אם לא תעשה עימנו טוב", שכן לא באו לחייב אותו באלה לעשות עמהם טוב; אין כוונתם אלא לרמוז ליצחק כי הם מצפים שהוא יעשה עמהם טובה כראות עיניו, אך לא כללו זאת בשבועת האלה.

אולם אם נפרש את המילים "אַתָּה עַתָּה בְּרוּךְ ה'" באופנים אחרים, ולא כרש"י, נמצא שאין בדברי אבימלך הבעת רצון בכך שיצחק יעשה עמהם טובה על ידי הדברים שהתברך בהם; ואם כן שבה ועולה השאלה, מהי המקבילה שהם מצפים לה מיצחק בעקבות זאת שעשו עימו טוב?

ניתן להסביר, שאכן הם מבקשים מיצחק שיעשה עמהם טוב – וטוב זה הוא עצם היענותו לבקשתם לכרות עמהם ברית; ושיעור דבריהם: "אִם תַּעֲשֵׂה עִמָּנוּ רָעָה, כַּאֲשֶׁר לֹא נְגַעֲנוּךָ" – דהיינו הישבע שלא תעשה עימנו רעה, כמו שאנחנו לא עשינו עמך רעה; ועשה עימנו טוב בכך שתיענה לבקשתנו לכרות עימך ברית זו, שבה תישבע כך, כמו שאנחנו עשינו עימך טוב – "וְכַאֲשֶׁר עָשִׂינוּ עִמְּךָ רַק טוֹב". וכך פירש ר' נפתלי הרץ ויזל: "הטובה תהיה שיבוא בְאָלָה ובשבועה לבלתי הרע עמהם ולכרות עמהם ברית".

באופן נוסף ניתן לפרש זאת על פי ההבנה שהוצעה לעיל, שלפיה המילים "כַּאֲשֶׁר לֹא נְגַעֲנוּךָ" מהוות כלל שלאחריו פרט: "וְכַאֲשֶׁר עָשִׂינוּ עִמְּךָ רַק טוֹב, וַנְּשַׁלֵּחֲךָ בְּשָׁלוֹם", דהיינו: לא נגענוך לרעה – הן בעת שבתך עימנו, והן בעת שילוחך מעימנו. אם כן, עיקר עניינן של המילים "וְכַאֲשֶׁר עָשִׂינוּ עִמְּךָ רַק טוֹב" הוא המילה "רַק" – עשינו עימך רק טוב, ולא רע; אין אבימלך טוען לתמורה בעבור ה'טוב' שהם עשו עימו, אלא כוונתו לומר – בעת שישבת עימנו והיו בינינו מעשים, היו אלו רק מעשים שמיטיבים איתך; וגם בעת ששילחנוך מאיתנו, היה זה "בְּשָׁלוֹם". אם לא נפרש שפסוק זה בנוי ככלל ופרט לא ניתן יהיה ליישב כך, שכן אם כל עניין האמירה "עָשִׂינוּ עִמְּךָ רַק טוֹב" הוא לשלול עשיית רע, מה מוסיפות מילים אלו על האמור לפניהן "לֹא נְגַעֲנוּךָ"? אך אם נפרש שזהו כלל ופרט, יובן כי הזכרת המעשים – "עָשִׂינוּ עִמְּךָ" – נועדה לציין התייחסות ספציפית לתקופת שבתו של יצחק בגרר, ולאחריה תופיע התייחסות ספציפית לעת שילוח יצחק מגרר, כך שיחד הן מבהירות כי "לֹא נְגַעֲנוּךָ" כלל, מתחילה ועד סוף.

באופן נוסף ניתן להסביר, שאכן אין כוונתם במילים "וְכַאֲשֶׁר עָשִׂינוּ עִמְּךָ רַק טוֹב" להביע בקשה שיצחק יעשה עמהם טוב, ואינם אומרים זאת אלא כחיזוק על גבי האמור קודם לכן "לֹא נְגַעֲנוּךָ"; כלומר: אנו לא נגענוך לרעה, ואף עשינו עימך טוב – ולכן מצידך מתבקש שלכל הפחות לא תעשה עימנו רעה. אם תעשה רעה למי שלא עשה לך רעה, יהיה זה מעשה שאינו הגון; אך אם תעשה רעה למי שעשה עימך טובה, יהיה זה מעשה נבלה; ולפיכך, הם מוסיפים כי מעבר לכך ש"לֹא נְגַעֲנוּךָ", גם "עָשִׂינוּ עִמְּךָ רַק טוֹב" – ומצופה ממך ביתר שאת שלא "תַּעֲשֵׂה עִמָּנוּ רָעָה".135

"וַנְּשַׁלֵּחֲךָ בְּשָׁלוֹם"

כהמשך לדבריהם על כך ש"לֹא נְגַעֲנוּךָ", הם מזכירים בפירוש גם את שילוחו מאיתם, ומציינים ששילוח זה היה "בְּשָׁלוֹם". ניתן לפרש שכוונת הדברים היא שלא נגעו בו לרעה בשלחם אותו מעימם, ולא הזיקו לו בגופו או בממונו; כך פירש הרמב"ן: "כי גם בקנאתנו בך לא לקחנו מכל העושר אשר עשית עימנו, ושילחנוך עם כל אשר לך בשלום", וכן רלב"ג: "וכאשר שילחנו אותך לא הזקנו לך לא בגופך ולא בקנייניך". לפי אמות המידה המוסריות של הפלשתים, הם ראו ציון לשבח לעצמם בכך שלא הזיקו ליצחק בשלחם אותו; וכפי שכתב רש"ר הירש (בפס' כו): "אופיינית היא גם הטענה 'וַנְּשַׁלֵּחֲךָ בְּשָׁלוֹם' – הם רואים עצמם כחסידים וישרים, על שגירשו אותו מבלי לגזול ממנו מקודם את רכושו".

במדרש בראשית רבה (סד י) על פסוקנו מובא משל הממחיש עניין זה: "ארי טָרַף טֶרֶף ועמד עצם בגרונו, אמר: כל דאתי מפיק ליה אנא יהיב ליה אגריה [=כל שיבוא ויוציא את העצם, אתן לו שכרו]; אתא הדין קורא מיצראה דמקוריה אריך, יהיב מקוריה ואפקיה [=בא אותו קורא מצרי שמקורו ארוך, הכניס מקורו והוציאה], אמר ליה: הב לי אגרי! אמר ליה: זיל תהא מלגלג ואומר דעילת לפומא דאריה בשלם ונפקת בשלם [=אמר לו: תן לי שכרי! אמר לו: לך תהא מלגלג ואומר שנכנסת לפי האריה בשלום ויצאת בשלום]" – כלומר: דייך בכך שלא אכלתיך כשהיית בפי, וזהו שכרך; כך גם הפלשתים רואים שבח לעצמם בכך ששילחו את יצחק בשלום, ולא גרשוהו מאיתם בחוסר כל.

ניתן לפרש אמירה זו באופן נוסף, כך שאין כוונתה לשילוח בשלום ללא פגע, אלא לשילוח שנעשה מתוך אווירה של שלום, ולא כגירוש; כך פירש רד"ק: "ולא גרשנוך מעימנו דרך קלון, אלא דרך כבוד שילחנוך בשלום; ובעבור קנאת הפלשתים בך אמרנו לך מה שאמרנו, כדי שלא יגעו בך מרב קנאתם", וכן ר' נפתלי הרץ ויזל:

גם השילוח ששילחנוך מאיתנו, שאתה מתרעם עליה, לא היה שילוח של שנאה אלא לטובה; לפי שקינאו אותך יושבי העיר יעצנוך ללכת מן העיר, ולא גרשנוך מן הארץ, ולא לקחנו מאומה מכל אשר לך; ואם כן ראוי שֶׁנֵחָשֵׁב בעיניך כאוהבים ושתכרות עימנו ברית.

לעיל בביאור לפסוק טז עסקנו בשתי אפשרויות בהבנת דברי אבימלך ליצחק "לֵךְ מֵעִמָּנוּ" – שניתנים להיקרא בנימה של ציווי וגירוש, וכמו כן גם בנימה רכה של עצה טובה. הפירוש האחרון מתאים לאפשרות השנייה, שלפיה אכן אבימלך שילח את יצחק בשלום ובטוב, ולא כגירוש; אך ייתכן כי גם לפי האפשרות הראשונה ניתן לפרש בפסוקנו בדרך זו, כך שאבימלך ואנשיו אינם מציגים כאן את הדברים כהוויתם אלא כפי שנוח להם להציגם, ואף על פי שגרשוהו הם טוענים ששילחוהו "בְּשָׁלוֹם".

גם מתוך דברי הרשב"ם ניתן להבין ש"וַנְּשַׁלֵּחֲךָ בְּשָׁלוֹם" היינו באווירה של שלום, שכן כתב: "'אַתָּה עַתָּה' כמו כן כרות ברית עימנו ושלחנו מאיתך בשלום כמו שעשינו לך. [...] וכן עשה יצחק, כדכתיב: 'וַיְשַׁלְּחֵם יִצְחָק וַיֵּלְכוּ מֵאִתּוֹ בְּשָׁלוֹם' (פס' לא)". אבימלך אמר ליצחק: "אִם תַּעֲשֵׂה עִמָּנוּ רָעָה, כַּאֲשֶׁר לֹא נְגַעֲנוּךָ" – דהיינו שלא תעשה לנו רעה, כשם שאנו לא עשינו לך רעה; "וְכַאֲשֶׁר עָשִׂינוּ עִמְּךָ רַק טוֹב" – כשם שעשינו עימך טוב, כך עשה עימנו טוב (כמבואר לעיל), וכהמשך לכך, מוסיף: "וַנְּשַׁלֵּחֲךָ בְּשָׁלוֹם", והמילה "וְכַאֲשֶׁר" מתייחסת גם לתוספת זו – "וְכַאֲשֶׁר [...] וַנְּשַׁלֵּחֲךָ בְּשָׁלוֹם", ואם כן מתבקש לפרש שגם על השילוח בשלום הם מצפים מיצחק לתמורה מקבילה: כאשר שילחנוך בשלום, כך עשה עימנו. לפיכך מפרש רשב"ם שכוונת דבריהם היתה "ושלחנו מאיתך בשלום, כמו שעשינו לך"; ומוסיף, שאכן כך היה, כאמור להלן לאחר כריתת הברית: "וַיְשַׁלְּחֵם יִצְחָק וַיֵּלְכוּ מֵאִתּוֹ בְּשָׁלוֹם".

ועל כך יש לשאול, מה פשר בקשתם מיצחק שישלחם מאיתו בשלום? הרי ודאי לא חששו מכך שיפגע בהם כעת, בעודם מתארחים אצלו כמשלחת דיפלומטית!136 מסתבר איפוא, שכוונת רשב"ם לפרש שביקשו מיצחק שישלחם באווירת שלום ופיוס, מתוך כריתת הברית עמהם; שלא יתווכח עימם בדברים קשים ויסרב לבקשתם, אלא ישתף עימם פעולה ובכך ייפרדו מתוך שלום. זוהי בוודאי משמעות התיאור האמור להלן "וַיְשַׁלְּחֵם יִצְחָק וַיֵּלְכוּ מֵאִתּוֹ בְּשָׁלוֹם", ורשב"ם משווה בין שני הפסוקים הללו המזכירים שילוח בשלום ומפרש אותן במשמעויות זהות, כך שהם מהווים שני צדדיו של אותו המטבע: הם ביקשו מיצחק שינהג עמהם כמו שהם נהגו עימו – "וְכַאֲשֶׁר [...] וַנְּשַׁלֵּחֲךָ בְּשָׁלוֹם", ומעיד הכתוב כי אכן כך עשה – "וַיְשַׁלְּחֵם יִצְחָק וַיֵּלְכוּ מֵאִתּוֹ בְּשָׁלוֹם".

כאמור, הדמיון הלשוני שבין שני הפסוקים עשוי לסייע לפירוש רשב"ם, שה'הליכה בשלום' בפסוקנו מתפרשת כאותה 'הליכה בשלום' האמורה להלן; אך אין בכך הכרח, שכן ניתן גם לומר שבכל אחד מהפסוקים מתפרשת ה'הליכה בשלום' בהתאם להקשר של אותו הפסוק: בפסוקנו מדובר על שילוח במובן של גירוש, שילוח אדם מארץ שבה הוא התיישב כגר, ובעניין זה היתה קיימת אפשרות ששילוח זה לא יהא 'בשלום' אלא תוך פגיעה בו או ברכושו – ובמובן זה יש לפרש את הביטוי האמור כאן "וַנְּשַׁלֵּחֲךָ בְּשָׁלוֹם"; ואילו להלן (בפס' לא) מדובר על שילוח במובן של ליווי או פרידה, ובעניין זה אין צריך לומר שיצחק לא פגע באבימלך וסיעתו, אלא שנפרדו מתוך שלום והסכמה ולא מתוך ריב ומחלוקת – וממילא מובן כי במשמעות זו יש לפרש את האמור שם "וַיְשַׁלְּחֵם יִצְחָק וַיֵּלְכוּ מֵאִתּוֹ בְּשָׁלוֹם".137

"אַתָּה עַתָּה בְּרוּךְ ה'"

לעיל הוזכרו פירושיהם של רש"י ושל הרמב"ן למילים אלו; רש"י פירש: "'אַתָּה' – גם אתה עשה עימנו כמו כן", וכך פירש את דבריו ה'מזרחי':

זהו פירוש "אַתָּה עַתָּה בְּרוּךְ ה'", והרצון בזה: כמו שהשווה עצמו עם יצחק בעניין הֶעְדֵּר הָרָעָה, ואמר "אִם תַּעֲשֵׂה עִמָּנוּ רָעָה כַּאֲשֶׁר לֹא נְגַעֲנוּךָ", שפירושו: כמו שלא עשינו גם אנחנו לך שום רעה, כן השווה עצמו עם יצחק גם בעניין הטוב, ואמר "וְכַאֲשֶׁר עָשִׂינוּ עִמְּךָ רַק טוֹב וַנְּשַׁלֵּחֲךָ בְּשָׁלוֹם" כן גם אתה עשה עימנו כן, עתה שאתה "בְּרוּךְ ה'" ויש לך כוח לעשות עימנו טוב; וכמוהו נאמר לאברהם [מפי אבימלך]: "כַּחֶסֶד אֲשֶׁר עָשִׂיתִי עִמְּךָ תַּעֲשֶׂה עִמָּדִי" (לעיל כא כג).

"בְּרוּךְ ה'" היינו מבורך מאת ה'; ולפי רש"י רומז לו אבימלך במילים אלו שמכיון שעתה הוא "בְּרוּךְ ה'" ויש בידו להיטיב ולגמול להם – הוא מצפה לקבל מאיתו טובה, כשם שהם עשו עימו "רַק טוֹב". וראה עוד מה שהתבאר בשיטתו לעיל (תחת הכותרת "וְכַאֲשֶׁר עָשִׂינוּ עִמְּךָ רַק טוֹב").

הרמב"ן פירש באופן אחר:

"אִם תַּעֲשֵׂה עִמָּנוּ רָעָה כַּאֲשֶׁר לֹא נְגַעֲנוּךָ" – יאמר: אם תעשה עימנו רעה כמו שאנחנו לא נגענוך – "אַתָּה עַתָּה בְּרוּךְ ה'", ואין לאל ידינו לעשות עימך רע, אבל ישתנה העת בעבור החמס שאתה עושה, ותצטרך לשוב אל ארצנו ונשיב לך גמול.

לפי פירושו, המילים "אַתָּה עַתָּה בְּרוּךְ ה'" משלימות את האמור לפניהן "אִם תַּעֲשֵׂה עִמָּנוּ רָעָה"; ובאמירה זו הם רומזים ליצחק: אמנם עתה אתה מבורך מאת ה' ואינך צריך לנו, אך אם תעשה עימנו רעה, על אף שאנחנו לא נגענוך ועשינו עימך רק טוב, ישתנה מזלך לרעה "ותצטרך לשוב אל ארצנו, ונשיב לך גמול" רע, על החמס שעשית עימנו. וראה לעיל (תחת הכותרת "אִם תַּעֲשֵׂה עִמָּנוּ רָעָה כַּאֲשֶׁר לֹא נְגַעֲנוּךָ") בעניין הקשיים העומדים על פירוש זה.

באופן נוסף פירש האלשיך, על פי שיטתו בפירוש המילים "רָאוֹ רָאִינוּ כִּי הָיָה ה' עִמָּךְ" שהובאה לעיל (בביאור למילים אלו):

"וַיֹּאמְרוּ רָאוֹ רָאִינוּ" וכו' – לומר: ראינו מה שגדלת עלינו בהיותך עימנו, ואחר כך ראינו שלא נוסף לנו שפע בלכתך, והכרנו בזה כי לא מעושר שלנו היית מושפע בהיותך עימנו, רק "כִּי הָיָה ה' עִמָּךְ" [...]. במה ראינו זה? הלא הוא כי "אַתָּה עַתָּה בְּרוּךְ ה'", כי הלא אחר לכתך מעימנו אתה לבדך "בְּרוּךְ ה'" ככל השפע שהיה לך עימנו, מה שאין כן אנחנו רק כאשר היינו בתחילה בעודך עימנו.

כך פירש גם המלבי"ם, בהתאם לפירושו שם: "'אַתָּה עַתָּה בְּרוּךְ ה'" – רוצה לומר שראינו שטעינו מה שחשבנו שאתה נתעשרת מארצנו ומגרע את השפע שלנו, שהלא ראינו שאתה גם עתה שאינך בינינו, אתה ברוך ה' בפני עצמך ואינך צריך אל ארצנו", וכן ה'משך חכמה': "שכאשר היית עימנו אמרנו 'כִּי עָצַמְתָּ מִמֶּנּוּ', היינו שלקחת שלנו ומאיתנו נתעשר[ת], אבל כעת רואים שאתה בעצמך 'בְּרוּךְ ה'' ומהשגחתו יתברך באה אליך ההצלחה והברכה הנפרצה מגבולי הטבע". ומכיון שראו שה' עימו ושהוא ברוך ה', הם מעוניינים לכרות עימו ברית, כמבואר לעיל שם.

באופן נוסף פירש 'בכור שור', שלפיו מילים אלו ממשיכות את טענת אבימלך שהם לא עשו עימו כל רע, אף על פי ששילחוהו מאיתם, ובכך כוונתו לנמק מדוע יצחק נדרש לקבל את בקשתם ולא לעשות עמהם רעה; ואלו דבריו:

"אַתָּה עַתָּה בְּרוּךְ ה'" – כלומר: אפילו גרמה [=נזק עקיף] לא עשינו עימך; שאם היית אומר: אתם שילחתם אותי ממקום שהייתי מתברך בו למקום שאינו מתברך, היית יכול לומר כי גרמנו לך רעה; אבל "אַתָּה עַתָּה בְּרוּךְ ה'" כמו לְשָׁם, נמצא שלא גרמנו לך [אלא] רק טוב.

כלומר, לא גרמנו לך כל נזק בכך ששילחנוך מאיתנו, שכן "אַתָּה עַתָּה בְּרוּךְ ה'" כמו שהיית ברוך ה' בעבר, בהיותך איתנו.

יתרה מזאת, ניתן לפרש שכוונתם לומר: דווקא עתה, שאינך בינינו, אתה ברוך ה', אך לא כן אם היית נשאר עימנו בגרר; ואם כן, לא זו בלבד שהשילוח לא גרם לך נזק, אלא שהוא גם הביא לך תועלת; וכך פירש 'בית הלוי' על התורה: "'אַתָּה עַתָּה בְּרוּךְ ה'', כאומר: כעת, אשר אתה מרוחק מהפלשתים ההמוניים, אתה ברוך ה' – כלומר איש מוצלח בברכות ה', ואם היית בתוכם מי יודע מה שהיו עושים לך בקנאתם, שהיו שודדים ושוללים כל אשר לך" – ובהתאם לאפשרות המובאת לעיל, שבאמרו "וַנְּשַׁלֵּחֲךָ בְּשָׁלוֹם" היתה כוונתו לומר ששילחוהו לטובתו כדי שלא ייפגע.

פירוש נוסף מופיע בדברי רשב"ם: "'אַתָּה עַתָּה בְּרוּךְ ה'' – סרס המקרא ודרשהו, עתה ראינו שאתה ברוך ה' ולא ישמעאל אחיך, כי כאשר ראינו באברהם אביך ראינו בך"; וכך גם בדברי החזקוני, ביתר פירוט:

"אַתָּה עַתָּה בְּרוּךְ ה'" – שלא תאמר: עד שאתם משביעים אותי, השביעו את ישמעאל אחי! לכך נאמר: "אַתָּה עַתָּה בְּרוּךְ ה'" – אין אנו צריכים להשביע רק אתה [=אלא אותך], לפי ש"אַתָּה עַתָּה בְּרוּךְ ה'" – שהקב"ה גזר שתירש הארץ, כדכתיב: "וַיְהִי אַחֲרֵי מוֹת אַבְרָהָם וַיְבָרֶךְ אֱלֹהִים אֶת יִצְחָק בְּנוֹ" (לעיל כה יא).

מכך שה' בירך את יצחק למדו הפלשתים כי בו בחר ה' לקיים בזרעו את ההבטחות שהבטיח לאברהם, שבכללן ירושת הארץ138– וממילא חששו פן הוא או זרעו יילחמו בהם כדי ליטול את ארצם; לפיכך, הם מבקשים לכרות ברית דווקא עימו, ומנמקים זאת בכך ש"אַתָּה עַתָּה בְּרוּךְ ה'" – אתה הוא המבורך מאת ה', ולא ישמעאל אחיך. בשונה מהפירושים הקודמים, שלפיהם הדגש מונח על המילה "עַתָּה", לפי פירוש הדגש מונח על המילה "אַתָּה" – ושיעור הכתוב: "עַתָּה [אנו רואים, כי] אַתָּה [הוא] בְּרוּךְ ה'".

    שם נאמר: "וְהֵם לֹא יָדְעוּ כִּי שֹׁמֵעַ יוֹסֵף כִּי הַמֵּלִיץ בֵּינֹתָם", וגם שם היה המליץ מתווך בין שני צדדים. כך גם בפסוקים "כִּי עֵד הוּא בֵּינֹתֵינוּ כִּי ה' הָאֱלֹהִים" (יהושע כב לד) – בין בני גד וראובן לבין יתר השבטים; "וַיֹּאמְרוּ זִקְנֵי גִלְעָד אֶל יִפְתָּח: ה' יִהְיֶה שֹׁמֵעַ בֵּינוֹתֵינוּ אִם לֹא כִדְבָרְךָ כֵּן נַעֲשֶׂה" (שופטים יא י) – בינינו לבינך.

  3. 131

    המילה "התנצלו" אינה מתפרשת כאן כמשמעותה בעברית המודרנית – הודאה בטעות ובקשת סליחה, אלא כפשט המילה: "התנצלו" – הצילו את עצמם, כלומר טענו טענת הצטדקות שמנקה אותם מאשמה.

  4. 132

    מפשט דברי הרמב"ן כאן משמע שברית אברהם עם אבימלך נכרתה בלא אלה, וכעת מבקש אבימלך להוסיף עליה אלה; אך בפירושו שם (שיצוטט להלן, בביאור המילים "אִם תַּעֲשֵׂה עִמָּנוּ רָעָה כַּאֲשֶׁר לֹא נְגַעֲנוּךָ") כתב שגם הברית עם אברהם כללה אלה, ואם כן שבה ועולה השאלה – כיצד הם הציגו את רצונם לכרות ברית עם יצחק אף על פי שברית אברהם לכאורה עודנה קיימת? ייתכן, שאכן לא התייחסו בדבריהם לעניין זה; הם לא הזכירו את המניע האמיתי שגורם להם לרצות לכרות ברית מחדש, כאמור, אך גם לא הציגו מניע אחר – אלא התעלמו מהיות ברית קיימת כבר ביניהם, וביקשו "תְּהִי נָא אָלָה בֵּינוֹתֵינוּ בֵּינֵינוּ וּבֵינֶךָ, וְנִכְרְתָה בְרִית עִמָּךְ".

  5. 133

    כאמור בעניין מלחמת המלכים: "וַיָּרֶק אֶת חֲנִיכָיו יְלִידֵי בֵיתוֹ שְׁמֹנָה עָשָׂר וּשְׁלֹשׁ מֵאוֹת" (לעיל יד יד).

  6. 134

    כלשון הכתוב לגבי דוד: "וְהוּא גַם בֶּן חַיִל אֲשֶׁר לִבּוֹ כְּלֵב הָאַרְיֵה הִמֵּס יִמָּס" (שמואל ב' יז י).

  7. 135

    רד"ק כתב על מילים אלו: "'וְכַאֲשֶׁר עָשִׂינוּ עִמְּךָ רַק טוֹב' – 'לֹא' שזכר, עומד במקום שנים, כאילו אמר: 'ובעבור שלא עשינו עימך רק טוב', וכמוהו: '[כִּי לֹא לָנֶצַח יִשָּׁכַח אֶבְיוֹן], (ו)תִּקְוַת ענוים [קרי: עֲנִיִּים] תֹּאבַד לָעַד' (תהלים ט יט) – ולא תקות ענוים [תאבד לעד], ורבים כמוהו"; וכך גם החזקוני: "'לֹא' קאי אף למטה, כאשר לא נגענוך וכאשר לא עשינו עימך רק טוב". לפירושם, המילה "לֹא" שמופיעה לפני כן, באמירה "לֹא נְגַעֲנוּךָ", מושכת עצמה ואחר עימה, כך שהיא מתייחסת גם למילים "עָשִׂינוּ עִמְּךָ רַק טוֹב" – ויוצרת את המשמעות "ולא עָשִׂינוּ עִמְּךָ רַק טוֹב", והם מפרשים את המילה "רַק" כ"אלא", כך שמשמעות הכתוב היא "לא עשינו עימך אלא טוב".

    אולם פירוש זה תמוה עד מאוד; שכן אמנם בלשונם של הפרשנים המילה 'רק' משמשת כ'אלא', אך לא בלשון המקרא, שבה מילה זו מתפרשת במובן של ייחוד ומיעוט, וכמו בלשון ימינו; ואם כן, אין כל צורך בהוספת המילה 'לא', ואדרבה – הוספה כזו תהפוך את משמעות הכתוב: לא עשינו עימך רק טוב, אלא גם רע.

  8. 136

    כפי שהקשה ר' נפתלי הרץ ויזל: "וכי היו אבימלך ושריו חושדים את יצחק שירע עמהם, איך הלכו אליו? ואם הלכו, למה לא לקחו עמהם גיבורי חיל להגן עליהם מרעתו?".

  9. 137

    עוד יש לציין, כי גם מבחינה לשונית קיים הבדל בין הפסוקים, שכן לא נאמר שם "וַיְשַׁלְּחֵם יִצְחָק בְּשָׁלוֹם", אלא "וַיְשַׁלְּחֵם יִצְחָק, וַיֵּלְכוּ מֵאִתּוֹ בְּשָׁלוֹם" – לא על השילוח נאמר "בְּשָׁלוֹם", אלא על ההליכה; אך ניתן גם להבין שהמילה "בְּשָׁלוֹם" מתייחסת לכל מה שלפניה, הן ל"וַיְשַׁלְּחֵם יִצְחָק" והן ל"וַיֵּלְכוּ מֵאִתּוֹ", ומלמד הכתוב שהן מצד יצחק והן מצידם שררה אווירה של שלום.

  10. 138

    מפשט לשון החזקוני ניתן להבין שכוונתו לכך שה' בירך את יצחק בברכת אברהם, דהיינו הבטחה לירושת הארץ, וכפי שהתפרש בבראשית רבה (סא ו) הפסוק שציטט בדבריו; אולם בדרך הפשט נראה שקשה לפרש שהפלשתים ידעו על הברכה שאמר ה' ליצחק, ומסתבר יותר לפרש שכוונתם לברכה שה' בירכו בה בפועל, דהיינו הצלחה ועושר, וכפי שהתפרש אותו הפסוק בדברי חלק מהפרשנים (ראה בספרי 'אם בישראל – פשט ועומק בפרשת חיי שרה', עמ' 413-412), ומכך הסיקו כי בו בחר ה' להיות המשך לאברהם, ולא בישמעאל.

פרק כו · פסוקים ל–לא

וַיַּעַשׂ לָהֶם מִשְׁתֶּה וַיֹּאכְלוּ וַיִּשְׁתּוּ. וַיַּשְׁכִּימוּ בַבֹּקֶר וַיִּשָּׁבְעוּ אִישׁ לְאָחִיו, וַיְשַׁלְּחֵם יִצְחָק וַיֵּלְכוּ מֵאִתּוֹ בְּשָׁלוֹם.

כתב רד"ק: "לקיים האהבה עשה להם משתה ואכלו ביחד ושתו" – שכן כך היה נהוג, שלאחר כריתת ברית שני הצדדים אוכלים יחד ובכך מבטאים את התקרבותם זה לזה ואת גמירות דעתם להסכם שביניהם, וכפי שמצינו בברית שכרתו ביניהם לבן ויעקב, שנאמר באותו עניין: "וַיֹּאכְלוּ שָׁם עַל הַגָּל" (להלן לא מו).

האברבנאל כתב: "הנה יצחק קיבל מאיתם טענתם, ויעש להם משתה ונשבעו איש לרעהו". אך נראה שאין הכרח לומר שיצחק קיבל את טענתם, וייתכן שהסכים לכרות עמהם ברית על אף שטענתם לא היתה מקובלת עליו – והדבר תלוי בשאלה האם דבריהם שיקפו את המציאות בצורה נכונה או לא; שכן הם אומרים: "כַּאֲשֶׁר לֹא נְגַעֲנוּךָ וְכַאֲשֶׁר עָשִׂינוּ עִמְּךָ רַק טוֹב" – והרי הכתוב מעיד שהם קינאו אותו, כאמור לעיל "וַיְקַנְאוּ אֹתוֹ פְּלִשְׁתִּים" (פס' יד), וסתמו את הבארות שחפר אביו, כאמור לעיל "וְכָל הַבְּאֵרֹת אֲשֶׁר חָפְרוּ עַבְדֵי אָבִיו בִּימֵי אַבְרָהָם אָבִיו סִתְּמוּם פְּלִשְׁתִּים וַיְמַלְאוּם עָפָר" (פס' טו); ואם כן, אם נפרש שקנאתם היתה בלב בלבד, ושסתימת הבארות לא נעשתה כדי להתנכל לו, אלא מפני סיבה אחרת, והיו אלו בארות נטושים שנסתמו על ידי פלשתים אחרי מות אברהם (כמבואר באריכות לעיל שם) – הרי שייתכן כי דבריהם כאן נכונים; אך אם הקנאה האמורה לעיל משמעותה היא התנכלות בפועל (כמבואר שם בעקבות חלק מהפרשנים), או שהבארות שסתמו פלשתים היו בארות שימושיות ליצחק – הרי שדבריהם כאן אינם אלא מצג שווא של הצטדקות, ולא ניתן לומר שיצחק "קיבל מאיתם טענתם". כמו כן, גם לגבי אמירתם "וַנְּשַׁלֵּחֲךָ בְּשָׁלוֹם" – אם משמעות הדברים היא שלא הזיקו לו בשלחם אותו, הרי שדבריהם נכונים; אך אם נפרש שמשמעותם היא ששילחוהו בכבוד ומתוך עצה טובה, ומאידך נפרש שבפועל הם גרשוהו בעל כרחו – הרי שדברים אלו אינם נכונים, ולא קיבלם יצחק.

מכל מקום, נאות יצחק לכרות עמהם ברית שלום, ואף עשה להם משתה; "כדרך הצדיקים להתפייס בנקל ולהימצא למי שמבקש שלום" ('העמק דבר'). בברית אברהם ואבימלך לא נאמר שעשה להם משתה; וייתכן, שדווקא מפני המתיחות שהיתה בין יצחק ואבימלך על רקע שילוחו מגרר, וכן על רקע ההתנכלויות לו מצד פלשתים (למפרשים שאכן היו כאלה) – ראה יצחק צורך לעשות להם משתה על מנת לחזק את האחווה והפיוס, ורק מתוך כך לכרות את הברית, כך שהברית תיכרת בלב שלם ומתוך קרבה הדדית; מה שאין כן באברהם, שבינו לבין אבימלך לא קדמה מתיחות כזו לכריתת הברית. ולפיכך, רק לאחר המשתה ולינת הלילה שאחריו – כרתו ביניהם את הברית, כאמור כאן: "וַיַּשְׁכִּימוּ בַבֹּקֶר, וַיִּשָּׁבְעוּ אִישׁ לְאָחִיו".

"וַיְשַׁלְּחֵם יִצְחָק" – נפרד מהם, ושמא גם ליווה אותם, וכאמור במדרש (מכילתא, פתיחתא לפרשת בשלח): "אין שילוח בכל מקום אלא ליווי, שנאמר [...]: 'וַיְשַׁלְּחֵם יִצְחָק'".

"וַיֵּלְכוּ מֵאִתּוֹ בְּשָׁלוֹם" – "זוהי שלום הנפש, שלא נשאר טינא בליבם, והלכו שמחים וטובי לב כהפרשת אוהבים ורעים זה מזה" (ר' נפתלי הרץ ויזל); וראה עוד מה שהתבאר על מילים אלו לעיל (בפסוק כט, תחת הכותרת "וַנְּשַׁלֵּחֲךָ בְּשָׁלוֹם").

פרק כו · פסוק לב

וַיְהִי בַּיּוֹם הַהוּא וַיָּבֹאוּ עַבְדֵי יִצְחָק וַיַּגִּדוּ לוֹ עַל אֹדוֹת הַבְּאֵר אֲשֶׁר חָפָרוּ, וַיֹּאמְרוּ לוֹ: מָצָאנוּ מָיִם.

"וַיְהִי בַּיּוֹם הַהוּא" – באותו היום שבו כרת ברית עם אבימלך, באו עבדיו ובישרו לו שמצאו מים. מהי משמעות הדבר שקרה כך דווקא באותו היום, ומדוע הכתוב רואה לנכון לציין זאת?

רד"ק כתב: "'וַיְהִי בַּיּוֹם הַהוּא' – קודם נסעם מאיתו; והיה זה סיבה מאת האל כדי שיראו עוד בהצלחתו וייראו ממנו". לפירושו, עבדי יצחק באו אליו ואמרו לו כן בעוד שהיו אבימלך ואנשיו עימו – וכך סובבה ההשגחה, על מנת שיווכחו עוד בהצלחתו של יצחק וייראו ממנו.

אולם ישנו קושי בפירוש זה, שכן לפיו הכתובים אינם מופיעים כסדרם; ומעבר לקושי שקיים בכך מצד עצם שינוי הסדר, הרי שבזה נעלם מעיני הקורא הטעם לכך שהיה כן דווקא "בַּיּוֹם הַהוּא"! אם כל עיקר האמירה שהיה זה באותו היום הוא כדברי רד"ק, היה מתבקש שהכתוב יבטא רעיון זה גם בסדר הדברים, וישבץ את פסוקנו לפני סוף הפסוק הקודם – "וַיְשַׁלְּחֵם יִצְחָק וַיֵּלְכוּ מֵאִתּוֹ בְּשָׁלוֹם". ועוד, מאחר וכבר כרתו עם יצחק ברית מפני שראו בהצלחתו ויראו ממנו – לשם מה היה צורך בכך "שיראו עוד בהצלחתו וייראו ממנו"?

נראה איפוא, כי מסתבר יותר לפרש שהכתובים מופיעים כסדרם (וכפי שמשמע מיתר המפרשים, שלא ציינו שקיים כאן שינוי סדר), ועבדי יצחק סיפרו לו על אודות הבאר רק לאחר הליכת אבימלך ואנשיו; ואף אם נאמר שהיה זה לפני הליכתם, לא היתה חשיבות לכך שסיפור הדברים היה דווקא בנוכחותם. ואם כן, עלינו למצוא טעם אחר לכך שמציין כאן הכתוב כי היה זה "בַּיּוֹם הַהוּא".

ניתן להסביר זאת באופן פשוט על פי פירוש רש"י שיובא להלן (בביאור הפסוק הבא), שכתב כי יצחק קרא לבאר 'שבעה' על שם הברית, שבה נשבעו זה לזה. אם כך, נֹאמַר כי הכתוב מציין שהיה זה "בַּיּוֹם הַהוּא" על מנת להסביר מדוע קרא יצחק לבאר על שם השבועה – שכן הוא התבשר אודותיה באותו היום שבו נשבעו.

"וַיַּגִּדוּ לוֹ עַל אֹדוֹת הַבְּאֵר אֲשֶׁר חָפָרוּ, וַיֹּאמְרוּ לוֹ מָצָאנוּ מָיִם" – כתב הרמב"ן: "הוא האמור בו למעלה 'וַיִּכְרוּ שָׁם עַבְדֵי יִצְחָק בְּאֵר' (פס' כה), כי החלו לחפור אותו, ובא אבימלך אליו בימים ההם, וביום הברית בלכתם ממנו בישרוהו כי מצאו מים". כפי שהתבאר לעיל שם, הפועל "וַיִּכְרוּ" מציין את תחילת החפירה; הם התחילו בחפירת הבאר וטרם מצאו בה מים, עד לאותו היום – שבו מצאו מים בבאר, וממילא סיימו את חפירתה; ובהתאם לכך – שם נאמר "וַיִּכְרוּ", וכאן נאמר "חָפָרוּ". וכך כתב הנצי"ב ב'העמק דבר': "'עַל אֹדוֹת הַבְּאֵר' – שהחלו לכרות אז, 'אֲשֶׁר חָפָרוּ' – כי גמרו החפירה".

מהו היחס בין שתי צלעות הפסוק, המקבילות זו לזו – "וַיַּגִּדוּ לוֹ עַל אֹדוֹת הַבְּאֵר אֲשֶׁר חָפָרוּ, וַיֹּאמְרוּ לוֹ מָצָאנוּ מָיִם"? כתב על כך ר' נפתלי הרץ ויזל:

וזהו שאמר "וַיַּגִּדוּ לוֹ עַל אֹדוֹת הַבְּאֵר אֲשֶׁר חָפָרוּ", וכל "הגדה" – סיפור דברים בכל פרטיהן; כי סיפרו ליצחק כל פרטי העמל והיגיעה שהיה להם לחפור הבאר הזה. ואחר שהגידו לו למה האריכו ימים טרם השיגו מבוקשם, אמרו לו "מָצָאנוּ מָיִם" – עתה מצאנו מים.

כלומר, הכתוב מתאר כי עבדי יצחק סיפרו לו באריכות על ענייני חפירת הבאר – וזוהי משמעות הפועל "וַיַּגִּדוּ", המציין 'הגדה' מפורטת; ותכלית הדברים היתה האמירה "מָצָאנוּ מָיִם", וזוהי משמעות הפועל "וַיֹּאמְרוּ", המציין 'אמירה' תכליתית.

פרק כו · פסוק לג

וַיִּקְרָא אֹתָהּ שִׁבְעָה, עַל כֵּן שֵׁם הָעִיר בְּאֵר שֶׁבַע עַד הַיּוֹם הַזֶּה.

"וַיִּקְרָא אֹתָהּ שִׁבְעָה" – לא התפרש בכתוב מה היה הטעם לקריאת שם זה; רש"י כתב: "'שִׁבְעָה' – על שם הברית", דהיינו השבועה שבברית, וכדברי מדרש 'לקח טוב': "על שם השבועה שנשבעו". כאמור לעיל, מכך שהכתוב מציין שהיה זה "בַּיּוֹם הַהוּא" ישנו סיוע לפירוש זה – שכן מטרת הכתוב לציין שישנו קשר בין השבועה שנשבעו באותו היום לבין קריאת שם הבאר, מאחר ושמה של הבאר נקרא כך לאור מאורעות אותו היום. גם מהאמור בעניין ברית אברהם ואבימלך ישנו סיוע לפירוש זה, שכן נאמר שם: "עַל כֵּן קָרָא לַמָּקוֹם הַהוּא בְּאֵר שָׁבַע, כִּי שָׁם נִשְׁבְּעוּ שְׁנֵיהֶם" (לעיל כא לא) – ואם כן, מסתבר לפרש שגם קריאת שם הבאר 'שבעה' היתה "כִּי שָׁם נִשְׁבְּעוּ שְׁנֵיהֶם".

לדרך זו, ייתכן שיצחק קרא לבאר על שם השבועה על מנת ששמה של הבאר יהווה עדות ומזכרת על הברית שכרת עם פלשתים, ויזכיר להם את מחוייבותם כלפיו. ייתכן גם שהודה לה' על כך שקיים לו הבטחתו – "כִּי אִתְּךָ אָנֹכִי, וּבֵרַכְתִּיךָ" (פס' כד), עד כדי שגדולי הפלשתים באו אליו לבקש ממנו שיכרות עמהם ברית, בנימוק ש"רָאוֹ רָאִינוּ כִּי הָיָה ה' עִמָּךְ" – והוא ביטא הודאה זו בקריאת שמה של הבאר 'שבעה', על שם השבועה שנשבעו הוא ואבימלך איש לאחיו.

ניתן להסביר באופן נוסף מדוע קרא יצחק לבאר על שם השבועה, שעל פיו גם יובן היטב מדוע מדגיש הכתוב שהיה זה "בַּיּוֹם הַהוּא" – שכן לא סמיכות זמן מקרית בלבד קשרה את השבועה עם בשורת הבאר, אלא קשר מהותי ראה יצחק ביניהן, שבסיבתו קרא לבאר על שם השבועה. יצחק ראה במציאת המים בתזמון זה כעין אות משמיים המהווה תגובה על מעשה כריתת הברית; שכן, היו לו סיבות רבות להיות בתרעומת על הפלשתים, ויכול היה להמשיך את הקו הביקורתי שביטא בתחילת דבריו אליהם: "מַדּוּעַ בָּאתֶם אֵלָי, וְאַתֶּם שְׂנֵאתֶם אֹתִי וַתְּשַׁלְּחוּנִי מֵאִתְּכֶם", ולסרב לכרות עמהם ברית או להתחמק מכך; אך לא כך עשה, אלא הניח את תרעומתו והקפדתו עליהם, ונענה לבקשתם; ולא עוד, אלא שהתפייס עימם ועשה להם משתה, כדי לחזק את הקרבה והריעות ביניהם – "כדרך הצדיקים להתפייס בנקל ולהימצא למי שמבקש שלום" ('העמק דבר' בפס' ל). והנה באותו היום, מיד לאחר כריתת הברית, מבשרים לו עבדיו כי מצאו מים; זה עתה התפייס, למען השלום, עם אלו ששילחוהו מאיתם וסתמו את בארותיו – ומיד הוא מקבל 'פיצוי' מאת יד ההשגחה, שמספקת לו באר חדשה, וכביכול אומרת לו: "יֵשׁ לַה' לָתֶת לְךָ הַרְבֵּה מִזֶּה" (דברי הימים ב' כה ט), ויפה עשית שלא הקפדת עם אבימלך וסיעתו על אודות הבארות, מעבר לתוכחה המרומזת שבתחילת הדברים; כדאי הוא הפיוס למען השלום, והנה המקום ממלא חסרונך.139

באופן אחר פירש הספורנו את טעם קריאת שם הבאר 'שבעה':

קרא את הבאר 'שבעה', מפני שהיה מקום שביעי שבו חפרו באר: ג' של אברהם שסתמו פלשתים, כאמרו "וְכָל הַבְּאֵרֹת [אֲשֶׁר חָפְרוּ עַבְדֵי אָבִיו בִּימֵי אַבְרָהָם אָבִיו] סִתְּמוּם פְּלִשְׁתִּים" (פס' טו), ולא יאמר 'כל' בפחות מג'; וג' של יצחק, שהן 'עשק' 'שטנה' ו'רחובות'; וזה היה השביעי, שקראוהו 'שבעה'.

לעומת זאת, הרמב"ן מפרש כי באר זו היתה באר שכבר נחפרה על ידי עבדי אברהם, והיא הבאר שעליה אמר אברהם לאבימלך בעת שכרתו ברית ביניהם: "כִּי אֶת שֶׁבַע כְּבָשׂת תִּקַּח מִיָּדִי בַּעֲבוּר תִּהְיֶה לִּי לְעֵדָה כִּי חָפַרְתִּי אֶת הַבְּאֵר הַזֹּאת" (לעיל כא ל), ולאחר מכן נאמר: "עַל כֵּן קָרָא לַמָּקוֹם הַהוּא בְּאֵר שָׁבַע, כִּי שָׁם נִשְׁבְּעוּ שְׁנֵיהֶם". על פי הנחה זו, כותב הרמב"ן כי יצחק קרא כך לבאר מפני שזהו השם שניתן לה על ידי אביו, וכאמור לעיל "וַיִּקְרָא לָהֶן שֵׁמוֹת כַּשֵּׁמֹת אֲשֶׁר קָרָא לָהֶן אָבִיו" (פס' יח); ואלו דבריו:

וקרוב לי, שהוא הבאר שכרה אברהם ונתן לו שבע כבשות לעדה, כי סתמוהו פלשתים עם האחרים, וישב יצחק ויחפור אותו, ויקרא לו גם הוא שֵׁם כַּשֵׁם אשר קרא לו אביו. ו"עַל כֵּן שֵׁם הָעִיר בְּאֵר שֶׁבַע", על שם הבאר אשר קראוהו כן האב והבן "כִּי שָׁם נִשְׁבְּעוּ שְׁנֵיהֶם".140

הרמב"ן כותב כי יצחק קרא לבאר בשם שקרא לה אביו; ואמנם, בעניין אברהם אומר הכתוב "עַל כֵּן קָרָא לַמָּקוֹם הַהוּא בְּאֵר שָׁבַע", וכאן יצחק קורא לבאר "שִׁבְעָה", ושמות אלו אינם זהים – אולם ניתן להסביר שלדעת הרמב"ן גם אברהם קרא לבאר זו 'שבעה', ואילו על המקום הסמוך לה נֶאֱמַר שנקרא 'באר שבע'; וכך ניתן ללמוד מלשון הכתוב, שכן כאן נאמר "וַיִּקְרָא אֹתָהּ שִׁבְעָה" – את הבאר עצמה, ואילו שם נאמר "קָרָא לַמָּקוֹם הַהוּא בְּאֵר שָׁבַע". וטעם הדבר, שהרי ידוע שהבאר היא באר, ואין צורך לכלול בשמה את המילה 'באר', אלא רק את המילה המורה על העניין שעל שמו היא נקראת – 'שבעה'; אך המקום שליד הבאר נקרא 'באר שבע', דהיינו: המקום שבו ישנה 'באר' בשם 'שבעה'.

מלשון הכתוב לעיל "וַיִּכְרוּ שָׁם עַבְדֵי יִצְחָק בְּאֵר" (פס' כה) משמע קצת שלא כדברי הרמב"ן, שכן משתמע מכך שהיתה זו באר סתמית, ואילו היתה זו בארו של אברהם היה לַכָּתוּב לציין זאת. מעבר לכך, ניתן גם להקשות על דבריו – הלוא אם היתה זו באר שכבר נחפרה בעבר, מהו החידוש באמירה "מָצָאנוּ מָיִם"? הרי מאחר וידוע שבעבר היתה שם באר, ניתן לצפות שגם עתה יימצאו בה מים! אך ניתן ליישב זאת, ולומר שבימי אברהם לא היתה זו 'באר מים חיים' שיש בה נביעה, אלא באר שהתנקזו אליה מי גשמים (וכמו הבארות האחרות שחפרו עבדי אברהם, כמבואר לעיל בפסוק יט), ומשכך – לא ידעו עבדי יצחק שימצאו בה מים; אך הם העמיקו לחפור יותר מעומקה המקורי של הבאר, וכך מצאו בה נביעה – ועל כך הם מבשרים ליצחק "מָצָאנוּ מָיִם".

מתוך דברי הרמב"ן עולה, כפי שניתן גם להבין מפשטות הכתובים, כי באר שבע הנזכרת בענייננו היא אותה באר שבע הנזכרת בעניין ברית אברהם ואבימלך; ואם כך, עולה מאליה השאלה – מה פשר דברי הכתוב "עַל כֵּן שֵׁם הָעִיר בְּאֵר שֶׁבַע עַד הַיּוֹם הַזֶּה"? הרי כבר לאחר ברית אברהם ואבימלך נקרא המקום ההוא 'באר שבע', ואם כן הוא היה נקרא כך גם לפני הברית שכרת יצחק!

רד"ק כתב על כך: "ומה שאמר 'עַל כֵּן שֵׁם הָעִיר בְּאֵר שֶׁבַע' – וכבר נקראת כן מימי אברהם, כמו שכתוב! על זה המעשה היה יותר קבוע"; כלומר, לאחר שאברהם קרא כך למקום,141 שם זה עדיין לא היה מושרש היטב בפי כל; אך לאחר שיצחק קרא את שם הבאר 'שבעה' השם השתרש יותר, כך שהוא נותר "עַד הַיּוֹם הַזֶּה".

הסבר זה, וכן ההסברים דלהלן, תואמים לכך שבעניין אברהם לא נאמר שהברית היא הסיבה לכך שזהו שם המקום כיום, אלא: "עַל כֵּן קָרָא לַמָּקוֹם הַהוּא בְּאֵר שָׁבַע" – הכתוב מעיד כי אברהם, או קורא אחר, קרא לאותו המקום על שם השבועה, אך לא נאמר שָׁם ששם זה התקיים לאורך זמן. לעומת זאת, בענייננו לא נאמר שיצחק קרא כך למקום, אלא שלבאר קרא 'שבעה', ומוסיף הכתוב ששמה של הבאר היווה סיבה לכך שהעיר הסמוכה לה נקראה "בְּאֵר שֶׁבַע, עַד הַיּוֹם הַזֶּה" – שכן מימי יצחק ואילך שם זה השתרש בפי כל, ונותר בשימוש "עַד הַיּוֹם הַזֶּה" – דהיינו עד ימי כתיבת התורה.

באופן דומה ניתן להסביר, שהשם 'באר שבע' שניתן למקום זה בימי אברהם נשכח בינתיים, וכעת יצחק חידש אותו, ובעקבות זאת שם זה נותר בשימוש "עַד הַיּוֹם הַזֶּה". ייתכן שהשם שקרא אברהם נשכח בעקבות סתימת הבאר על ידי הפלשתים, שמאחר ונסתמה הבאר חדלו לקרוא למקום על שמה, או שחדלו אנשים לעבור באותו המקום מאחר וכבר לא יכולים היו להשתמש בבאר, וכך נשתכח שמו. כעת, יצחק קורא למקום מחדש באותו השם, על שם הבאר שנחפרה עתה, ושם זה נשאר "עַד הַיּוֹם הַזֶּה".

הנצי"ב ב'העמק דבר' הסביר באופן אחר מדוע לא השתמר שם המקום:

כי מאז שהפר אבימלך ושריו הברית, שינו השם כדי לשכח השבועה ולא יהא להם למזכיר עוון, ועתה כשחזרו וכרתו ברית שיש בה אלה, וגם הזדמן שחפר הבאר ההיא השביעית וקרא אותה 'שבעה', חזר השם לכמו שהיה ונשאר כן לדורות. וזהו שכתוב כאן "עַל כֵּן שֵׁם הָעִיר", ולא כמו לעיל באברהם, דכתיב "עַל כֵּן קָרָא לַמָּקוֹם הַהוּא" – דשָׁם הטעם על קריאת השם, וכאן הטעם על שלא נשתנה עוד השם.

יש שיישבו קושי זה בדרך אחרת, על סמך ההבדל בין האמור בעניין אברהם "עַל כֵּן קָרָא לַמָּקוֹם הַהוּא בְּאֵר שָׁבַע" לבין האמור בענייננו "עַל כֵּן שֵׁם הָעִיר בְּאֵר שֶׁבַע". כך כתב 'בכור שור': "שם המקום היה 'באר שבע' בימי אברהם, ועתה אף שם העיר נקראת 'באר שבע'", וכן המלבי"ם:

והנה בימי אברהם קרא למקום ההוא 'באר שבע', רק להמקום של הבאר והאשל, והעיר לא נקראת עדיין בשם זה; ועתה שיצחק קרא גם כן בשם 'שבעה' נעשה 'באר שבע' שם כולל לכל העיר, ושם זה נשאר "עַד הַיּוֹם הַזֶּה".

פירוש זה מעוגן היטב בפשט הכתובים, שכן מהם נראה כי המקום שקרא לו אברהם 'באר שבע' לא היה בתוך עיר, אלא בשטחי מרעה ובסמוך לדרך – שהרי נאמר שם "וַיִּטַּע אֵשֶׁל בִּבְאֵר שָׁבַע, וַיִּקְרָא שָׁם בְּשֵׁם ה' אֵל עוֹלָם" (פס' לג) – כמנהגו של אברהם, לייסד מקום להכנסת אורחים וקריאה בשם ה' במקום שיש בו הולכי דרכים;142 ואם כן, מובן שה"מָקוֹם הַהוּא" שקרא לו אברהם "בְּאֵר שָׁבַע" לא היה עיר, אלא 'מקום' כפשוטו – המקום שסמוך לאותה באר, ושמאוחר יותר נטע בו אשל וייסד בו מרכז לקריאה בשם ה'. לעומת זאת, כאן נאמר "עַל כֵּן שֵׁם הָעִיר בְּאֵר שֶׁבַע" – בעקבות קריאת יצחק את שם הבאר 'שבעה' נקראה על שמה גם העיר הסמוכה לאותו המקום – 'באר שבע', ולפני כן לא נקראה כך. וייתכן, שהיתה זו עיר חדשה שלא היתה קיימת בימי אברהם, ונבנתה בסמוך למאורע זה, כך שבאופן טבעי דבק בה השם 'באר שבע' על שם הבאר הסמוכה לה, מאחר וטרם השתרש לה שם אחר.

לפי פירוש זה, ייתכן שהשם 'באר שבע' שקרא אברהם לאותו המקום השתמר עד ימי יצחק, ושמא גם לאחר מכן; ומכל מקום לא ציין שָׁם הכתוב שזהו שמו "עַד הַיּוֹם הַזֶּה", שכן עד ימי כתיבת התורה לא השתמר שם זה לאותו המקום אלא רק לעיר, וייתכן גם שהמקום נבלע בשטחה של העיר הסמוכה לו ברבות הימים, וממילא לא היתה עוד משמעות לקריאת שמו על ידי אברהם.

לעומת כל האמור עד כה, ניתן גם לומר ש'באר שבע' הנזכרת בענייננו אינה אותה 'באר שבע' הנזכרת בעניין אברהם; ואף על פי שנאמר לעיל בתחילת העניין "וַיַּעַל מִשָּׁם בְּאֵר שָׁבַע" (פס' כג) – באותה שעה לא נקרא המקום כך, אלא הכתוב מכנהו בשם זה על שם סופו, וכמנהגו במקומות אחרים;143 ולפי אפשרות זו, השאלה שהצבנו בראש הדברים אינה עולה כלל. כך פירש רשב"ם: "'עַל כֵּן שֵׁם הָעִיר בְּאֵר שֶׁבַע' – ואין זו אותו באר שבע של אברהם, כי שני 'באר שבע' היו, כדכתיב: 'וַיָּבֹא בְּאֵר שֶׁבַע אֲשֶׁר לִיהוּדָה' (מלכים א' יט ג)".

רשב"ם מוכיח מהפסוק האמור בעניין אליהו – "וַיָּבֹא בְּאֵר שֶׁבַע אֲשֶׁר לִיהוּדָה" – שהיתה גם 'באר שבע' אחרת, שלא היתה בנחלת יהודה, ולפיכך נצרך הכתוב להבהיר שאותה 'באר שבע' האמורה שם היא זו שליהודה; ולהבנתו, שתי 'באר שבע' אלו הן זו של אברהם וזו של יצחק. לשיטתו, בהכרח עלינו לומר כי 'באר שבע' האחרת היתה לשמעון, שהרי אברהם ויצחק גרו בארץ הנגב שהיתה כולה בנחלת יהודה, ובתוכה היו ערים שניתנו לשבט שמעון, ואם כן לא ייתכן ש'באר שבע' של אברהם או של יצחק היתה בנחלת שבט אחר. ואכן, עיר בשם 'באר שבע' מופיעה הן ברשימת ערי שבט יהודה (יהושע טו כח), והן ברשימת ערי שבט שמעון (שם יט ב).

אולם ניתן לדחות ראיה זו, ולפרש שלא הוסיף שָׁם הכתוב את המילים "אֲשֶׁר לִיהוּדָה" מפני שישנה 'באר שבע' אחרת, אלא כדי לנמק מדוע בחר אליהו ללכת לשם, וכפי שיתבאר להלן; ואף מהאמור בספר יהושע אין ראיה לכך, שכן הכתוב מונה ברשימת ערי שבט יהודה גם את הערים שניתנו לשבט שמעון, שהרי אף הן היו בתוך נחלתו. וכך פירש רד"ק במלכים שם:

לא נאמר "אֲשֶׁר לִיהוּדָה" אלא בעבור [=ביחס ל]באר שבע אשר נמצא בנחלת בני שמעון, כי הוא אותו הכתוב גם כן בנחלת בני יהודה, וכבר פירשתי כי שתיהן אחד,144 ולפי שנחלת בני שמעון בתוך נחלת בני יהודה מנה אותו עם יהודה ומנאו עם שמעון. אלא אמר "אֲשֶׁר לִיהוּדָה", כלומר: יצא לו מתחת רשות אחאב ומארצו, והלך לו אל ארץ יהודה מפחדו מאיזבל.

מכל מקום, גם אם נאמר שרק 'באר שבע' אחת היא הנזכרת בספר יהושע ובספר מלכים, ניתן לומר ש'באר שבע' של יצחק אינה זו של אברהם, ושתי 'באר שבע' היו, אף כי רק אחת מהן נזכרת בהמשך ספרי המקרא. וייתכן גם שבתקופתם של ספרים אלו רק 'באר שבע' אחת היתה, שהיא זו של יצחק, ואילו זו של אברהם לא השתמר שמה לאורך שנים, הגם שבימי יצחק עדיין נקראה בשם זה; ואם אכן כך, ניתן לומר שמשום זאת לא נאמר בעניין אברהם שזהו שם המקום "עַד הַיּוֹם הַזֶּה" כמו בענייננו. יצחק קרא כך לְעִיר, שנותרה קיימת לאורך שנים, וממילא שמה השתמר; אך אברהם קרא כך לְמָקוֹם, שהיה מחוץ לעיר, כמבואר לעיל, וממילא אבד שמו ברבות הימים.

פרק כו · פסוק לד

וַיְהִי עֵשָׂו בֶּן אַרְבָּעִים שָׁנָה וַיִּקַּח אִשָּׁה אֶת יְהוּדִית בַּת בְּאֵרִי הַחִתִּי וְאֶת בָּשְׂמַת בַּת אֵילֹן הַחִתִּי.

הכתוב מספר כאן על נישואי עשו ועל התנהגותן של נשיו, כהקדמה הנצרכת עבור הבנת דברי רבקה ליצחק בסוף הפרק הבא, כפי שכתב רשב"ם: "לפי שעתיד לומר '[וַתֹּאמֶר רִבְקָה אֶל יִצְחָק: קַצְתִּי בְחַיַּי מִפְּנֵי בְּנוֹת חֵת] אִם לֹקֵחַ יַעֲקֹב אִשָּׁה מִבְּנוֹת חֵת כָּאֵלֶּה [מִבְּנוֹת הָאָרֶץ, לָמָּה לִּי חַיִּים]' (להלן כז מו), לכך הוצרך לפרש תחילה מעשיהם הרעים" – וכפי שייאמר בפסוק הבא: "וַתִּהְיֶיןָ מֹרַת רוּחַ לְיִצְחָק וּלְרִבְקָה". בנוסף לכך, כתב רד"ק: "ובא הסיפור הזה להודיע פחזותו של עשו, בקחתו נשים שהיו מכעיסות אביו ואימו וְלֹא כִהָה בָּם145 ולא הוציאם מבית אביו".

מדוע מציין הכתוב שהיה אז עשו "בֶּן אַרְבָּעִים שָׁנָה"?

רשב"ם כתב על כך: "'וַיְהִי עֵשָׂו בֶּן אַרְבָּעִים שָׁנָה' – ואז היה יצחק בן מאה, כי בן שישים היה בלדת רבקה אותם,146להודיענו בא מה שכתוב אחריו: 'וַיְהִי כִּי זָקֵן יִצְחָק' (להלן כז א)". אולם ישנו קושי בהסבר זה, שכן מאחר ונאמר שם "וַיְהִי כִּי זָקֵן יִצְחָק" בידוע שהיה אז זקן, ודי בכך להבנת העניין, ולשם מה נצרך הכתוב ללמד שהיה אז בן מאה (או יותר ממאה)?

מתוך דברי המדרש (בראשית רבה סה א) עולה הסבר אחר לעניין זה, וכך נאמר שם:

"וַיְהִי עֵשָׂו בֶּן אַרְבָּעִים שָׁנָה" – הדא הוא דכתיב: "יְכַרְסְמֶנָּה חֲזִיר מִיָּעַר" (תהלים פ יד); רבי פינחס בשם רבי סימון [אמר]: מכל הנביאים לא פרסמוה [את מלכות רומי] אלא שנים, משה ואסף; משה אמר: "וְאֶת הַחֲזִיר כִּי מַפְרִיס פַּרְסָה הוּא" (דברים יד ח), אסף אמר: "יְכַרְסְמֶנָּה חֲזִיר מִיָּעַר". למה הוא מוֹשְׁלָהּ בחזיר? אלא מה חזיר הזה בשעה שהוא רובץ הוא מפשט את טלפיו כלומר שאני טהור, כך מלכות הזאת הרשעה גוזלת וחומסת, נראית כאילו מצעת את הבימה.147כך עֵשָׂו – כל ארבעים שנה צד נשי אנשים ומענה אותן, וכיון שהגיע לארבעים שנה דימה עצמו לאביו, אמר: מה אבא נשא אישה בן ארבעים אף אני נושא אישה בן ארבעים שנה.148

ניתן לסייע להבנה זו מלשון הכתוב כאן: "וַיְהִי עֵשָׂו בֶּן אַרְבָּעִים שָׁנָה, וַיִּקַּח אִשָּׁה אֶת יְהוּדִית בַּת בְּאֵרִי הַחִתִּי"; לשם מה מוסיף הכתוב את המילה "אִשָּׁה"? לא היה לו אלא לומר "וַיִּקַּח אֶת יְהוּדִית" וכו'! אך לפי דברי המדרש, מובן כי כוונת הכתוב לומר: "וַיְהִי עֵשָׂו בֶּן אַרְבָּעִים שָׁנָה – וַיִּקַּח אִשָּׁה"; כלומר, הכתוב מעיד על כך שבגיל ארבעים נשא עשו אישה, כעניין בפני עצמו, ואין משמעות לעניין זה לזהותה של אותה אישה, אלא לעצם הנישואין – כשם שאביו נשא אישה בגיל ארבעים, כך גם הוא. רק לאחר מכן, כעניין נוסף, מפרש הכתוב גם את זהותה של אותה אישה: "אֶת יְהוּדִית בַּת בְּאֵרִי הַחִתִּי, וְאֶת בָּשְׂמַת בַּת אֵילֹן הַחִתִּי"; שימוש הכתוב במילה "אִשָּׁה" בלשון יחיד, הגם שהוא מזכיר שמותיהן של שתי נשים, מחזק את הרעיון האמור – שלפיו מטרת הכתוב במילה זו היא להבליט את הדמיון בין נישואי עשו לנישואי יצחק; כשם שיצחק לקח אישה (אחת) בגיל ארבעים, כך גם עשו – "וַיִּקַּח אִשָּׁה"; ואמנם הוא הוסיף ולקח שתי נשים, אך אין זה נוגע לעצם העניין.

"אֶת יְהוּדִית בַּת בְּאֵרִי הַחִתִּי, וְאֶת בָּשְׂמַת בַּת אֵילֹן הַחִתִּי" – כתב רד"ק: "מגדוליהם לקח, כי אחר שמזכיר שמותיהם נראה כי בעלי השם היו, ולכבוד יצחק שהיה גדול ונכבד בעיניהם נתנום לו".

הספורנו מעיר על העולה מפסוקנו, שבשונה מאברהם שהקפיד על כך שיצחק לא יישא מבנות כנען, כפי שאמר לעבדו: "אֲשֶׁר לֹא תִקַּח אִשָּׁה לִבְנִי מִבְּנוֹת הַכְּנַעֲנִי אֲשֶׁר אָנֹכִי יוֹשֵׁב בְּקִרְבּוֹ" (לעיל כד ג), יצחק לא הקפיד כך עם עשו: "ולא הקפיד יצחק על נישואי החתיות כמו שעשה אביו"; ובטעם הדבר, ראה מה שהתבאר להלן עמ' 434-433.

פרק כו · פסוק לה

וַתִּהְיֶיןָ מֹרַת רוּחַ לְיִצְחָק וּלְרִבְקָה.

"וַתִּהְיֶיןָ – נשות עשו, שנזכרו בפסוק הקודם – מֹרַת רוּחַ לְיִצְחָק וּלְרִבְקָה". מה פירוש "מֹרַת רוּחַ"? ניתן לפרש ביטוי זה באופנים שונים, הן מבחינת זהות השורש של המילה "מֹרַת", והן מבחינת אופי המילה וצורתה; וכפי שיתבאר.

רש"י פירש: "לשון המראת רוח, כמו 'מַמְרִים הֱיִיתֶם [עִם ה']' (דברים ט ז), כל מעשיהן היו לעצבון ליצחק ולרבקה". לפירושו, מילה זו היא משורש 'מרה' – במשמעות של התנגדות, עשייה הפכית; כשם ש'להמרות פה' היינו לעשות להפך מדיבורו של המְצַוֶּה, כך 'להמרות רוח' היינו לעשות להפך מרוחו, כלומר מרצונו.

באופן אחד ניתן להבין שלפי פירוש רש"י הביטוי "מֹרַת רוּחַ" הוא שם פעולה, כלשונו "המראת רוח", ולפי זה כוונת הכתוב לומר: "וַתִּהְיֶיןָ נשים אלו, בעצם הוויתן ומעשיהן – המראת רוח לְיִצְחָק וּלְרִבְקָה". באופן אחר ניתן להבין, שלפירושו הביטוי "מֹרַת רוּחַ" הוא פועל – "מַמְרָה רוח", ולפי זה שיעור הכתוב: "ותהיין נשים אלו ממרות את רוח יצחק ורבקה";149 ואמנם נאמר "מֹרַת" בלשון יחיד, אך כוונת הכתוב לכל אחת מהן, וכפי שכתב אבן עזרא: "'מֹרַת רוּחַ' – כל אחת מהן", וכן רד"ק (בפירושו השני): "כל אחת ואחת היתה מורה רוחם".

לפי פירוש זה, לכאורה היה צריך לומר "מוֹרָה", דהיינו "מַמְרָה", כמו "בֵּן סוֹרֵר וּמוֹרֶה" (דברים כא יח) בלשון זכר, דהיינו "מַמְרֶה"; אך מפני שמילה זו נסמכת על המילה "רוּחַ" היא מסתיימת באות ת', וכפי שמפרש רשב"ם:

"מֹרַת רוּחַ" – כמו מן "עוֹשֶׂה" לזכר יאמר "עֹשָׂה דְבָרוֹ" (תהלים קמח ח) לנקבה, וכשהוא דבוק יאמר "עוֹשַׂת", כן יאמר מן "מוֹרֶה" לשון "מַמְרֶה", מן "יַמְרֶה אֶת פִּיךָ" (יהושע א יח), יאמר "מוֹרָה" לנקבה, וכשהוא דבוק יאמר "מֹרַת רוּחַ".

לעומת זאת, ניתן גם לפרש שמילה זו היא משורש 'מרר'; וגם לפי אפשרות זו ניתן לפרש באופנים שונים את צורתה של המילה, כשם פעולה – "המררת רוח", כשם עצם – "מרירות רוח", או כפועל – "ממררת רוח".

בדרך זו פירש רד"ק (בפירושו הראשון): "'מֹרַת רוּחַ' – שם, כמו: '[לֵב יוֹדֵעַ] מַָת נַפְשׁוֹ' (משלי יד י), שורשם 'מרר', אלא שזה המ"ם בחולם וזה המ"ם בקמץ חטף", וכן ר' יונה אבן ג'נאח ב'ספר השורשים' (שורש 'מרה'): "ויכשר להיות ממנו [-מלשון מרירות] עוד, 'מֹרַת רוּחַ' – על משקל 'וּלְדַבֵּר אֶל ה' תּוֹעָה' (ישעיהו לב ו);150 ושב מלעיל, מפני 'רוּחַ'".151 לפירושם, זהו שם עצם – מרירות רוח, וכמו הביטוי הדומה "מַָת נַפְשׁוֹ" – מרירות נפשו.

בדרך זו פירש גם אבן עזרא: "'מֹרַת רוּחַ' – כל אחת מהן [...], ועל דעתי שהוא מגזרת 'מָרָה כַלַּעֲנָה' (משלי ה ד), כטעם מרירות נפש", אך לפירושו משמע שזהו פועל – כל אחת מהן היתה ממררת את רוח יצחק ורבקה.152

בהתאם לפירושים השונים, גם ה"רוּחַ" האמורה כאן מתפרשת במובנים שונים:

לפי הדרך הראשונה, שהמילה "מֹרַת" היא מלשון 'להמרות', הרוח האמורה כאן מתפרשת במובן של רצון,153 וכוונת הכתוב לומר שנשות עשו נהגו באופן שהיה הפך רצונם של יצחק ורבקה, ובכך המרו את רוחם; וכפי שכתב רבינו בחיי: "החפץ והדעת קרא בלשוננו 'רוח', [...] והכתוב יאמר באלו כי כל אחת היתה מסרבת הרוח ליצחק ולרבקה, כלומר שהיתה מכעסת אותם ועושה כל מה שעושה כנגד דעתם".

לעומת זאת, לפי הדרך האחרונה – שהמילה "מֹרַת" היא מלשון 'למרר' – הרוח האמורה כאן מתפרשת במובן של החיות הפנימית, המצב הנפשי של האדם,154 וכוונת הכתוב לומר שנשות עשו נהגו באופן שמירר את רוחם של יצחק ורבקה.

סָתַם הכתוב ולא פירש במה היו נשות עשו "מֹרַת רוּחַ לְיִצְחָק וּלְרִבְקָה". רש"י כתב (על פי בראשית רבה סה ד): "שהיו עובדות עבודה זרה"; וכך בתרגום המיוחס ליונתן, המוסיף רשעה על רשעתן: "וַהֲוָון מְגַחְנָן בְּפוּלְחָנָא נוּכְרָאָה וּמִתְכַּוֵינָן לְאַמְרָדָא בְּעוֹבָדֵיהוֹן בִּישַׁיָא לְיִצְחָק וּלְרִבְקָה [=ותהיינה כורעות בעבודה זרה ומתכוונות למרוד במעשיהן הרעים כלפי יצחק ורבקה]".

בנוסף לכך ניתן גם לפרש שהן היו להם "מֹרַת רוּחַ" בהתנהלותן הכללית ובדברים שבין אדם לחבירו, וכפי שכתב רד"ק: "מכעיסות וממררות רוחן במעשיהן הרעים, וגם בעסקי הבית היו עושות הפך רצונם להכעיסם".

    לפירוש זה, משמעות המילה "מֹרַת" האמורה כאן שווה למשמעות המילה "מַָת" האמורה במשלי, בביטוי "מַָת נַפְשׁוֹ"; ואם כן, עלינו לתת טעם לשני ההבדלים שביניהם – שכן כאן המ' מנוקדת בחולם והמילה נהגית מלעיל, ואילו שם המ' מנוקדת בקמץ קטן והמילה נהגית מלרע. כמובא כאן, ר' יונה אבן ג'נאח כותב כי מפני המילה "רוּחַ" שבאה לאחר המילה "מֹרַת", היא נהגית מלעיל, והיינו על פי כלל 'נסוג אחור' – שכאשר המילה השנייה נהגית מלעיל, והטעמתה מונחת על ההגייה הקודמת לאחרונה (או על ההגייה היחידה, במקרה זה), גם המילה שלפניה נהגית מלעיל; מה שאין כן במשלי, ששם לאחר המילה "מַָת" מופיעה המילה "נַפְשׁוֹ", הנהגית מלרע, ולכן גם המילה "מַָת" נהגית כך. הבדל זה יוצר ממילא את ההבדל השני בעניין הניקוד, שכן מאחר ובפסוקנו ההטעמה היא על האות מ', שבתחילת המילה – לא ניתן לנקדה בקמץ קטן, שנהגה כחצי תנועה, שכן ההטעמה לעולם תהיה על תנועה שלימה; ולפיכך כאן האות מ' מנוקדת בחולם, הנהגה כתנועה גדולה, בעוד שבמשלי הוא מנוקדת בקמץ קטן.

    רד"ק כותב שגם לשון 'המראת רוח' (שבה מתפרשת המילה לפי הדרך הראשונה) היא מלשון מרירות: "או יהיה 'מֹרַת רוּחַ' שורשו 'מרה', מעניין 'מָרוּ וְעִצְּבוּ אֶת רוּחַ קָדְשׁוֹ' (ישעיהו סג י), שהוא גם כן עניין מרירות והכעסה", וכך גם שד"ל: "וגם שורש 'מרה' וכל לשון מרי, עיקרו משורש 'מרר', כי הבלתי שומע בקולנו גורם לנו עִצָּבוֹן [...], וכן בישעיהו סג י: 'וְהֵמָּה מָרוּ וְעִצְּבוּ אֶת רוּחַ קָדְשׁוֹ', 'מָרוּ' כמו 'מֵרְרוּ', כחבירו 'וְעִצְּבוּ'". אולם ניתן גם לפרש שיסוד לשון זו אינו לשון מרירות אלא לשון המרה והחלפה; שכן המראת רוח, או המראת פה, היא עשיית הפכו וחילופו של רצונו או דיבורו של האחר; וכך משמע מדברי רד"ק ב'ספר השורשים' (שורש 'מרה'), שכתב: "'וַתִּהְיֶיןָ מֹרַת רוּחַ' – כל אחת ואחת היתה מורה רוחם, כלומר מחלפת ומשנה רצונם".
