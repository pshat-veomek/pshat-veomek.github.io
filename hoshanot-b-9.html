נוסח הפיוט

רַחֶם נָא קְהַל עֲדַת יְשֻׁרוּן, סְלַח וּמְחַל עֲוֹנָם,

וְהוֹשִׁיעֵנוּ אֱלֹהֵי יִשְׁעֵנוּ.

אָז כְּעֵינֵי עֲבָדִים אֶל יַד אֲדוֹנִים,

בָּאנוּ לְפָנֶיךָ נִדּוֹנִים, וְהוֹשִׁיעֵנוּ אֱלֹהֵי יִשְׁעֵנוּ.

גֵּאֶה אֲדוֹנֵי הָאֲדוֹנִים, נִתְגָּרוּ בָנוּ מְדָנִים,

דָּשׁוּנוּ וּבְעָלוּנוּ זוּלָתְךָ אֲדוֹנִים, וְהוֹשִׁיעֵנוּ אֱלֹהֵי יִשְׁעֵנוּ.

הֵן גַּשְׁנוּ הַיּוֹם בְּתַחֲנוּן, עָדֶיךָ רַחוּם וְחַנּוּן,

וְסִפַּרְנוּ נִפְלְאוֹתֶיךָ בְּשִׁנּוּן, וְהוֹשִׁיעֵנוּ אֱלֹהֵי יִשְׁעֵנוּ.

זָבַת חָלָב וּדְבָשׁ, נָא אַל תִּיבָשׁ,

חַשְׁרַת מַיִם כְּאֵבָהּ תֶּחְבָּשׁ, וְהוֹשִׁיעֵנוּ אֱלֹהֵי יִשְׁעֵנוּ.

טְעֵנוּ בִשְׁמֵנָה, בְּיַד שִׁבְעָה וּשְׁמוֹנָה,

יָשָׁר צַדִּיק אֵל אֱמוּנָה, וְהוֹשִׁיעֵנוּ אֱלֹהֵי יִשְׁעֵנוּ.

כָּרַתָּ בְרִית לָאָרֶץ, עוֹד כָּל יְמֵי הָאָרֶץ,

לְבִלְתִּי פְרָץ בָּהּ פָּרֶץ, וְהוֹשִׁיעֵנוּ אֱלֹהֵי יִשְׁעֵנוּ.

מִתְחַנְּנִים עֲלֵי מָיִם, כַּעֲרָבִים עַל יִבְלֵי מָיִם,

נָא זְכֹר לָמוֹ נִסּוּךְ הַמָּיִם, וְהוֹשִׁיעֵנוּ אֱלֹהֵי יִשְׁעֵנוּ.

שִׂיחִים בְּדֶרֶךְ מַטָּעָתָם, עוֹמְסִים בְּשַׁוְעָתָם,

עֲנֵם בְּקוֹל פְּגִיעָתָם, וְהוֹשִׁיעֵנוּ אֱלֹהֵי יִשְׁעֵנוּ.

פּוֹעֵל יְשׁוּעוֹת, פְּנֵה לְפִלּוּלָם שְׁעוֹת,

צַדְּקֵם אֵל לְמוֹשָׁעוֹת, וְהוֹשִׁיעֵנוּ אֱלֹהֵי יִשְׁעֵנוּ.

קוֹל רִגְשָׁם תִּשַׁע, תִּפְתַּח אֶרֶץ וְיִפְרוּ יֶשַׁע,

רַב לְהוֹשִׁיעַ וְלֹא חָפֵץ רֶשַׁע, וְהוֹשִׁיעֵנוּ אֱלֹהֵי יִשְׁעֵנוּ.

שַׁעֲרֵי שָׁמַיִם פְּתַח, וְאוֹצָרְךָ הַטּוֹב לָנוּ תִפְתַּח,

תּוֹשִׁיעֵנוּ וְרִיב אַל תִּמְתַּח, וְהוֹשִׁיעֵנוּ אֱלֹהֵי יִשְׁעֵנוּ.

על הפיוט

פיוט זה מהווה סיכום לכל תפילותינו ובקשותינו עד כה. אנו מציגים לפני הקב"ה סקירה של סדרי התפילות שערכנו לפניו: בראש השנה וביום הכיפורים, בהם באנו לפניו 'כְּעֵינֵי עֲבָדִים אֶל יַד אֲדוֹנִים'; בימי חג הסוכות, בהם באנו לפניו 'מִתְחַנְּנִים עֲלֵי מַיִם', כשבידינו ארבעת המינים – 'שִׂיחִים בְּדֶרֶךְ מַטָּעָתָם' – שתפקידם לרַצות על המים,412 ו'סִפַּרְנוּ נִפְלְאוֹתֶיךָ בְּשִׁנּוּן' – באמירת ההלל; ואת תפילותנו ביום זה – 'הֵן גַּשְׁנוּ הַיּוֹם בְּתַחֲנוּן, עָדֶיךָ רַחוּם וְחַנּוּן'; אף את ניסוך המים, שנבצר מאיתנו לקיים, אנו מזכירים ומבקשים: 'נָא זְכוֹר לָנוּ נִסּוּךְ הַמַּיִם' – שקיימו אבותינו בזמן המקדש. אנו מבקשים מהקב"ה שיקבל ברצון את תפילותינו ועבודותינו אלה, ומתוך כך – 'וְהוֹשִׁיעֵנוּ אֱלֹהֵי יִשְׁעֵנוּ'. בשונה מרוב הפיוטים דלעיל, שכל אחד מהם התמקד בנושא מסויים, פיוט זה מסכם את הנושאים עליהם ביקשנו: גשמי ברכה, הצלחת החקלאות, ישועה מעול הגויים ומעול הגלויות, קיבוץ נידחי ישראל לארצם, ביאת המשיח, ותחיית המתים.

רַחֶם נָא קְהַל עֲדַת יְשֻׁרוּן – ישראל המכונים 'יְשֻׁרוּן', כמו שכתוב: "וַיְהִי בִישֻׁרוּן מֶלֶךְ" (דברים לג ה), על שם ישרותם;413סְלַח וּמְחַל עֲוֹנָם, וְהוֹשִׁיעֵנוּ אֱלֹהֵי יִשְׁעֵנוּ – האלוהים המושיע אותנו תמיד מצרותינו;414 כלשון הכתוב: "וְאִמְרוּ: הוֹשִׁיעֵנוּ אֱלֹהֵי יִשְׁעֵנוּ, וְקַבְּצֵנוּ וְהַצִּילֵנוּ מִן הַגּוֹיִם, לְהֹדוֹת לְשֵׁם קָדְשֶׁךָ, לְהִשְׁתַּבֵּחַ בִּתְהִלָּתֶךָ" (דברי הימים א' טז לה). פסוק זה נאמר בחתימת השיר שהיו הלוויים אומרים בבית המקדש, ואנו משלבים אותו בכל אחת משורותיו של פיוט זה, המהווה אף הוא חתימה וסיכום לפיוטי ה'הושענות'.

אָז כְּעֵינֵי עֲבָדִים אֶל יַד אֲדוֹנִים, בָּאנוּ לְפָנֶיךָ נִדּוֹנִים – אז, בראש השנה וביום הכיפורים,415 עמדנו לפניך בדין ונקבע כל אשר יארע לנו במשך השנה, ועינינו היו נשואות אליך כעיני עבדים אל יד אדוניהם; על פי לשון הכתוב: "הִנֵּה כְעֵינֵי עֲבָדִים אֶל יַד אֲדוֹנֵיהֶם, כְּעֵינֵי שִׁפְחָה אֶל יַד גְּבִרְתָּהּ, כֵּן עֵינֵינוּ אֶל ה' אֱלֹהֵינוּ עַד שֶׁיְּחָנֵּנוּ" (תהילים קכג ב). לעבד אין דבר משל עצמו, והוא נושא עיניו אל אדונו, המהווה עבורו מקור לכל צרכיו; כמו כן, גם אנו מכירים בכך שכל חיינו וצרכינו תלויים בקב"ה, וכשעמדנו לפניו בדין ביום הכיפורים, היו עינינו נשואות אליו בתפילה להחתם בספר חיים, ברכה, ושלום, ופרנסה טובה, אנחנו וכל בית ישראל; בזכות זאת: וְהוֹשִׁיעֵנוּ אֱלֹהֵי יִשְׁעֵנוּ – בכך שתברכנו בשנה גשומה, ובכך שתצילנו מצרות הגלות, כמפורט בהמשך הפיוט.

    על פי מחזור ויטרי

גֵּאֶה אֲדוֹנֵי הָאֲדוֹנִים – אתה ה', המתגאה (=מתעלה)416 מעל כל הכוחות הקיימים בעולם, ואדון לכל השליטים המושלים בו, ובידך להשפילם, כמו שכתוב: "אָשִׁירָה לַּה' כִּי גָאֹה גָּאָה" (שמות טו א), ותירגם יונתן: "אֲרֵי אִתְגָאֵי עַל גֵוְתָנַיָּא וְגֵאוּתָא דִּילֵהּ, דְמִתְגָּאֵי עַל גֵּיוְותָנַיָּא וּמִתְנַטֵּל עַל מְנַטְלַיָּא, כָּל מַאן דְּמִתְגָּאֵי קֳדָמוֹי – הוּא בְּמֵימְרֵיהּ פָּרַע מִנֵּיהּ" (תרגום חוזר: כי התגאה על גאים, והגאווה שלו, המתגאה על גאים ומתנשא על מתנשאים, כל מי שמתגאה לפניו – הוא במאמרו נפרע ממנו).

נִתְגָּרוּ בָנוּ מְדָנִים – אומות העולם שאנו שרויים ביניהן מתגרות בנו ומעוררות עלינו מדנים (=מריבות), דָּשׁוּנוּ וּבְעָלוּנוּ זוּלָתְךָ אֲדֹנִים – על פי לשון הכתוב: "ה' אֱלֹהֵינוּ, בְּעָלוּנוּ אֲדֹנִים זוּלָתֶךָ, לְבַד בְּךָ נַזְכִּיר שְׁמֶךָ" (ישעיהו כו יג); הגויים המושלים בנו רומסים אותנו ושולטים בנו כבעלים, ומנסים לכופנו לאמונתם, אך אנו שומרים על נאמנותנו לך – "לְבַד בְּךָ נַזְכִּיר שְׁמֶךָ", ולפיכך – גלה את גאוותך ועליונותך עליהם, וְהוֹשִׁיעֵנוּ אֱלֹהֵי יִשְׁעֵנוּ.

זָבַת חָלָב וּדְבָשׁ – ארץ ישראל, כלשון הכתוב: "וַיִּתֶּן לָנוּ אֶת הָאָרֶץ הַזֹּאת אֶרֶץ זָבַת חָלָב וּדְבָשׁ" (דברים כו ט), נָא אַל תִּיבָשׁ – אלא תתברך בגשמי ברכה רבים; חַשְׁרַת מַיִם – ענני המים המקושרים זה בזה,418 כלשון הכתוב: "חַשְׁרַת מַיִם עָבֵי שְׁחָקִים" (שמואל ב' כב יב) כְּאֵבָהּ תֶּחְבָּשׁ – חשרת המים תרפא (מלשון 'תחבושת') את 'כְּאֵבָהּ' של אדמת ארץ ישראל, בכך שתוריד עליה גשמי ברכה רבים, וְהוֹשִׁיעֵנוּ אֱלֹהֵי יִשְׁעֵנוּ.

    כמו: "וְחִשֻּׁקֵיהֶם וְחִשֻּׁרֵיהֶם" (מלכים א' ז לג) – מצודת ציון על הפסוק דלהלן

טְעֵנוּ בִשְׁמֵנָה – נטע אותנו בארץ ישראל, שהיא שמנה ומשובחת, כמאמר הכתוב: "וּבְטוּבְךָ הָרָב אֲשֶׁר נָתַתָּ לָהֶם, וּבְאֶרֶץ הָרְחָבָה וְהַשְּׁמֵנָה אֲשֶׁר נָתַתָּ לִפְנֵיהֶם" (נחמיה ט לה), בְּיַד שִׁבְעָה וּשְׁמוֹנָה – על פי דברי הגמרא:

"וְהָיָה זֶה שָׁלוֹם, אַשּׁוּר כִּי יָבוֹא בְאַרְצֵנוּ וְכִי יִדְרֹךְ בְּאַרְמְנֹתֵינוּ, וַהֲקֵמֹנוּ עָלָיו שִׁבְעָה רֹעִים וּשְׁמֹנָה נְסִיכֵי אָדָם" (מיכה ה ד) – מאן נינהו [=מי הם] שבעה רועים? דוד באמצע, אדם שת ומתושלח – מימינו, אברהם יעקב ומשה – בשמאלו.419 ומאן נינהו שמונה נסיכי אדם? ישי ושאול ושמואל, עמוס וצפניה, חזקיהו ואליהו ומשיח.420

אנו מתפללים להיות נטועים בארץ ישראל תחת ממשלתם של שבעת רועי ישראל ושמונת הנסיכים (=מלכים)421 הללו, שינהיגו את עם ישראל לאחר שיקומו לתחייה לעתיד לבוא, ויצאו בראש העם להילחם במלחמת גוג ומגוג; וכך נאמר במדרש:

"הִנֵּה יוֹם בָּא לַה'" (זכריה יד א) – אותו היום שבא גוג ומגוג על ישראל, מרגיז ארץ ושמיים, ורעש גדול באדמת ישראל, ועולים ללכוד את ירושלים, שנאמר: "וְאָסַפְתִּי אֶת כָּל הַגּוֹיִם אֶל יְרוּשָׁלַםִ לַמִּלְחָמָה" (שם שם ב). באותה שעה, עושה [הקב"ה] מלחמה בגוג ומגוג, ויוצאין עימו שבעה רועים ושמונה נסיכי אדם.422

שַׁעֲרֵי שָׁמַיִם פְּתַח – לקבל את תפילותינו, וכן להוריד לנו גשמים, וְאוֹצָרְךָ הַטּוֹב לָנוּ תִפְתַּח – על פי לשון הכתוב: "יִפְתַּח ה' לְךָ אֶת אוֹצָרוֹ הַטּוֹב אֶת הַשָּׁמַיִם, לָתֵת מְטַר אַרְצְךָ בְּעִתּוֹ וּלְבָרֵךְ אֵת כָּל מַעֲשֵׂה יָדֶךָ" (דברים כח יב). בעניין משמעות המושג 'וְאוֹצָרְךָ הַטּוֹב' נאמר בגמרא:

אמר רפרם בר פפא אמר רב חסדא: מיום שחרב בית המקדש אין הגשמים יורדין מאוצר טוב, שנאמר: "יִפְתַּח ה' לְךָ אֶת אוֹצָרוֹ הַטּוֹב": בזמן שישראל עושין רצונו של מקום וישראל שרויין על אדמתם – גשמים יורדין מאוצר טוב; בזמן שאין ישראל שרויין על אדמתם – אין גשמים יורדין מאוצר טוב.437

משמעות ירידת הגשמים מ'אוצר טוב' היא, שהקב"ה מורידם בעצמו, כביכול, מתוך 'אוצרו הפרטי', בהארת פנים לישראל; דבר זה מתבטא בכך שהם יורדים בשפע ('יִפְתַּח'), בזמנים נוחים (בְּעִתּוֹ'), לא יוצרים נזקים ('וּלְבָרֵךְ אֵת כָּל מַעֲשֵׂה יָדֶךָ'), וכד'. כשהגשמים אינם יורדים מ'אוצר טוב', הרי הם יורדים, כביכול, על ידי שליח, רק כדי לענות על הצורך הטכני בירידת גשמים, אך אינם יורדים באופן של הארת פנים מאת הקב"ה; במצב כזה, הגשמים יורדים בכמות מצומצמת, לאו דווקא בזמנים נוחים, ואף עלולים ליצור נזקים.438 חז"ל לומדים מהמילה 'אַרְצְךָ' שבפסוק, שרק כאשר ישראל שרויים על אדמתם, שהיא "אֶרֶץ אֲשֶׁר ה' אֱלֹהֶיךָ דֹּרֵשׁ אֹתָהּ, תָּמִיד עֵינֵי ה' אֱלֹהֶיךָ בָּהּ" (דברים יא יב), ורק כאשר הם עושים רצונו של מקום, כפי שכתוב לפני כן: "וְהָיָה אִם שָׁמוֹעַ תִּשְׁמַע בְּקוֹל ה' אֱלֹהֶיךָ" (שם כח א) – הם זוכים לירידת גשמים כזו, המבטאת הארת פנים מאת הקב"ה. על פי זה, בקשתנו מתפרשת כבקשה להיות נטועים בארצנו, וכן לקבל סיוע מאת הקב"ה לקיים את רצונו, ומתוך שני אלו נזכה לשפע גשמי ברכה מאוצרו הטוב.

תּוֹשִׁיעֵנוּ וְרִיב אַל תִּמְתַּח – אל תשפטנו במידת הדין, המותחת בינינו 'רִיב', כלומר: ריחוק, אלא קרבנו אליך במידת החסד, וְהוֹשִׁיעֵנוּ אֱלֹהֵי יִשְׁעֵנוּ.
