כְּהוֹשַׁעְתָּ אָדָם יְצִיר כַּפֶּיךָ לְגוֹנְנָה,

בְּשַׁבַּת קֹדֶשׁ הִמְצֵאתוֹ כֹּפֶר וַחֲנִינָה, כֵּן הוֹשַׁע נָא.

כְּהוֹשַׁעְתָּ גּוֹי מְצֻיָּן מְקַוִּים חֹפֶשׁ,

דֵּעָה כִּוְּנוּ לָבוּר שְׁבִיעִי לְנֹפֶשׁ, כֵּן הוֹשַׁע נָא.

כְּהוֹשַׁעְתָּ הָעָם נִהַגְתָּ כַּצֹּאן לְהַנְחוֹת,

וְחֹק שַׂמְתָּ בְּמָרָה עַל מֵי מְנוּחוֹת, כֵּן הוֹשַׁע נָא.

כְּהוֹשַׁעְתָּ זְבוּדֶיךָ בְּמִדְבַּר סִין בַּמַחֲנֶה,

חָכְמוּ וְלָקְטוּ בַּשִּׁשִּׁי לֶחֶם מִשְׁנֶה, כֵּן הוֹשַׁע נָא.

כְּהוֹשַׁעְתָּ טְפוּלֶיךָ הוֹרוּ הֲכָנָה בְּמַדָּעָם,

יִשַּׁר כֹּחָם וְהוֹדָה לָמוֹ רוֹעָם, כֵּן הוֹשַׁע נָא.

כְּהוֹשַׁעְתָּ כֻּלְכְּלוּ בְּעֹנֶג מָן הַמְּשֻׁמָּר,

לֹא הָפַךְ עֵינוֹ וְרֵיחוֹ לֹא נָמָר, כֵּן הוֹשַׁע נָא.

כְּהוֹשַׁעְתָּ מִשְׁפְּטֵי מַשְׂאֹת שַׁבָּת גָּמָרוּ,

נָחוּ וְשָׁבְתוּ, רְשׁוּיוֹת וּתְחוּמִים שָׁמָרוּ, כֵּן הוֹשַׁע נָא.

כְּהוֹשַׁעְתָּ סִינַי הָשְׁמְעוּ בְּדִבּוּר רְבִיעִי,

עִנְיַן זָכוֹר וְשָׁמוֹר לְקַדֵּשׁ שְׁבִיעִי, כֵּן הוֹשַׁע נָא.

כְּהוֹשַׁעְתָּ פֻּקְּדוּ יְרִיחוֹ שֶׁבַע לְהַקֵּף,

צָרוּ עַד רִדְתָּהּ בַּשַּׁבָּת לְתַקֵּף, כֵּן הוֹשַׁע נָא.

כְּהוֹשַׁעְתָּ קֹהֶלֶת וְעַמּוֹ בְּבֵית עוֹלָמִים,

רִצּוּךָ בְּחָגְגָם שִׁבְעָה וְשִׁבְעָה יָמִים, כֵּן הוֹשַׁע נָא.

כְּהוֹשַׁעְתָּ שָׁבִים עוֹלֵי גוֹלָה לְפִדְיוֹם,

תּוֹרָתְךָ בְּקָרְאָם בֶּחָג יוֹם יוֹם, כֵּן הוֹשַׁע נָא.

כְּהוֹשַׁעְתָּ מְשַׂמְּחֶיךָ בְּבִנְיַן שֵׁנִי הַמְחֻדָּשׁ,

נוֹטְלִין לוּלָב כָּל שִׁבְעָה בַּמִּקְדָּשׁ, כֵּן הוֹשַׁע נָא.

כְּהוֹשַׁעְתָּ חִבּוּט עֲרָבָה שַׁבָּת מַדְחִים,

מֻרְבִּיּוֹת מוֹצָא לִיסוֹד מִזְבֵּחַ מַנִּיחִים, כֵּן הוֹשַׁע נָא.

כְּהוֹשַׁעְתָּ בְּרַכּוֹת וַאֲרוּכּוֹת וּגְבוֹהוֹת מְעַלְּסִים,

בִּפְטִירָתָן יֹפִי לְךָ מִזְבֵּחַ מְקַלְּסִים, כֵּן הוֹשַׁע נָא.

כְּהוֹשַׁעְתָּ מוֹדִים וּמְיַחֲלִים וְלֹא מְשַׁנִּים,

כֻּלָּנוּ אָנוּ לְיָהּ וְעֵינֵינוּ לְיָהּ שׁוֹנִים, כֵּן הוֹשַׁע נָא.

כְּהוֹשַׁעְתָּ יֶקֶב מַחֲצָבֶיךָ סוֹבְבִים בְּרַעֲנָנָה,

רוֹנְנִים אֲנִי וָהוּ הוֹשִׁיעָה נָּא, כֵּן הוֹשַׁע נָא.

כְּהוֹשַׁעְתָּ חֵיל זְרִיזִים מְשָׁרְתִים בִּמְנוּחָה,

קָרְבַּן שַׁבָּת כָּפוּל עוֹלָה וּמִנְחָה, כֵּן הוֹשַׁע נָא.

כְּהוֹשַׁעְתָּ לְוִיֶּךָ עַל דּוּכָנָם לְהַרְבָּת,

אוֹמְרִים מִזְמוֹר שִׁיר לְיוֹם הַשַּׁבָּת, כֵּן הוֹשַׁע נָא.

כְּהוֹשַׁעְתָּ נִחוּמֶיךָ בְּמִצְוֹתֶיךָ תָּמִיד יִשְׁתַּעְשְׁעוּן,

וּרְצֵם וְהַחֲלִיצֵם, בְּשׁוּבָה וָנַחַת יִוָּשֵׁעוּן, כֵּן הוֹשַׁע נָא.

כְּהוֹשַׁעְתָּ שְׁבוּת שִׁבְטֵי יַעֲקֹב,

תָּשׁוּב וְתָשִׁיב שְׁבוּת אָהֳלֵי יַעֲקֹב, וְהוֹשִׁיעָה נָּא.

כְּהוֹשַׁעְתָּ שׁוֹמְרֵי מִצְוֹת וְחוֹכֵי יְשׁוּעוֹת,

אֵל לְמוֹשָׁעוֹת, וְהוֹשִׁיעָה נָּא.

על הפיוט

בפיוט זה אנו מזכירים ישועות שונות שהקב"ה פעל במשך ההיסטוריה, שכל אחת מהן קשורה בעניינה של השבת או בעניינו של חג הסוכות. אנו מבקשים ישועות בתחומים שונים, בהתאם לסוגי הישועות שנעשו בעבר, אותן אנו מזכירים. בשונה משאר ה'הושענות' שחוברו על ידי רבי אלעזר הקליר, הושענא זו נתחברה בתקופה מאוחרת יותר, על ידי רבי מנחם בן רבי מכיר,390 ושמו חתום בראשי החרוזים, לאחר סדר הא"ב ('מנחם ברבי מכיר חזק לעד אמן').

כְּהוֹשַׁעְתָּ אָדָם יְצִיר כַּפֶּיךָ לְגוֹנְנָה, בְּשַׁבַּת קֹדֶשׁ הִמְצֵאתוֹ כֹּפֶר וַחֲנִינָה – אדם הראשון היה יציר כפיו של הקב"ה, כמאמר הכתוב: "וַיִּיצֶר ה' אֱלֹהִים אֶת הָאָדָם עָפָר מִן הָאֲדָמָה, וַיִּפַּח בְּאַפָּיו נִשְׁמַת חַיִּים, וַיְהִי הָאָדָם לְנֶפֶשׁ חַיָּה" (בראשית ב ז). לאחר שחטא ואכל מעץ הדעת טוב ורע – היה ראוי למות, כפי שהזהירו הקב"ה: "וּמֵעֵץ הַדַּעַת טוֹב וָרָע לֹא תֹאכַל מִמֶּנּוּ, כִּי בְּיוֹם אֲכָלְךָ מִמֶּנּוּ מוֹת תָּמוּת" (שם שם יז). האכילה היתה ביום השישי בו נברא, ולאחר כמה שעות, ביום השבת, הקב"ה הושיעו מעונש זה 'לְגוֹנְנָה' = להגן עליו מהמיתה, והמציאו (=המציא לידו, נתן לו) כפרה וחנינה, כדברי המדרש:391

"מִזְמוֹר שִׁיר לְיוֹם הַשַּׁבָּת" (תהילים צב) – רבי שמעון אומר: המזמור הזה אדם הראשון אמרו. לפי שבא יום השבת ונעשה סניגור לאדם לפני הקב"ה, ואמר: "ריבון העולם! בששת ימי המעשה לא נהרג איש בעולם, ובי אתה מתחיל?! זו היא קדושתי?! וזו היא ברכתי?!"; וניצל אדם הראשון מדינה של גיהנם בזכות השבת. וכשראה אדם כוחה של שבת – התחיל משורר: "מִזְמוֹר שִׁיר לְיוֹם הַשַּׁבָּת".392

על פי ההבנה הפשוטה בלשון אזהרת הקב"ה לאדם, היה עליו למות ביום בו אכל מהעץ; אך בזכות השבת, הקב"ה פסק את דינו לפי הבנה אחרת המתאפשרת בלשון האזהרה, לפיה כוונת העונש 'מוֹת תָּמוּת' היא שייעשה לבן מוות,393 ובפרט שבמידה מסויימת גם יש לכך משמעות מיידית, שכן כבר מאז התחיל תהליך המיתה, שהרי כל יום שעובר מקרב אותו אל מיתתו.394 ניתן גם להסביר שעונש המיתה מתייחס לירידת נשמתו ממדרגתה העליונה בה היתה קודם החטא, ודבר זה נחשב כמיתה, כלומר: ירידת מדרגת חיי הנשמה, שהם חיי האדם האמיתיים.395 ניתן להסביר את עומק כוונת המדרש, שוודאי לא היתה כאן רק טענה חיצונית של יום השבת, שמפאת קדושתו לא ראוי שהמיתה הראשונה בהיסטוריה תתרחש בו; שאם כן, לא מובן מדוע לא מת אדם הראשון לאחר השבת. כוונת המדרש היא, שיום השבת הוא זמן של התבוננות פנימית – האדם שובת ממלאכתו ומפעולותיו המעשיות, ומפנה את מחשבתו להתבוננות במשמעות הפנימית של חייו ושל העולם. טענת יום השבת שמציג המדרש, היא עצם ההסתכלות על המציאות מתוך מבט של שבת, מבט פנימי ומעמיק. במבט כזה, חטא האדם הוא חיצוני לו, תוצאת השפעת היצר עליו, ואינו חלק ממהותו העצמית. לפיכך – אף על פי שבפועל האדם חטא, אין בכך בכדי להותיר את חייו חסרי ערך ולהצדיק עונש מיתה עבורו, שהרי עצמיותו נותרה כשהיתה, ולא היתה כאן כי אם סטייה חיצונית מהדרך, אותה יש לתקן בתוך מסגרת חיי האדם. גם 'מִזְמוֹר שִׁיר לְיוֹם הַשַּׁבָּת', אותו אמר אדם הראשון באותה שעה, מתבונן על המציאות במבט פנימי:

מַה גָּדְלוּ מַעֲשֶׂיךָ ה', מְאֹד עָמְקוּ מַחְשְׁבֹתֶיךָ. אִישׁ בַּעַר לֹא יֵדָע, וּכְסִיל לֹא יָבִין אֶת זֹאת.

הבער והכסיל, שהם אנשים חומריים,396 רגילים להסתכל על המציאות בצורה שטחית – כשם שהם חשים שייכות אל העולם רק מצד חיצוניותו: החומר. מי שמסתכל כך על המציאות, אינו יכול להבין את פשר מעשי ה' בהנהגת העולם ואת מחשבותיו העומדות מאחוריהם.

בִּפְרֹחַ רְשָׁעִים כְּמוֹ עֵשֶׂב, וַיָּצִיצוּ כָּל פֹּעֲלֵי אָוֶן – לְהִשָּׁמְדָם עֲדֵי עַד. וְאַתָּה מָרוֹם לְעֹלָם, ה'. כִּי הִנֵּה אֹיְבֶיךָ ה', כִּי הִנֵּה אֹיְבֶיךָ יֹאבֵדוּ, יִתְפָּרְדוּ כָּל פֹּעֲלֵי אָוֶן. [...] צַדִּיק כַּתָּמָר יִפְרָח, כְּאֶרֶז בַּלְּבָנוֹן יִשְׂגֶּה. שְׁתוּלִים בְּבֵית ה' – בְּחַצְרוֹת אֱלֹהֵינוּ יַפְרִיחוּ. עוֹד יְנוּבוּן בְּשֵׂיבָה, דְּשֵׁנִים וְרַעֲנַנִּים יִהְיוּ. לְהַגִּיד כִּי יָשָׁר ה', צוּרִי וְלֹא עלתה (קרֵי: עַוְלָתָה) בּוֹ.

על פי המציאות הנראית לעין, הרשעים פורחים ומצליחים, והצדיקים נרדפים – דבר שאינו מובן במבט חיצוני, והבער והכסיל אף יכולים לקבל מכך אישוש לכדאיות אורח חייהם. רק אדם רוחני ופנימי, שמבין שמעל כל הבריאה ניצב הקב"ה ומנהיגה בצדק – "וְאַתָּה מָרוֹם לְעֹלָם, ה'" – יכול לשאת מבטו למרחוק ולהבין את המשמעות האמיתית של המאורעות המתרחשים לעיניו. מי שלא רואה בעולם הזה, החומרי, את חזות הכול, אלא יודע להעריך כראוי את חשיבותו של העולם הבא, הנצחי, שלעומתו העולם הזה הוא כאין וכאפס, כחלום יעוף – מבין ששלוות הרשעים בעולם הזה נועדה כדי לגמול להם על מעט הזכויות שבידם, ועל ידי כך להכחידם מהעולם הנצחי – "לְהִשָּׁמְדָם עֲדֵי עַד". כך גם מפורש בתורה: "וּמְשַׁלֵּם לְשֹׂנְאָיו אֶל פָּנָיו – לְהַאֲבִידוֹ" (דברים ז י). לעומת זאת, הצדיק נענש בעולם הזה על מעט העוונות שבידו, כדי שיזכה לעולם הבא, שהוא הטוב האמיתי והנצחי. הסתכלות כזו, בכוחה "לְהַגִּיד כִּי יָשָׁר ה', צוּרִי וְלֹא עַוְלָתָה בּוֹ" – כל מעשיו הם ביושר ובצדק, ואין בהם כל עוולה. "הַצּוּר תָּמִים פָּעֳלוֹ כִּי כָל דְּרָכָיו מִשְׁפָּט, אֵל אֱמוּנָה וְאֵין עָוֶל, צַדִּיק וְיָשָׁר הוּא" (שם לב ד). בעל המאמר במדרש שהובא, הוא רבי שמעון בר יוחאי; שיטתו של רבי שמעון היא ההסתכלות הפנימית. הדבר בא לידי ביטוי הן בזיקתו לתורת הסוד – פנימיות התורה, הן בשיטתו ההלכתית, בה הוא נוטה לפסוק על פי ראיית המציאות בפנימיותה, כגון לייחס יותר חשיבות לכוונת האדם בעשיית הפעולה מאשר לפעולה עצמה.397 גם בוויכוחו עם רבי יהודה לגבי היחס למעשי הרומאים,398 הוא פוסל אותם אף על פי שבפועל הם יעילים, משום שכוונת הרומאים בעשייתם היא פסולה.399 במאמר זה מגלה רבי שמעון את המבט הפנימי, מבט השבת, על חטאי האדם, הרואה בהם דבר חיצוני בלבד. הביטוי 'יְצִיר כַּפֶּיךָ' בו משתמש הפייטן, מהווה אף הוא טעם לכך שהקב"ה הגן על אדם הראשון מעונש המיתה. כפי שבואר לעיל,400 משמעותה של מידת 'חַנּוּן', שמכוחה העניק הקב"ה לאדם הראשון 'חֲנִינָה' היא, שכשם שכל יוצר חש קשר והזדהות עם יצירתו, כך הקב"ה חש, כביכול, קשר רגשי והזדהות עם האדם – יציר כפיו. הדברים נכונים ביתר שאת לגבי אדם הראשון, שהיה יציר כפיו של הקב"ה באופן ישיר; מכוח מידה זו – העניק לו הקב"ה חנינה. 'כֹּפֶר וַחֲנִינָה' – בביאור שני מושגים אלו והיחס שביניהם ניתן להסביר, ש'כֹּפֶר' = כפרה, הוא המתקת גזר הדין; אך גם לאחריה, הנפש עדיין מוכתמת מהחטא, והיחס בין האדם לבוראו אינו כבתחילה. 'כפרה' היא לשון כיסוי401 – החטא עצמו נשאר תחת הכיסוי, אלא שאינו גורר בעקבותיו עונש. לפיכך דרושה גם ה'חֲנִינָה' – שנובעת ממציאת 'חן' של הנאשם בעיני השופט ומקשר רגשי עמוק ביניהם; היא, באשר איננה מוגבלת לחוקים הפורמליים, יכולה למחוק לגמרי את רושם החטא, ולהשיב את מערכת היחסים שבין האדם לבוראו על כנה. זוהי גם הסיבה שהחוטא צריך להקריב גם קורבן עולה לאחר הקרבת קורבן החטאת; קורבן החטאת נועד למנוע את העונש, וקורבן העולה נועד לפייס, כביכול, את הקב"ה, כפי שכתב רש"י: "'וְהִקְרִיב אֶת אֲשֶׁר לַחַטָּאת רִאשׁוֹנָה' – חטאת קודמת לעולה; למה הדבר דומה? לפרקליט שנכנס לרַצות, ריצה פרקליט – נכנס דורון אחריו".402

אנו מתפללים לישועה ולהגנה על חיינו, כפי שהושיע הקב"ה את אדם הראשון והגן עליו מן המיתה, ואף אנו נזכה ל'כֹּפֶר וַחֲנִינָה' – ביטול תוצאות החטאים ורשמיהם השליליים, ותיקון מערכת היחסים שביננו לבין הקב"ה, ומתוך כך – נהיה זכאים לשנה טובה וברוכה ברוחניות ובגשמיות.

אִם יֵשׁ עָלָיו מַלְאָךְ מֵלִיץ אֶחָד מִנִּי אָלֶף, לְהַגִּיד לְאָדָם יָשְׁרוֹ. וַיְחֻנֶּנּוּ, וַיֹּאמֶר פְּדָעֵהוּ [=פְּדֵהוּ] מֵרֶדֶת שָׁחַת, מָצָאתִי כֹפֶר. [...] לְהָשִׁיב נַפְשׁוֹ מִנִּי שָׁחַת, לֵאוֹר בְּאוֹר הַחַיִּים.403

שייסרם בחוקי המדבר, לסבול הרעב והצמא, לקרוא בהם אל ה', לא דרך תלונה. ומשפטים שיחיו בהם: לאהוב איש את רעהו, ולהתנהג בעצת הזקנים, והצנע לכת באהליהם בעניין הנשים והילדים, ושינהגו שלום עם הבאים במחנה למכור להם דבר, ותוכחות מוסר שלא יהיו כמחנות השוללים אשר יעשו כל תועבה ולא יתבוששו, וכעניין שציוה בתורה: "כִּי תֵצֵא מַחֲנֶה עַל אֹיְבֶיךָ – וְנִשְׁמַרְתָּ מִכֹּל דָּבָר רָע" (דברים כג י).

'עַל מֵי מְנוּחוֹת' – ליד מי מנוחות = מים צלולים,408 והם המים שמצאו במרה, שנמתקו על ידי הנס. טעם הדבר שמשה לימד את ישראל 'חֹק וּמִשְׁפָּט' דווקא 'עַל מֵי מְנוּחוֹת', רמוז בפסוק שלאחר מכן:

וַיֹּאמֶר: אִם שָׁמוֹעַ תִּשְׁמַע לְקוֹל ה' אֱלֹהֶיךָ, וְהַיָּשָׁר בְּעֵינָיו תַּעֲשֶׂה, וְהַאֲזַנְתָּ לְמִצְוֹתָיו, וְשָׁמַרְתָּ כָּל חֻקָּיו, כָּל הַמַּחֲלָה אֲשֶׁר שַׂמְתִּי בְמִצְרַיִם לֹא אָשִׂים עָלֶיךָ, כִּי אֲנִי ה' רֹפְאֶךָ.

משה משתמש בנס המתקת המים להמחיש באופן מציאותי לעם ישראל את הכדאיות שבשמירה על אותו החוק והמשפט שלימדם אז ועל שאר מצוות ה': כשם שהקב"ה המתיק את המים המרים על ידי העץ שהשליך לתוכם משה, על אף שאין סיבה טבעית לכך שהעץ ימתיק את המים – כך גם שמירת מצוותיו מגינה על האדם מפגעים וממחלות, על אף שאין סיבה טבעית הנראית לעין שגורמת לכך. יסוד מוסרי זה, נקל לאדם להבינו בשכלו, אך כדי שהוא יהיה חי בליבו כנתון ריאלי וממשי, קיים צורך להמחיש זאת לעיניו במציאות.

סיכום

שורה זו היא המשך לקודמתה, ועוסקת בלקיטת ישראל כמות מן כפולה ביום השישי: כְּהוֹשַׁעְתָּ טְפוּלֶיךָ – ישראל, הדבקים בך,411הוֹרוּ הֲכָנָה בְּמַדָּעָם – שהורו לעצמם, במדעם = מתוך דעתם ותבונתם האישית, להכין מן ביום השישי עבור השבת, מבלי שמשה הורה להם לעשות כן. כשנשיאי העדה ראו זאת, סברו שהעם עובר בכך על ציווי משה שלא להותיר מהמן ליום המחרת, וכשאמרו זאת למשה, השיב להם שציווי זה אינו חל על יום השבת: יִשַּׁר כֹּחָם וְהוֹדָה לָמוֹ רוֹעָם – משה, רועה ישראל, יישר את כוחם = חיזק את כוחם ואישר את מה שעשו, והודה למו = הסכים עמהם, כמו שכתוב:

וַיֹּאמֶר אֲלֵהֶם: הוּא אֲשֶׁר דִּבֶּר ה': שַׁבָּתוֹן שַׁבַּת קֹדֶשׁ לַה' מָחָר, אֵת אֲשֶׁר תֹּאפוּ אֵפוּ, וְאֵת אֲשֶׁר תְּבַשְּׁלוּ בַּשֵּׁלוּ, וְאֵת כָּל הָעֹדֵף הַנִּיחוּ לָכֶם לְמִשְׁמֶרֶת עַד הַבֹּקֶר. [...] וַיֹּאמֶר מֹשֶׁה: אִכְלֻהוּ הַיּוֹם כִּי שַׁבָּת הַיּוֹם לַה', הַיּוֹם לֹא תִמְצָאֻהוּ בַּשָּׂדֶה. שֵׁשֶׁת יָמִים תִּלְקְטֻהוּ, וּבַיּוֹם הַשְּׁבִיעִי – שַׁבָּת, לֹא יִהְיֶה בּוֹ.412

משה לא הורה לעם ללקוט כמות כפולה, משום שאכן לא נצטווה לומר להם זאת, אלא שהקב"ה תיאר לו שכך יהיה: "וְהָיָה בַּיּוֹם הַשִּׁשִּׁי, וְהֵכִינוּ אֵת אֲשֶׁר יָבִיאוּ, וְהָיָה מִשְׁנֶה עַל אֲשֶׁר יִלְקְטוּ יוֹם יוֹם" (שם שם ה).413 ייתכן שהמטרה בכך היתה שהם יעשו זאת דווקא מדעתם, כדי שבכך ירגישו יותר קשר וחיבור לשבת ולקדושתה; שכן, בכך שאדם מבצע פעולה מדעתו לצורך דבר מסויים, הוא מפתח בקרבו אחריות אישית על הדבר שעבורו הוא פועל, ומחזק את תחושת הקשר אליו. מאחר וראו בני ישראל שירדה כמות כפולה של מן, הבינו מדעתם שאין זה מכבוד השבת לצאת וללקוט בה מן, אלא יש להכין עבורה כבר ביום השישי, ולשם כך הוריד להם הקב"ה כמות כפולה. לאחר מעשה, לימדם משה שלקיטת המן בשבת אסורה מן הדין, מחמת איסור 'הוצאה מרשות לרשות'.

כיבוש יריחו נעשה בשילוב של פעילות מצד ישראל ושל נס מצד הקב"ה; ישראל הקיפו את יריחו במשך שבעה ימים כפי שנצטוו, ונפילת החומה – שאיפשרה בפועל את הכיבוש – היתה בנס. אופן כיבוש זה, שהוא הכיבוש הראשון שנעשה לאחר הכניסה לארץ,424 נועד ללמד על מהותו של תהליך הכיבוש כולו ושל חיי עם ישראל בארצו באופן כללי: אף על פי שמצוות כיבוש הארץ ויישובה, מתממשות על ידי פעולות חומריות – מלחמות, בניין, וחקלאות, יש בהן מימד של 'נס', של קדושה על-טבעית. פעולות אלו אצל ישראל שונות באופן מהותי מאותן פעולות עצמן אצל אומות העולם, שכן אצל ישראל גם מדרגת הטבע והחומר שייכת לקדושה. כיבוש יריחו, שנעשה בנס על-טבעי, מלמד שכל חיי המעשה של עם ישראל בארצו שייכים למדרגה שמעל הטבע, למדרגת הקדושה. זוהי גם הסיבה שהכיבוש נעשה דווקא בשבת, כפי שמציין הפייטן; בדומה לארץ ישראל, המשלבת קדושה וטבע, כך גם השבת: מצד עצמה, שייכת היא למדרגה העל-טבעית – 'מעין עולם הבא', אך היא משפיעה מקדושתה גם על ששת ימי המעשה. זאת, על ידי שהיא מזכירה לאדם "כִּי שֵׁשֶׁת יָמִים עָשָׂה ה' אֶת הַשָּׁמַיִם וְאֶת הָאָרֶץ" (שמות כ יא), ושמשום כך – גם פעולותיו החומריות במשך ימי המעשה, צריכות להעשות מתוך מטרה לרומם את עולם החומר ולקדשו, ולממש את המטרות האלוהיות לשמן הוא נברא, על ידי הכוונתו לייעודיו הנכונים. על פי זה יובן, שאותה 'הוראת שעה' שניתנה אז, להילחם בשבת על אף שכרוכות בכך פעולות שיש בהן איסור מלאכה – אינה מהווה סתירה לשמירת השבת; אדרבה, לשבת ולכיבוש הארץ עניין אחד – העלאת מדרגת החומר ושילובו במארג אחד עם הקדושה, כך שהיא תאיר ותשפיע עליו. בתור נקודת הפתיחה של הפעילות המעשית של ישראל בארץ, המהווה סמל לה, התיר כיבוש יריחו את איסורי השבת, גם בפועל ממש. לרעיון זה, של שילוב הנס עם הפעילות האנושית בכיבוש יריחו, גם פן מוסרי: ישראל הקיפו את יריחו שבעה ימים על פי סדר מפורט שנצטוו עליו, עם ארון הברית, הכוהנים, ותקיעה בשופרות; אך בפועל, מצד הטבע – פעולות אלו לא הועילו מאומה לכיבוש העיר, שנעשה בנס גלוי: חומת יריחו נבלעה באדמה – "וַתִּפֹּל הַחוֹמָה תַּחְתֶּיהָ". עם כניסתם של ישראל לארץ והתחלת חיי המעשה הטבעיים בה (בניגוד לשנים בהן היו במדבר, שאז החיים התבססו על ניסים, כגון המן, השליו, ובאר מִרְיָם), אירוע פתיחה זה של כיבוש יריחו מסמל את המסר שכל הפעולות המעשיות שיפעלו ישראל בכיבוש הארץ, בניינה, ופיתוחה החקלאי, הצלחתן תלויה בקב"ה. כשם שבכיבוש יריחו ישראל עשו את המוטל עליהם, ועל ידי כך אירע הנס ונכבשה העיר, על אף שאין להקפות אלו יכולת מצד הטבע להפיל את החומה – כך גם בפעולות שמצד הטבע נראה שאכן הן הגורמות לתוצאה, יש לזכור כי מצד האמת אין הדבר כן; גם בפעולות אלו הקב"ה הוא זה שיוצר את ההצלחה, ופעילותנו היא רק עשיית ההשתדלות המוטלת עלינו. "וְאָמַרְתָּ בִּלְבָבֶךָ: כֹּחִי וְעֹצֶם יָדִי עָשָׂה לִי אֶת הַחַיִל הַזֶּה; וְזָכַרְתָּ אֶת ה' אֱלֹהֶיךָ כִּי הוּא הַנֹּתֵן לְךָ כֹּחַ לַעֲשׂוֹת חָיִל" (דברים ח יז-יח).425 ייתכן, שזוהי הסיבה לכך שטקס הקפות יריחו התבצע באופן הדומה לעריכת מלחמה: המקיפים היו 'אנשי המלחמה', עם 'החלוץ' ו'המאסף', כשהם מלֻווים בארון הברית ובכוהנים שתוקעים בשופרות; דימיון זה עשוי לחזק אצל ישראל את המסר האמור, המשווה בין הקפות אלו לבין המלחמות הממשיות שנכונו להם בעתיד. על פי זה ניתן להסביר באופן נוסף את הקשר שבין כיבוש יריחו לבין השבת: כמבואר לעיל,426 השבת מבטאת הסתכלות פנימית ואמיתית על המציאות; בהסתכלות כזו רואים את יד ה' בכל פרט מפרטי הטבע, אף באלו שהיא נסתרת בהם. השבת, שהיא 'זִכָּרוֹן לְמַעֲשֶׂה בְרֵאשִׁית', מזכירה לנו כי כל עולם הטבע הסובב אותנו, ונראה לנו כמובן מאליו וכנתון תחת שליטתנו, הוא בריאתו של הקב"ה ונתון לשליטתו הבלעדית. משום כך, השבת היא מקור הברכה וההצלחה של ימות השבוע; כשהאדם נפגש עם תוכנה של השבת, ולומד לייחס את פעולותיו למקורן האמיתי, ממילא הן קרובות יותר לאותו מקור, ויכולות לינוק ממנו יותר ברכה, כוח, והצלחה.

בגמרא428 מבואר שגם ביום הכיפורים, שחל בשבוע הראשון, המשיכו ישראל בשמחת האכילה והשתייה, ולמדו את ההיתר לכך מ'קל וחומר': אם קרבנות הנשיאים שהוקרבו בשנים עשר הימים של חנוכת המשכן, דחו את השבת, על אף שקדושת המשכן היתה זמנית, וקרבנות אלו היו קרבנות יחיד – כל שכן שהקרבת קרבנות ציבור, במקדש, שקדושתו היא נצחית – תדחה את יום הכיפורים, שקדושתו קלה מקדושת השבת, ומותר להקריב את קרבנות חנוכת המקדש ולאוכלם. לאחר מעשה, היה ליבם נוקפם, שכן חששו שמא דווקא הקרבת קרבנות, שהיא 'צורך גבוה' = לצורך הקב"ה, כביכול – דוחה את השבת, אך אכילה, שהיא 'צורך הדיוט' = לצורך האדם – אינה דוחה אף את יום הכיפורים. יצאה בת קול ואמרה להם: "כולכם מזומנים לחיי העולם הבא". 'בת קול' היא קול פנימי ששומע האדם מתוכו, כעין נבואה ורוח הקודש, "והיא מדרגה למטה מן החזון והדיבור האלוהי".429 באותו מעמד, כל ישראל שמעו מתוכם בת קול זו, שאישרה את האופן בו נהגו, והניחה את דעתם. על כך נאמר בפיוט: "כּ¿הוֹשׁ¿ע»תּ קֹהֶלֶת וְעַמּוֹ בְּבֵית עוֹלָמִים" – הושעת אותם מהמצוקה הנפשית שהיו שרויים בה מחמת החשש הנזכר, "רƒצּוּךָ בְּחָגְגָם שִׁבְעָה וְשִׁבְעָה יָמִים" – על אף שחגיגה זו כללה אכילה ביום הכיפורים, היא היתה רצויה לפני הקב"ה. רבינו חננאל פירש שם: "כולכם מזומנים לחיי העולם הבא – שלא נתכוונתם אלא למצווה". ישראל חששו שמא אכילתם לא היתה בכוונה מוחלטת לשם שמיים, ולפיכך דאגו, שהרי דבר שהוא לצורך האדם אינו דוחה את יום הכיפורים; בת הקול הודיעה להם שכוונתם היתה טהורה דיה כדי שאכילתם עצמה תחשב 'צורך גבוה', וממילא לא היה בכך כל איסור או פגם בקדושת יום הכיפורים. מטרתה של התענית ביום הכיפורים היא שהאדם יתנתק מנטיותיו החומריות ויתעלה מעליהן; באותו מעמד של חנוכת המקדש, התעלו ישראל למדרגה כה גבוהה, עד שגם אכילתם היתה טהורה מנטיות חומריות והיתה כולה מתוך קדושה ושמחה של מצוה.430 כשנשמת האדם מתעלה והיא נמצאת קרובה למקורה האלוהי – גם כשהוא בא במגע עם דברים חומריים ופעולות גשמיות, הוא פוגש את הפן הטוב והאלוהי שבהם; ממילא, אותן פעולות נחשבת כ'צורך גבוה', כיוון שגם דרכן האדם מתקרב לקב"ה. זוהי המדרגה אליה הגיעו ישראל בזמן חנוכת המקדש, מדרגה שבה גם תענוג גשמי כאכילה ושתייה היה עשוי לרומם אותם. משום כך, אכילתם היתה שוות ערך לקורבן ויכלה לדחות את יום הכיפורים, שהרי זהו תפקידו של הקורבן – לקרב את האדם לבוראו. הגמרא שם מקשה: מדוע אכן אכלו ישראל את בשר קרבנותיהם ביום הכיפורים, ולא הסתפקו בעצם הקרבתם? ומשיבה: "אין שמחה בלא אכילה ושתייה". אפילו שמחה כה גדולה ועליונה של חנוכת בית המקדש, אינה שלימה ללא אכילה ושתייה; וליתר דיוק: דווקא שמחה עליונה כזו, אינה שלימה מבלי שהיא כוללת בתוכה גם את רגשי הקודש שיכולים להתעורר בקרב האדם מתוך העונג הגופני.

והנה האיש המתקדש בקדושת בוראו, אפילו מעשיו הגשמיים חוזרים להיות ענייני קדושה ממש. וסימנך: אכילת קודשים, שהיא עצמה מצוַת עשה, ואמרו זכרונם לברכה: "כוהנים אוכלים ובעלים מתכפרים"431[...] כללו של דבר: עניין הקדושה הוא שיהיה האדם דבק כל כך באלוהיו עד שבשום מעשה אשר יעשה לא ייפרד ולא יזוז ממנו יתברך, עד שיותר יתעלו הדברים הגשמיים הנחשבים לאחד מתשמישיו, במה שהוא משתמש בהם, ממה שירד הוא מדבקותו ומעלתו בהשתמשו בדברים גשמיים [...] ואז יהיה כמלאך ה' ממש, וכל מעשיו – אפילו השפלים והגשמיים – כקורבנות ועבודות.432

סיכום

לשם הבנת השורה הבאה יש להקדים,435 שמצוַת נטילת ערבה במקדש היא 'הלכה למשה מסיני', ובה מביאים ערבות באורך אחת עשרה אמות, זוקפים אותן סביב המזבח, ולאחר מכן נוטלים אותן ומקיימים את 'חיבוט הערבה'; הראשונים נחלקו מהו ה'חיבוט': לדעת רש"י הוא נענוע הערבות,436 ולדעת הרמב"ם437– חבטת הערבות על הקרקע או על כלי.438 מדין התורה, מצוַת נטילת הערבה נוהגת בכל שבעת ימי החג, אך חכמים תיקנו שבשבת לא יקיימוה, אלא אם כן חל היום השביעי של החג בשבת, שאז מצוַת הערבה נוהגת גם ביום זה. הטעם לכך הוא, שחכמים ביטלו את מצוַת נטילת לולב בשבת, מחשש פן ישכחו ויטלטלוהו ברשות הרבים, ואילו מצוַת ערבה היתה נוהגת בשבת – מצוַת הלולב היתה נראית לעומתה כבעלת חשיבות מועטה, ואנשים היו נוטים לזלזל בה באופן כללי. משום כך, תיקנו חכמים שאף מצוַת הערבה לא תתקיים בשבת, מלבד ביום השביעי של החג, שהרי גם הלולב דוחה את השבת באחד מימי החג – ביום-טוב, ולפיכך לא יהיה בכך עדיפות של מצוַת הערבה על מצוַת הלולב.

כְּהוֹשַׁעְתָּ חִבּוּט עֲרָבָה שַׁבָּת מַדְחִים – כשם שהושעת את ישראל בזמן בית המקדש, שהיו מקיימים בו את חיבוט הערבה, וכשחל היום השביעי בשבת היו 'מַדְחִים' (=דוחים) את השבת בחיבוט הערבה, דהיינו שקיימו מצוה זו על אף השבת, ובאותו מעמד גם היו: מֻרְבִּיּוֹת מוֹצָא לִיסוֹד מִזְבֵּחַ מַנִּיחִים – על פי האמור במשנה: "מצוַת ערבה כיצד? מקום היה למטה מירושלים, ונקרא 'מוצא'. יורדים לשם ומלקטים משם מורביות של ערבה, ובאין וזוקפין אותן בצדי המזבח, וראשיהן כפופין על גבי המזבח".439'מֻרְבִּיּוֹת' הן ענפים ארוכים,440 או עצים שלמים שנעקרו מהשורש.441 בגמרא נאמר שאת הערבות היו מניחים על גבי יסוד המזבח, שהוא בליטה סביב המזבח סמוך לרצפה, בגובה אמה ורוחב אמה; זאת, כדי שהערבות תהיינה גבוהות יותר וראשיהן יוכלו להיות כפופים על גבי המזבח בגובה אמה מעליו, על מנת לנאות את המראה. אנו מבקשים מהקב"ה שכשם שהושיע את ישראל וענה לתפילות שנשאו בעת שקיימו את מצוַת הערבה – כך יושיע אף אותנו, למרות שאיננו יכולים לקיים מצוה זו, שעל ידה מושפעים לישראל ברכה וישועה. זוהי גם משמעות הבקשה: "כְּהוֹשַׁעְתָּ ... כֵּן הוֹשַׁע נָא" שבשורות הבאות, המתייחסות אף הן לעובדים במקדש, שעל ידי עבודתם השפיעו ברכה וישועה לישראל.

הערבה מכונה כאן 'רַעֲנָנָה', לא רק מפני שאכן ערבה יבשה פסולה למצוַה, אלא גם משום שהרעננות הוא מאפיין מהותי של הערבות, של ארבעת המינים, ושל חג הסוכות בכלל. בימי הדין שבראשית השנה, האדם שרוי במצב של יראה וכובד ראש, הגורמים במידה מסויימת לצמצום נפשי ולהתכנסות פנימית; זהו בסיס הכרחי לשנה העומדת בפתח ולהתנהלות האדם בה מתוך ביקורת עצמית ותשומת לב למצבו הרוחני. אך כדי לאזן את ההרגשה הנפשית, בסוכות אנו מדגישים את הצד ההפוך, שאף הוא בסיס הכרחי לעבודת ה' במשך השנה כולה: התרחבות הנפש מתוך שמחה, רעננות, וחיבור אל הטבע. הדבר מתבטא בכך שאנו יוצאים מהבית – המבטא את הקביעות והצמצום, אל החוץ והמרחב, דרים שם בסוכה המסוככת בגידולי קרקע טבעיים, ומתפללים בבית הכנסת – או, בזמן הבית – בבית המקדש, כשבידינו צמחים רעננים. וכך נאמר במדרש:

"תִּכָּתֶב זֹאת לְדוֹר אַחֲרוֹן, וְעַם נִבְרָא יְהַלֶּל יָהּ" (תהילים קב יט) – אלו הדורות שחייבין במעשיהם הרעים, ובאין ועושין תשובה ומתפללין לפניך בראש השנה וביום הכיפורים. ולפי שהם מחדשים את מעשיהם – הקב"ה בורא אותם ברייה חדשה. ומה להם לעשות? יש להם ליטול אתרוגיהן בידיהם, ולולביהן וערבי נחל והדסים, להלל אותך; הוֵי אומר: "וְעַם נִבְרָא יְהַלֶּל יָהּ".453

"וּלְקַחְתֶּם לָכֶם בַּיּוֹם הָרִאשׁוֹן" – הדא הוא דכתיב: "יַעֲלֹז שָׂדַי וְכָל אֲשֶׁר בּוֹ" (תהילים צו יב); "יַעֲלֹז שָׂדַי" – זה העולם, [...] "וְכָל אֲשֶׁר בּוֹ" – אלו הבריות [...]. "אָז יְרַנְּנוּ עֲצֵי הַיָּעַר" (דברי הימים א' טז לג) – אמר רבי אחא: [...] "עֲצֵי הַיָּעַר" – אלו אילנות שעושין פירות, ו"כָּל עֲצֵי יָעַר" (תהילים שם) – אלו אילנות שאינן עושין פירות. לפני מי? "לִפְנֵי ה'", למה? "כִּי בָא" – בראש השנה וביום הכיפורים, מה לעשות? "יִשְׁפֹּט תֵּבֵל בְּצֶדֶק וְעַמִּים בְּמֵישָׁרִים" (שם צח ט).454

לאחר ימי הדין, בהם ישראל זוכו ונוקו מעוונותיהם, הם כביכול נבראים מחדש, ומתחדשים ומתרעננים יחד עם העולם כולו; ההתחדשות היא החזרה למצב הטבעי, שבו האדם נקי וזך, והחטאים אינם מהווים מחיצה בינו לבין בוראו. מתוך מצב זה, עם ישראל עושה את הפעולה הטבעית לו ביותר: מהלל את הקב"ה. על ידי נטילת ארבעת המינים, מצרף הוא להילולו גם את עולם הטבע, ובנענועו בהם – כאילו נופח בתוכם נשמת חיים, עד שהם מתפללים ומהללים עימו; כך הבריאה המתחדשת כולה, על שני רבדיה: האדם והצומח, אומרת שירה לפני בוראה בטבעיות וברעננות, מתוך שמחת התחדשותה. רעיון זה עולה אף מבין השיטין של לשון הגמרא, בה אגודת ארבעת המינים מכונה 'הושענא', כביכול בעצם מציאותה ביד המתפלל הרי היא אומרת "הוֹשַׁע נָא". גם מדברי התלמוד הירושלמי,455 בו פְּסוּל 'יבש' בארבעת המינים נלמד מהפסוק: "לֹא הַמֵּתִים יְהַלְלוּ יָהּ" (תהילים קטו יז) – עולה מבט על ארבעת המינים כישויות עצמאיות המהללות את הקב"ה יחד עם האדם. רעיון זה מתבטא גם בדברי המדרש,456 המשווה כל אחד מארבעת המינים לאחד מאברי האדם: לולב – לעמוד השדרה, אתרוג – ללב, הדס – לעין, וערבה – לפה. אגודת ארבעת המינים היא כעצם חי העומד לפני ה', רואה את המציאות עליה יש להתפלל ולהודות, ומודה ומתפלל לקב"ה בפיו, מתוך קירות ליבו. היא הנציג של עולם הצומח והטבע בבית המקדש ובבית הכנסת, ועל ידי נטילתה – האדם מתאחד עם עולם הטבע בעמידתו לפני ה'.

על האדם לראות מעין הזדהות עם המינים שבידו, לקלוט מהם תחושת רעננות וטבעיות, ובכך להרחיב את כוחות נפשו ולהתפלל מתוך שמחה טבעית.457 הטבע הוא תוצר ישיר של הקב"ה, וכשהאדם נחשף לטבע ומתחבר אליו, ממילא הוא קרוב יותר בהרגשתו לקב"ה – מייצר הטבע, והרגש הטבעי והעמוק ביותר שבנפשו – המשיכה לקרבת אלוהים, עולה אל פני השטח ופועל בקרבו את פעולתו. מזמור סה בספר תהילים מתאר שני שלבים אלו. בתחילתו: את התפילה, היראה, וכפרת העוונות:

שֹׁמֵעַ תְּפִלָּה, עָדֶיךָ כָּל בָּשָׂר יָבֹאוּ. דִּבְרֵי עֲוֹנֹת גָּבְרוּ מֶנִּי, פְּשָׁעֵינוּ אַתָּה תְכַפְּרֵם.[...] נוֹרָאוֹת בְּצֶדֶק תַּעֲנֵנוּ אֱלֹהֵי יִשְׁעֵנוּ, מִבְטָח כָּל קַצְוֵי אֶרֶץ וְיָם רְחֹקִים. מֵכִין הָרִים בְּכֹחוֹ, נֶאְזָר בִּגְבוּרָה. מַשְׁבִּיחַ שְׁאוֹן יַמִּים, שְׁאוֹן גַּלֵּיהֶם, וַהֲמוֹן לְאֻמִּים, וַיִּירְאוּ יֹשְׁבֵי קְצָוֹת מֵאוֹתֹתֶיךָ.458

ולאחר מכן, את שלב פריחת הטבע ושירתו:

פָּקַדְתָּ הָאָרֶץ וַתְּשֹׁקְקֶהָ, רַבַּת תַּעְשְׁרֶנָּה, פֶּלֶג אֱלֹהִים מָלֵא מָיִם, תָּכִין דְּגָנָם כִּי כֵן תְּכִינֶהָ. תְּלָמֶיהָ רַוֵּה, נַחֵת גְּדוּדֶיהָ, בִּרְבִיבִים תְּמֹגְגֶנָּה, צִמְחָהּ תְּבָרֵךְ. עִטַּרְתָּ שְׁנַת טוֹבָתֶךָ, וּמַעְגָּלֶיךָ יִרְעֲפוּן דָּשֶׁן. יִרְעֲפוּ נְאוֹת מִדְבָּר, וְגִיל גְּבָעוֹת תַּחְגֹּרְנָה. לָבְשׁוּ כָרִים הַצֹּאן, וַעֲמָקִים יַעַטְפוּ בָר, יִתְרוֹעֲעוּ אַף יָשִׁירוּ.459

הנה במוסף השבת אין בו קורבן חטאת כלל, כי אם שני כבשים לכפל התמיד, להיות בו לחם משנה למזבח; להורות על מעלת השבת בברכה וקידוש על כל שאר הימים.462

עניינו של מוסף השבת אינו להוסיף, אלא להכפיל; כשם שבשבת יש לאדם 'נשמה יתירה', ובסעודות השבת בוצעים על לחם משנה, כך גם בקורבן המוסף, המבטא את עניינו של היום, באות לידי ביטוי ברכתו וקדושתו המיוחדות של יום השבת, באמצעות ההכפלה.

    שורה זו היתה צריכה להיות מנוסחת כך: "כְּהוֹשַׁעְתָּ נִחוּמֶיךָ בְּמִצְוֹתֶיךָ תָּמִיד יִשְׁתַּעַשְׁעוּן - כֵּן הוֹשַׁע נָא, וּרְצֵם וְהַחֲלִיצֵם, בְּשׁוּבָה וָנַחַת יִוָּשֵׁעוּן", שהרי 'וּרְצֵם וְהַחֲלִיצֵם' היא בקשה נוספת ואינה המשך ל'כְּהוֹשַׁעְתָּ' כמו בשאר שורות הפיוט; אך כדי לא לשנות שורה זו מחברותיה, בחר הפייטן לנסחה כבפנים.

    'מצודת ציון' על אתר

כְּהוֹשַׁעְתָּ שְׁבוּת שִׁבְטֵי יַעֲקֹב, תָּשׁוּב וְתָשִׁיב שְׁבוּת אָהֳלֵי יַעֲקֹב וְהוֹשִׁיעָה נָּא.

כְּהוֹשַׁעְתָּ שׁוֹמְרֵי מִצְוֹת וְחוֹכֵי יְשׁוּעוֹת אֵל לְמוֹשָׁעוֹת, וְהוֹשִׁיעָה נָּא – שורות אלו אינן חלק מהפיוט: השורה הראשונה לקוחה מסוף הפיוט 'כְּהוֹשַׁעְתָּ אֵלִים', הנאמר בכל יום לאחר הפיוט המיוחד לו, ובשבת הוא מוחלף בפיוט זה, שנתחבר בתקופה מאוחרת יותר. אך כדי לא לוותר לגמרי על אותו פיוט קדום – שובצה כאן השורה החותמת אותו. השורה השנייה לקוחה מהפיוט 'כְּהוֹשַׁעְתָּ אָב הֲמוֹן' (אמירתו אינה נהוגה כיום ברוב הקהילות), שחובר על ידי רבי שמואל החסיד בן רבי קלונימוס הזקן,467 ושמו רמוז בה: "שׁוֹמְרֵי ƒמצְוֹת ו¿חוֹכֵי יְשׁוּעוֹת ≈אל לְמוֹשָׁעוֹת"; היא נאמרת גם בשאר ימי החג לאחר הפיוט 'כְּהוֹשַׁעְתָּ אֵלִים' – ראה שם ביאור שתי שורות אלו, וכן ביאור הפסוקים הנאמרים לאחריהן (עמ' 163-161).

סיכום

בפיוט זה ביקשנו על ישועות בתחומי חיינו השונים, בדומה לישועות שפעל הקב"ה בעבר: אדם הראשון נושע בזכות השבת מהמיתה שהיתה ראויה לבוא עליו בעקבות האכילה מעץ הדעת טוב ורע, ואף אנו מתפללים להגנה על חיינו מפגעי הזמן, לסליחה, ולתיקון מערכת היחסים שביננו לבין הקב"ה. עם ישראל נושע ממצרים, ואף אנו מתפללים להיות עם חופשי בארצנו, ללא שיעבוד לאומות העולם ולרצונותיהן. עם ישראל הונהג כצאן על ידי הקב"ה ביציאתו ממצרים, נושע מפגעי המדבר 'הַגָּדוֹל וְהַנּוֹרָא הַהוּא' (דברים א יט), ומאיום הרעב, על ידי ירידת המן – ואף אנו מתפללים להינצל מכל פגע ולסיפוק צרכינו הכלכליים ברווח ובנחת. יהושע ועמו זכו לישועה ולניצחון בכיבוש יריחו, שהיתה "סֹגֶרֶת וּמְסֻגֶּרֶת מִפְּנֵי בְּנֵי יִשְׂרָאֵל, אֵין יוֹצֵא וְאֵין בָּא" (יהושע ו א), ולא ניתן היה לכובשה בדרך הטבע, ואף אנו מתפללים לישועה ולניצחון על אויבינו. קהלת ועמו נושעו בחנוכת בית המקדש מהמצוקה הנפשית שנגרמה מאותו חשש פן לא נהגו כשורה בכך שלא התענו ביום הכיפורים, ואף אנו מתפללים גם לישועה מסוג זה – לרוגע נפשי ותחושת ביטחון בדרכנו בעבודת ה'. עולי הגולה נושעו ונפדו מעול הגלות, חזרו לשמירת התורה והמצוות, ובנו את 'בִּנְיַן שֵׁנִי הַמְחֻדָּשׁ' – ואף אנו מתפללים להיוושע מעול הגויים, לראות בתשובתו השלימה של עם ישראל כולו, ולזכות במהרה לבניין הבית השלישי. המאורעות בפיוט מסודרים על פי סדרם הכרונולוגי: אדם הראשון, שיעבוד מצרים והיציאה ממנה, החנייה במרה, ירידת המן, ומעמד הר סיני. לאחר הכניסה לארץ: כיבוש יריחו, חנוכת בית ראשון, השיבה מגלות בבל, בנין בית שני, והעבודה בו. בכמה שורות בפיוט מופיע עניין הסגולה הפנימית שבישראל ובאדם: "כְּהוֹשַׁעְתָּ אָדָם יְצִיר כַּפֶּיךָ לְגוֹנְנָה, בְּשַׁבַּת קֹדֶשׁ הִמְצֵאתוֹ כּוֹפֶר וַחֲנִינָה" – מחמת ראיית נקודת הטוב שבתוככי האדם, שאינה נפגמת מהחטא; "דֵּעָה כִּוְּנוּ לָבוּר שְׁבִיעִי לְנֹפֶשׁ" – מכוח הסגולה הפנימית שבהם, שזיהתה את קדושתו של יום השבת, וכך בררו לעצמם באופן תת-מודע את השבת ליום המנוחה השבועי שאישר להם פרעה; "חָכְמוּ וְלָקְטוּ בַּשִּׁשִּׁי לֶחֶם מִשְׁנֶה ... הוֹרוּ הֲכָנָה בְּמַדָּעָם, יִשַּׁר כֹּחָם וְהוֹדָה לָמוֹ רוֹעָם" – גם בזה הצליחו ישראל לכוון מעצמם לרצון ה', וליבם הורה להם כי עבור השבת יש להכין מן מראש; "קֹהֶלֶת וְעַמּוֹ בְּבֵית עוֹלָמִים, רִצּוּךָ בְּחָגְגָם שִׁבְעָה וְשִׁבְעָה יָמִים" – גם אז, עם ישראל כולו הבין מדעתו, מנקודת האמת שבפנימיותו, שהאופן הנכון להעביר את אותו יום הכיפורים הוא דווקא על ידי שמחת אכילה ושתייה. מלבד דרשת ה'קל וחומר' ההלכתית שהתירה את הדבר, ההיתר התבסס, כפי שנתבאר לעיל, על ההנחה ש"אין שמחה בלא אכילה ושתייה" – הנחה הנובעת מהכרה פנימית בערכה העליון של אותה שמחה מיוחדת. מכוח סגולה פנימית זו של ישראל, העומדת תמיד בטהרתה בקרבו של כל יהודי באשר הוא, אנו מתפללים לישועה ולגאולה השלימה; אף אם איננו ראויים לכך מצד מעשינו – הרי שסגולתנו הפנימית כעם ה' מצדיקה זאת. סגנון הבקשה: "כְּהוֹשַׁעְתָּ ... כֵּן הוֹשַׁע נָא" אין עניינו, כמובן, 'להזכיר' לקב"ה שהוא יכול לעשות זאת, אלא לקבוע בתודעת המתפלל – קודם שהוא ניגש לתוכן הבקשה עצמה – שאכן הקב"ה הוא בעל היכולת הבלעדית לקיים את מבוקשו.468 על מנת לחזק הכרה זו, ולהמירה מאמונה שכלית גרידא להרגשה ברורה ומוחשית, אנו מזכירים ומציירים בדעתנו מאורעות שאירעו לאורך ההיסטוריה, ואת הישועות השונות שפעל בהם הקב"ה.
