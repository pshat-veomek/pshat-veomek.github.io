א וְיוֹסֵף הוּרַד מִצְרָיְמָה וַיִּקְנֵהוּ פּוֹטִיפַר סְרִיס פַּרְעֹה שַׂר הַטַּבָּחִים אִישׁ מִצְרִי מִיַּד הַיִּשְׁמְעֵאלִים אֲשֶׁר הוֹרִדֻהוּ שָׁמָּה. ב וַיְהִי יְהֹוָה אֶת יוֹסֵף וַיְהִי אִישׁ מַצְלִיחַ וַיְהִי בְּבֵית אֲדֹנָיו הַמִּצְרִי. ג וַיַּרְא אֲדֹנָיו כִּי יְהֹוָה אִתּוֹ וְכֹל אֲשֶׁר הוּא עֹשֶׂה יְהֹוָה מַצְלִיחַ בְּיָדוֹ. ד וַיִּמְצָא יוֹסֵף חֵן בְּעֵינָיו וַיְשָׁרֶת אֹתוֹ וַיַּפְקִדֵהוּ עַל בֵּיתוֹ וְכָל יֶשׁ לוֹ נָתַן בְּיָדוֹ. ה וַיְהִי מֵאָז הִפְקִיד אֹתוֹ בְּבֵיתוֹ וְעַל כָּל אֲשֶׁר יֶשׁ לוֹ וַיְבָרֶךְ יְהֹוָה אֶת בֵּית הַמִּצְרִי בִּגְלַל יוֹסֵף וַיְהִי בִּרְכַּת יְהֹוָה בְּכָל אֲשֶׁר יֶשׁ לוֹ בַּבַּיִת וּבַשָּׂדֶה. ו וַיַּעֲזֹב כָּל אֲשֶׁר לוֹ בְּיַד יוֹסֵף וְלֹא יָדַע אִתּוֹ מְאוּמָה כִּי אִם הַלֶּחֶם אֲשֶׁר הוּא אוֹכֵל וַיְהִי יוֹסֵף יְפֵה תֹאַר וִיפֵה מַרְאֶה. ז וַיְהִי אַחַר הַדְּבָרִים הָאֵלֶּה וַתִּשָּׂא אֵשֶׁת אֲדֹנָיו אֶת עֵינֶיהָ אֶל יוֹסֵף וַתֹּאמֶר שִׁכְבָה עִמִּי. ח וַיְמָאֵן וַיֹּאמֶר אֶל אֵשֶׁת אֲדֹנָיו הֵן אֲדֹנִי לֹא יָדַע אִתִּי מַה בַּבָּיִת וְכֹל אֲשֶׁר יֶשׁ לוֹ נָתַן בְּיָדִי. ט אֵינֶנּוּ גָדוֹל בַּבַּיִת הַזֶּה מִמֶּנִּי וְלֹא חָשַׂךְ מִמֶּנִּי מְאוּמָה כִּי אִם אוֹתָךְ בַּאֲשֶׁר אַתְּ אִשְׁתּוֹ וְאֵיךְ אֶעֱשֶׂה הָרָעָה הַגְּדֹלָה הַזֹּאת וְחָטָאתִי לֵאלֹהִים. י וַיְהִי כְּדַבְּרָהּ אֶל יוֹסֵף יוֹם יוֹם וְלֹא שָׁמַע אֵלֶיהָ לִשְׁכַּב אֶצְלָהּ לִהְיוֹת עִמָּהּ. יא וַיְהִי כְּהַיּוֹם הַזֶּה וַיָּבֹא הַבַּיְתָה לַעֲשׂוֹת מְלַאכְתּוֹ וְאֵין אִישׁ מֵאַנְשֵׁי הַבַּיִת שָׁם בַּבָּיִת. יב וַתִּתְפְּשֵׂהוּ בְּבִגְדוֹ לֵאמֹר שִׁכְבָה עִמִּי וַיַּעֲזֹב בִּגְדוֹ בְּיָדָהּ וַיָּנָס וַיֵּצֵא הַחוּצָה. יג וַיְהִי כִּרְאוֹתָהּ כִּי עָזַב בִּגְדוֹ בְּיָדָהּ וַיָּנָס הַחוּצָה. יד וַתִּקְרָא לְאַנְשֵׁי בֵיתָהּ וַתֹּאמֶר לָהֶם לֵאמֹר רְאוּ הֵבִיא לָנוּ אִישׁ עִבְרִי לְצַחֶק בָּנוּ בָּא אֵלַי לִשְׁכַּב עִמִּי וָאֶקְרָא בְּקוֹל גָּדוֹל. טו וַיְהִי כְשָׁמְעוֹ כִּי הֲרִימֹתִי קוֹלִי וָאֶקְרָא וַיַּעֲזֹב בִּגְדוֹ אֶצְלִי וַיָּנָס וַיֵּצֵא הַחוּצָה. טז וַתַּנַּח בִּגְדוֹ אֶצְלָהּ עַד בּוֹא אֲדֹנָיו אֶל בֵּיתוֹ. יז וַתְּדַבֵּר אֵלָיו כַּדְּבָרִים הָאֵלֶּה לֵאמֹר בָּא אֵלַי הָעֶבֶד הָעִבְרִי אֲשֶׁר הֵבֵאתָ לָּנוּ לְצַחֶק בִּי. יח וַיְהִי כַּהֲרִימִי קוֹלִי וָאֶקְרָא וַיַּעֲזֹב בִּגְדוֹ אֶצְלִי וַיָּנָס הַחוּצָה. יט וַיְהִי כִשְׁמֹעַ אֲדֹנָיו אֶת דִּבְרֵי אִשְׁתּוֹ אֲשֶׁר דִּבְּרָה אֵלָיו לֵאמֹר כַּדְּבָרִים הָאֵלֶּה עָשָׂה לִי עַבְדֶּךָ וַיִּחַר אַפּוֹ. כ וַיִּקַּח אֲדֹנֵי יוֹסֵף אֹתוֹ וַיִּתְּנֵהוּ אֶל בֵּית הַסֹּהַר מְקוֹם אֲשֶׁר (אסורי) אֲסִירֵי הַמֶּלֶךְ אֲסוּרִים וַיְהִי שָׁם בְּבֵית הַסֹּהַר. כא וַיְהִי יְהֹוָה אֶת יוֹסֵף וַיֵּט אֵלָיו חָסֶד וַיִּתֵּן חִנּוֹ בְּעֵינֵי שַׂר בֵּית הַסֹּהַר. כב וַיִּתֵּן שַׂר בֵּית הַסֹּהַר בְּיַד יוֹסֵף אֵת כָּל הָאֲסִירִם אֲשֶׁר בְּבֵית הַסֹּהַר וְאֵת כָּל אֲשֶׁר עֹשִׂים שָׁם הוּא הָיָה עֹשֶׂה. כג אֵין שַׂר בֵּית הַסֹּהַר רֹאֶה אֶת כָּל מְאוּמָה בְּיָדוֹ בַּאֲשֶׁר יְהֹוָה אִתּוֹ וַאֲשֶׁר הוּא עֹשֶׂה יְהֹוָה מַצְלִיחַ.

פרק לט · פסוק א

וְיוֹסֵף הוּרַד מִצְרָיְמָה, וַיִּקְנֵהוּ פּוֹטִיפַר סְרִיס פַּרְעֹה שַׂר הַטַּבָּחִים אִישׁ מִצְרִי מִיַּד הַיִּשְׁמְעֵאלִים אֲשֶׁר הוֹרִדֻהוּ שָׁמָּה.

בפרקנו חוזר הכתוב לסיפורו של יוסף, מאותה נקודת הזמן שבה הפסיק בו ועבר לספר על קורותיו של יהודה שהתרחשו במקביל; "לאחר שסיפר תולדותיו של יהודה, חזר על יוסף וגילגוליו איך אירע לו במצרים" ('בכור שור'). פסוקנו חוזר על מה שנאמר לעיל בסוף פרק לז: "וְהַמְּדָנִים מָכְרוּ אֹתוֹ אֶל מִצְרָיִם לְפוֹטִיפַר סְרִיס פַּרְעֹה שַׂר הַטַּבָּחִים", "כאדם האומר: נחזור על הראשונות"211 לשם יצירת רצף סיפורי, ולפיכך הוא פותח בלשון עבר רגיל: "וְיוֹסֵף הוּרַד מִצְרָיְמָה" (ולא בו' ההיפוך "וַיּוּרַד יוסף", כדרך המקרא), כלומר: "וְ[כאמור לעיל], יוֹסֵף הוּרַד מִצְרָיְמָה [...]".

רש"י כתב: "חוזר לעניין ראשון, אלא שהפסיק בו כדי לסמוך ירידתו של יהודה למכירתו של יוסף, לומר לך שבשבילו הורידוהו מגדולתו", וכמבואר לעיל (לח א). רש"י ציין את הקשר הנסיבתי שלדבריו הכתוב מלמד באמצעות סדר הפרשיות, אולם גם בלא להניח שהסמיכות נועדה ללמד מסר מיוחד, ניתן להסבירה בפשטות בכך שזה אכן היה סדרן הכרונולוגי של ההתרחשויות – ומן ההכרח היה להפסיק בתוך סיפור קורות יוסף במצרים בסיפורו של יהודה, שכן הם התרחשו במקביל.

"וַיִּקְנֵהוּ פּוֹטִיפַר סְרִיס פַּרְעֹה" – "בהשגחת השם עליו נמכר לאיש שר וגדול, לא להדיוט העובד בו עבודת עבד" (ר' נפתלי הרץ ויזל), ודבר זה איפשר את המסופר בהמשך הפרק, שיוסף זכה שם לשררה: "וַיִּמְצָא יוֹסֵף חֵן בְּעֵינָיו וַיְשָׁרֶת אֹתוֹ וַיַּפְקִדֵהוּ עַל בֵּיתוֹ וְכָל יֶשׁ לוֹ נָתַן בְּיָדוֹ" (פס' ד).

"שַׂר הַטַּבָּחִים" – בעניין משמעות תואר זה, ראה מה שהתבאר לעיל (לז לו).

"אִישׁ מִצְרִי" – כתב רש"ר הירש:

מודגש כאן שוב ושוב שפוטיפר היה "מצרי" (פס' א, ב, ה). ונשאלת השאלה: מה הטעם לכך? בוודאי שנושא משרה בחצר המלכות המצרית יהיה מצרי!

אולם יודעים אנו על הניגוד שהיה קיים בין המצרים לבין כל שאר האומות, ובמיוחד ידועה לנו הגאווה בה הביטו המצרים על השבטים הנוודים. ניגוד גדול עוד יותר היה קיים בין מוסריותם ואורח חייהם של המצרים לבין אלה של "הנער העברי", כפי שמכונה יוסף להלן (מא יב). תיבת "מצרי" בפרשה הזאת מדגישה ניגוד זה, ומעמידה בפנינו את הפיתויים שיוסף היה חייב בוודאי להתגבר עליהם, כדי לשמור על טהרתו בתוך בית מצרי תחת אדון מצרי, כך שה' יראה אותו כְּרָאוּי לקרבתו.

תיבת "מצרי" גם רומזת לכשרונות ולכוחות הנפש הגדולים, שעבד צעיר – בן שבט אסייתי נווד שנקנה מידי סוחרים אסייתיים נוודים – היה צריך להם כדי למצוא חן בעיני אדם כפוטיפר, שהיה שר למלך, ובנוסף היה "מצרי".

"מִיַּד הַיִּשְׁמְעֵאלִים אֲשֶׁר הוֹרִדֻהוּ שָׁמָּה" – בסוף פרק לז נאמר: "וְהַמְּדָנִים מָכְרוּ אֹתוֹ אֶל מִצְרָיִם לְפוֹטִיפַר סְרִיס פַּרְעֹה שַׂר הַטַּבָּחִים", וראה לעיל (לז כח) ביאור עניין זה בהרחבה.

פרק לט · פסוק ב

וַיְהִי ה' אֶת יוֹסֵף וַיְהִי אִישׁ מַצְלִיחַ, וַיְהִי בְּבֵית אֲדֹנָיו הַמִּצְרִי.

"וַיְהִי ה' אֶת יוֹסֵף" – ה' היה עם יוסף, ודבר זה התבטא בפועל בכך שהוא היה "אִישׁ מַצְלִיחַ".

"וַיְהִי אִישׁ מַצְלִיחַ" – הכתוב מוסיף כאן ליוסף את הכינוי "אִישׁ" מלבד עצם תיאורו כ"מַצְלִיחַ". כינוי זה עשוי לרמוז לשינוי שחל בו בשלב הנוכחי: בפרק לז הוא מתואר כ"נַעַר" ה"רֹעֶה אֶת אֶחָיו בַּצֹּאן" (לז ב), המצוי בתוך המסגרת המשפחתית ותחת חסותם של המבוגרים ממנו; לעומת זאת, כאן – לאחר שנמכר למצרים והורחק ממשפחתו, הוא מתחיל את דרכו העצמאית ונדרש לעמוד בכוחות עצמו, והרי הוא עובר ממעמד 'נער' למעמד 'איש'. וּמֵעֵבֶר לכך, מעבר זה מ'נער' ל'איש' אינו מתבטא רק בשינוי נסיבות חייו, אלא גם באופן התנהלותו: הליווי האלוהי הצמוד שהוא זוכה לו ממלא אותו בכוחות ובעוצמה פנימית, ובעטיו מתנהל יוסף בביטחון ובזקיפות קומה, כ'איש' – אדם בעל אישיות, ולא כעבד כנוע. כך, דווקא בצילה של העבדות והניכור ממשפחתו – יוסף מתבגר ואישיותו מתפתחת, והוא מופיע כ'איש' במלוא משמעות המושג.

"וַיְהִי [יוסף] בְּבֵית אֲדֹנָיו הַמִּצְרִי" – דהיינו שהיה משרת בביתו הפרטי של פוטיפר, ולא בשדה וכדומה; בכך מתבטאת השגחת ה' לטובה על יוסף – הן בהיבט הנקודתי של עובדה זו, שכן העבודה בבית נוחה יותר מבשדה, וכפי שכתב האברבנאל: "הנה השם היטיב עימו בבית אדוניו להניחו בביתו במצרים אשר שם ינוח לו, ולא שלחו השדה לעמול שמה"; והן בהיבט של הסיפור בכללותו – שכן על ידי כך שיוסף שובץ לעבודה דווקא "בְּבֵית אֲדֹנָיו" יכול היה פוטיפר להבחין בו מקרוב ולעמוד על טיבו, ובעקבות זאת הוא מצא חן בעיניו וזכה מאיתו לקידום, כפי שיתואר בהמשך הפסוקים; וכך כתב בעל 'עקידת יצחק': "שכבר היתה סיבה אלוהית להצלחתו [...] מה ששם אותו משרת בביתו ולא נתן עליו עבודת שדה וכרם שהם מלאכות נעשות בחוץ, וזה כדי שיראה אדוניו 'כִּי ה' אִתּוֹ וְכֹל אֲשֶׁר הוּא עֹשֶׂה ה' מַצְלִיחַ בְּיָדוֹ' (פס' ג)".

יש שפירשו כי מלכתחילה הופקד יוסף בעבודת השדה וכדומה, ורק בעקבות היותו "אִישׁ מַצְלִיחַ" העביר אותו אדוניו לעבוד בביתו, כגון 'אור החיים' שכתב: "הנה יסדר הכתוב הדרגות המעלות אשר העלה ה' אותו, כי מקודם כשקנאו נתנו לעבוד עבודת עבדים העובדים עבודת חוץ, כי הגדולים לא יעמידו לשרת בבתיהם אלא אשר יהיה איש ראוי ומצליח, ולזה ציווה ה' איתו את ההצלחה, ובזה היה 'בְּבֵית אֲדֹנָיו הַמִּצְרִי'". לפי זה, מובן מדוע המילים "וַיְהִי בְּבֵית אֲדֹנָיו הַמִּצְרִי" מופיעות דווקא בסוף הפסוק ולא בתחילתו, כפי שלכאורה היה מתבקש – בתחילה תיאור תנאי המצב הטכניים, ולאחר מכן תיאור התוכן של אותו מצב – שכן הן מתארות את המצב שנוצר דווקא בעקבות התיאור שבתחילת הפסוק.

אולם ניתן גם להבין שמלכתחילה הושם יוסף "בְּבֵית אֲדֹנָיו הַמִּצְרִי", וכאמור היה זה בהשגחת ה' – שכן רק בפסוק הבא נאמר "וַיַּרְא אֲדֹנָיו כִּי ה' אִתּוֹ וְכֹל אֲשֶׁר הוּא עֹשֶׂה ה' מַצְלִיחַ בְּיָדוֹ", ובעקבות זאת: "וַיִּמְצָא יוֹסֵף חֵן בְּעֵינָיו וַיְשָׁרֶת אֹתוֹ וַיַּפְקִדֵהוּ עַל בֵּיתוֹ וְכָל יֶשׁ לוֹ נָתַן בְּיָדוֹ" – ומכך משמע שהמתואר לפני כן, בפסוקנו, אירע בטרם ראה אדוניו כי ה' איתו והוא מצא חן בעיניו.

לפי דרכו של הפירוש הקודם ניתן ליישב זאת בכך שאכן לא פוטיפר הוא שהעביר את יוסף מעבודת השדה לעבודת הבית, כפי שכתב שד"ל: "על ידי שהיה מצליח בכל אשר היה עושה נשאר בבית עם אדוניו, ולא שלחוהו השדה לעבוד את האדמה, וזה תחילת עלייתו; והנה העלייה הזו לא היתה על ידי פוטיפר (כי הוא לא השגיח עליו מתחילה), אבל היתה על ידי אחרים, אולי היה שָׁם איש אשר על הבית".

פרק לט · פסוק ג

וַיַּרְא אֲדֹנָיו כִּי ה' אִתּוֹ וְכֹל אֲשֶׁר הוּא עֹשֶׂה ה' מַצְלִיחַ בְּיָדוֹ.

באופן פשוט, כוונת הכתוב היא: "וַיַּרְא אֲדֹנָיו כִּי ה' אִתּוֹ" – ובמה התבטא שה' איתו? בכך ש"וְכֹל אֲשֶׁר הוּא עֹשֶׂה ה' מַצְלִיחַ בְּיָדוֹ"; אדוניו ראה שיוסף מצליח במעשי ידיו בפועל, ומכך הבין כי ה' איתו. מבנה זה מקביל לתיאור שבפסוק הקודם: "וַיְהִי ה' אֶת יוֹסֵף" – ודבר זה התבטא בכך ש"וַיְהִי אִישׁ מַצְלִיחַ".212 וכך פירש הרמב"ן: "'וַיַּרְא אֲדֹנָיו כִּי ה' אִתּוֹ' – כי יצליחו מעשיו בכל עת יותר מכל אדם, וידע כי מאת אלוהים הוא לו, וכן 'רָאוֹ רָאִינוּ כִּי הָיָה ה' עִמָּךְ' (לעיל כו כח)" – אבימלך ופיכל אומרים כך ליצחק, משנוכחו בהצלחתו היוצאת מגדר הרגיל,213 ואף בענייננו זהו משמעותו של היות ה' עם יוסף.

אולם במדרש בראשית רבה (פו ה) התבאר באופן אחר, על דרך הדרש, מהי אותה ראייה שראה אדוניו "כִּי ה' אִתּוֹ", כך שמדובר בראייה ממש ולא בהבנה בלבד:

רב הונא בשם רב אחא אמר: [יוסף היה] מלחש ונכנס מלחש ויוצא,214 הוה אמר ליה מזוג רותחין – והא רותחין, פושרין – והא פושרין,215 אמר ליה: מה יוסף, תבן לעפריים אתה מכניס [...], באתר דחרשין חרשין?!216 אתמהא. עד היכן? אמר ר' חייא: עד שראה שכינה עומדת על גביו, הדא הוא דכתיב: "וַיַּרְא אֲדֹנָיו כִּי ה' אִתּוֹ".

וכך הסביר זאת הרמב"ן:

ולרבותינו בו מדרש [...] והעניין, כי בעבור היות אדוניו מצרי לא ידע את ה', אמרו כי בראותו הצלחתו הגדולה היה חושש כי דרך כשפים היא כאשר [מקובל] בבני ארצו, עד שראה שהיה מאת עליון – במראה שנראה לו בחלום, או בהקיץ בעמוד ענן וכיוצא בו לכבוד הצדיק.

במדרש תנחומא (אות ח) התבאר העניין באופן דומה אך שונה, וכך נאמר שם:

"וַיַּרְא אֲדֹנָיו כִּי ה' אִתּוֹ" וגו', וכי פוטיפר [ה]רשע, היה רואה שהקב"ה עימו?! ומה הוא "כִּי ה' אִתּוֹ"? אלא שלא היה שמו של הקדוש ברוך הוא זז מפיו: היה נכנס לשמשו והוא היה מלחש ואומר: ריבון העולם, אתה הוא ביטחוני אתה הוא פטרוני, תנני לחן ולחסד ולרחמים בעיניך ובעיני כל רואיי ובעיני פוטיפר אדוני. ופוטיפר אומר לו: מה אתה מלחש, שמא כשפים אתה עושה לי? והוא משיבו: לא, אלא אני מתפלל שאמצא חן בעיניך, לפיכך כתיב "וַיַּרְא אֲדֹנָיו כִּי ה' אִתּוֹ".

בדומה לבראשית רבה, גם לפי התנחומא פוטיפר חשד שיוסף נעזר בכשפים בעקבות היותו "מלחש", אלא שהתנחומא הזכיר רק את הליחושים ובבראשית רבה הוזכרה גם הצלחתו החריגה של יוסף שהיתה נראית כמעשי כשפים. מה גרם לפוטיפר לחדול מלחשוד כך ביוסף? לפי בראשית רבה – התרחשות ניסית מסויימת של ראיית דבר מה שהוכיחה לו כי ה' עם יוסף, "במראה שנראה לו בחלום, או בהקיץ בעמוד ענן וכיוצא בו לכבוד הצדיק" כדברי הרמב"ן, ועל כך נאמר: "וַיַּרְא אֲדֹנָיו כִּי ה' אִתּוֹ"; ולפי התנחומא – הסברו של יוסף עצמו, שאמר לו כי לחישותיו אינן אלא תפילות, לימד את פוטיפר כי אין אלו כשפים. אולם את משמעות הראייה המתוארת במילים "וַיַּרְא אֲדֹנָיו כִּי ה' אִתּוֹ", לפי התנחומא, ניתן להבין בשני אופנים:

האחד, שמאחר שיוסף הסביר לו כי הוא מתפלל תדיר שימצא חן בעיניו – הבין פוטיפר שהלחישות אינן כשפים, וממילא הסיק מכך כי הצלחתו של יוסף היא מכך שה' איתו – שהרי הוא אלוהיו ואליו הוא מתפלל. התנחומא פתח בשאלה: "וכי פוטיפר [ה]רשע, היה רואה שהקב"ה עימו?!", כלומר: אף אם ראה שיוסף מצליח, וכי מדעתו הבין שיש לייחס זאת לה'?! ועל כך הוא מפרט את התיאור כולו, שבסופו מובן כי אכן כך הבין פוטיפר, אך לא מדעתו אלא בעקבות הסברו של יוסף.

באופן שני ניתן להבין, שעצם זה "שלא היה שמו של הקדוש ברוך הוא זז מפיו" הוא היות ה' איתו – כך שהנושא הפעיל במשוואה זו אינו ה', שמתלווה מעצמו ליוסף, אלא יוסף עצמו – שמנכיח את מציאות ה' בתודעתו ובפיו. לפי זה, הרי שתשובת המדרש לתמיהתו מופיעה כבר בתחילת התיאור, ולאחריה הוא מפתח אותו יותר (בשונה מההסבר הקודם, שלפיו רק בסופו של התיאור מיושבת התמיהה).

כך ככל הנראה הבין רש"י בכוונת המדרש, שכן כתב בפירושו לפסוקנו: "'כִּי ה' אִתּוֹ' – שם שמיים שגור בפיו",217 ומדבריו משתמע שראה את המשפט "שלא היה שמו של הקדוש ברוך הוא זז מפיו" שבדברי התנחומא כפירוש למילות הפסוק עצמן, ולא רק כפתיחת התיאור.

אולם ההבנה האחרונה אינה מיושבת היטב, שכן היות שם שמיים שגור בפיו של יוסף אינו מהווה סיבה למה שיספר הכתוב בפסוק הבא: "וַיִּמְצָא יוֹסֵף חֵן בְּעֵינָיו וַיְשָׁרֶת אֹתוֹ וַיַּפְקִדֵהוּ עַל בֵּיתוֹ וְכָל יֶשׁ לוֹ נָתַן בְּיָדוֹ", אלא דווקא היות ה' מלווה אותו ומסייע לו – והלוא התיאור שבפסוקנו מוצג בכתוב כגורם המוביל לתיאור שלאחריו, ואין כל טעם לכאורה להזכיר כאן את זאת שאדוניו ראה ששם שמיים שגור בפיו. ועוד, שבכל המקרא תיאור שה' היה עם אדם – משמעו הוא סיוע אלוהי, כמו בפסוק "רָאוֹ רָאִינוּ כִּי הָיָה ה' עִמָּךְ" שהזכיר הרמב"ן, וכן "וְהִנֵּה אָנֹכִי עִמָּךְ וּשְׁמַרְתִּיךָ בְּכֹל אֲשֶׁר תֵּלֵךְ" (לעיל כח טו), "כִּי אֶהְיֶה עִמָּךְ" (שמות ג יב, שופטים ו טז), "ה' עִמָּכֶם בִּהְיוֹתְכֶם עִמּוֹ וְאִם תִּדְרְשֻׁהוּ יִמָּצֵא לָכֶם" (דברי הימים ב' טו ב), ועוד רבים. ואכן הרמב"ן כתב על פירוש רש"י: "ואינו נראה".218

    מהר"ל ב'גור אריה' הסביר שרש"י פירש כך משום "דאין לומר כמשמעו, כמו 'רָאוֹ רָאִינוּ כִּי הָיָה ה' עִמָּךְ' (לעיל כו כח), ורוצה לומר כי היה מצליח על ידי הקב"ה, דזה כתיב אחריו: 'וְכֹל אֲשֶׁר [הוּא] עֹשֶׂה ה' מַצְלִיחַ בְּיָדוֹ', אלא רוצה לומר כי שם שמיים היה שגור בפיו". אולם לכאורה אין בכך קושי על הפירוש שעל דרך הפשט, שהרי כאמור תבנית זו מופיעה גם בפסוק הקודם ובסוף הפרק, ואין בה כפילות מיותרת אלא תיאור כפול של המצב: ההצלחה בפועל, והסיוע האלוהי שגרם לה. אילו נאמר כאן רק "וַיַּרְא אֲדֹנָיו כִּי כֹל אֲשֶׁר הוּא עֹשֶׂה ה' מַצְלִיחַ בְּיָדוֹ" היה מתבטא רק ההיבט המעשי של ההצלחה, בלא ההתרשמות מעצם זה שיוסף זוכה לליווי אלוהי.

    ה'מזרחי' ביקש לפרש הן את בראשית רבה והן את התנחומא כדעת רש"י, ואלו דבריו: "ועד היכן היה טועה אדוניו בזה, עד שראה שכינה עומדת על גביו – פירוש, עד ששמע שכל אותן הלחישות שהיה לוחש בפיו לא היו אלא שם שמיים שהיה שגור בפיו, וזו היא השכינה העומדת על גביו. זהו הפירוש הנראה מלשון הבראשית רבה. אבל הרמב"ן ז"ל כתב [...]. ומה יעשה [הרמב"ן] במאמר 'לוחש ונכנס לוחש ויוצא', אותן הלחישות מה היה, אם לא שהיה שם שמיים שגור בפיו, כדברי הרב [רש"י]? והרי בתנחומא אמרו בהדיא שכשראהו פוטיפר מתפלל אמר לו מה אתה מלחש, שמא כשפים אתה עושה, השיבו: אני מתפלל לפני הקב"ה למצוא חן בעיניך, לפיכך כתב 'וַיַּרְא אֲדֹנָיו כִּי ה' אִתּוֹ'". אולם פירושו לבראשית רבה נראה דחוק, ואף קושייתו על הרמב"ן אינה מובנת – שהרי גם לפי הרמב"ן ניתן לומר שהיה שם שמיים שגור בפיו, וכך אכן מסתבר לפרש את היותו "מלחש ונכנס מלחש ויוצא", אך לא זוהי כוונת הכתוב במילים "וַיַּרְא אֲדֹנָיו כִּי ה' אִתּוֹ", וכמבואר לעיל. הוכחתו לפירוש רש"י מהתנחומא, הרי היא מבוססת על האופן השני בהבנת דבריו, כפי שהתבאר כאן.

פרק לט · פסוק ד

וַיִּמְצָא יוֹסֵף חֵן בְּעֵינָיו וַיְשָׁרֶת אֹתוֹ, וַיַּפְקִדֵהוּ עַל בֵּיתוֹ וְכָל יֶשׁ לוֹ נָתַן בְּיָדוֹ.

בפסוק הקודם ובפסוקנו מתוארים שני עניינים ביחס למבטו של פוטיפר על יוסף, ושתי תוצאות שלהם:" [א] וַיַּרְא אֲדֹנָיו כִּי ה' אִתּוֹ וְכֹל אֲשֶׁר הוּא עֹשֶׂה ה' מַצְלִיחַ בְּיָדוֹ; [ב] וַיִּמְצָא יוֹסֵף חֵן בְּעֵינָיו" – פוטיפר ראה כי ה' מצוי עם יוסף ומצליח בידו, ומלבד זאת, גם אישיותו של יוסף מצאה חן בעיניו; והיו לכך שתי תוצאות: האחת – "וַיְשָׁרֶת אֹתוֹ", דהיינו שיוסף התמנה על ידו להיות משרתו האישי, והשנייה – "וַיַּפְקִדֵהוּ עַל בֵּיתוֹ וְכָל יֶשׁ לוֹ נָתַן בְּיָדוֹ".

נראה להסביר, שכל אחד מהמינויים של יוסף היה תוצאה של סיבה אחרת: מפני שיוסף היה מצליח במעשיו, מינה אותו אדוניו לנהל את כל ענייני הבית והרכוש, על מנת שאלו ינוהלו כהלכה ובתבונה; ומפני שיוסף מצא חן בעיניו מינה אותו למשרתו, שכן מציאת חן אינה תלויה בכשרון מעשי אלא באישיות כשלעצמה – ובסיבת חינו של יוסף בעיניו, חפץ בקרבתו ומינה אותו לתפקיד זה.

אם כן, הרי שפסוקים אלו ערוכים במבנה כיאסטי של סיבה ותוצאה: "[א] וַיַּרְא אֲדֹנָיו כִּי ה' אִתּוֹ וְכֹל אֲשֶׁר הוּא עֹשֶׂה ה' מַצְלִיחַ בְּיָדוֹ, [ב] וַיִּמְצָא יוֹסֵף חֵן בְּעֵינָיו – [ב] וַיְשָׁרֶת אֹתוֹ, [א] וַיַּפְקִדֵהוּ עַל בֵּיתוֹ וְכָל יֶשׁ לוֹ נָתַן בְּיָדוֹ". ההצלחה המעשית והמינוי לתפקיד המעשי מופיעים כמסגרת החיצונית, בעוד שהחן האישי והמינוי האישי מופיעים בתווך, כגרעין הפנימי של מערכת היחסים שבין פוטיפר ליוסף.

"וְכָל יֶשׁ לוֹ נָתַן בְּיָדוֹ" – כתב רש"י: "'וְכָל יֶשׁ לוֹ' – הרי לשון קצר, חסר 'אשר'", ומשמעות הכתוב היא: "וְכָל אשר יֶשׁ לוֹ – נָתַן בְּיָדוֹ". אולם ניתן לפרש זאת גם באופן אחר: פוטיפר נתן בידו את כל ה"יש" שלו, ה"ישות" של כל קנייניו ונכסיו; וכך פירש בעל 'הכתב והקבלה':

ולדעתי מילת "יֶשׁ" כאן מעניין "לְהַנְחִיל אֹהֲבַי יֵשׁ" (משלי ח כא), שהוא שם כולל לקנייני השלמות בטוב ובמועיל, והלמ"ד במילת "לוֹ" משמשת במקום "של", כמו "וְכֹל אֲשֶׁר אַתָּה רֹאֶה לִי הוּא" (לעיל לא מג) – שלי הוא [...], ועל דרך זה: "אַף אֵין יֶשׁ רוּחַ בְּפִיהֶם" (תהלים קלה יז), כלומר אין ישיות רוח בפיהם שיהיה לתועלת הדיבור [...], וטעמו כאן: ישיות קניינים שלו נתן בידו.

פרק לט · פסוק ה

וַיְהִי מֵאָז הִפְקִיד אֹתוֹ בְּבֵיתוֹ וְעַל כָּל אֲשֶׁר יֶשׁ לוֹ – וַיְבָרֶךְ ה' אֶת בֵּית הַמִּצְרִי בִּגְלַל יוֹסֵף, וַיְהִי בִּרְכַּת ה' בְּכָל אֲשֶׁר יֶשׁ לוֹ בַּבַּיִת וּבַשָּׂדֶה.

כתב ר' נפתלי הרץ ויזל:

טרם הפקידוֹ על ביתו וטרם נתן בידו כל אשר יש לו, היה במעשהו "אִישׁ מַצְלִיחַ"; כל אשר עשה יוסף ה' היה מצליח בידו, ואין כן שאר אנשי פוטיפר שלא דבקה השגחת ה' במעשיהם. אבל מאז נעשה יוסף פקיד על בית פוטיפר, ומאז נתן הכל ביד יוסף, ראה ה' טוב להצליח מעשיהם וכל אשר הם עושים מצליח, כאילו יוסף העושה, לפי שהכל [תחת] ידו ונעשה על פיו.

פסוקנו מתאר את ברכת ה' ששרתה בנכסי פוטיפר מאז המינויים שהוזכרו בפסוק הקודם, ואכן תחילת הפסוק נראית מקבילה למה שנאמר שם: "[א] וַיַּפְקִדֵהוּ עַל בֵּיתוֹ, [ב] וְכָל יֶשׁ לוֹ נָתַן בְּיָדוֹ – [א] וַיְהִי מֵאָז הִפְקִיד אֹתוֹ בְּבֵיתוֹ, [ב] וְעַל כָּל אֲשֶׁר יֶשׁ לוֹ". אולם ניתן לשאול – מדוע נאמר כאן "מֵאָז הִפְקִיד אֹתוֹ בְּבֵיתוֹ", ולא "על ביתו" כפי שנאמר בפסוק הקודם? הנוסח "על ביתו" נראה מתאים יותר – שכן הוא מציין את מהות המינוי והסמכות שניתנה במסגרתו, בעוד ש"בביתו" אינו מציין אלא את המקום שבו היה יוסף.

הנצי"ב ב'העמק דבר' פירש שאכן הפקדת יוסף "בְּבֵיתוֹ" היא שלב קודם להפקדתו "עַל בֵּיתוֹ": "'הִפְקִיד אֹתוֹ בְּבֵיתוֹ' – משמעות 'בְּבֵיתוֹ': בעוד לא הפקידו לאדון 'עַל בֵּיתוֹ', אלא היה עוד בביתו ממש, היינו לעבודת הבית", וכפי שנאמר לפני כן "וַיְהִי בְּבֵית אֲדֹנָיו הַמִּצְרִי" – שאז היה יוסף בבית פוטיפר ומופקד בו על עניינים מסויימים, אך לא על הבית כולו.

אך פירוש זה מעורר קושי, שכן בפסוקנו נאמר "וַיְהִי מֵאָז הִפְקִיד אֹתוֹ בְּבֵיתוֹ וְעַל כָּל אֲשֶׁר יֶשׁ לוֹ", והרי ההפקדה "עַל כָּל אֲשֶׁר יֶשׁ לוֹ" נעשתה בשלב מאוחר יותר להפקדה "בְּבֵיתוֹ" (כפי שהיא מתפרשת לפי דבריו) – ואילו מלשון הכתוב משמע שמדובר בו על נקודת זמן אחת.

הנצי"ב עצמו (ב'הרחב דבר') פירש שאכן ברכת ה' התחילה בנקודת זמן אחת, המוקדמת יותר: "והגיד הכתוב שהברכה החלה מאז שנכנס יוסף בעסקי ביתו, אפילו בעוד לא הפקידו על ביתו להיות למושל [...] – גם אז נתברך בית המצרי בכל אשר יש לו בבית ובשדה, מה שלא היה נוגע כלל ליוסף, אלא בזכותו נגללה הברכה". אך גם אפשרות זו מעוררת קושי, שכן אם הברכה המתוארת בפסוקנו החלה כבר "מֵאָז הִפְקִיד אֹתוֹ בְּבֵיתוֹ", דהיינו בשלב המוקדם יותר של "וַיְהִי בְּבֵית אֲדֹנָיו הַמִּצְרִי" – לשם מה מזכיר כאן הכתוב את השלב המאוחר של "מֵאָז הִפְקִיד אֹתוֹ [...] וְעַל כָּל אֲשֶׁר יֶשׁ לוֹ", מאחר ולא התחדש בו דבר?

בנוסף לכל זאת, כאמור לעיל – פסוקנו מנוסח באופן מובהק כתקבולת לפסוק הקודם: "[א] וַיַּפְקִדֵהוּ עַל בֵּיתוֹ, [ב] וְכָל יֶשׁ לוֹ נָתַן בְּיָדוֹ – [א] וַיְהִי מֵאָז הִפְקִיד אֹתוֹ בְּבֵיתוֹ [ב], וְעַל כָּל אֲשֶׁר יֶשׁ לוֹ", ומכך נראה שמדובר בהם על אותם המינויים, וקשה לפרש שהמילים "מֵאָז הִפְקִיד אֹתוֹ בְּבֵיתוֹ" אינן מתייחסות ל"וַיַּפְקִדֵהוּ עַל בֵּיתוֹ" אלא למינוי אחר שהיה בשלב מוקדם יותר.

ניתן להסביר באופן אחר את פשר ההבדל בין לשונות הפסוקים, כך שאין כוונת הכתוב במילים "מֵאָז הִפְקִיד אֹתוֹ בְּבֵיתוֹ", רק למינוי יוסף "עַל בֵּיתוֹ", אלא לכלל התפקידים שניתנו לו בבית פוטיפר; שכן בפסוק הקודם הוזכרו שני מינויים שקיבל יוסף בביתו של פוטיפר: "וַיְשָׁרֶת אֹתוֹ" – מינויו למשרתו האישי, ובנוסף לכך "וַיַּפְקִדֵהוּ עַל בֵּיתוֹ" – מינויו לאחראי על ענייני הבית. לפיכך נקט כאן הכתוב בלשון הכוללת את שניהם: "מֵאָז הִפְקִיד אֹתוֹ בְּבֵיתוֹ" – הפקדתו כמשרתו האישי, והפקדתו על ביתו. בכך מובן גם שינוי הלשון, וגם פשר ההתעלמות הקיימת לכאורה בפסוקנו מהמינוי "וַיְשָׁרֶת אֹתוֹ" שהוזכר בפסוק הקודם.

הסבר זה שופך אור גם על הבנת המשך הפסוק: "וַיְבָרֶךְ ה' אֶת בֵּית הַמִּצְרִי בִּגְלַל יוֹסֵף, וַיְהִי בִּרְכַּת ה' בְּכָל אֲשֶׁר יֶשׁ לוֹ בַּבַּיִת וּבַשָּׂדֶה" – מפני מה מוזכרת פעמיים ברכת ה' בבית פוטיפר?

ר' נפתלי הרץ ויזל פירש שהמילים "וַיְבָרֶךְ ה' אֶת בֵּית הַמִּצְרִי בִּגְלַל יוֹסֵף" אינן מתייחסות למלאכת הבית (שעליה מדובר בסוף הפסוק), אלא למעגל המשפחתי של פוטיפר: "ברכת הבריאות והבנים, כמו 'וַיְבָרֶךְ ה' אֶת עֹבֵד אֱדֹם וְאֶת כָּל בֵּיתוֹ' (שמואל ב' ו יא)",219 וכן הנצי"ב ב'העמק דבר': "'אֶת בֵּית הַמִּצְרִי' היינו ביתו ממש, היינו נשיו ובניו". כך מדויק גם בלשון הכתוב, שכן בפעם הראשונה מיוחסת הברכה לבית עצמו – "וַיְבָרֶךְ ה' אֶת בֵּית הַמִּצְרִי בִּגְלַל יוֹסֵף", והיינו המשפחה, ואילו בפעם השנייה נאמר: "וַיְהִי בִּרְכַּת ה' בְּכָל אֲשֶׁר יֶשׁ לוֹ בַּבַּיִת וּבַשָּׂדֶה" – לא בבית עצמו, אלא "בְּכָל אֲשֶׁר יֶשׁ לוֹ בַּבַּיִת", דהיינו הנכסים והמלאכה שבבית.

על פי זה, הרי שההסבר האמור לעיל – כי הכתוב כולל במילים "מֵאָז הִפְקִיד אֹתוֹ בְּבֵיתוֹ" גם את המינוי "וַיְשָׁרֶת אֹתוֹ" – מקבל תוספת משמעות; שכן מעתה יובן כי אותו מינוי הוא היה הגורם לברכה במשפחתו של פוטיפר, הנזכרת כאן במילים "וַיְבָרֶךְ ה' אֶת בֵּית הַמִּצְרִי בִּגְלַל יוֹסֵף". עצם נוכחותו ומעורבתו של יוסף בחייו האישיים של פוטיפר – הביא ברכה לבני ביתו, המעגל הקרוב אליו באופן אישי; וזאת מלבד מערכת הגורמים המפורשת בכתוב באופן גלוי: ניהולו את ענייני הבית, שהביא ברכה ל"כָל אֲשֶׁר יֶשׁ לוֹ בַּבַּיִת", וניהולו את ענייני השדה, שהביא ברכה ל"כָל אֲשֶׁר יֶשׁ לוֹ [...] וּבַשָּׂדֶה".

בכך יובן גם מדוע רק על "וַיְבָרֶךְ ה' אֶת בֵּית הַמִּצְרִי" פירש הכתוב "בִּגְלַל יוֹסֵף", מה שאין כן לגבי "וַיְהִי בִּרְכַּת ה' בְּכָל אֲשֶׁר יֶשׁ לוֹ בַּבַּיִת וּבַשָּׂדֶה", ששם סמך הכתוב על הבנת הקורא שמאחר וברכה זו החלה מאז שהופקד על הבית והשדה – בידוע שזו באה בגללו. שכן, בעניין מלאכת הבית והשדה – הרי שיוסף הופקד על ניהולה, ומובן שאם שרתה בה ברכה מאז – הרי שזו באה לרגלו; אך ענייני בני משפחתו של פוטיפר, פוריותם ובריאותם, היו תלויים בידי שמיים ולא מנוהלים על ידי יוסף – שתפקידו במעגל זה היה רק לשרת את פוטיפר, ולכאורה אין קשר נסיבתי בין זה לזה. לפיכך, לגבי ברכת "בֵּית הַמִּצְרִי" נדרש הכתוב לציין בפירוש שאף היא היתה "בִּגְלַל יוֹסֵף" – לא כתוצאה של ניהול מושכל ומצליח ששורה בו ברכת ה', כפי שהיה במלאכת הבית והשדה, אלא כתוצאה של עצם נוכחותו ומעורבתו בחייו האישיים של פוטיפר, אשר בעקבות זאת הופיעה בהם ברכה אלוהית.

"וַיְהִי בִּרְכַּת ה' בְּכָל אֲשֶׁר יֶשׁ לוֹ בַּבַּיִת וּבַשָּׂדֶה" – באופן פשוט, המילה "וַיְהִי" מתייחסת ל"בִּרְכַּת ה'", הגם שלפי כללי הדקדוק היה ראוי להכתב "וַתְּהִי" בלשון נקבה. אולם בעל מדרש 'שכל טוב' פירש (בעקבות מדרש 'לקח טוב') שאכן לשון הזכר שבמילה זו מצביעה על כך שאין היא מתייחסת לברכה, אלא ליוסף: "חזר הדיבור ליוסף, כלומר: ויהי יוסף ברכת ה' – הוא היה הברכה בכל אשר לו, לפוטיפר". התנסחות דומה המגדירה אדם כ'ברכה' מצינו לגבי אברהם, שהובטח מפי ה' "וֶהְיֵה בְּרָכָה" (לעיל יב ב), דהיינו שיהיה מקור ברכה לאחרים, כפי שפירש שם רש"י: "הברכות נתונות בידך [...] אתה תברך את אשר תחפוץ", והמלבי"ם: "יהיה הוא הבאר שממנו יביע מי הברכה על כל העולם". אף כאן, היה יוסף מקור לברכת ה' ששרתה בכל אשר לפוטיפר, "בַּבַּיִת וּבַשָּׂדֶה".

טבלה זו מסכמת וממחישה את המבואר לעיל בעניין מינוייו של יוסף והברכה שבאה בעקבותיהם:

פרק לט · פסוק ו

וַיַּעֲזֹב כָּל אֲשֶׁר לוֹ בְּיַד יוֹסֵף וְלֹא יָדַע אִתּוֹ מְאוּמָה כִּי אִם הַלֶּחֶם אֲשֶׁר הוּא אוֹכֵל, וַיְהִי יוֹסֵף יְפֵה תֹאַר וִיפֵה מַרְאֶה.

בפסוקים הקודמים תיאר הכתוב את מציאת חינו של יוסף בעיני פוטיפר, את זאת שראה כי ה' איתו, ואת המינויים שמינהו בעקבות אלו. פסוקנו מוסיף על זאת את מימד הנאמנות – יוסף לא היה בעיני אדוניו רק "אִישׁ מַצְלִיחַ" המביא ברכה, אלא גם עבד נאמן וישר; ומשכך – לא זו בלבד שמינה אותו לאחראי על כל אשר לו, אלא גם לא ראה צורך לפקח עליו – לא על טיב עבודתו ולא על ישרות מעשיו, וסמך עליו באופן מוחלט.

"וַיַּעֲזֹב כָּל אֲשֶׁר לוֹ בְּיַד יוֹסֵף"

ניתן לשים לב להבדלים בלשונות הפסוקים האחרונים בהתייחסותם לרכושו של פוטיפר: בפסוק ד נאמר: "וְכָל יֵשׁ לוֹ נָתַן בְּיָדוֹ", בפסוק ה: "מֵאָז הִפְקִיד אֹתוֹ בְּבֵיתוֹ וְעַל כָּל אֲשֶׁר יֶשׁ לוֹ", ובפסוקנו: "וַיַּעֲזֹב כָּל אֲשֶׁר לוֹ בְּיַד יוֹסֵף". לעיל הובא פירושו של ה'כתב והקבלה', שלפיו המילה "יֵשׁ" שבפסוק ד משמשת כשם עצם, ה'ישות' של נכסי פוטיפר, ההון והרכוש עצמם, שאותה נתן ביד יוסף.

על פי דרך זו, ניתן להסביר את טעם שינויי הלשון שבשלושת הפסוקים:

בפסוק ד נאמר: "וְכָל יֵשׁ לוֹ נָתַן בְּיָדוֹ" – שכן מופיעה שם לשון 'נתינה', המציינת את עצם מסירת נכסיו של פוטיפר לידי יוסף; בהתאם לכך, מדגיש הכתוב את ה"יֵשׁ" – הרכוש שניתן בידו, ומעמיד אותו במרכז התיאור.

בפסוק ה נאמר: "מֵאָז הִפְקִיד אֹתוֹ בְּבֵיתוֹ וְעַל כָּל אֲשֶׁר יֶשׁ לוֹ", שכן בו מזכיר הכתוב את המינוי בלשון 'הפקדה' ולא בלשון 'נתינה', כך שאמנם הרכוש הוא שעומד במרכז התיאור, אך אין הוא מושא הפעולה – הוא אינו 'ניתן', אלא יוסף הוא שמופקד עליו; ולפיכך, ה"יש" אינו עומד שם כמושא עצמאי, ישות, אלא כחלק מן הצירוף "אֲשֶׁר יֶשׁ לוֹ". אין הדגש על ההון מצד עצמו, אלא על תחום האחריות שהופקד עליו יוסף.

בפסוקנו נאמר: "וַיַּעֲזֹב כָּל אֲשֶׁר לוֹ בְּיַד יוֹסֵף", ללא הזכרת הנכסים כ"יש" כלל, מפני שכאן אין התיאור מתמקד בנכסים עצמם ואינו מתאר פעולה שנעשתה בהם, אלא את יחסו של פוטיפר אליהם – שהניחם ביד יוסף וסילק את עצמו מן העיסוק בהם; מוקד התיאור בענייננו אינו הנכסים אלא ייתור הצורך בעיני פוטיפר לפקח עליהם, ולפיכך מסתפק כאן הכתוב בניסוח הסתמי "כָּל אֲשֶׁר לוֹ".220

"וְלֹא יָדַע אִתּוֹ מְאוּמָה כִּי אִם הַלֶּחֶם אֲשֶׁר הוּא אוֹכֵל"

פירש רש"י: "'וְלֹא יָדַע אִתּוֹ מְאוּמָה' – לא היה נותן ליבו לכלום"; פוטיפר לא ידע ולא התעניין באופן שבו מנהל יוסף את ביתו ונכסיו, מאחר וסמך באופן מוחלט על תבונתו ויושרו.

"כִּי אִם הַלֶּחֶם אֲשֶׁר הוּא אוֹכֵל" – באופן פשוט, משמעות הכתוב היא שהדבר היחיד שידע פוטיפר מענייני ביתו הוא מה שהיה אוכל, וכפי שפירש רשב"ם: "לא היה מעיין ומביט כלל בכל צורכי ביתו מהיכן בא לו, עד שעה שרואה את מאכלו על השולחן ואוכלו. כל מיני מאכל קורא אותו 'לחם', כדכתיב: 'עֲבַד לְחֶם רַב' (דניאל ה א), וכן: 'כִּי לִפְנֵי לַחְמִי אַנְחָתִי תָבֹא' (איוב ג כד) – מאכלי". לפי פירוש זה, הלשון "כִּי אִם" אינה מצביעה על הסתייגות של ממש, וכוונת הכתוב לומר שפוטיפר סמך על יוסף באופן כה מוחלט עד שלא ידע איתו מאומה – והדבר היחיד שידע הוא מה שמובן מאליו שידע: מה שמוגש לפניו על השולחן.

באופן דומה פירש בעל הטורים (בפירוש הארוך), אלא שלפיו מילים אלו אינן מציינות את המינימום המובן מאליו, אלא הסתייגות – הצבעה על דבר שבו אכן היה פוטיפר מעורב: "שלא היה מצווה עליו שום דבר מענייני הבית, אלא הכל עשה כרצונו, מלבד הלחם, שהיה אומר: זה אני רוצה כן לאכול מדי יום יום"; וכך גם ר' נפתלי הרץ ויזל: "שמדרך השרים לצוות על המאכלים שיכינו לו כאשר יתאווה להן, ועל זה לא יוכל יוסף לצוות, כי איך ידע המאכל שהוא מתאווה אליו היום? ומה שיתאווה אליו מחר? והוא לבדו יצווה עליו אדוניו". לפי פירוש זה, כוונת הכתוב אינה שידע רק מה שהוגש לפניו (כרשב"ם), אלא גם "ידע" באופן פעיל, דהיינו שהיה מצווה על כך את יוסף בהתאם לרצונו שלו. ומכל מקום, מובן שאין זו הסתייגות אלא ממידת אי מעורבתו של פוטיפר בענייני הבית, ולא ממידת נאמנותו של יוסף בעיני פוטיפר; הכתוב מדגיש את הטוטאליות של נאמנות זו בכך שמציין כי הדבר היחיד שעליו היה פוטיפר נותן הוראות ליוסף הוא מה שברצונו לאכול.

אבן עזרא פירש באופן אחר: "שכל אשר לו היה בידו, חוץ מהלחם שלא היה אפילו נוגע בו, בעבור היותו עברי, כי הנה מפורש: 'כִּי לֹא יוּכְלוּן הַמִּצְרִים לֶאֱכֹל אֶת הָעִבְרִים [לֶחֶם כִּי תוֹעֵבָה הִוא לְמִצְרָיִם]' (להלן מג לב); כי פוטיפר ידע שיוסף עברי הוא, וכן אשתו אומרת: 'רְאוּ הֵבִיא לָנוּ אִישׁ עִבְרִי' (פס' יד)". בפסוק שציטט, מעיד הכתוב על כך שהמצרים נמנעו לאכול לחם עם העברים, "כִּי תוֹעֵבָה הִוא לְמִצְרָיִם", ולהבנת אבן עזרא מנהג זה כלל גם הימנעות מלאכול לחם שנגע בו עברי – ולפיכך מציין כאן הכתוב שמכל רכושו של פוטיפר רק לחמו לא היה ביד יוסף.

אולם פירוש זה מעורר מספר קשיים: האחד, שזהו ענין צדדי שאינו נוגע כלל לנושא המדובר בפסוקים אלו שהוא מעמדו של יוסף ונאמנותו בעיני אדוניו, וקשה למצוא טעם לכך שראה הכתוב לציינו כאן; ועוד, שבוודאי יכול היה פוטיפר לסמוך על יוסף שלא ייגע בלחמו, ואם כן הרי שאין טעם לומר שרק "הַלֶּחֶם אֲשֶׁר הוּא אוֹכֵל" לא היה "בְּיַד יוֹסֵף", שהרי אמנם לא היה בידו הפיזית, אך היה אף הוא תחת אחריותו ככל ענייני הבית; ועוד, שהרי המילים "כִּי אִם הַלֶּחֶם אֲשֶׁר הוּא אוֹכֵל" אינן מופיעות כהסתייגות מהכלל "וַיַּעֲזֹב כָּל אֲשֶׁר לוֹ בְּיַד יוֹסֵף" (כפי שהציג זאת אבן עזרא: "שכל אשר לו היה בידו, חוץ מהלחם"), אלא מהכלל "וְלֹא יָדַע אִתּוֹ מְאוּמָה" – ואם כן הרי שלא נאמר כאן שלחמו לא היה ביד יוסף, אלא שבעניין הלחם "יָדַע אִתּוֹ מְאוּמָה" – ואין זה מיושב לכאורה עם פירושו, שכן אף אם נאמר שאכן פוטיפר הקפיד על דבר זה – הרי די היה בכך שיזהיר על כך יוסף פעם אחת, ושוב לא היה נדרש לפקח על קיום ההוראה, כשם שלא פיקח על העניינים האחרים, ואף בזה לא היה לו צורך "לדעת איתו מאומה".

הרמב"ן הציע פירוש נוסף:

או יהיה פירוש הכתוב: כי לא ידע אדוניו את יוסף מאומה שייקח מכל אשר לו רק את הלחם לבדו אשר יאכל יוסף, לא לשאר הנאות כאשר יעשו הבחורים, ולא לאסוף לו עושר ונכסים; כעניין שנאמר "וְלֹא מָצָאתִי בוֹ מְאוּמָה מִיּוֹם נפְלוֹ עַד הַיּוֹם הַזֶּה" (שמואל א' כט ג).221 ו"לֹא יָדַע אִתִּי מַה בַּבָּיִת" (פס' ח) – עניין אחר, שלא ישגיח לדעת מה איתו בבית;222 כי "לֹא יָדַע אִתּוֹ מְאוּמָה" – לשון שולל, ו"מַה בַּבָּיִת" – יקיים.223

לפי פירוש זה, כוונת הכתוב לומר: פוטיפר לא ראה מאומה מרכושו שיוסף לקח לעצמו והוא היה "אִתּוֹ", "כִּי אִם הַלֶּחֶם אֲשֶׁר הוּא אוֹכֵל" – חוץ מהלחם אשר יוסף אוכל, אשר אותו בוודאי היה רשאי לקחת.

אולם, דומה כי הלשון "וְלֹא יָדַע אִתּוֹ מְאוּמָה" משתמעת יותר כפי הפירושים הקודמים, שכן המילה "מְאוּמָה" משמשת כאן כמושא של הפועל "יָדַע", ומכך מובן שהיא מציינת 'ידיעה', כלומר: לא היתה שום ידיעה מענייני הבית שידע האדון; אך לפירוש הרמב"ן המילה "מְאוּמָה" אינה מציינת מושג מופשט של ידיעה, אלא מושג מוחשי – פריט כל שהוא מרכושו של פוטיפר, והרי זה אינו עשוי להוות מושא של הפועל "יָדַע", שכן אין פוטיפר יודע את הפריט הגנוב, אלא יודע על הימצאותו אצל מי שלקחו.

מלבד פירושים אלו שעל דרך הפשט, מופיע בדברי המפרשים פירוש נוסף, שניתן לראותו כקרוב לדרך הדרש; כך פירש רש"י (על פי בראשית רבה ע ד, פו ו; שמות רבה א לב): "'כִּי אִם הַלֶּחֶם' – היא אשתו, אלא שדיבר בלשון נקייה"; וכך בתרגום המיוחס ליונתן: "ולא ידע עימיה מדעם, אלהין אינתתיה דשכיב גבה [=ולא ידע איתו מאומה, כי אם אשתו ששוכב אצלה]".

פרשנות זו מתבססת על מספר מקומות במקרא שבהם נדרשת לשון 'לחם' כרמז לאישה, או לשון 'אכילה' כרמז ליחסי אישות; ובנוסף לכך, גם על השוואה למה שיאמר יוסף להלן (פס' ט) לאשת פוטיפר: "וְלֹא חָשַׂךְ מִמֶּנִּי מְאוּמָה כִּי אִם אוֹתָךְ בַּאֲשֶׁר אַתְּ אִשְׁתּוֹ"; וכפי שציינו כמה מן המפרשים כך, כגון ר' שמואל בן חפני גאון: "ואמרו 'כִּי אִם הַלֶּחֶם' – ויש עימו ההוצאה-מן-הכלל השנייה, שהיא אמרו 'כִּי אִם אוֹתָךְ בַּאֲשֶׁר אַתְּ אִשְׁתּוֹ'; וכך נֹאמַר כי המטרה בשני הדיבורים היא אחת, ויהיה אמרו 'כִּי אִם הַלֶּחֶם' – אשתו, בדומה לאמרו 'אָכְלָה וּמָחֲתָה פִיהָ' וגו' (משלי ל כ)",224 וכן רבינו בחיי: "ועוד יכלול 'הַלֶּחֶם' ההוא כינוי על אשתו, כעניין שכתוב 'קִרְאֶן לוֹ וְיֹאכַל לָחֶם' (שמות ב כ),225 וכתיב: 'אָכְלָה וּמָחֲתָה פִיהָ'. וכן ביאר בסמוך: 'כִּי אִם אוֹתָךְ בַּאֲשֶׁר אַתְּ אִשְׁתּוֹ'".

השוואה זו בין שני הפסוקים, כשלעצמה אינה מהווה בהכרח סימוכין לפירוש זה, על אף הדמיון האסוציאטיבי ביניהם, שכן בכל אחד מהפסוקים מדובר על עניין שונה; כאן נאמר "וְלֹא יָדַע אִתּוֹ מְאוּמָה כִּי אִם הַלֶּחֶם אֲשֶׁר הוּא אוֹכֵל" – הנושא הוא ידיעתו ומעורבותו של האדון במה שנעשה בביתו, ואילו שם נאמר "וְלֹא חָשַׂךְ מִמֶּנִּי מְאוּמָה כִּי אִם אוֹתָךְ בַּאֲשֶׁר אַתְּ אִשְׁתּוֹ" – הנושא הוא מה עומד לרשותו של יוסף ומה אינו עומד לרשותו. גם ההנחה כי 'לחם' עשוי להוות רמז לאישה, ו'אכילה' רמז ליחסי אישות, אינה מעוגנת ברובד הפשט שבמקרא, שכן התקדימים שהביאו לכך המפרשים הם עצמם על דרך הדרש.226

מכל מקום, אם נפרש כדברי מפרשים אלו, שכוונת הכתוב לומר: "וְלֹא יָדַע אִתּוֹ מְאוּמָה כִּי אִם את אשתו", הרי שמשמעות הדבר אינה רק שפוטיפר "חָשַׂךְ" (=מנע) מיוסף את אשתו, אלא גם ש"יָדַע" בדבר, כלומר פיקח על כך שהם רחוקים זה מזה. אם כן, הרי שאף על פי שהוא סמך על יוסף לחלוטין בכל הקשור לענייני מלאכתו ונכסיו, לא כך היה בעניין אשתו – וניתן להסביר זאת הן בכך שהיה קנאי יותר לאשתו מאשר לממונו, והן בכך שהיה מודע לפוטנציאל של המכשול בעניין זה, שהיה גדול יותר מזה של מכשול בענייני ממון – בשל כוחו של היצר, וכן בשל יופיו ותדמיתו של יוסף, ושמא גם חשד באשתו כי היא נושאת עיניה אליו. לפיכך, לא רק ש"חָשַׂךְ" את אשתו מיוסף, כפי שמובן מאליו, אלא גם "יָדַע אִתּוֹ מְאוּמָה" בעניין זה.

"וַיְהִי יוֹסֵף יְפֵה תֹאַר וִיפֵה מַרְאֶה"

נתון זה מופיע בין תיאור הצלחתו של יוסף בבית פוטיפר לבין המעשה עם אשת פוטיפר, וניתן לראותו כשייך לשניהם. מבחינת המבנה הספרותי של הפרק, הרי שהוא שייך לחטיבת הפסוקים הקודמת לו, שכן הפסוק הבא פותח במילים "וַיְהִי אַחַר הַדְּבָרִים הָאֵלֶּה" המציינות פתיחת עניין; וניתן להסביר שזהו חלק מתיאור הצלחתו – היות ה' עם יוסף התבטא לא רק בהצלחתו ומעמדו בבית פוטיפר, אלא גם ביפי תארו ומראהו; השפע האלוהי שליווה אותו הקיף את כל תחומי חייו – המעשיים והגופניים. עם זאת, ניתן לראות בהוספה זו גם הקדמה לחטיבת הפסוקים הבאה, שכן היא מופיעה כאן לאחר סיכום כל התפתחות המינויים והתקדמות מעמדו של יוסף, וכביכול עומדת לעצמה; וכך פירש הרמב"ן: "הזכיר זה בכאן, לומר כי בעבור יופיו נשאה אשת אדוניו אליו את עיניה".

ישנו קושי להניח שכל עניינו של היגד זה הוא להוות רקע לכך שאשת אדוניו נשאה אליו עיניה, כפי שהעיר ר' נפתלי הרץ ויזל על דברי הרמב"ן: "ואיני רואה מה צורך לזה, אם נשאה אליו את עיניה – נודע שחמדתו בליבה!"; ולפיכך נראה שעיקר עניינו הוא כאמור השלמת תיאור הצלחתו של יוסף; ועם זאת, ניתן להבין שהכתוב שיבץ אותו דווקא בסיום חטיבה זו, על מנת שישמש גם כהקדמה למעשה הבא, שבו נעשה יופיו של יוסף גורם מרכזי בהתפתחות העלילה.

"יְפֵה תֹאַר וִיפֵה מַרְאֶה" – פירש בעל מדרש 'לקח טוב': "'יְפֵה תֹאַר' הוא הקומה [דהיינו תואר הגוף בכללותו], 'וִיפֵה מַרְאֶה' זו צורת פנים"; ויש שפירשו שהמונח 'תואר' מציין את קווי המתאר של הפנים, והמונח 'מראה' מציין את גוון העור או את זיו הקלסתר (רש"י ורשב"ם על הפסוק "וְרָחֵל הָיְתָה יְפַת תֹּאַר וִיפַת מַרְאֶה", לעיל כט יז). וראה הרחבה בעניין זה בספרי 'אבן ישראל' על פרשת ויצא, ח"א עמ' 138-137.

    פסוק זה אומר אכיש מלך גת על דוד, על תקופת שהותו עימו בגת ובצקלג; הרמב"ן מדמה את הפסוקים זה לזה, אך דימוי זה עשוי להטעות: שכן שם, משמעות הדברים היא "לא מצאתי בדוד עצמו מאומה רע", אך כאן אין כוונת הכתוב לומר שפוטיפר לא מצא דבר רע ביוסף, אלא שלא ידע על קיומו של רכוש מבית האדון שנמצא ברשותו של יוסף, ולפיכך לא נאמר כאן "בּוֹ", דהיינו בו עצמו, אלא "אִתּוֹ" – ברשותו; והמילה 'מאומה' מתפרשת כפי שהיא, ולא 'מאומה רע' כמשמעותה בדברי אכיש. אילו מילה זו היתה מתפרשת כאן כ'מאומה רע', לא היה טעם לומר לאחר מכן כהסתייגות "כִּי אִם הַלֶּחֶם אֲשֶׁר הוּא אוֹכֵל", שהרי אף בכך אין כל רע. מסתבר איפוא שכוונת הרמב"ן לדמיון כללי בין הפסוקים, ולא לדמיון מילולי או מבני.

  3. 222

  4. 223

  5. 224

  6. 225

  7. 226

    כפי שהתבאר בהערות הקודמות. במדרש תנחומא (שמות יא) מופיעים הדברים המצוטטים בהערה הקודמת משמות רבה, ומוזכרים שם פסוקים נוספים בהקשר זה: "וכן הוא אומר: 'וְלֶחֶם סְתָרִים יִנְעָם' (משלי ט יז), וכן הוא אומר: 'עַד כִּכַּר לָחֶם וְאֵשֶׁת אִישׁ' וגו' (שם ו כו)". אולם גם בפסוקים אלו, על דרך הפשט מדובר בלחם כפשוטו; הפסוק "מַיִם גְּנוּבִים יִמְתָּקוּ וְלֶחֶם סְתָרִים יִנְעָם" אמנם מופיע שם בתוך דברי הפיתוי שאומרת "אֵשֶׁת כְּסִילוּת הֹמִיָּה" המבקשת לפתות עוברי דרך, אך הלחם אינו משמש בדבריה אלא כמשל, ולא ככינוי לזנות עצמה – וכשם שהמים אינם משמשים ככינוי לכך; הפסוק השני שהזכיר התנחומא, אומר: "כִּי בְעַד אִשָּׁה זוֹנָה עַד כִּכַּר לָחֶם, וְאֵשֶׁת אִישׁ נֶפֶשׁ יְקָרָה תָצוּד" – ומובן כי על פי פשוטו כוונת הדברים היא ש"בעבור חמדת אישה זונה בא האדם לרוב העוני עד כי ישאל כיכר לחם" ('מצודת דוד' שם).

פרק לט · פסוק ז

וַיְהִי אַחַר הַדְּבָרִים הָאֵלֶּה וַתִּשָּׂא אֵשֶׁת אֲדֹנָיו אֶת עֵינֶיהָ אֶל יוֹסֵף, וַתֹּאמֶר: שִׁכְבָה עִמִּי.

"וַיְהִי אַחַר הַדְּבָרִים הָאֵלֶּה"

חכמי המדרש (בראשית רבה מד ה) נחלקו בעניין משמעות ביטוי זה: "ר' יודן אמר: בכל מקום שנאמר 'אַחֲרֵי' – סמוך, 'אַחַר' – מופלג, ור' הונא אמר: בכל מקום שנאמר 'אַחַר' – סמוך, 'אַחֲרֵי' – מופלג"; כך שלפי ר' הונא האירוע שבפסוקנו היה סמוך למתואר לעיל – מינויי יוסף והתעלותו בבית פוטיפר, ואילו לפי ר' יודן הוא היה מופלג מהם, דהיינו שעבר מאז זמן רב.

מכל מקום, ביטוי זה מצביע על קיומו של קשר בין "הַדְּבָרִים הָאֵלֶּה" שתוארו לעיל לבין המסופר בענייננו. ניתן להבין מהותו של קשר זה מפנים שונים: מצד אשת פוטיפר, הרי ש'נשיאת עיניה' ליוסף היתה תוצאה של כל מה שתואר עד כה: התעלותו ממעמד עבד פשוט למעמד של בעל שררה, היותו איש מצליח, והיותו "יְפֵה תֹאַר וִיפֵה מַרְאֶה", וכפי שכתב שד"ל: "אחר שנתעלה יוסף ולא היה עוד כעבד נבזה, אז נשאה אליו עיניה והביטה אל יופיו ואהבתו"; וגם מצד יוסף – שבהיותו במעמד כזה הרי שגדל הניסיון והפיתוי, וכפי שכתב רבינו בחיי: "'וַיְהִי אַחַר הַדְּבָרִים הָאֵלֶּה' – להגיד מעלת יוסף, כי אחר שהשליטו האדון ו'שָׂמוֹ אָדוֹן לְבֵיתוֹ וּמֹשֵׁל בְּכָל קִנְיָנוֹ' (תהלים קה כא), כעניין הנזכר, והיה רואה עצמו אדון – תבעתו גבירתו, ולא קיבל".

"וַתִּשָּׂא אֵשֶׁת אֲדֹנָיו אֶת עֵינֶיהָ אֶל יוֹסֵף"

באופן פשוט, משמעות 'נשיאת עיניים' זו היא מבט של רצון וחמדנות, כפי שפירש 'בכור שור': "ושמה עליו עיניה וליבה, כי היה יוסף איש תואר [...] ותבער בה אהבתו".

אולם ייתכן שביטוי זה נושא כאן משמעות נוספת – מבט של התמקדות אובססיבית, דהיינו שכל מחשבותיה ומאוויה היו ממוקדים ביוסף ובציפייה לקבל ממנו את מבוקשה; וכך פירש ר' נפתלי הרץ ויזל:

זאת המליצה [-'נשיאת עיניים'] כוללת בלשון הקודש האהבה הגמורה העזה, שלא יחשוק הלב עניין אחר זולת אותו הדבר שהוא אוהב, ולא תשבע רצון עד שתשיג הדבר הנאהב. כמו "אֶשָּׂא עֵינַי אֶל הֶהָרִים מֵאַיִן יָבֹא עֶזְרִי, עֶזְרִי מֵעִם ה'" (תהלים קכא א-ב),227"אֵלֶיךָ נָשָׂאתִי אֶת עֵינַי הַיֹּשְׁבִי בַּשָּׁמָיִם" (שם קכג א); כלומר שאין עזר נחשב בעיניו זולתי כְּשֶׁיֵעָזֵר מיושב שמיים. והמשיל הדבר "הִנֵּה כְעֵינֵי עֲבָדִים אֶל יַד אֲדוֹנֵיהֶם כְּעֵינֵי שִׁפְחָה אֶל יַד גְּבִרְתָּהּ כֵּן עֵינֵינוּ אֶל ה' אֱלֹהֵינוּ עַד שֶׁיְּחָנֵּנוּ" (שם פס' ב), וידוע שאין לעבד כי אם רצון אדוניו, ואין לשפחה רק מה שתפיק מיד גברתה. וכן לעניין האהבה, אין לאוהבת אהבה עזה כזאת השקט והמנוחה, עד שתשיג רצונה מן האהוב לה.228

פרק לט · פסוקים ח–ט

וַיְמָאֵן, וַיֹּאמֶר אֶל אֵשֶׁת אֲדֹנָיו: הֵן אֲדֹנִי לֹא יָדַע אִתִּי מַה בַּבָּיִת וְכֹל אֲשֶׁר יֶשׁ לוֹ נָתַן בְּיָדִי. אֵינֶנּוּ גָדוֹל בַּבַּיִת הַזֶּה מִמֶּנִּי וְלֹא חָשַׂךְ מִמֶּנִּי מְאוּמָה כִּי אִם אוֹתָךְ בַּאֲשֶׁר אַתְּ אִשְׁתּוֹ, וְאֵיךְ אֶעֱשֶׂה הָרָעָה הַגְּדֹלָה הַזֹּאת וְחָטָאתִי לֵאלֹהִים.

"וַיְמָאֵן" – כתב ר' נפתלי הרץ ויזל:

כבר ביארנו ההבדל שבין "וימאן" ל"לא אבה",229 ואמרו שכל לשון "מיאון" יורה שתחפץ הנפש בדבר ההוא אבל נמנע מעשותו בעבור טענת השכל, ולשון "לא אבה" יורה שהנפש בטבעה [סולדת ממנו]. וכן כאן, אילו נאמר "ולא אבה" היינו שומעים שגעלה נפשו בה, שהיתה מכוערת וסרת טעם; עכשיו שנאמר "וימאן" ידענו שכפי טבע נפשו היה חושק בה, אלא נשמר מרע ביראת ה' ושכלו הטוב.

"וַיְמָאֵן, וַיֹּאמֶר אֶל אֵשֶׁת אֲדֹנָיו [...]" – באופן פשוט ניתן להבין שזהו כלל ופרט: הכתוב מציין, ראשית כל, שהוא מיאן; ולאחר מכן מפרט כיצד מיאן ומה היו דברי המיאון שאמר. אך ניתן גם לפרש שאלו עניינים שונים: הפועל "וַיְמָאֵן" מציין מיאון מוחלט, ולאחר מכן מובא הנימוק המפורט שאמר יוסף; וכך פירש האלשיך:

והנה יוסף התנהג בחכמה, והוא כי דרך אנשים בבוא רשע או אשת כסילות לפתותו לדבר עבירה והוא לא כן ידמה, כי יתן טענות נגד המפתה להשתיקו. אך לא זו הדרך להציל את נפשו מעשות רע, כי יקרה יהיה המפתה איש לשון, שבשפת חלקות ישחית דברי הנפתה הנעימים, ויילכד ברשתו. אך אשר מוח לו בקדקדו לא יעשה כן, כי אם מיד יחליט אומר למאן ולומר שלא יעשה בשום פנים גם כי ינוצח, ואחרי כן אם ירצה יסדר גם כן טענות נגד המפתה. וזה היה עניין יוסף כי ראשונה החליט "וַיְמָאֵן" כמי שקושר עצמו בשלשלת, ואחר כך אמר טענות.230

"וַיֹּאמֶר אֶל אֵשֶׁת אֲדֹנָיו" – בטעם הדבר שהזכיר אותה כאן הכתוב שנית בתואר "אֵשֶׁת אֲדֹנָיו" (ולא אמר "אליה", בהיסמך על איזכורה בפסוק הקודם), כתב הרמב"ן: "סיפר הכתוב כי מיאן לעשות כרצונה אף על פי שהיא גבירתו אשת אדוניו והוא ירא ממנה, כי היה ירא את השם יותר, וזה טעם 'אֶל אֵשֶׁת אֲדֹנָיו'". לפי פירושו, הרי שביטוי זה נועד לציין ניגוד: למרות שהיא "אֵשֶׁת אֲדֹנָיו", אמר לה כך.

אך ניתן לפרש זאת גם בכיוון הפוך: הוא אמר לה כך מפני שראה בה "אֵשֶׁת אֲדֹנָיו" – "אֵשֶׁת" איש, ולא איש סתם אלא "אֲדֹנָיו", פטרונו ומיטיבו; ומבט זה יתבטא בדברים שיאמר. וכך פירש ר' נפתלי הרץ ויזל:

זכר "אֵשֶׁת אֲדֹנָיו" לרמוז על שתי הרעות שעלו על לב הצדיק הזה שהיה עובר עליהן: האחד, שהיא אשת זולתו והשוכב עימה עושה עוון פלילי; והשני, שהיא אשת אדוניו שהיטיב עימו וגידלו, והוא צריך וחייב בכבודו וביראתו ולהכיר טובתו עליו. ושתיהן כלולים בדברי יוסף אלו.231

יוסף נושא נאום מפורט ומנומק: "הֵן אֲדֹנִי לֹא יָדַע אִתִּי מַה בַּבָּיִת" – כפי שנאמר לעיל: "וְלֹא יָדַע אִתּוֹ מְאוּמָה", אינו מפקח על ענייני הבית המנוהלים על ידי; "וְכֹל אֲשֶׁר יֶשׁ לוֹ נָתַן בְּיָדִי" – כפי שנאמר לעיל: "וְכָל יֶשׁ לוֹ נָתַן בְּיָדוֹ"; "אֵינֶנּוּ גָדוֹל בַּבַּיִת הַזֶּה מִמֶּנִּי" – כלומר: אין בבית הזה מי שהוא גדול ממני (מלבד האדון עצמו, כמובן), הכל נתון תחת ידי, וברשותי לעשות בכל דבר כאשר אראה לנכון – חוץ מדבר אחד: "וְלֹא חָשַׂךְ מִמֶּנִּי מְאוּמָה כִּי אִם אוֹתָךְ" – ואין כוונת הדברים לאזהרה מפורשת מפוטיפר ליוסף אודות אשתו, אלא שהדבר מובן מאליו – "בַּאֲשֶׁר אַתְּ אִשְׁתּוֹ", מעצם זה שאת אשתו הרי שאת מנועה ממני, וכפי שכתב ר' אברהם בן הרמב"ם: "אין טעמו שמנע אותו מזה והזהיר אותו על זה, אלא טעמו שקנאתו על זה ידועה אע"פ שלא פירש אותה, שהרי השכל והטבע והחוק מונעים מזה".

שני הפסוקים מתמקדים בהיבטים שונים: פסוק ח מתמקד במעילה באמונו של האדון – "הֵן אֲדֹנִי לֹא יָדַע אִתִּי מַה בַּבָּיִת, וְכֹל אֲשֶׁר יֶשׁ לוֹ נָתַן בְּיָדִי" – הוא נותן בי את אמונו וסומך עליי, ויהיה זה מעשה נבלה אם אנצל אמון זה עצמו למעול בו ולפעול בניגוד לרצונו.232פסוק ט מתמקד בתחילתו בכך שאין זה ראוי שדווקא את הדבר היחיד שהאדון מנע מיוסף, בעוד ששאר נכסיו מסורים לרשותו באופן מוחלט, ייקח יוסף מרשותו: "אֵינֶנּוּ גָדוֹל בַּבַּיִת הַזֶּה מִמֶּנִּי, וְלֹא חָשַׂךְ מִמֶּנִּי מְאוּמָה כִּי אִם אוֹתָךְ"; ועל כך הוא מוסיף, כי גם לולא זאת שהיא הדבר היחיד שאדונו מנע ממנו – הרי שהיא מנועה ממנו מצד עצמה, "בַּאֲשֶׁר אַתְּ אִשְׁתּוֹ".

את כל זאת מסכם יוסף במילות הסיום: "וְאֵיךְ אֶעֱשֶׂה הָרָעָה הַגְּדֹלָה הַזֹּאת" – לאור כל מה שאמרתי, הרי מובן שזוהי רעה גדולה; ואם כן, איך אוכל לעשותה?

על כך הוא מוסיף: "וְחָטָאתִי לֵאלֹהִים". הוספה זו ניתנת להבנה בשני אופנים: האחד, שזהו נימוק נוסף – מלבד הרעה כלפי האדון, הרי שזהו גם חטא כלפי אלוהים מצד איסור עריות של אשת איש; והשני, שאין זו אלא מסקנת כל האמור עד כה – דהיינו: אותה הרעה הגדולה כלפי האדון, הרי שהיא חטא גם כלפי אלוהים.

רש"י פירש כאופן הראשון: "'וְחָטָאתִי לֵאלֹהִים' – בני נח נצטוו על העריות"; וכתב על כך הרמב"ן: "ונכון הוא; רק בעבור חסרון דעת הנשים הקדים אליה כי הדבר בגידה באדוניו אשר בוטח בו, ואחרי כן אמר כי יש בעניין עוד חטא לאלוהים". הרמב"ן מסביר מדוע מקדים יוסף את עניין החטא לאדוניו לפני עניין החטא לאלוהים – שכן היות הדבר "בגידה באדוניו אשר בוטח בו" מהווה ההסבר העיקרי, באשר הוא שייך למישור השכלי-אנושי ועשוי להיות מובן לאשת פוטיפר יותר מן הנימוק הדתי. כך פירש גם ר' אברהם בן הרמב"ם:

והקדים במאמרו את צד הנימוס [=החוק והסדר החברתי] ופירש אותו לה וביאר את עניינו, ואיחר את צד העבירה וקיצר את המאמר בעניינה, לפי שזה היה מתאים יותר עם דיבורו עימה – "עֲנֵה כְסִיל כְּאִוַּלְתּוֹ" (משלי כו ה), לפי שהנימוס הוא ודאי טבעי ומצוי אפילו אצל הכופרים, ולכן דחה אותה תחילה בצד הנימוס וביאר לה שהוא משגיח בו, והיא גם היא ראוי לה להשגיח בו. שוב הזכיר לה את צד הדת שהוא דבוק בה, הגם שהיא היתה מובדלת מלהיות דבוקה בה.

בהמשך דבריו, מזכיר הרמב"ן את האופן השני: "וייתכן לפרש: 'וְחָטָאתִי לֵאלֹהִים' בבגידה הזאת, כי רעה גדולה היא שיהיה בה לי חטא לפני האלוהים, כי עיניו בנאמני ארץ, ולא לפניו בוגד יבוא, ודברו אמת, רק לא הזכיר איסור הערווה כי דיבר עימה כדרך הנשים". לפי הבנה זו, הרי שיוסף לא הזכיר כלל את האיסור הדתי, שלא היה שייך לעולמה של שומעת הדברים, אלא שהוסיף לומר כי רעה זו שכלפי אדונו הרי היא גם חטא כלפי אלוהים – ובכך גדלים גם חומרת המעשה כשלעצמו, וגם החשש פן ייענש על כך; "ואפילו הייתי מאותם שאינם משגיחים בנימוס, היתה יראת החטא ועונשו יתעלה מונעת מזה" (ר' אברהם בן הרמב"ם).

מפשט לשון הכתוב משתמע יותר כאופן השני, שהמילים "וְחָטָאתִי לֵאלֹהִים" אינן מהוות דבר נוסף אלא תוצאה ומסקנה של כל מה שהוסבר קודם לכן – "וְאֵיךְ אֶעֱשֶׂה הָרָעָה הַגְּדֹלָה הַזֹּאת, וְ[בכך] חָטָאתִי לֵאלֹהִים"; אולם אינו מן הנמנע לפרש שאכן מילים אלו כוללת הן את מימד החטא שבבגידה באדוניו והן את מימד חטא העריות.

פרק לט · פסוק י

וַיְהִי כְּדַבְּרָהּ אֶל יוֹסֵף יוֹם יוֹם וְלֹא שָׁמַע אֵלֶיהָ לִשְׁכַּב אֶצְלָהּ לִהְיוֹת עִמָּהּ.

פסוקנו והפסוק שלאחריו מהווים רצף אחד; פסוקנו מציין את הרקע למאורע, והפסוק הבא מתחיל לתאר את המאורע עצמו; ושיעור הכתוב: "וַיְהִי כְּדַבְּרָהּ אֶל יוֹסֵף יוֹם יוֹם וְלֹא שָׁמַע אֵלֶיהָ לִשְׁכַּב אֶצְלָהּ לִהְיוֹת עִמָּהּ" – כלומר, על רקע זאת שדיברה אליו יום יום ולא שמע אליה, התרחש האירוע הבא: "וַיְהִי כְּהַיּוֹם הַזֶּה, וַיָּבֹא הַבַּיְתָה" וכו'.233

הקדמת רקע זה נחוצה עבור הבנה מלאה של המאורע עצמו, הן מצד הזווית של יוסף והן מצד הזווית של אשת פוטיפר: מצד הזווית של יוסף, שכן מכך אנו למדים שקדמה למאורע תקופה ממושכת של הפצרות שמתחו את החבל עוד ועוד, ובכך הניסיון שבאותו המאורע המתואר להלן היה קשה ומאתגר יותר עבור יוסף, באשר נשא עימו את כל כובד משקלה של התקופה שקדמה לה; ומצד הזווית של אשת פוטיפר, שכן בכך מובן כיצד הגיעה לנקוט בצעד כה דרמטי של "וַתִּתְפְּשֵׂהוּ בְּבִגְדוֹ" – לאחר שניסתה "יוֹם יוֹם" לפתות את יוסף בדברים, נואשה מדרך זו ועברה לדרך בוטה יותר.

"וְלֹא שָׁמַע אֵלֶיהָ לִשְׁכַּב אֶצְלָהּ לִהְיוֹת עִמָּהּ" – אין הכתוב אומר "לשכב עימה", כדרך שאמרה היא "שִׁכְבָה עִמִּי", אלא "לִשְׁכַּב אֶצְלָהּ, לִהְיוֹת עִמָּהּ", ומלשון זו משמע שהכתוב מלמד כי אפילו להצעה הפחותה מדרישתה המקורית לא שמע אליה; וכפי שכתב הרמב"ן: "טעם 'לִשְׁכַּב אֶצְלָהּ' – כדברי ר' אברהם [אבן עזרא]: אפילו שישכב אצלה בסמוך, היא בבגדה והוא בבגדו, או 'לִהְיוֹת עִמָּהּ' – לדבר שיחות. כי לא מצאנו 'אצלה' על המשגל, רק 'עימה' או 'אותה': 'שִׁכְבָה עִמִּי', 'וְאִם שָׁכֹב יִשְׁכַּב אִישׁ אֹתָהּ' (ויקרא טו כד), 'וְהַנָּשִׁים 'תִּשָּׁכַבְנָה' (זכריה יד ב)". נראה להסביר זאת בכך שהמילה 'אצל' מורה על סמיכות המקום כשלעצמה, בעוד שהמילה 'עם' מצביעה בהקשר זה על אינטראקציה בין השניים.

בכך הכתוב מזכיר את שתי ה'פשרות' שהיא הציעה לו, מתוך תקווה שמימושן יתפתח מעבר להן עצמן, ואת סירובו לשתיהן – ואין צריך לומר שסירב לדרישה המקורית "שִׁכְבָה עִמִּי".234

ניתן לשים לב לכך שהכתוב מפצל לשניים את הביטוי המקורי שאמרה – "שִׁכְבָה עִמִּי": "לִשְׁכַּב אֶצְלָהּ, לִהְיוֹת עִמָּהּ", כשכל אחד מהשניים נושא את אחד המרכיבים שלו; האחד, "לִשְׁכַּב אֶצְלָהּ" אך לא "עִמָּהּ", והשני, "לִהְיוֹת עִמָּהּ", אך בלי "לִשְׁכַּב", דהיינו "לדבר שיחות" – להיות יחד ולנהל שיחה במצב אינטימי. פיצול המרכיבים הלשוניים משקף את פיצול הפונקציות המבוקשות: "לִשְׁכַּב אֶצְלָהּ" היינו לשכב לידה בקרבת מקום גם ללא אינטראקציה ביניהם, ואילו "לִהְיוֹת עִמָּהּ" היינו לקיים איתה אינטראקציה אינטימית גם מבלי לשכב; כל אחת מהשתיים נושאת מטען אינטימי אחר, והיתה עשויה להגיע להתפתחות נוספת – אשר מפני כך דרשה זאת היא, ומאותה סיבה עצמה מיאן בכך הוא.235

רבינו בחיי פירש באופן אחר: "'וְלֹא שָׁמַע אֵלֶיהָ לִשְׁכַּב אֶצְלָהּ' – על דרך הפשט: אפילו בלא תשמיש; 'לִהְיוֹת עִמָּהּ' בתשמיש, שכן מצינו ביאה בלשון הוויה, כעניין שכתוב 'גַּם הָיִיתִי הַלַּיְלָה לְאִישׁ' (רות א יב)". לפי פירושו, הרי שהכתוב מנוסח בדרך 'זו, ואין צריך לומר זו': "'וְלֹא שָׁמַע אֵלֶיהָ לִשְׁכַּב אֶצְלָהּ", וכל שכן "לִהְיוֹת [=לשכב] עִמָּהּ".

אולם ישנו קושי בהנחה הלשונית שעליה הוא מתבסס, ש"מצינו ביאה בלשון הוויה", שכן בפסוק "גַּם הָיִיתִי הַלַּיְלָה לְאִישׁ" שאומרת נעמי לכלותיה משמעותם הפשוטה של הדברים היא "נישאתי לאיש", ואין מדובר על המשכב עצמו – שהרי לא נאמר "הייתי הלילה עם איש", כפי שנאמר בפסוקנו "לִהְיוֹת עִמָּהּ", אלא "לְאִישׁ" – נשואה לאיש, וכלשון המקובלת במקרא "וַתְּהִי לוֹ לְאִשָּׁה" (לעיל כד סז, רות ד יג, ועוד), והרי זו הוויה במובן של שייכות ונישואין, ואין מכך ראיה לפסוקנו. ראיה אפשרית נוספת שלא הזכיר בדבריו היא מדברי אבשלום לתמר "הַאֲמִינוֹן אָחִיךְ הָיָה עִמָּךְ" (שמואל ב' יג כ), אך גם שם נראה שאין זו משמעות מילולית אלא ניסוח מרומז ומעודן, שעשוי להיות שימושי בלשון בני אדם; אך בדברי הכתוב עצמו לא מצינו שיתבטא כך.

פרק לט · פסוק יא

וַיְהִי כְּהַיּוֹם הַזֶּה וַיָּבֹא הַבַּיְתָה לַעֲשׂוֹת מְלַאכְתּוֹ, וְאֵין אִישׁ מֵאַנְשֵׁי הַבַּיִת שָׁם בַּבָּיִת.

"וַיְהִי כְּהַיּוֹם הַזֶּה"

ביטוי זה הינו יחידאי במקרא ומשמעותו אינה ברורה דיה; מכל מקום נראה שזוהי הקדמה שנועדה לציין את המעבר מתיאור של שגרה חוזרת לתיאור אירוע שהתרחש ביום מסויים. בפסוק הקודם נאמר "וַיְהִי כְּדַבְּרָהּ אֶל יוֹסֵף יוֹם יוֹם וְלֹא שָׁמַע אֵלֶיהָ לִשְׁכַּב אֶצְלָהּ לִהְיוֹת עִמָּהּ", וכמבואר שם הוא נסמך על פסוקנו ונועד להוות רקע עבורו; ולאחר הקדמת הרקע מתחיל הכתוב בתיאור האירוע עצמו, בפתיחה "וַיְהִי כְּהַיּוֹם הַזֶּה".

רש"י פירש שביטוי זה מציין את היות אותו היום יום בעל ייחודיות – שהיא זו שהכשירה את הקרקע למה שיתרחש בהמשך: "'וַיְהִי כְּהַיּוֹם הַזֶּה' – כלומר: ויהי כאשר הגיע יום מיוחד, יום צחוק, יום איד שלהם שהלכו כולם לבית עבודה זרה, אמרה: אין לי יום הגון להזקק ליוסף כהיום הזה, אמרה להם: חולה אני ואיני יכולה לילך". דבריו מבוססים על האמור בתלמוד (סוטה לו ע"ב) ובמדרשים (בראשית רבה פז ז, תנחומא ט), אך שם מופיעים הדברים כהסבר ריאלי לכך ש"אֵין אִישׁ מֵאַנְשֵׁי הַבַּיִת שָׁם בַּבָּיִת" ולא כהסבר לשוני למילים "וַיְהִי כְּהַיּוֹם הַזֶּה"; אולם רש"י מרחיב את השימוש בפירוש זה, כך שיסביר גם את הלשון שבפתיחת הפסוק.

"וַיָּבֹא הַבַּיְתָה לַעֲשׂוֹת מְלַאכְתּוֹ"

לשון זו נוסכת על הפסוק אווירה של שגרה סתמית: יוסף בא לבית פוטיפר לעשות מלאכתו כמדי יום ביום, ואינו נותן דעתו כלל על אשת האדון – ואז, נכפית עליו בהפתעה הסיטואציה שתתואר בפסוק הבא: "וַתִּתְפְּשֵׂהוּ בְּבִגְדוֹ לֵאמֹר שִׁכְבָה עִמִּי".

נראה שאותו המקום בבית שאליו בא יוסף היה נפרד ורחוק מחדרה, כפי המשתמע מלשונות הכתובים, וכדברי ר' נפתלי הרץ ויזל:

"וַיָּבֹא הַבַּיְתָה" – רחוק מחלק הבית שהיא יושבת שם עם אנשיה; שכפי מנהגם היה בית הגבירה ואנשיה לבד, ובית האדון לבד, וכמו שעד היום המנהג כן בכמה ארצות; ויוסף בא הביתה שהוא יושב בו [...], "וְאֵין אִישׁ מֵאַנְשֵׁי הַבַּיִת שָׁם בַּבָּיִת" – ידקדק הכתוב לומר "שָׁם בַּבָּיִת", כלומר: שם בחלק הבית שבא יוסף לשם, אבל בחלק הבית שהיא יושבת בו היו אנשיה אִתָּהּ, כמו שאמר למטה בעניין "וַתִּקְרָא לְאַנְשֵׁי בֵיתָהּ".236

בתלמוד (סוטה לו ע"ב) מובאת מחלוקת בעניין הבנת מילים אלו:

"וַיָּבֹא הַבַּיְתָה לַעֲשׂוֹת מְלַאכְתּוֹ" – אמר רבי יוחנן: מלמד ששניהם לדבר עבירה נתכוונו ["והאי 'מְלַאכְתּוֹ' – תשמיש" (רש"י)]. "וַיָּבֹא הַבַּיְתָה לַעֲשׂוֹת מְלַאכְתּוֹ" – רב ושמואל, חד אמר: לעשות מלאכתו ממש, וחד אמר: לעשות צרכיו נכנס ["תשמיש" (רש"י)]".

לפי דעת ר' יוחנן והאומר "לעשות צרכיו", הרי שהפסוק מתפרש בכיוון הפוך לחלוטין מזה שעל דרך הפשט, כך ש"לַעֲשׂוֹת מְלַאכְתּוֹֹ" הוא לשון נקייה למעשה העבירה; ואם כן, הרי שעמידתו האיתנה של יוסף הגיעה בשלב זה לנקודת שבירה, וסיוע אלוהי מיוחד הוא שסייע לו להתגבר על יצרו, כפי שמתואר שם בהמשך הדברים: "'וַתִּתְפְּשֵׂהוּ בְּבִגְדוֹ לֵאמֹר' וגו', באותה שעה באתה דיוקנו של אביו ונראתה לו בחלון, אמר לו: יוסף, עתידין אחיך שייכתבו על אבני אפוד ואתה ביניהם; רצונך שיימחה שמך מביניהם ותיקרא רועה זונות?! דכתיב 'וְרֹעֶה זוֹנוֹת יְאַבֶּד הוֹן' (משלי כט ג)".

תיאור זה מנוסח בסגנון דרשני, אך הוא עשוי לשפוך אור על הבנת הרקע של הסיפור גם ברובד הפשט, ולחשוף תהליכי עומק בנפשו של יוסף, שהתרחשו במשך תקופה זו מאז נמכר למצרים. נראה כי חלק מכובד המשקל שהקשה עליו לעמוד בניסיון נבע מהריחוק והניתוק ממשפחתו; המחוייבות למוסר של בית אביו היא שמנעה אותו מלהתפתות במשך כל אותם הימים, אך כנגדה עמד המשקל של הניתוק ותחושת הבגידה שהיה יוסף עשוי לחוש כלפי בית אביו, המייצג את עברו והרקע שממנו בא. שני אלו עמדו על כף המאזניים, ובנקודת השבר – כשגברה תחושת הניתוק שהובילה לכך שיוסף נכנע והסכים להיענות לדרישת גבירתו – הופיעה לעיניו דמות דיוקנו של אביו, ועלתה בו הידיעה העמוקה על כך שלמרות מעשה הבגידה שחווה מצד אחיו, ולמרות ריחוקו ממשפחתו, שלכאורה יימשך כל חייו – מכל מקום הרי הוא עדיין אחד מהם, ואף לו יהיה חלק בעתיד הנצחי שנכון לשנים עשר שבטי ישראל. מתחת לפני השטח הקודרים, מאחורי המציאות הריאלית שגוזרת עליו חיים של עבדות נצחית, ללא יכולת לחבור אל משפחתו ואף להקים משפחה משלו – עולה בתודעתו הבנה עמוקה ונסתרת על תקווה רחוקה שממתינה לו – אשר אם יאמין בה ויתגבר על הפיתוי הזמני, יזכה שיהיה שמו חקוק לנצח באבני החושן, כאחד שבטי ישראל.

מבחינת תוכן הסיפור, הרי שגישה זו מוסיפה מימד של מורכבות לסיפור התמודדותו של יוסף; היא מציגה גם רגעי חולשה אנושית שלו, ומאידך – גם רבדי עומק במודעותו, שצפו ועלו לנגד עיניו באותו הרגע וגרמו לו לשוב לאיתנו. אך מבחינת ניתוח לשון הכתוב ומבנהו, ישנו קושי לקבל פרשנות זו כפשט הכתוב; הן מצד משמעותן הפשוטה של המילים "לַעֲשׂוֹת מְלַאכְתּוֹ", והן מצד התפתחות ההתרחשות – שכן אם נפרש שלשם כך הוא בא הביתה, כפי שעולה מפסוקנו לפי פרשנות זו, הרי שהיה נכון לכך מרצונו, ואם כן מדוע נצרכה גבירתו לנקוט באלימות, כמתואר בפסוק הבא: "וַתִּתְפְּשֵׂהוּ בְּבִגְדוֹ לֵאמֹר שִׁכְבָה עִמִּי"? ואף כי מבחינה ריאלית ניתן להבין זאת בכך שעשתה כן בטרם זיהתה את כוונותיו, מבחינה ספרותית הרי שהכתוב אינו בונה את הסצנה כאילו יוסף נכנס לשם מטרה זו – שכן הפסוק הבא מציג את תפיסתה של אשת פוטיפר כפעולה יזומה ופתאומית כלפיו.

אם כן, הרי שברובד הפשט ניתן לקבל את ההנחה כי באותה סיטואציה היה גם רגע של שבר שבו יוסף כמעט והתפתה – אך אין הוא מוזכר בכתוב בפירוש. שילוב של שני הפירושים למילים "לַעֲשׂוֹת מְלַאכְתּוֹ", עשוי ליצור את ההבנה שכאשר בא יוסף הביתה לא היה בכוונתו אלא לעסוק במלאכתו השגרתית – אולם במהלך המשך הסיטואציה הופיע גם שלב של "לעשות צרכיו", והוא רמוז בפרשנות הדרש למילים אלו.

"וְאֵין אִישׁ מֵאַנְשֵׁי הַבַּיִת שָׁם בַּבָּיִת"

ייתכן שבמקרה אירע כך, ש"אֵין אִישׁ מֵאַנְשֵׁי הַבַּיִת שָׁם בַּבָּיִת", כפי שכתב רשב"ם (בפס י'): "עד שאירע מעשה שנשאר יחידי בעל כרחו לעשות מלאכת צורכי הבית כדרכו, ואירע באותו יום שלא נותר איש בבית"; וייתכן גם, כפי שהובא לעיל מדברי רש"י על פי חז"ל, שהיתה סיבה מיוחדת לכך שבאותו היום התרוקן הבית מאנשיו – ולפי זה, הרי שאשת פוטיפר צפתה זאת מראש ותכננה את פעולתה בהתאם לכך, על מנת לנצל את ההזדמנות.

וכך נאמר בבראשית רבה (פז ז): "אפשר ביתו של אותו האיש משתייר בלא איש?! רבי יהודה ורבי נחמיה, רבי יהודה אומר: יום ניבול [-כינוי לעבודה זרה]237 של נילוס היה, והלכו הכל לראות ['שהלכו לראות נילוס נהר מצרים שעלה על כל גדותיו' (רשב"ם כאן)], והוא לא הלך. ורבי נחמיה אמר: יום תיאטירון היה, והלכו הכל לראותו, והוא לא הלך, אלא 'וַיָּבֹא הַבַּיְתָה לַעֲשׂוֹת מְלַאכְתּוֹ', לחשב חשבונות של רבו". מעבר לכך שתיאור זה מספק הסבר להיעדרות אנשי הבית, הוא גם מדגיש את נאמנותו של יוסף לאלוהיו ולאדוניו – מתיאורו של ר' יהודה על "יום ניבול של נילוס" ניתן ללמוד על התרחקות יוסף מאירועי פולחן אליליים, הגם שאלו היו נחלתם של כל סובביו; ומתיאורו של ר' נחמיה על "יום תיאטירון" ניתן ללמוד על נאמנותו לאדוניו ולעבודתו, שאף כאשר הולכים הכל לראותו – הוא דבק בשגרת חובותיו, "לחשב חשבונות של רבו".

מהמשך הסיפור עולה סתירה לכאורה לאמור בפסוקנו "וְאֵין אִישׁ מֵאַנְשֵׁי הַבַּיִת שָׁם בַּבָּיִת", שהרי נאמר "וַתִּקְרָא לְאַנְשֵׁי בֵיתָהּ" (פס' יד) – הרי שהיו שם אנשים בבית באותו הזמן. לפי הפשט ניתן ליישב סתירה זו בכך שאנשי ביתה אכן היו בבית, אך לא באותו חלק של הבית שבו היו היא ויוסף, שעליו נאמר "וְאֵין אִישׁ מֵאַנְשֵׁי הַבַּיִת שָׁם בַּבָּיִת" (וכדברי ר' נפתלי הרץ ויזל שהובאו לעיל); אך לדברי המדרש ש"הלכו הכל לראותו" – הרי שלכאורה לא היו אנשים בבית כלל. ניתן ליישב זאת ולומר שאכן כל אנשי הבית הלכו, ורק משרתיה האישיים נשארו כדי לשמשה, ובפרט לפי המתואר בתלמוד (שם) בהקשר זה, ש"היא אמרה להן חולה היא", ועליהם נאמר "וַתִּקְרָא לְאַנְשֵׁי בֵיתָהּ".

פרק לט · פסוק יב

וַתִּתְפְּשֵׂהוּ בְּבִגְדוֹ לֵאמֹר שִׁכְבָה עִמִּי, וַיַּעֲזֹב בִּגְדוֹ בְּיָדָהּ וַיָּנָס וַיֵּצֵא הַחוּצָה.

"וַתִּתְפְּשֵׂהוּ בְּבִגְדוֹ" – אשת פוטיפר מנצלת את ההזדמנות ש"אֵין אִישׁ מֵאַנְשֵׁי הַבַּיִת שָׁם בַּבָּיִת", והולכת צעד נוסף בניסיונותיה מול יוסף – כעת אינה מסתפקת באמירה אלא עוברת למעשה, בניסיון לכפות עליו את רצונה. מדוע, אם כן, היא תופסת רק בבגדו ולא בו עצמו? ייתכן שאכן תפסה בו, אך הוא נחלץ מאחיזתה ורק בגדו נשאר בידה, והכתוב הזכיר רק את התפיסה בבגדו כיון שרק היא רלוונטית להבנת הסיפור, שכן בגדו הוא שנעזב בידה לבסוף. ייתכן גם שלא רצתה להתחיל במעשה בוטה ואלים מדי, ולפיכך תפסה רק בבגדו כשלב ראשון, המרמז לו שהפעם לא תרפה ממנו, אך הוא התחכם לה ועזב בגדו בידה.238

כהמשך לדבריו שהובאו בפסוק הקודם, שההתרחשות אירעה בחלק אחר מן הבית ולא בחדרה של הגברת, כתב כאן ר' נפתלי הרץ ויזל: "הכתוב מקצר וסומך על המובן מן העניין; ידענו כי יוסף לא הלך אצלה, שכבר נאמר 'וְלֹא שָׁמַע אֵלֶיהָ לִשְׁכַּב אֶצְלָהּ לִהְיוֹת עִמָּהּ'; ועוד, כי שם אנשי ביתה, וכמו שפירשנו. אבל היא הלכה אל מקום-הבית המיוחד ליוסף, ולא יָרְאָה, כי 'אֵין אִישׁ מֵאַנְשֵׁי הַבַּיִת שָׁם בַּבָּיִת'".

"לֵאמֹר שִׁכְבָה עִמִּי" – ניתן להבין שאמרה כך בפירוש, ו"לֵאמֹר" היינו שזוהי אמירה שליוותה את המעשה, וניתן גם להבין שעצם המעשה הרי הוא כבר 'אומר' כך,239 כפירוש 'אור החיים': "התורה מגדת כוונתה בתפיסת הבגד, שכוונתה 'לֵאמֹר' וגו', אבל היא לא אמרה כלום אלא שלחה ידה לתופסו להכריחו קום עשה"; ומיד כשתפסתו והבין את מטרתה – עזבו בידה ונס החוצה, "שלא רצה להשהות בבית עד שאמרה, אלא כיון שראה שתפסתו בבגדו הבין שרצונה לאמר לא הספיק להמתין עד שאמרה, ועזב בגדו בידה [...], שזה עניין התגברות על יצרו להיות בורח במהירות האפשרי כדי שלא יגבור עליו יצרו" ('פנים יפות').240

"וַיַּעֲזֹב בִּגְדוֹ בְּיָדָהּ" – כתב הרמב"ן: "לכבוד גבירתו לא רצה להוציאו מידה בכחו הגדול ממנה, והסיר אותו מעליו, כי היה בגד [עליון ש]יתעטף בו כמעיל וצניף". אולם ניתן גם להסביר שאף אם היה רוצה להוציאו מידה, הרי שהדבר היה כרוך בעיכוב, בעוד שהוא ביקש להימלט מהסיטואציה מיד; ונראה, שמבחינה ספרותית זוהי מגמת הכתוב בתיאור זה: הוא לא התעכב לחלץ מידה את בגדו כפי שהיה מתבקש לכאורה, אלא נקט בפעולה מפתיעה – הוא עזב בידה את מה שתפסה, בעוד שהוא עצמו נס החוצה. פעולה זו ממחישה את המיידיות של מנוסת יוסף, באשר חשש פן כל עיכוב עלול לאפשר לה להתגבר עליו יותר, או פן ייכנע ליצרו. טקטיקת ההתמודדות של יוסף בנקודה מכריעה זו נשענה על עיקרון המידיות – מבלי לתת לעצמו שהות להתפתות, הוא 'חתך' את המצב המאתגר – ונס החוצה.

"וַיָּנָס וַיֵּצֵא הַחוּצָה" – בעניין היחס בין שני פעלים אלו ניתן להסביר, שהפועל "וַיָּנָס" מציין את המנוסה מהסיטואציה עצמה, שהיתה בפועל יציאה מאותו החדר, ואילו הפועל "וַיֵּצֵא [הַחוּצָה]" מוסיף על כך שגם יצא מן הבית כולו. מנוסתו של יוסף מוזכרת עוד שלוש פעמים נוספות בהמשך הפסוקים: באחת מהן נאמר כמו כאן: "וַיָּנָס וַיֵּצֵא הַחוּצָה", ובשתיים האחרות נאמר רק "וַיָּנָס הַחוּצָה" – וראה להלן בביאור לפסוק יח ביאור עניין זה.241242

הסמליות שבמוטיב הבגד

כפי שהתבאר בפסוק הקודם, הניסיון שעמד כאן בפני יוסף נשא משקל רב, מעבר לאירוע הנקודתי שעמד בו; היתה זו נקודת מבחן והכרעה בין צורות הסתכלות שונות של יוסף על עצמו, על זהותו ועל גורלו: האם עודנו רואה את עצמו כשייך לבית אביו, לעתיד הנצחי של עם ישראל, למוסרו ולקדושתו, או שמא יראה עצמו כמנותק מכל אלו, כמי שאבדה תקוותו ונגזר מארץ חיים. תחושות הבגידה והריחוק ממשפחתו היו עשויות ליצור בו ניכור כלפי בית אביו וכל אשר הוא מייצג, ולהפנות אליו עורף – גם אם לא מתוך בחירה מודעת, מתוך רפיון ידיים מול עוצמת הפיתוי.

על רקע זה, ניתן לראות במוטיב הבגד – העומד במרכז הסיטואציה – משמעות סמלית: הבגד מכסה את הגוף, והרי הוא 'בוגד' ומתעתע בעיני הרואה; יוצר הוא מצג של חזות חיצונית המסתירה את האמת הפנימית בטהרתה. אשת פוטיפר תופסת ב'בגדו' של יוסף – כביכול היא נאחזת באותה הסתכלות חיצונית על המציאות, שמבעדה עשוי יוסף לראות את עצמו כנבגד ודחוי השקוע במצולות הייאוש, ומכוח זאת מבקשת למשכו אליה, "לֵאמֹר שִׁכְבָה עִמִּי", ולפתותו לבגוד באלוהיו ובאדוניו, בזהותו ובקדושתו.

אולם יוסף עוזב את בגדו בידה – הוא מסיר מעליו את המעטה החיצוני והמטעה, המוכתב על ידי תחושות המרירות, התסכול והנבגדות, ואינו מניח לו להגדיר את זהותו ואת גורלו; דבק הוא במהותו הפנימית, ומתוך כך "וַיָּנָס וַיֵּצֵא הַחוּצָה" – הוא מתנתק מצורת ההסתכלות החיצונית והשגויה, ויוצא "הַחוּצָה" – אל מרחבי האמת הפשוטה, אל זהותו האמיתית, ואל מחוייבותו לדבר ה'.243

  1. 238

    ר' נפתלי הרץ ויזל כתב: "ומיד העיזה פניה לתופשו בבגדו להסירו מעליו", ובכך מיושבת השאלה דלעיל. אך ישנו קושי בפירושו, שכן לפיו היה מתבקש שייכתב "ותתפוש בבגדו", שמכך משתמע כי מטרת התפיסה היתה הבגד; מה שאין כן לגבי האפשרויות הקודמות, שלפיהן מובן מדוע לא נכתב כך אלא "וַתִּתְפְּשֵׂהוּ בְּבִגְדוֹ", שכן מטרתה לא היתה הבגד אלא יוסף: היא תפסה ביוסף באמצעות התפיסה בבגדו.

  2. 239

    אפשרויות דומות הוזכרו לעיל לגבי הפסוק "וַתִּקְשֹׁר עַל יָדוֹ שָׁנִי לֵאמֹר זֶה יָצָא רִאשֹׁנָה" (לח כח).

  3. 240

    לפי דברי המפרשים הללו, יוסף נס מיד כשהבין את משמעות הסיטואציה. אך אין הכרח להבין כך – ייתכן שאשת פוטיפר אמרה בפועל "שִׁכְבָה עִמִּי", ולאחר מכן עוד היו כמה רגעי שהייה עד שנמלט יוסף. לעיל, בביאור המילים" וַיָּבֹא הַבַּיְתָה לַעֲשׂוֹת מְלַאכְתּוֹ" שבפסוק הקודם, הועלתה האפשרות (על פי דברי התלמוד) שיוסף חווה גם נקודת שבירה במהלך אירוע זה, וכמעט שהתפתה; ולפי זה, עלינו לומר שבין רגע התפיסה בבגדו לבין החלטתו להימלט עברו כמה רגעים הרי גורל של מאבק פנימי עז – ובסופם, משגמלה בליבו ההחלטה לסרב ולהימלט, ביצע זאת במיידיות, כפי שמתאר כאן הכתוב, וכדלהלן.

  4. 241

    הספורנו פירש זאת באופן אחר: "בהיותו חוץ מן החדר יצא לאיטו בלתי תנועת ניסה, שלא ישאלוהו: מה לך כי תנוס, ומי רודפך". כלומר, עד פתח הבית יוסף נס במהירות, אך יציאתו עצמה היתה בנחת, פן ייראה בחוץ באופן מחשיד; לפיכך, אומר הכתוב: "וַיָּנָס" – עד פתח הבית, "וַיֵּצֵא הַחוּצָה" – בנחת. וראה בדבריו כיצד פירש לפי שיטתו את שלושת האיזכורים האחרים.

  5. 242

    ר' נפתלי הרץ ויזל כתב: "'וַיָּנָס וַיֵּצֵא הַחוּצָה' – לא שפחד פן תטהו ברוב לקחה, אבל ידע כי מרוב זנוניה תוסיף לדבר אליו דברי תחנונים ויצטרך להשיב לה, ובין כה יבואו האנשים הביתה, ויראו כי גבירתו עימו בחדרו, ויתגלה הדבר ויהי לו לקלון ולסכנה בעיני אדוניו. לכן נס ויצא החוצה". אולם מפשט הכתוב לא משמע כך, שכן אם היתה סבירות לכך שבכל רגע יופיע שם אדם, הרי שהנתון "וְאֵין אִישׁ מֵאַנְשֵׁי הַבַּיִת שָׁם בַּבָּיִת" אינו כה משמעותי, בעוד שמכך שהכתוב מציין זאת נראה שהאירוע התבסס על פרטיות מובטחת, ולא היה אדם עשוי להופיע שם. גם מבחינה ספרותית, פירושו למניע של יוסף במנוסתו מסיט את ההתמקדות מהתגברות יוסף על גבירתו ועל יצרו לשיקול צדדי שאינו הציר המרכזי של הסיפור. משכך, נראה יותר שאכן יוסף חשש פן ייכשל, ולפיכך נס במיידיות מיד גבירתו.

  6. 243

    בהקשר זה ניתן להוסיף, כי גם הבגד כשלעצמו משמש בסיפורו של יוסף כאמצעי ליצירת מצג שווא, ה'בוגד' בָּאֱמֶת ומסלף אותה: אחי יוסף הביאו לאביהם את כתונת הפסים כשהיא טבולה בדם, ובכך הציגו אותו כמי שנטרף, ובהמשך הסיפור – אשת פוטיפר תשתמש בבגדו של יוסף כראיה המוכיחה כביכול את גירסתה ומציגה את יוסף כמי שבא לשכב עם אשת אדוניו. בשני המקומות, הבגד משמש ליצירת תדמית שקרית ליוסף; ובענייננו, נקודת המבחן שעמדה בפניו היא האם אף הוא יניח ל'בגד' להטעות אותו בנוגע לזהותו שלו, וישתכנע ממצג השווא שיצרו נסיבות חייו – שכביכול הוא נדחה מבית אביו ומעתידו של עם ישראל, או שישכיל לעזוב את בגדו – ובכך לא לעזוב את עצמו.

פרק לט · פסוקים יג–טו

וַיְהִי כִּרְאוֹתָהּ כִּי עָזַב בִּגְדוֹ בְּיָדָהּ וַיָּנָס הַחוּצָה. וַתִּקְרָא לְאַנְשֵׁי בֵיתָהּ וַתֹּאמֶר לָהֶם לֵאמֹר: רְאוּ הֵבִיא לָנוּ אִישׁ עִבְרִי לְצַחֶק בָּנוּ, בָּא אֵלַי לִשְׁכַּב עִמִּי וָאֶקְרָא בְּקוֹל גָּדוֹל. וַיְהִי כְשָׁמְעוֹ כִּי הֲרִימֹתִי קוֹלִי וָאֶקְרָא, וַיַּעֲזֹב בִּגְדוֹ אֶצְלִי וַיָּנָס וַיֵּצֵא הַחוּצָה.

כתב הרמב"ן: "והיא בראותה כי הניח בגדו בידה, פחדה פן יגלה עליה לבני הבית או לאדוניו, והקדימתו אליהם לאמר כי הפשיט בגדו לשכב עימה, ובראותו כי הרימותי קולי – נבהל לברוח".

"וַיְהִי כִּרְאוֹתָהּ כִּי עָזַב בִּגְדוֹ בְּיָדָהּ וַיָּנָס הַחוּצָה" – המילים "וַיְהִי כִּרְאוֹתָהּ" מצביעות על כך שהנתונים המפורטים בפסוקנו מהווים סיבה למה שיתואר בהמשך; דהיינו: מפני שראתה ש"עָזַב בִּגְדוֹ בְּיָדָהּ וַיָּנָס הַחוּצָה" – "וַתִּקְרָא לְאַנְשֵׁי בֵיתָהּ וַתֹּאמֶר לָהֶם לֵאמֹר [...]". שני נתונים מופיעים כאן: עזיבת הבגד בידה, והמנוסה החוצה; וניתן להסביר שכל אחד מהם משמש כגורם לדרך הפעולה שנקטה בה אשת פוטיפר, מצד חמישה היבטים:

א. מצד תקוותה להצליח במזימתה – ההיבט הבסיסי ביותר מבחינת התפתחות העלילה הוא, שכאשר ראתה שיוסף נס מפניה החוצה, ולא עוד אלא שהיה כה נחוש בדעתו עד שאפילו עזב בגדו בידה כדי לא להתעכב אך לרגע – נואשה מתקוותה להצליח לפתותו, ומשכך עברה לכיוון פעולה אחר.

ב. מצד חששה מפני רצונו של יוסף לספר על כך – משראתה שהוא נס החוצה, חששה פן יספר לאחרים את מה שאירע; שכן אילו היה נשאר בבית, הרי שפעם זו לא היתה שונה מהפעמים הקודמות שבהן ניסתה לפתותו ולא עלה בידה, ולא היתה חוששת מכך; אך במנוסתו החוצה ראתה תגובה דרמטית יותר, וחששה פן כעת ימהר לדווח על האירוע – ומשכך, מיהרה היא להקדים אותו ולהפיץ את גירסתה שלה. וכך כתב הנצי"ב ב'העמק דבר': "היא ראתה שנס חוץ לחדר, וכסבורה וחששה שמא נס במהומה הלאה ויעשה פרסום, וכן כתב הספורנו,244 ולולא זאת היתה מסתרת כל העניין".

ג. מצד חששה פן יוסף יישאל על כך, וישיב – שכן אם יראוהו נס לאחר שיצא מן הבית, וכן כשיראוהו מבלי לבושו העליון שעימו הוא רגיל ללכת, ישאלוהו הרואים לפשר הדבר, העשוי להיראות מחשיד – ואזי הוא יספר להם על שהיה, אף אילו לא היה עושה כן מדעתו; וכדברי בעל 'עקידת יצחק': "וַיְהִי כִּרְאוֹתָהּ כִּי עָזַב בִּגְדוֹ בְּיָדָהּ וַיָּנָס' [...] משולל מבגדו העליון, שלא כמנהגו, חשבה בליבה שהוא יגיד הדבר לתת טעם לצאתו כדמות בורח".

ד. מצד אמינות דברי יוסף – מלבד מה שמנוסתו כשלעצמה גרמה לה לחשוש פן יספר על כך, הרי שהיא גם עשויה לתמוך בדבריו – שכן אם ראוהו נס לאחר שיצא מן הבית, הרואים עשויים להאמין לו – שכן עצם מנוסתו החפוזה תוסיף אמינות לסיפורו. אף עזיבת הבגד הצטרפה לחשש זה, שכן היותו בלי בגדו והיות בגדו בידה עשויים לתמוך בסיפורו.

ה. מצד אפשרות הפעולה שעמדה לרשותה – מאידך, נתונים אלו עשויים גם לתמוך בגירסה שבדתה היא, ומשכך – "וַיְהִי כִּרְאוֹתָהּ" שכך אירע, התחכמה לבדות גירסה שקרית שתיתמך על ידי העובדות הללו, כדבריה: "וַיְהִי כְשָׁמְעוֹ כִּי הֲרִימֹתִי קוֹלִי וָאֶקְרָא, וַיַּעֲזֹב בִּגְדוֹ אֶצְלִי וַיָּנָס וַיֵּצֵא הַחוּצָה" – והנה בגדו כאן, כראיה לדבריה, שכן "הפשיט בגדו לשכב עימה" (כלשון הרמב"ן דלעיל), ואף מנוסתו – אם נראתה לעיני אחרים שהיו בחוץ, הרי שהיא ראיה לכך.

"וַתִּקְרָא לְאַנְשֵׁי בֵיתָהּ" – שהיו באותו הזמן בחלק אחר של הבית, כפי שהתבאר לעיל (פס' יא) – על מנת להפיץ באזניהם את גירסתה לאירוע בטרם תתקבל גירסתו של יוסף.

"וַתֹּאמֶר לָהֶם לֵאמֹר: רְאוּ הֵבִיא לָנוּ" – כתב רש"י: "הרי זו לשון קצרה: הביא לנו, ולא פירש מי הֱבִיאוֹ; ועל בעלה אומרת כן". ומדוע אכן "לא פירש מי הביאו"? כתב על כך הרמב"ן: "ואמרה 'הֵבִיא לָנוּ' על בעלה, לא הזכירתו – לכבוד [שכן היא מותחת כאן ביקורת על המביא], או דרך מוסר לנשים [שלא להזכיר בפירוש את בעליהן], או בעבור שהמביא ידוע [ואין צורך לפרש מי הוא]". בנוסף לכך ניתן לומר, שלא זו בלבד שמובן מאליו כי על בעלה היא אומרת כן, אלא שנתון זה גם אינו רלוונטי לאמירתה – שכן מוקד דבריה אינו פוטיפר אלא יוסף, ולפיכך אינה אומרת "הביא לנו בעלי", שכן ניסוח כזה היה מסיט את תשומת הלב ממוקד האמירה – אשמתו של יוסף, ויוצר את הרושם כאילו המוקד הוא תלונה על בעלה. ומכיון שכך, מובנת צורת הניסוח "הֵבִיא לָנוּ [מי שהביא, שכידוע הוא בעלי] אִישׁ עִבְרִי, לְצַחֶק בָּנוּ [...]".

"אִישׁ עִבְרִי" – צאצאי בית אברהם נקראו 'עברים', על שם היותו מֵעֵבֶר הנהר או על שם היותו ממשיכו של עֵבֶר, מבחינה אתנית, תרבותית והשקפתית.245 ומדוע היא מזכירה בהקשר זה את היות יוסף "אִישׁ עִבְרִי"?

כתב על כך הרמב"ן:

וטעם "רְאוּ הֵבִיא לָנוּ אִישׁ עִבְרִי" – כי העברים שנואי המצרים המה, ולא יוכלו לאכול איתם לחם כי תועבה היא להם,246 לא יקנו מהם [אלא] רק לכורמים וליוגבים, ולא יבואו בביתם. ולכן אמרה: הנה עושה עימנו רעה להביא בביתנו עברי, ועשאו פקיד נגיד, וראוי לו שישחק בנו, כעניין שנאמר: "מְפַנֵּק מִנֹּעַר עַבְדּוֹ" וגו' (משלי כט כא).247

הרמב"ן מסביר כי היות יוסף "עִבְרִי" הוא עיקר התלונה; אם היה זה עבד ככל העבדים, כגון מצרי או כנעני, שהיה נוהג כך – לא היתה מותחת ביקורת על בעלה בשל כך שהביא אותו לביתם; אך העברים היו שנואים עליהם, ובשל כך היא מבקרת את פוטיפר על עצם זה ש"הֵבִיא" לביתנו עבד עברי. וכוונתה לומר: אילו היה קונה אותו ומפקידו על עבודה חקלאית, לא היתה לי תלונה – וגם לא היה מתרחש עימו דבר; אך הוא "הֵבִיא לָנוּ" לתוך ביתנו "אִישׁ עִבְרִי" – והרי בידוע שהוא ינהג בפחזות ויצחק בנו, שכן הוא עברי, כך שכביכול כבר מאז שהביאו היה זה "לְצַחֶק בָּנוּ"; ועל כך היא אומרת: "רְאוּ, הֵבִיא לָנוּ אִישׁ עִבְרִי לְצַחֶק בָּנוּ" – "וראוי לו שישחק בנו!", כלומר: מראש היה צפוי כי כך יעשה.

בדבריה להלן לפוטיפר, תאמר: "בָּא אֵלַי הָעֶבֶד הָעִבְרִי אֲשֶׁר הֵבֵאתָ לָּנוּ לְצַחֶק בִּי", ואילו כאן היא מכנה אותו "אִישׁ עִבְרִי"; וטעם הדבר מובן, שכן הכינוי 'עבד' משתמע כזלזול, וזו אכן מגמתה בדבריה לבעלה; אך אנשי ביתה, שאליהם היא פונה בפסוקנו, הרי שאף הם עבדים – ומשכך מסתבר שלא תשתמש באזניהם במונח 'עבד' ככינוי גנאי למישהו אחר, שכן לדידם זהו מעמד כשלהם ואינם רואים בו סיבה לזלזול; ומאידך, אם אכן כינוי זה ליוסף מפי הגבירה יתפרש בעיניהם כזלזול, הרי שיחושו אף הם כלולים באותו זלזול – שהרי גם הם עבדיה, ודבר זה ירחיק אותם ממנה, בעוד שהיא חפצה דווקא לגייסם לצידה.

"לְצַחֶק בָּנוּ" – ביטוי דומה תאמר לאחר מכן גם לבעלה: "בָּא אֵלַי הָעֶבֶד הָעִבְרִי אֲשֶׁר הֵבֵאתָ לָּנוּ לְצַחֶק בִּי", ויש לעמוד על משמעותו של שם הפועל 'לצחק' בהקשר זה.

על הפסוק "וַתֵּרֶא שָׂרָה אֶת בֶּן הָגָר הַמִּצְרִית אֲשֶׁר יָלְדָה לְאַבְרָהָם מְצַחֵק" (לעיל כא ט) נאמר בבראשית רבה (נג יא): "דרש רבי עקיבא [...] אין 'מְצַחֵק' אלא גילוי עריות, היך מה דאת אמר: 'בָּא אֵלַי הָעֶבֶד הָעִבְרִי אֲשֶׁר הֵבֵאתָ לָּנוּ לְצַחֶק בִּי'"; הרי שהוא מניח כדבר ידוע שהלשון "מְצַחֵק" האמורה בענייננו היא לשון גילוי עריות. באופן דומה פירש רשב"ם על הפסוק "וַיַּרְא וְהִנֵּה יִצְחָק מְצַחֵק אֵת רִבְקָה אִשְׁתּוֹ" (לעיל כו ח) – "'מְצַחֵק' – תשמיש, כדכתיב: 'בָּא אֵלַי הָעֶבֶד [הָעִבְרִי] אֲשֶׁר הֵבֵאתָ לָּנוּ לְצַחֶק בִּי', ובמקום אחר הוא אומר: בָּא אֵלַי לִשְׁכַּב עִמִּי'".

אולם ברובד הפשט ישנו קושי בפירוש זה, שכן היא אומרת: "רְאוּ הֵבִיא לָנוּ אִישׁ עִבְרִי לְצַחֶק בָּנוּ, בָּא אֵלַי לִשְׁכַּב עִמִּי וָאֶקְרָא בְּקוֹל גָּדוֹל" – ואם כן, הרי ש"לְצַחֶק" אינו זהה ל"לִשְׁכַּב", אלא שבהקשר הזה ה'צחוק' התבטא באופן של "בָּא אֵלַי לִשְׁכַּב עִמִּי"; ובהתאם לכך יש לפרש גם את דבריה לאחר מכן לפוטיפר.

אם כן, באופן פשוט נראה להסביר שכוונתה לומר: העבד העברי בא לעשות ממני צחוק, מתוך לעג וזלזול בכבודי, בכך ש"בָּא אֵלַי לִשְׁכַּב עִמִּי"; וכפי שכתב בעל מדרש 'שכל טוב': "'לְצַחֶק בָּנוּ' ולהלעיג עלינו". זוהי גם משמעות המילה "מְצַחֵק" במקומות אחרים במקרא, על דרך הפשט – "וַיְהִי כִמְצַחֵק בְּעֵינֵי חֲתָנָיו" (לעיל יט יד), "וַתֵּרֶא שָׂרָה אֶת בֶּן הָגָר הַמִּצְרִית אֲשֶׁר יָלְדָה לְאַבְרָהָם מְצַחֵק" (שם כא ט), "וַיָּקֻמוּ לְצַחֵק" (שמות לב ו) – לשון המציינת לעג או מעשי שחוק.

אלא שאם כן, ניתן לשאול על כך – לפי גירסתה, הרי שאשמתו של יוסף חמורה בהרבה מאשר עצם הצחוק והלעג לגבירתו – שהרי "בָּא אֵלַי לִשְׁכַּב עִמִּי"; ואם כן מדוע היא ממסגרת את האירוע כ"לְצַחֶק בָּנוּ" (או בניסוח שתאמר לבעלה: "בָּא אֵלַי הָעֶבֶד הָעִבְרִי אֲשֶׁר הֵבֵאתָ לָּנוּ לְצַחֶק בִּי") ותו לא?

נראה להסביר, שמפני כבודה היא בוחרת להציג את הדברים כאילו לא היה עולה על הדעת שיוסף יצליח בפועל לאונסה, שהעבד העברי הנקלה יצליח לגבור על גבירתו; מובן שהיו אנשיה באים לעזרתה, וכל מזימתו הינה מופרכת מעיקרה ואינה אלא כדברי שחוק שפגעו בכבוד גבירתו.

"לְצַחֶק בָּנוּ" – מדוע היא משתמשת בלשון רבים, מאחר ולא בא אלא אליה לבדה? לכאורה היה מתבקש שתאמר "לְצַחֶק בִּי", כפי שאכן אמרה לאחר מכן לבעלה: "בָּא אֵלַי הָעֶבֶד הָעִבְרִי אֲשֶׁר הֵבֵאתָ לָּנוּ לְצַחֶק בִּי".

נראה להסביר, כי הלעג לגבירה והפגיעה בכבודה הרי שהם 'צחוק' גם באנשיה, שכן מעשה זה משקף יחס מזלזל כלפי כל המוסד של בית פוטיפר; ואם גבירתו מזולזלת בעיניו, כל שכן שגם עבדיה ואנשי הבית, השייכים לאותה מערכת. על כך היא אומרת: "הֵבִיא לָנוּ אִישׁ עִבְרִי לְצַחֶק בָּנוּ" – אני ואתם, איננו כלום בעיניו; בכולנו הוא בא לצחק, ואף כבודכם נפגע מכך; ולכשיזדמן לו, גם בכם יְצַחֵק. היא בוחרת להתנסח דווקא באופן זה, על מנת לעורר באנשיה כעס על יוסף ולהעמיד אותם לצידה.248

לעומת זאת, לבעלה אמרה "לְצַחֶק בִּי", כדי לבטא את הפגיעה האישית בכבודה; ומובן מאליו שבתור אדון ובעל מעשה זה מהווה פגיעה גם בו ובכבודו, ומשכך אינה נדרשת לומר לו "לצחק בנו" באופן שיכלול גם אותו כדי להעמיד אותו לצידה. מעבר לכך, בתלונה זו הרי שהיא מציגה את פוטיפר דווקא כְּגוֹרֵם הַמִּפְגָּע, ולא בתור הנפגע, ולפיכך היא אומרת "בִּי" בלשון יחיד – "אתה הבאת לנו עבד עברי שבא לצחק בי".

"בָּא אֵלַי לִשְׁכַּב עִמִּי" – "לאנסי, אלא שקראתי בקול גדול וברח" ('בכור שור'), "וָאֶקְרָא בְּקוֹל גָּדוֹל" – "לפי שאתם רחוקים ממני, שאילו לא הרימותי קולי היה שוכב עימי בעל כרחי" (רשב"ם).

"וַיְהִי כְשָׁמְעוֹ כִּי הֲרִימֹתִי קוֹלִי וָאֶקְרָא" – "לבני הבית, שיבואו ויסייעוני מידו" ('שכל טוב'), "וַיַּעֲזֹב בִּגְדוֹ אֶצְלִי" – "שהסיר מעליו לאנסני, ולא הספיק ליקח אותו מפני שצעקתי, והניחו וברח" ('בכור שור'); מובן שאין היא אומרת "בידי", כפי שאמר הכתוב בתיאור המעשה "וַיַּעֲזֹב בִּגְדוֹ בְּיָדָהּ", אלא "אֶצְלִי", שכן לדבריה היא לא אחזה בבגדו כלל אלא שהוא פשט אותו מרצונו והניחו אצלה, דהיינו בסמוך לה, "וַיָּנָס וַיֵּצֵא הַחוּצָה".

פרק לט · פסוקים טז–יח

וַתַּנַּח בִּגְדוֹ אֶצְלָהּ עַד בּוֹא אֲדֹנָיו אֶל בֵּיתוֹ. וַתְּדַבֵּר אֵלָיו כַּדְּבָרִים הָאֵלֶּה לֵאמֹר: בָּא אֵלַי הָעֶבֶד הָעִבְרִי אֲשֶׁר הֵבֵאתָ לָּנוּ לְצַחֶק בִּי. וַיְהִי כַּהֲרִימִי קוֹלִי וָאֶקְרָא וַיַּעֲזֹב בִּגְדוֹ אֶצְלִי וַיָּנָס הַחוּצָה.

"וַתַּנַּח בִּגְדוֹ אֶצְלָהּ עַד בּוֹא אֲדֹנָיו אֶל בֵּיתוֹ" – כדי להציגו כראיה לדבריה; וכשבא אדוני יוסף אל ביתו, "וַתְּדַבֵּר אֵלָיו כַּדְּבָרִים הָאֵלֶּה לֵאמֹר [...]" – לשם מה מוסיף הכתוב את המילים "כַּדְּבָרִים הָאֵלֶּה", ומה משמעותן?

ייתכן שהביטוי "כַּדְּבָרִים הָאֵלֶּה" מתייחס למה שנאמר לפניו, ש"כדברים האלה שדיברה אל אנשי הבית" (רד"ק) דיברה גם אליו;249 וייתכן שהוא מתייחס למה שנאמר לאחריו, "כדברים האלה שמפרש הכתוב" (הנצי"ב ב'העמק דבר'), דהיינו מה שיצוטט כעת. האפשרות האחרונה נראית יותר, שכן אם כוונת הכתוב לומר שאמרה לו דברים דומים למה שאמרה לאנשי ביתה – הרי הקורא רואה זאת, ולשם מה נדרש הכתוב לציין כך? אולם לפירוש הנצי"ב הרי שניתן להבין כי במילים אלו מתכוון הכתוב להעיר את תשומת הלב דווקא להבדלים שבין הדברים שאמרה לאנשי ביתה לבין אלו שאמרה לבעלה, ולפיכך הוא מדגיש: "כַּדְּבָרִים הָאֵלֶּה [דלהלן]" – ולא כדברים שאמרה לאנשי ביתה, וכפי שכתב הנצי"ב בהמשך דבריו: "כדברים האלה שמפרש הכתוב, באופן אחר ממה שאמרה לאנשי הבית", וכפי שיתבאר בהמשך.

"בָּא אֵלַי הָעֶבֶד הָעִבְרִי אֲשֶׁר הֵבֵאתָ לָּנוּ לְצַחֶק בִּי"250– פיסוק המילים הנכון הוא: "בָּא אֵלַי הָעֶבֶד הָעִבְרִי אֲשֶׁר הֵבֵאתָ לָּנוּ – לְצַחֶק בִּי", כפי שמוכח מטעמי המקרא, וכפי שפירש רש"י: "בא אלי לצחק בי, העבד העברי אשר הבאת לנו"; שהרי לא כדי לצחק בה הוא הביא אותו.

ואמנם, בדבריה לאנשי ביתה אמרה: "רְאוּ הֵבִיא לָנוּ אִישׁ עִבְרִי לְצַחֶק בָּנוּ", ומשמע כביכול שלשם כך הביא אותו – אך אין הפסוקים דומים לזה לזה, כך שיהיה ניתן לפרש גם כאן "הֵבֵאתָ לָּנוּ לְצַחֶק בִּי"; שכן בדבריה לעיל, כמבואר שם, עיקר תלונתה הוא על כך ש"הֵבִיא לָנוּ אִישׁ עִבְרִי", משום שכביכול בידוע היה שסופו "לְצַחֶק בָּנוּ"; אך כאן אין היא אומרת "הבאת לנו עבד עברי לצחק בי", אלא "בָּא אֵלַי הָעֶבֶד הָעִבְרִי אֲשֶׁר הֵבֵאתָ לָּנוּ לְצַחֶק בִּי" – עיקר האמירה הוא דיווח על המאורע: "בָּא אֵלַי הָעֶבֶד הָעִבְרִי", והיא מוסיפה תיאור על אותו "הָעֶבֶד הָעִבְרִי", הכולל מורת רוח מרומזת: "אֲשֶׁר הֵבֵאתָ לָּנוּ", ובהקשר זה אין מקום לומר "אֲשֶׁר הֵבֵאתָ לָּנוּ לְצַחֶק בִּי" כאילו ידוע שלשם כך הביא אותו. ועוד, שאם הפסוק יתפרש כך – הרי שלא נאמר בו לשם מה בא אליה יוסף, והעיקר חסר מן הדברים.

בשונה מדבריה לאנשי ביתה, שבהם אמרה בפירוש "בָּא אֵלַי לִשְׁכַּב עִמִּי", בדבריה לפוטיפר אינה אומרת כך, אלא רק "לְצַחֶק בִּי". הנצי"ב ב'העמק דבר' מפרש שאכן כלפיו היא הציגה גירסה רכה יותר של האירוע, "ולא אמרה 'לשכב עימי', אלא אמרה כי היקל הרבה בכבודה ונתגרה עימה בצחוק מכאיב לב, כאילו רוצה לבוא עליה ופושט בגדו לפניה, והוא קלות ראש באדונו". לפי זה, הוא מפרש גם את שינוי הלשון שבפסוק הבא: "וַיְהִי כַּהֲרִימִי קוֹלִי וָאֶקְרָא" – "לא אמרה 'ואקרא בקול גדול', שהרי אין דרך לצעוק מר על דבר ליצנות ובזיון, אלא הרימה קול לגעור בו; 'וָאֶקְרָא' – לבני הבית שיגערו גם המה בו".

אם כן, מדוע דווקא בדבריה לבעלה בחרה להציג גירסה כזו? ניתן להסביר זאת בכך שככל שההאשמה היתה חמורה יותר, כך חששה פן הוא לא יאמין לדבריה, מתוך היכרותו עם יוסף. לפיכך, בתחילת האירוע, בדבריה לאנשי ביתה, הפעילה את מלוא כובד משקלה של עלילת השווא, מתוך בהלתה למהר להטות את דעתם לטובתה בטרם יספיק יוסף לספר את שאירע; ומשכך אמרה בפירוש: "בָּא אֵלַי לִשְׁכַּב עִמִּי". אולם לאחר מכן, בדברה עם בעלה, שהכיר את יוסף כדמות חיובית והיה עלול להתקשות להאמין להאשמה חמורה כל כך, מיתנה את ניסוח דבריה, ואמרה רק "לְצַחֵק בִּי" – ובכך הציגה את יוסף כמי שפגע בכבודה, מבלי לנקוב במפורש בטענה הקיצונית יותר של ניסיון לשכב עימה בפועל.

"וַיַּעֲזֹב בִּגְדוֹ אֶצְלִי וַיָּנָס הַחוּצָה" – תיאור מנוסתו של יוסף מופיע ארבע פעמים בפרקנו, בשינויי לשון:

א. בתיאור האירוע עצמו – "וַיָּנָס וַיֵּצֵא הַחוּצָה".

ב. בתיאור מה שראתה אשת פוטיפר – "וַיָּנָס הַחוּצָה".

ג. בדבריה לאנשי ביתה – "וַיָּנָס וַיֵּצֵא הַחוּצָה".

ד. ובדבריה לבעלה – "וַיָּנָס הַחוּצָה".

עלינו לעמוד איפוא על משמעות כפל הפעלים "וַיָּנָס וַיֵּצֵא", וכמו כן על טעם השינויים בין הפסוקים ומהות החוקיות שלפיה מתנסח הכתוב כך או כך.

ניתן להסביר (כפי שהתבאר לעיל בפס' יב), שהתיאור "וַיָּנָס הַחוּצָה" מציין את המנוסה מן החדר עצמו, ואילו התיאור "וַיָּנָס וַיֵּצֵא הַחוּצָה" מוסיף על כך שמעבר למנוסה מהחדר, גם יצא מן הבית כולו.251 לפיכך, בתיאור האירוע עצמו נאמר "וַיָּנָס וַיֵּצֵא הַחוּצָה", שכן בפועל אכן יוסף יצא מן הבית כולו, ברצונו להתרחק ככל האפשר מהסיטואציה של הניסיון; אולם אשת פוטיפר ראתה רק את מנוסתו מן החדר ולא ידעה את דרכו לאחר מכן, ולפיכך בעניינה נאמר "וַיְהִי כִּרְאוֹתָהּ כִּי עָזַב בִּגְדוֹ בְּיָדָהּ וַיָּנָס הַחוּצָה". בדבריה לאנשי ביתה, היא אומרת "וַיָּנָס וַיֵּצֵא הַחוּצָה", שכן היא טופלת עליו האשמה של ניסיון אונס – "בָּא אֵלַי לִשְׁכַּב עִמִּי", ובהתאם לכך מספרת כי כאשר קראה "בְּקוֹל גָּדוֹל" והוא הבין כי הוא עלול להיתפס באשמה זו – נס ויצא מן הבית כולו, פן ייתפס וייענש בחומרה; אך בדבריה לבעלה, היא אומרת "וַיָּנָס הַחוּצָה" מן החדר, שכן שם היא ממתנת את גירסתה, כפי שהתבאר לעיל, ואינה טוענת שיוסף בא לשכב עימה אלא לְצַחֶק בה. בהתאם לכך, אינה מתארת אותו כמי שנמלט על נפשו מן הבית, אלא כמי שנס מן החדר והוציא את עצמו מן הסיטואציה, כדי שלא להיתפס בקלקלתו, ולהמנע מעימות עם אנשי הבית שקראה להם.

פרק לט · פסוק יט

וַיְהִי כִשְׁמֹעַ אֲדֹנָיו אֶת דִּבְרֵי אִשְׁתּוֹ אֲשֶׁר דִּבְּרָה אֵלָיו לֵאמֹר כַּדְּבָרִים הָאֵלֶּה עָשָׂה לִי עַבְדֶּךָ, וַיִּחַר אַפּוֹ.

"כַּדְּבָרִים הָאֵלֶּה"

המפרשים הסבירו בדרכים שונות את משמעות הלשון "כַּדְּבָרִים הָאֵלֶּה עָשָׂה לִי עַבְדֶּךָ":

רש"י כתב (על פי בראשית רבה פז ט): "בשעת תשמיש אמרה לו. וזהו שאמרה: 'כַּדְּבָרִים הָאֵלֶּה עָשָׂה לִי עַבְדֶּךָ', ענייני תשמיש כאלה". בכך הוא מסביר הן את משמעות האות כ' בהקשר זה והן את מהותם של ה"דְּבָרִים הָאֵלֶּה" שאליהן הכ' נוספת; ומשמעות דבריה היא: דברים שהם כמו ה"דְּבָרִים הָאֵלֶּה" שאנו עושים כעת, "עָשָׂה לִי עַבְדֶּךָ".

נראה שאין כוונת הדברים לתשמיש ממש בפועל, שהרי לא זו היתה האשמתה כלפי יוסף – הן בדבריה לאנשי ביתה והן בדבריה לפוטיפר בפסוקים הקודמים – שהרי אמרה כי הרימה קולה והוא נס בטרם הצליח למלא את מבוקשו; ועוד, שגם לא היה כדאי לה לטעון כך, כפי שכתב הרמב"ן: "ואני תמה [...] כי איך תפגום את עצמה ותימאס בעיני בעלה, לאמר לו כי זנתה באונס או ברצון? וראוי היה שיהרגנה, כי למה לא צעקה מתחילה ויברח, כאשר עשתה בסוף? [...] ואם תאמר [שאמרה כך] כדי שיחר אפו ויהרגנו – ו[הרי] די חיוב מיתה לעבד שיבוא לאנוס אשת השר אדוניו [אף אם לא עשה כן בפועל]".

אם כן, נראה להסביר שכוונת המדרש ורש"י היא לעצם כוונתו של יוסף, כפי שכתב מהר"ל ב'גור אריה': "'ענייני תשמיש כאלה' – פירוש, לא תשמיש ממש, דהא אמרה 'בָּא אֵלַי לִשְׁכַּב', ולא ששכב עימה, אלא ענייני תשמיש – הבא לשמש – עשה לי [...], שבא לשכב עימה, ולא רצתה"; או למעשים מיניים אחרים שעשה לפי דבריה, כפי שהציע הרמב"ן לפרש את כוונתם: "ואפשר שרצו לפרש 'כַּדְּבָרִים הָאֵלֶּה' – ענייני תשמיש בגילוי ערווה ומשמוש ידיים, לא תשמיש ממש".

כפי שנאמר לעיל, דרשה זו מתבססת על הלשון "כַּדְּבָרִים הָאֵלֶּה" המתפרשת לפיה כדימוי; מעבר לכך, מהר"ל מוסיף כי גם עצם הייתור בלשון הפסוק הוביל לדרשה זו, והוא מסביר גם מהי משמעות הדברים במסגרת הסיפור:

ומה שדרשו רז"ל [ש]בשעת תשמיש אמרה כן – משום שהכתוב מיותר לגמרי, דלמה לי למכתב עוד "כַּדְּבָרִים הָאֵלֶּה" וגו', דלא הוי ליה למכתב [אלא] רק "וַיְהִי כִשְׁמֹעַ [אֲדֹנָיו] אֶת דִּבְרֵי אִשְׁתּוֹ", אלא לומר שבשעת תשמיש גם כן אמרה לו כך, כדי שיהיה כועס על יוסף, דבשעת תשמיש שבא לשכב עימה – ומאוס הוא על האדם האישה הנשכבת עם אחר – והיא אמרה לו שהוא רצה לשכב עימה עד שעשה לה כך וכך, ולפיכך היה כועס עליו יותר.

לעומת זאת, הרמב"ן כתב:

ועל דרך הפשט אין צורך [בפירוש המדרש ורש"י], כי אין הכ"ף לדמיון בדבר אחר, ופתרונו: "הדברים האלה", וכמוהו "וַתַּגֵּד לְבֵית אִמָּהּ כַּדְּבָרִים הָאֵלֶּה" (לעיל כד כח), וכן "וּבְדַבְּרוֹ עִמִּי כַּדְּבָרִים הָאֵלֶּה נָתַתִּי פָנַי אַרְצָה" (דניאל י טו), וכן רבים.

או יאמר: כי בשמוע אדוניו את דברי אשתו שאמרה לו "כה עשה לי עבדך ככל הדברים האלה שהגדתי מיד לאנשי הבית", חרה אפו.

וייתכן ששימוש הכ"ף בזה להפלגה, וכן "לָמָּה יְדַבֵּר אֲדֹנִי כַּדְּבָרִים הָאֵלֶּה" (להלן מד ז), "וַתִּקְרֶאנָה אֹתִי כָּאֵלֶּה" (ויקרא י יט).

שלושה הסברים מציע הרמב"ן למשמעות הכ' בביטוי "כַּדְּבָרִים הָאֵלֶּה":

לפי ההסבר הראשון, אין זו כ' הדמיון אלא אמצעי לשוני שגרתי המחליף את ה' הידיעה, ומשמעות הביטוי שווה ל"הדברים האלה". הרמב"ן מצטט פסוקים נוספים שבהם מופיע הביטוי "כַּדְּבָרִים הָאֵלֶּה" שלא במשמעות דימוי, כראיה לביסוס הסבר זה.

לפי ההסבר השני, אכן זוהי כ' הדמיון, והיא מתייחסת למה שאמרה לפני כן לאנשי ביתה; "שאמרה לו: כה עשה לי עבדך ככל הדברים האלה שהגדתי מיד לאנשי הבית".

אולם לפי הסבר זה קשה להבין מאיזה טעם תידרש לומר לו שהדברים שהיא מספרת לו הם כמו מה שסיפרה לאנשי ביתה; ועוד, שלא הוזכר כלל בכתוב שאמרה לו כי סיפרה זאת לאנשי ביתה, כך שלאותם דברים יתייחס הדימוי "כַּדְּבָרִים הָאֵלֶּה".

לפי ההסבר השלישי, "שימוש הכ"ף בזה, להפלגה", דהיינו לשם הדגשת חומרת הדבר. הרמב"ן מביא דוגמה לשימוש כזה מהפסוק "לָמָּה יְדַבֵּר אֲדֹנִי כַּדְּבָרִים הָאֵלֶּה" שאמרו אחי יוסף לשליחו של יוסף כאשר האשים אותם בגניבת הגביע, ומהפסוק "וַתִּקְרֶאנָה אֹתִי כָּאֵלֶּה" שאמר אהרן על מות בניו נדב ואביהוא; האות כ' בביטויים "כאלה" "כדברים האלה" "כדבר הזה", וכדומה – אינה מציינת דימוי, לדברי הרמב"ן, אלא הפלגה בחומרתו או במשמעותו של הדבר.252 כך גם בפסוקנו, עניינה של הכ' הוא להעצים את חומרתם של ה"דברים האלה".253

אולם הסבר זה מעורר קושי; שכן את חומרת הדברים בעיני מי מציינת כאן כ' זו? אם בעיני אשת פוטיפר – הרי היא עצמה לא אמרה את המילים "כַּדְּבָרִים הָאֵלֶּה", אלא זהו קיצור לשון של הכתוב (לפי הסבר זה) – ואם כן, קשה לומר שצורת הניסוח של מילים אלו משקף את נקודת מבטה שלה; ואם בעיני פוטיפר עצמו – הרי לא את נקודת מבטו מתאר הכתוב במילים אלו, אלא את מה שאמרה היא. הוספת הכ' עשויה לציין 'הפלגה' שקיימת בעיני הדובר – אך במילים אלו הדובר אינו אלא הכתוב עצמו, ומצידו אין טעם בהבעת עמדה ביחס לחומרת דברים אלו, שלא התרחשו כלל.

הנצי"ב ב'העמק דבר' פירש, בהתאם לדבריו שהובאו לעיל שהאשמתה כלפי יוסף היתה רק על כך שבא "לְצַחֶק" בה, ולא שהתכוון לשכב עימה בפועל, שהביטוי "כַּדְּבָרִים הָאֵלֶּה" בפסוקנו מכוון לכך שיוסף עשה דברים נוספים כאלו בעבר, לדבריה; וכך כתב: "באשר חקר האדון האם אפשר שהחל להתלוצץ בראשונה בצחוק מגונה כזה,254 ענתה: 'כַּדְּבָרִים הָאֵלֶּה עָשָׂה לִי', כמה פעמים היקל בי בליצנות אלא שלא שמתי ליבי על זה, עד שבא לזה הצחוק" – ופירוש אמירה זו הוא: כמו הדברים האלה שעשה לי עכשיו, עשה לי כבר בעבר.

אך ברובד הפשט ישנו קושי בהבנה זו, שכן ממהלך הכתובים משמע שפסוקנו חוזר על מה שכבר ידוע שאמרה לו, ולא שמוזכר כאן דבר חדש שלא נאמר לפני כן. המילים "אֲשֶׁר דִּבְּרָה אֵלָיו לֵאמֹר כַּדְּבָרִים הָאֵלֶּה עָשָׂה לִי עַבְדֶּךָ" אינן מופיעות כשלב נוסף בהתפתחות הסיפור, אלא כמאמר מוסגר החוזר על הידוע, ונועד לאפיין את הגורם לחרון אפו של פוטיפר: "כִשְׁמֹעַ אֲדֹנָיו אֶת דִּבְרֵי אִשְׁתּוֹ – אֲשֶׁר דִּבְּרָה אֵלָיו לֵאמֹר כַּדְּבָרִים הָאֵלֶּה עָשָׂה לִי עַבְדֶּךָ – וַיִּחַר אַפּוֹ". משכך, קשה לפרש שהמילים "כַּדְּבָרִים הָאֵלֶּה עָשָׂה לִי עַבְדֶּךָ" אינן מתייחסות לאותם הדברים שהוזכרו לפני כן – "בָּא אֵלַי הָעֶבֶד הָעִבְרִי אֲשֶׁר הֵבֵאתָ לָּנוּ לְצַחֶק בִּי", אלא לדברים הדומים להם, שעד כה לא הוזכרו כלל.

אם כן, נראה כי ההסבר הקרוב ביותר לפשוטו של מקרא הוא שהמילים "כַּדְּבָרִים הָאֵלֶּה עָשָׂה לִי עַבְדֶּךָ" מהוות סיכום מקוצר של דבריה. אין כאן דימוי ל'דברים' אחרים, אלא המקרא מצטט את דבריה בקיצור באמצעות הביטוי "כַּדְּבָרִים הָאֵלֶּה", מבלי לחזור על כל תוכן הדברים; וכפי שכתב שד"ל: "'כַּדְּבָרִים הָאֵלֶּה' – כך וכך, וכמוהו 'כָּזֹאת וְכָזֹאת דִּבְּרָה הַנַּעֲרָה' (מלכים ב' ה ד)", וכך גם בפסוקים נוספים, כמו "כָּזֹאת וְכָזֹאת יָעַץ אֲחִיתֹפֶל אֶת אַבְשָׁלֹם וְאֵת זִקְנֵי יִשְׂרָאֵל וְכָזֹאת וְכָזֹאת יָעַצְתִּי אָנִי" (שמואל ב' יז טו), "כָּזֹה וְכָזֶה תְּדַבֵּר אֵלֶיהָ" (מלכים א' יד ה), ועוד.

לפי הסבר זה, ניתן אמנם לפרש את משמעות האות כ' שנוספה כאן כך שאין בה משמעות מיוחדת והרי היא כמו ה' הידיעה, וכהסברו הראשון של הרמב"ן; אך ניתן גם לומר שאף זו משמשת ככ' הדמיון – ומשמעותה היא השוואה בין תוכן דיבורו של המספר לבין המעשה שהוא מתאר; הכתוב מקצר את סיפור מעשי יוסף בדברי אשת פוטיפר, במילים "כַּדְּבָרִים הָאֵלֶּה" – דהיינו: הדברים שעשה לי עבדך הם כמו הדברים שאני אומרת; שכן המעשה והאמירה, גם אם הם זהים, מהווים דברים נפרדים – ומשכך, נכון לומר כי מעשי יוסף היו "כַּדְּבָרִים הָאֵלֶּה" – כמו הדברים שבפיה.

כך גם בכל המקומות הנוספים שבהם מופיעה כ' זו בהקשר סתמי לכאורה, כגון בפסוקים שציטט הרמב"ן בהסברו הראשון: "וַתַּגֵּד לְבֵית אִמָּהּ כַּדְּבָרִים הָאֵלֶּה" – רבקה לא הגידה את ה"דְּבָרִים הָאֵלֶּה", דהיינו את ההתרחשות שסופר עליה בכתוב, אלא הגידה דברים שהם כמו ההתרחשות, דהיינו שמתארים אותה. כך גם בפסוק השני שהזכיר, "וּבְדַבְּרוֹ עִמִּי כַּדְּבָרִים הָאֵלֶּה" – לפני כן ציטט דניאל את הדברים שאמר לו המלאך, וכאן הוא אומר: וכאשר דיבר אלי דברים השווים לציטוט שאמרתי קודם. וכך גם בפסוקים שהוזכרו לעיל: "כָּזֹאת וְכָזֹאת דִּבְּרָה הַנַּעֲרָה", "כָּזֹאת וְכָזֹאת יָעַץ אֲחִיתֹפֶל אֶת אַבְשָׁלֹם" – משמעות הביטוי היא: כמו זאת, כמו האמירה הזאת שאני אומר כעת כציטוט – כך אמרה הדמות המדוברת. אף אם תוכן הדברים שווה, הרי שכל דיבור הוא דיבור אחר, והכ' משווה ביניהם.

"כַּדְּבָרִים הָאֵלֶּה עָשָׂה לִי עַבְדֶּךָ, וַיִּחַר אַפּוֹ"

על פי הביאור האחרון, הרי שהפסוק מצטט כאן את דבריה בקיצור – אך לא בקיצור מוחלט; שהרי ניתן היה לקצר יותר ולומר "וַיְהִי כִשְׁמֹעַ אֲדֹנָיו אֶת דִּבְרֵי אִשְׁתּוֹ אֲשֶׁר דִּבְּרָה אֵלָיו כַּדְּבָרִים הָאֵלֶּה", או אף לקצר יותר ולומר רק "וַיְהִי כִשְׁמֹעַ אֲדֹנָיו אֶת דִּבְרֵי אִשְׁתּוֹ". אולם הכתוב בחר להאריך מעבר לנדרש, ובצטטו את "דִּבְרֵי אִשְׁתּוֹ", הוא מקפל את תיאור המעשה עצמו במילים "כַּדְּבָרִים הָאֵלֶּה", ואת תיאור הפעולה במילים "עָשָׂה לִי" – ומוסיף עליהם בפירוש את המילה "עַבְדֶּךָ". מכך ניתן ללמוד, כי את המילה האחרונה מבקש הכתוב להעמיד במוקד תשומת הלב, והיא תכלית כל האריכות המיותרת-לכאורה שלפניה. מהי משמעות הדבר?

בדבריה, לא כינתה אשת פוטיפר את יוסף "עַבְדֶּךָ", אלא "הָעֶבֶד הָעִבְרִי אֲשֶׁר הֵבֵאתָ לָּנוּ"; בהתבטאות זו היא מבקשת להלין על בעלה ולהטיל עליו את האשמה על מה שעשה העבד – כאומרת: מדוע זה הבאת לביתנו עבד עברי? הרי בידוע שאם תכניס לביתנו עבד עברי נקלה, ואף תתן לו שררה, לא יהא זה אלא לתקלה!255 לפיכך, בפסוקנו אין הכתוב אומר "עשה לי העבד", אלא "עָשָׂה לִי עַבְדֶּךָ" – כאומרת: עבדך ולא עבדי, אתה הוא האחראי עליו ועל מעשיו, ובסיבתך נגרם אירוע זה.

על פי זה, ניתן גם להבין את כוונת הכתוב בהוספת המילים "אֲשֶׁר דִּבְּרָה אֵלָיו לֵאמֹר" – דברי הביקורת הופנו "אֵלָיו", שהרי אמרה "עַבְדֶּךָ" – ביטוי מקוצר בלשון הכתוב למה שאמרה בפועל: "הָעֶבֶד הָעִבְרִי אֲשֶׁר הֵבֵאתָ לָּנוּ" – קיצור אשר מחדד את העוקץ שבאמירה יותר מאשר הנוסח המקורי.

אם כנים הדברים, הרי שניתן לומר כי כל עניינה של אריכות הלשון בפסוקנו הוא מיקוד תשומת הלב במה שגרם לחרון אפו של פוטיפר: הטלת האשמה עליו בפי אשתו – וכעסו היה כלפיה. ואכן כך פירש הספורנו: "'וַיִּחַר אַפּוֹ' – על שהתרעמה מפני ש'הביא לה איש עברי לצחק בה', כי אמנם לא חרה אפו על יוסף בזה, שהאמין יותר לדברי יוסף, אבל נתנו בבית הסוהר להראות שהאמין לה, לכבודה".

לפי פירוש זה של הספורנו, פוטיפר לא האמין כלל לעלילה שסיפרה אשתו על יוסף, וחרון אפו היה עליה בלבד. בדומה לכך פירש האברבנאל, אלא שלדבריו הכעס לא היה בשל כך שהאשימה אותו; ואלו דבריו: "לא אמר 'ויחר אפו על יוסף' אלא 'ויחר אפו' בלבד, כי חרה אפו במה שדיברה אשתו, ומה לעשות לה, ואשתו אומרת כך היה הדבר" – ונראה שכוונת הדברים היא, שדווקא מפני שלא האמין לה חרה אפו על שהכניסה אותו לסיטואציה זו, שלא יוכל לצאת ממנה בשלום – שכן אינו יכול להתעלם מתלונתה, ובכך לפגוע בכבודה ולעורר עליו את כעסה, ואף הוא יהיה לבוז בעיני אנשי ביתו אם לא יעניש את העבד; ומאידך, אינו רוצה להמית את יוסף עבדו הנאמן או להענישו שלא כדין.

אולם מקריאה פשוטה של הפסוקים עולה הרושם שחרון אפו היה על יוסף: "וַיְהִי כִשְׁמֹעַ אֲדֹנָיו אֶת דִּבְרֵי אִשְׁתּוֹ אֲשֶׁר דִּבְּרָה אֵלָיו לֵאמֹר כַּדְּבָרִים הָאֵלֶּה עָשָׂה לִי עַבְדֶּךָ – וַיִּחַר אַפּוֹ – וַיִּקַּח אֲדֹנֵי יוֹסֵף אֹתוֹ וַיִּתְּנֵהוּ אֶל בֵּית הַסֹּהַר"; מופיע כאן מהלך לוגי של סיבות ותוצאות: דברי אשתו גרמו לו לחרון האף, וחרון האף גרם לכך שאסר את יוסף. אך אם על אשתו כעס ולא על יוסף, לשם מה מספר זאת הכתוב, מאחר ולכאורה לא היתה לכך משמעות במסגרת השתלשלות העניינים?

ניתן להסביר, כי אכן לחרון אפו על אשתו היה תפקיד בהתרחשות זו; לפי הפירוש שהתבאר לעיל, שהגורם לחרון האף היה התרעומת שהביעה כלפיו, הרי שניתן לומר כי מפני כך נוצר בו ריחוק רגשי ממנה ומהרצון לעמוד לצידה, ולעומת זאת התעורר בו הרצון דווקא שלא להעניש את יוסף במלוא החומרה כפי שהיא מצפה, ובכך לבטא את הסתייגותו מדבריה, ואת תרעומתו על כך שהיא מתרעמת עליו. אם כן, הרי ש"וַיִּחַר אַפּוֹ" אכן מהווה סיבה ל"וַיִּקַּח אֲדֹנֵי יוֹסֵף אֹתוֹ וַיִּתְּנֵהוּ אֶל בֵּית הַסֹּהַר", אך לא כפי ההבנה הפשוטה שלכן הענישו, אלא להפך: לכן הענישו רק במאסר ולא במיתה.256

לפי פירוש האברבנאל ניתן לומר, שהתיאור "וַיִּחַר אַפּוֹ" מלמד כי לא היה ישר בעיני פוטיפר להמית את יוסף שלא כדין, ובכך מובן מדוע בפסוק הבא הוא מעניש אותו במאסר בלבד. לפי זה, הרי ש"וַיִּחַר אַפּוֹ" אינו סיבה לענישה המקילה (כפי שהתבאר לפי הפירוש הקודם), אלא סימן לכך שהיה רע בעיניו להמיתו חינם, ומכך תובן התוצאה שהיא הענישה המקילה. התיאור "וַיִּקַּח אֲדֹנֵי יוֹסֵף אֹתוֹ וַיִּתְּנֵהוּ אֶל בֵּית הַסֹּהַר" מציג את הפתרון שמצא פוטיפר למציאות הבעייתית שנקלע אליה, אשר אותה ביטא הכתוב בתיאור הקודם לו "וַיִּחַר אַפּוֹ".

לעומת זאת, מפרשים אחרים הבינו כפי המשמעות הפשוטה של הכתוב, שחרון אפו היה על יוסף, ומכך משתמע שהאמין לדברי אשתו, לפחות באופן חלקי. משכך, נדרשו לשאלה מדוע הענישו רק במאסר ולא במיתה; אבן עזרא כתב על כך: "'וַיִּחַר אַפּוֹ' – אף על פי כן לא הרגו, כי בספק היה הדבר אצלו", והרמב"ן כתב: "והנה מאהבת אדוניו את יוסף לא הרגו; או שהיה נס מאת השם; או שנסתפק בדבריה, מדעתו צדקת יוסף. וכך אמרו בבראשית רבה (פז י): 'אמר ליה: ידע אנא דלית הוא מינך, אלא שלא לערב פסולת בבניי'".257

מכל מקום, מהלך הדברים משקף את ההשגחה האלוהית שליוותה את יוסף – אשר הטילה כעס או אי אמון בין איש לאשתו, על מנת שיוסף יישאר בחיים ויגיע אל בית הסוהר, המקום שממנו עשויה לצמוח לו הישועה, ומשם יגיע אל התפקיד שלשמו נשלח – "כִּי לְמִחְיָה שְׁלָחַנִי אֱלֹהִים לִפְנֵיכֶם" (להלן מה ה).

פרק לט · פסוק כ

וַיִּקַּח אֲדֹנֵי יוֹסֵף אֹתוֹ וַיִּתְּנֵהוּ אֶל בֵּית הַסֹּהַר מְקוֹם אֲשֶׁר אסורי [קרי: אֲסִירֵי] הַמֶּלֶךְ אֲסוּרִים, וַיְהִי שָׁם בְּבֵית הַסֹּהַר.

"וַיִּקַּח אֲדֹנֵי יוֹסֵף אֹתוֹ וַיִּתְּנֵהוּ אֶל בֵּית הַסֹּהַר"

כתב האברבנאל: "לא נתנו ביד עבדי השוטרים לשיוליכוהו בחרפה ובוז בבית הסוהר כמנהג, אלא אדוניו בעצמו לקחו בידו והוליכו שם, מאשר יקר בעיניו, וזהו 'וַיִּקַּח אֲדֹנֵי יוֹסֵף אֹתוֹ וַיִּתְּנֵהוּ אֶל בֵּית הַסֹּהַר'" – שכן, כפי שהתבאר לעיל, פוטיפר לא האמין לדברי אשתו (לפחות לא באופן מוחלט), ולפיכך יחסו אל יוסף נותר חיובי, וגם כשנאלץ לאסרו בבית הסוהר עשה זאת בדרך שמבטאת יחס זה.

המונח "בֵּית הַסֹּהַר" אינו מופיע עוד במקרא מלבד בענייננו, והמפרשים הציעו הבנות שונות בדבר מקורו הלשוני; רד"ק בספר השורשים (ערך 'סהר') כתב: "'בֵּית הַסֹּהַר', שם המאסר, שהוא עגול בדמות הסהר [=ירח]"; והרמב"ן כתב: "ולפי דעתי, שהוא הבור258– בית בנוי תחת הקרקע, ולו פתח קטן מלמעלה יכניסו בו האסורים וממנו להם אורה, והוא מלשון סיהרא בארמית [=ירח],259 כאשר אמר 'צֹהַר תַּעֲשֶׂה לַתֵּבָה' (לעיל ו טז) מלשון צהרים,260 ושינו בהם – זה לרוב אורו וזה למיעוט"; לפירושו, "בֵּית הַסֹּהַר" היינו "בית החלון" – המקום שבו מקור האור היחיד הוא חלון בודד המשמש גם כפתח. באופן נוסף ניתן להסביר ש"בֵּית הַסֹּהַר" הוא מלשון השורש 'סחר', דהיינו היקף – מקום מוקף וגדור.

"מְקוֹם אֲשֶׁר (אסורי) אֲסִירֵי הַמֶּלֶךְ אֲסוּרִים"

בתוך בית הסוהר, פוטיפר נתן אותו באגף המיועד לאסירים שנאסרו במצות המלך, המובדל משל האסירים האחרים; או שכל אותו בית הסוהר היה מיועד לאסירי המלך, כפי שפירש הרמב"ן: "'וַיִּתְּנֵהוּ אֶל בֵּית הַסֹּהַר' הידוע למלך, שהוא מקום אשר יאסרו שם אסירי המלך". נתון זה מהווה הקדמה למה שיסופר להלן, ששר המשקים ושר האופים – שנאסרו במצות המלך – הובאו "אֶל בֵּית הַסֹּהַר, מְקוֹם אֲשֶׁר יוֹסֵף אָסוּר שָׁם" (מ ג), וכפי שכתב הרמב"ן בהמשך דבריו: "לומר כי היה זה סיבה שייכנסו עימו המשקה והאופה".

ניתן להסביר שלאסירי המלך יועד מקום נפרד מפני שלא נאסרו בגלל עוונות חמורים בהכרח, אלא שלא נהגו כראוי לפי הציפיות הגבוהות שישנן כלפי משרתי המלך – ומשכך ניתן גם להניח שהתנאים שם היו קשים פחות מאשר בבית הסוהר של העבריינים הרגילים; וכך כתב הרמב"ן: "וייתכן שטעם 'אֲסִירֵי הַמֶּלֶךְ' – עבדיו ומשרתיו החוטאים לו במשפט המלוכה, כי שאר האסירים לעם ביד השופטים והשוטרים בבית סוהר אחר יינתנו; וסיפר הכתוב ששמו שם יוסף מאהבת אותו אדוניו, והכל סיבה מאת השם". על סמך ההנחה שהתנאים בבית הסוהר של אסירי המלך היו קלים יותר, מוסיף הרמב"ן כי זהו הטעם לכך שפוטיפר שם את יוסף דווקא שם, מאהבתו אותו.

ניתן גם לומר כי מאחר שפוטיפר היה "סְרִיס פַּרְעֹה, שַׂר הַטַּבָּחִים" (פס' א) ובעל מעמד גבוה בבית המלוכה – הרי שגם אלו שרצה לאסור בשל כך שחטאו לו באופן אישי נחשבו בכלל "אֲסִירֵי הַמֶּלֶךְ", וממילא זה היה המקום הטבעי שבו היה צפוי לאסור את יוסף. מעבר לכך, להלן (מ ג) יכנה הכתוב את המקום שבו נאסר יוסף: "בֵּית שַׂר הַטַּבָּחִים", והיינו (כפי שיתבאר שם) אגף מסויים מבית הסוהר שעליו היה ממונה שר הטבחים – ואם כן, הרי מובן מדוע שם אסר את יוסף, שכן מקום זה היה תחת אחריותו.

רש"ר הירש מתייחס למשמעות ההבדל בין הקרי והכתיב במילה "אֲסוּרֵי / אֲסִירֵי":

'אסיר' הוא שם־עצם, ומשום כך מציין תכונה קבועה; ו'אסור' הוא פועל [סביל] ומציין רק תכונה זמנית. ה'אסור' נעצר ומעוכב עד העמדתו לדין; אך ה'אסיר' נכלא על מנת שיכפר על חטאו. שני המובנים נכללים ב'אסורי' על ידי הקרי והכתיב: בית סוהר זה שימש גם לעצירים וגם לאסירים – דבר המהווה בסיס להמשך הסיפור; באופן זה יכול היה יוסף לעמוד בקשר עם אנשים שנעצרו לזמן קצר בלבד, ושלאחר מכן חזרו לעמדותיהם החשובות; ואכן נראה ששר המשקים ושר האופים היו עצורים בלבד, והמתינו להעמדתם לדין.261

"וַיְהִי שָׁם בְּבֵית הַסֹּהַר"

מילים אלו נראות כמיותרות, שהרי מובן כי שם היה – מאחר שניתן שם, כאמור בתחילת הפסוק. ר' נפתלי הרץ ויזל הסביר זאת כך:

הכתוב מודיע שלא נתנוהו שם במשמר עד יביאוהו למשפט לפני השופטים על הַנְּבָלָה שעשה, כמו שנאמר במקושש,262 אלא זהו גמר דינו: שֶׁיֵּשֵׁב בבית האסורים תמיד לעונש על מעשהו, וזהו "וַיְהִי שָׁם בְּבֵית הַסֹּהַר", ולא הוציאוהו עוד משם עד שֶׁיָּצָא במצות פרעה. וכמו שביארנו כי מעולם לא רצה אדוניו להרע עימו, רק בעל כרחו הוצרך לשים אותו בבור.

המלבי"ם כתב על כך: "ומבואר שהוא לא היה אסור, רק היה שם חופשי עובד עבודתו, ועל זה אמר 'וַיְהִי שָׁם בְּבֵית הַסֹּהַר', כי בית הסוהר היה שייך לבית פוטיפר263 והיה צריך עבדים גם שם, והיה יכול לתתו שם גם בלי פשע". לפי פירושו, משמעות מילים אלו היא שיוסף רק 'היה' שם, אך לא היה אסור בתנאי אסיר רגילים, אלא היה עוסק בעבודות שנצרכו שם לתחזוק המקום וכדומה, חופשי בתוך כותלי בית הסוהר.

אולם ניתן להקשות על הבנה זו, שהרי בספר תהלים (קה יח) נאמר על יוסף בעת מאסרו: "עִנּוּ בַכֶּבֶל רגליו [קרי: רַגְלוֹ], בַּרְזֶל בָּאָה נַפְשׁוֹ" – הרי שלא היה חופשי אלא כבול ומעונה;264 ואף את לשון הכתוב בפסוקנו אין הכרח לפרש כדברי המלבי"ם, וייתכן שמשמעותו היא "וַיְהִי שָׁם אסיר בְּבֵית הַסֹּהַר".265

באופן נוסף ניתן להסביר, שהמילים "וַיְהִי שָׁם בְּבֵית הַסֹּהַר" נועדו להוות הקדמה לפסוק הבא: "וַיְהִי ה' אֶת יוֹסֵף וַיֵּט אֵלָיו חָסֶד, וַיִּתֵּן חִנּוֹ בְּעֵינֵי שַׂר בֵּית הַסֹּהַר"; לא מיד כאשר ראה שר בית הסוהר את יוסף הוא מצא חן בעיניו, אלא רק לאחר שהיה שם זמן ממושך, שבו הוא עקב אחרי התנהלותו והתרשם מאישיותו. פסוקנו משמש מצע לתוכן שבפסוק שלאחריו: יוסף היה "שָׁם בְּבֵית הַסֹּהַר" – ובתקופת היותו שם היה מלווה בהשגחת ה' ובחסד אלוהי – כאמור לאחר מכן: "וַיְהִי ה' אֶת יוֹסֵף, וַיֵּט אֵלָיו חָסֶד", ובעקבות זאת: "וַיִּתֵּן חִנּוֹ בְּעֵינֵי שַׂר בֵּית הַסֹּהַר".

פרק לט · פסוק כא

וַיְהִי ה' אֶת יוֹסֵף וַיֵּט אֵלָיו חָסֶד, וַיִּתֵּן חִנּוֹ בְּעֵינֵי שַׂר בֵּית הַסֹּהַר.

השגחת ה' ממשיכה ללוות את יוסף בבית הסוהר, כשם שליוותה אותו בבית פוטיפר; וכשם ששם הצליח בכל מעשיו ומצא חן בעיני אדוניו, עד שהלה הפקידו על כל אשר לו וסמך עליו לחלוטין – כך גם כאן, והכתוב מבליט היטב הקבלה זו באמצעות מוטיבים חוזרים המופיעים בשני המקומות, כפי שניכר בפסוקנו ובפסוקים שלאחריו: "וַיִּתֵּן שַׂר בֵּית הַסֹּהַר בְּיַד יוֹסֵף אֵת כָּל הָאֲסִירִם אֲשֶׁר בְּבֵית הַסֹּהַר, וְאֵת כָּל אֲשֶׁר עֹשִׂים שָׁם הוּא הָיָה עֹשֶׂה. אֵין שַׂר בֵּית הַסֹּהַר רֹאֶה אֶת כָּל מְאוּמָה בְּיָדוֹ בַּאֲשֶׁר ה' אִתּוֹ, וַאֲשֶׁר הוּא עֹשֶׂה ה' מַצְלִיחַ".

כתב ר' נפתלי הרץ ויזל:

זה מפליא כי כאשר פתאום הושפל יוסף ממעלתו [...] וְיָרַד אל הבור באין תקווה, שהיה ראוי שליבו דווי ויימלא יגון, גם יעלו על ליבו אחיו ובית אביו, כי רחוק ממנו מנחם, וזה יוסיף יגון על יגונו, ואין השכינה שורה מתוך עצבות. ואף על פי כן: "וַיְהִי ה' אֶת יוֹסֵף", שנחה עליו רוח חכמה גם בבור הזה, ופתר שם חלומות סריסי פרעה ב"רוּחַ אֱלָהִין קַדִּישִׁין".266 כי נפשו תמיד דבקה באור החיים. צרה ויגון ימצא, ובשם ה' יקרא.

בהיותו בבית פוטיפר, נאמר על יוסף: "וַיְהִי ה' אֶת יוֹסֵף, וַיְהִי אִישׁ מַצְלִיחַ" – היות ה' עם יוסף התבטא בכך שהיה "אִישׁ מַצְלִיחַ"; אולם בענייננו לא נאמר "וַיְהִי אִישׁ מַצְלִיחַ", שכן שם עבד יוסף את עבודות האדון והצליח בעבודתו, אך בבית הסוהר לא היה ליוסף במה להיות 'מצליח', שכן אף אם היה שותף בעבודת האסירים, לא היו אלו עסקים שניתן להבחין בהם ב'הצלחה'. בהתאם לכך, היות ה' עם יוסף התבטא כאן באופן אחר: "וַיְהִי ה' אֶת יוֹסֵף וַיֵּט אֵלָיו חָסֶד, וַיִּתֵּן חִנּוֹ בְּעֵינֵי שַׂר בֵּית הַסֹּהַר".

"וַיֵּט אֵלָיו חָסֶד" – פירש רש"י: "שיהא מקובל לכל רואיו, לשון 'כלה נאה וחסודה' שבמשנה".267 גם במקרא מצינו ביטוי זה במקומות נוספים, כגון בדברי עזרא: "וְעָלַי הִטָּה חֶסֶד לִפְנֵי הַמֶּלֶךְ וְיוֹעֲצָיו" (עזרא ז כח), "כִּי עֲבָדִים אֲנַחְנוּ וּבְעַבְדֻתֵנוּ לֹא עֲזָבָנוּ אֱלֹהֵינוּ וַיַּט עָלֵינוּ חֶסֶד לִפְנֵי מַלְכֵי פָרַס" (שם ט ט). אין זה במובן של עשיית חסד, כמו ברוב הופעותיה של מילה זו במקרא, אלא במובן של רושם חיובי של אדם בעיני אחרים, בדומה ל'מציאת חן'268 (וראה להלן ביאור ההבדל ביניהם);269 ה' היטה חסד לעבר יוסף, כך שהיה אפוף חסד אלוהי שגרם לכל רואיו לראות בו אדם רצוי ומקובל.270

אם כן, הרי ששני מושגים דומים מופיעים בפסוקנו: חסד, וחן; את החסד נשא יוסף בעיני כל רואיו ("וַיֵּט אֵלָיו חָסֶד"), ואילו על החן נאמר רק שהיה "בְּעֵינֵי שַׂר בֵּית הַסֹּהַר" ("וַיִּתֵּן חִנּוֹ בְּעֵינֵי שַׂר בֵּית הַסֹּהַר"). מהי משמעותו הייחודית של כל אחד מהם, ומדוע החסד היה בעיני כל, והחן – רק בעיני שר בית הסוהר?

ניתן להסביר, שהמונח 'חסד' מציין, כאמור, יחס חיובי יוצא דופן – 'נושא החסד' מקובל ורצוי על סובביו; לעומת זאת, ה'חן' מציין רמה גבוהה יותר, שבה היחס החיובי גם נוגע לליבו של האחר באופן אישי, והוא מפתח כלפי 'נושא החן' רגשות חיוביים של אהבה, רצון לקרבה וכדומה (ראה הערה).271 כך בתחילת הפרק, שבה סופר כי יוסף מצא חן בעיני אדוניו: "וַיִּמְצָא יוֹסֵף חֵן בְּעֵינָיו, וַיְשָׁרֶת אֹתוֹ" (פס' ד), שכמבואר שם – מפני שמצא חן בעיניו, בחר לקרבו אליו לשירותו האישי, וזהו עניין נוסף מלבד הקידום ה'מקצועי' שנתן לו בעקבות היותו איש מצליח. בענייננו, ה' היטה חסד אל יוסף, שאותו נשא בעיני כל רואיו, האסירים והסוהרים; אולם לגבי שר בית הסוהר הוסיף ה' גם לתת בעיניו את חינו של יוסף, דהיינו שמעבר לראייתו אותו כאדם טוב וחיובי – גם יאהב אותו באופן אישי וקסם אישיותו ייגע לליבו.

הן החסד שנשא יוסף בעיני כל רואיו והן החן שנשא בעיני שר בית הסוהר, הכשירו את הקרקע לשלב הבא בתוכנית האלוהית – המינוי שהתמנה לו יוסף על ידו: "וַיִּתֵּן שַׂר בֵּית הַסֹּהַר בְּיַד יוֹסֵף אֵת כָּל הָאֲסִירִם אֲשֶׁר בְּבֵית הַסֹּהַר". מובן כי מינוי זה הוא תוצאה של נתינת חינו בעיני שר בית הסוהר, שכן מפני כך חפץ להיטיב לו; אך גם נטיית החסד אליו בעיני רואיו היוותה סיבה לכך – שכן מאחר ושר בית הסוהר, שאף הוא בכלל "כל רואיו", ראה בו אדם טוב והגון, ומאחר ויוסף היה מקובל ורצוי על יתר האסירים – הרי שבעיניו היה הוא האדם המתאים להתמנות כאחראי עליהם.272

ניתן להבחין בהבדל בין התיאור שבפסוקנו לבין זה שהופיע בתחילת הפרק בעניין פוטיפר; שכן שם נאמר "וַיִּמְצָא יוֹסֵף חֵן בְּעֵינָיו" ללא הזכרת התערבות אלוהית, ואילו כאן הן ה'חסד' והן ה'חן' מיוחסים לפעולה מצד ה': "וַיְהִי ה' אֶת יוֹסֵף – וַיֵּט אֵלָיו חָסֶד, וַיִּתֵּן חִנּוֹ בְּעֵינֵי שַׂר בֵּית הַסֹּהַר"

ייתכן להסביר זאת בכך שאף כי כל מציאת חן היא מאת ה', ובכלל זה גם מציאת חינו של יוסף בעיני פוטיפר – אין הכתוב נדרש לפרש זאת; אך בענייננו הכתוב מייחס לה' את הטיית החסד ונתינת החן עבור יוסף, כאמצעי רטורי שנועד להדגיש את קרבת ה' אליו ואת עוצמת ההשגחה המלווה אותו. כעת, משירד יוסף מדרגה נוספת – ממעמד עבד למעמד אסיר, הרי שכנגד זה התעצם הליווי האלוהי החופף עליו, שכן "קָרוֹב ה' לְנִשְׁבְּרֵי לֵב" (תהלים לד יט) – וככל שאדם מצוי במדרגה שפלה יותר כך הוא זקוק יותר לסיוע אלוהי. לבל יהא הקורא סבור שיוסף נעזב מעם ההשגחה – אדרבה, דווקא בעת זו יד ה' כביכול מתערבת יותר במציאות, ומטה אליו את ליבם של הסובבים אותו.

ר' נפתלי הרץ ויזל הסביר זאת באופן אחר:

"וַיֵּט אֵלָיו חָסֶד" – לא נאמר כן למעלה כשסיפר בואו אל יד פוטיפר, אבל אמר "וַיְהִי ה' אֶת יוֹסֵף וַיְהִי אִישׁ מַצְלִיחַ", לפי שבהיותו עובד עבודתו ראו שהוא איש מצליח במעשיו, וכן כבואו לבית אדוניו שמע מפיו דברי תבונות ומועצות בכל אשר שאל ממנו על עסקיו, גם ראה שהוא איש מצליח בעסקיו. על כן אמר "וַיַּרְא אֲדֹנָיו כִּי ה' אִתּוֹ וְכֹל אֲשֶׁר הוּא עֹשֶׂה ה' מַצְלִיחַ בְּיָדוֹ" ונמשך מזה שנשא חן וחסד בעיניו, וכאמרוֹ "וַיִּמְצָא יוֹסֵף חֵן בְּעֵינָיו" וכמו שביארנו שם. אבל בבית הסוהר, מקום שעמום ובטלה, אין סוֹדֵר אליו דבר ואינו עושה עסקים, במה יִוָּדַע לשר בית הסוהר כי ה' איתו, ושה' מצליח מעשיו, עד שימצא על ידי כן חן בעיניו? לכן הוצרך ה' להפליא עימו להטות אליו חסד, שהיה חוט של חסד משוך עליו, כל רואהו ישמח ליבו בו. וזהו "וַיֵּט אֵלָיו חָסֶד וַיִּתֵּן חִנּוֹ" וגו', שה' נתן חינו בדרך פלא, לא מחמת דבריו ומעשיו.

הסבר זה מתבסס על ההנחה שמציאת חינו של יוסף בעיני פוטיפר היתה בסיבת היותו איש מצליח בעסקיו – ומשכך, בבית הסוהר – שבו לא היה מקום ביטוי לכשרון ההצלחה של יוסף בעסקים – היטה עליו ה' חן וחסד באופן ישיר. אולם אם נניח שגם מציאת חינו בעיני פוטיפר לא היתה דווקא כתוצאה מהיותו איש מצליח – אלא בעיקר מחמת התרשמותו מקסם אישיותו והתנהלותו היום-יומית, הרי שאלו יכלו להתבטא גם בין כותלי בית הסוהר; מכוחם – בצירוף הכוונת יד ההשגחה – נשא יוסף חסד וחן, הן בבית פוטיפר והן בבית הסוהר, אלא שדווקא כאן בחר הכתוב להבליט את התערבות ההשגחה, מהטעם שהתבאר לעיל.

  4. 269

    המושג 'חסד' מופיע במקרא גם במובן של חרפה, כמו בפסוקים "וְאִישׁ אֲשֶׁר יִקַּח אֶת אֲחֹתוֹ [...] חֶסֶד הוּא" (ויקרא כ יז), "פֶּן יְחַסֶּדְךָ שֹׁמֵעַ וְדִבָּתְךָ לֹא תָשׁוּב" (משלי כה י); ונראה שישנה משמעות משותפת למובן זה ולמובן המקובל של חסד, שכן שניהם מציינים חריגה מהראוי והמקובל; מעשה חסד – הרי הוא חורג משורת הדין, ולפנים ממנה, שכן אין חובה לעשותו, ואילו מעשים החורגים משורת הדין לכיוון השני, השלילי, הרי שאף הם 'חסד'. כעין זה כתב רד"ק בספר השורשים (ערך 'חסד'): "ושני העניינים האלו שהזכרנו בחסד הם עניין אחד, כי כמו שהחסד הוא תוספת הטוב ויתרון הגמול כן 'חֶסֶד הוּא' רצה לומר תוספת הזנות ויתרון הנבלה". לפי זה, ניתן לומר שגם המובן השלישי של 'חסד', המופיע בפסוקנו, שייך למשמעות משותפת זו – הוא מבטא יחס מועדף שמקבל נושא החסד מאת רואיו, שרואים בו אדם רצוי וחיובי מעבר לכפי שהיה מצופה.

  5. 270

    כך עולה גם מתרגום אונקלוס, שתרגם "וַיֵּט אֵלָיו חָסֶד": "וּנְגַד לֵיהּ חִסְדָּא" – משך אליו חסד, ולא תרגם את המילה "חָסֶד" כ"טֵיבוּ", כמנהגו במקומות אחרים; "כי בפסוקנו אין חֶסֶד כמשמעו הרגיל, טובה, אלא בהוראת חיבה [...], וראה כעין זה בפסוק 'שֹׁמֵר הַבְּרִית וְהַחֶסֶד' (דברים ז ט) – נָטַר קְיָמָא וְחִסְדָּא'" ('פרשגן').

  6. 271

    'הטיית חסד' מופיעה בשני מקומות נוספים במקרא, שהוזכרו לעיל, אשר בהם מתאר עזרא כי ה' היטה עליו חסד לפני המלך ויועציו ושריו – וכך הם מילאו את בקשותיו וסייעו לו במימון תחזוקת המקדש, וכן שה' היטה חסד על עם ישראל "לִפְנֵי מַלְכֵי פָרַס – לָתֶת לָנוּ מִחְיָה, לְרוֹמֵם אֶת בֵּית אֱלֹהֵינוּ וּלְהַעֲמִיד אֶת חָרְבֹתָיו, וְלָתֶת לָנוּ גָדֵר בִּיהוּדָה וּבִירוּשָׁלִָם". באלו אין מדובר ב'מציאת חן' אישית, אלא בכך שמלכי פרס ראו בעין חיובית את שיבת יהודה לציון ואת בניית המקדש ועבודתו, ולפיכך מופיע שם המושג 'חסד'. לעומת זאת, הביטוי 'מציאת חן' מופיע בהקשרים שבהם מדובר על יחס ברמה האישית כלפי 'נושא החן', כגון "וְנֹחַ מָצָא חֵן בְּעֵינֵי ה'" (לעיל ו ח), "וְאַתָּה אָמַרְתָּ יְדַעְתִּיךָ בְשֵׁם וְגַם מָצָאתָ חֵן בְּעֵינָי" (שמות לג יב), ועוד רבים, או במחווה אישית לפנים משורת הדין שמתבקשת גם כלפי ציבור בעקבות מציאתם חן בעיני האחר, כגון בדברי בני גד ובני ראובן למשה: "וַיֹּאמְרוּ אִם מָצָאנוּ חֵן בְּעֵינֶיךָ יֻתַּן אֶת הָאָרֶץ הַזֹּאת לַעֲבָדֶיךָ לַאֲחֻזָּה" (במדבר לב ה), ובדומה לכך בדברי המצרים ליוסף: "נִמְצָא חֵן בְּעֵינֵי אֲדֹנִי וְהָיִינוּ עֲבָדִים לְפַרְעֹה" (להלן מז כה).

    הבחנה זו עשויה להתחדד מתוך התבוננות בלשון המקרא בעניין אסתר: "וַתִּיטַב הַנַּעֲרָה בְעֵינָיו [של הגי שומר הנשים] וַתִּשָּׂא חֶסֶד לְפָנָיו וַיְבַהֵל אֶת תַּמְרוּקֶיהָ וְאֶת מָנוֹתֶהָ לָתֵת לָהּ וְאֵת שֶׁבַע הַנְּעָרוֹת הָרְאֻיוֹת לָתֶת לָהּ מִבֵּית הַמֶּלֶךְ [...] וַתְּהִי אֶסְתֵּר נֹשֵׂאת חֵן בְּעֵינֵי כָּל רֹאֶיהָ [...] וַיֶּאֱהַב הַמֶּלֶךְ אֶת אֶסְתֵּר מִכָּל הַנָּשִׁים וַתִּשָּׂא חֵן וָחֶסֶד לְפָנָיו מִכָּל הַבְּתוּלֹת" (אסתר ב ט-יז). כשבאה לבית הנשים, נאמר "וַתִּיטַב הַנַּעֲרָה בְעֵינָיו, וַתִּשָּׂא חֶסֶד לְפָנָיו" – שני ביטויים המלמדים זה על זה – היא היתה טובה בעיני הגי יותר מהאחרות, שכן החסד המשוך עליה גרם לו לראותה בעין חיובית, ולפיכך הקדיש לה תשומת לב מיוחדת ודאג היטב לצרכיה, על מנת שתצלח למלוכה; זהו עניין של טיב ו'איכות', ולא של חן אישי. לאחר מכן, "וּבְהַגִּיעַ תֹּר אֶסְתֵּר [...] לָבוֹא אֶל הַמֶּלֶךְ" – אזי נסך עליה ה' חן מיוחד, "וַתְּהִי אֶסְתֵּר נֹשֵׂאת חֵן בְּעֵינֵי כָּל רֹאֶיהָ" – קסם אישיותה נגע לליבו של כל מי שראה אותה, ובכללם אחשוורוש, שעליו נאמר: "וַיֶּאֱהַב הַמֶּלֶךְ אֶת אֶסְתֵּר מִכָּל הַנָּשִׁים, וַתִּשָּׂא חֵן וָחֶסֶד לְפָנָיו מִכָּל הַבְּתוּלֹת" – מלבד החסד שנשאה לפניו, דהיינו שהיתה טובה בעיניו יותר מהאחרות, היא גם נשאה חן בעיניו – והוא אהב אותה יותר מכולן, כפי שנאמר לפני כן: "וַיֶּאֱהַב הַמֶּלֶךְ אֶת אֶסְתֵּר".

    בלשון חכמים הבחנה זו אינה נשמרת, והביטוי "חוט של חסד" מציין את המושג המוגדר לדברינו כ"חן"; כך אמרו בתלמוד על אסתר, ש"חוט של חסד משוך עליה" (מגילה יג ע"א) "מאת הקדוש ברוך הוא, לכך נראית יפה לאומות ולאחשוורוש" (רש"י שם), והיינו בהתאם לדברי הכתוב "וַתְּהִי אֶסְתֵּר נֹשֵׂאת חֵן בְּעֵינֵי כָּל רֹאֶיהָ"; ועוד אמרו (תמיד כח ע"א): "כל המוכיח את חבירו לשם שמיים [...] מושכין עליו חוט של חסד, שנאמר: [מוֹכִיחַ אָדָם אַחֲרַי] חֵן יִמְצָא מִמַּחֲלִיק לָשׁוֹן' (משלי כח כג)".

    על פי כל האמור, ניתן גם להבין מדוע לגבי החן דרך הכתוב להזכיר את העיניים, כגון בביטוי הנפוץ במקרא: "וימצא... חן בעיניו" וכדומה, בעוד שלגבי החסד לא נזכרות עיני הרואה, ונאמר רק "לִפְנֵי" או "לְפָנָיו" (כמצוטט לעיל מעזרא ומאסתר), או אף בלי ציון הרואה עצמו כלל, כמו בפסוקנו – שכן החן נוגע במקום אישי ופנימי, וכביכול מצוי בתוך עיניו של הרואה; בעוד שהחסד אינו כזה, ואינו אלא "לפני" הרואה, כלומר לנגד פניו.

  7. 272

    אסירים עשויים באופן טבעי לפתח יחס שלילי או מתמרד כלפי מי שממונה עליהם, ואם זהו אחד מתוכם הרי שעשויות להתעורר גם קנאה ויריבות, ולפיכך כאשר ראה שר בית הסוהר שיוסף נושא חסד בעיני כל – ידע כי מינויו על האסירים יביא ברכה, ויועיל להתנהלות התקינה בבית הסוהר, שכן הממונה עליהם הוא אדם המקובל ורצוי להם, וייקל עליהם לסור למשמעתו.

וַיִּתֵּן שַׂר בֵּית הַסֹּהַר בְּיַד יוֹסֵף אֵת כָּל הָאֲסִירִם אֲשֶׁר בְּבֵית הַסֹּהַר, וְאֵת כָּל אֲשֶׁר עֹשִׂים שָׁם הוּא הָיָה עֹשֶׂה. אֵין שַׂר בֵּית הַסֹּהַר רֹאֶה אֶת כָּל מְאוּמָה בְּיָדוֹ בַּאֲשֶׁר ה' אִתּוֹ, וַאֲשֶׁר הוּא עֹשֶׂה ה' מַצְלִיחַ.

"וַיִּתֵּן שַׂר בֵּית הַסֹּהַר בְּיַד יוֹסֵף אֵת כָּל הָאֲסִירִם אֲשֶׁר בְּבֵית הַסֹּהַר" – שמינה אותו כאחראי עליהם, "וְאֵת כָּל אֲשֶׁר עֹשִׂים שָׁם הוּא הָיָה עֹשֶׂה" – דהיינו: "וְאֵת כָּל אֲשֶׁר עֹשִׂים שָׁם" בפועל, כל סדרי ההתנהלות ועבודותיהם של האסירים, "הוּא הָיָה עֹשֶׂה" – מנהל ומְתַכְלֵל; וכפי שפירש רש"י: "'הוּא הָיָה עֹשֶׂה' – כתרגומו: 'במימריה הוה מתעביד'" – על פי דיבורו היה נעשה.273

"אֵין שַׂר בֵּית הַסֹּהַר רֹאֶה אֶת כָּל מְאוּמָה בְּיָדוֹ"

תיאור זה מקביל למה שנאמר לעיל על פוטיפר: "וַיַּעֲזֹב כָּל אֲשֶׁר לוֹ בְּיַד יוֹסֵף וְלֹא יָדַע אִתּוֹ מְאוּמָה כִּי אִם הַלֶּחֶם אֲשֶׁר הוּא אוֹכֵל" (פס' ו). שם משמעות הדברים היא שפוטיפר לא פיקח כלל על מעשיו של יוסף ולא התעניין בטיב האופן שבו הוא מנהל את ענייני הבית, וכך ניתן להבין גם כאן – שר בית הסוהר לא פיקח על התנהלותו של יוסף במאומה, וסמך עליו לחלוטין. כך כתבו כמה מהמפרשים, כגון ר' נפתלי הרץ ויזל: "לא השגיח עוד על דבר, אלא עזב הכל בידו, ולא ראה כל מאומה; וזהו שאמר הכתוב: 'אֵין שַׂר בֵּית הַסֹּהַר רֹאֶה אֶת כָּל מְאוּמָה בְּיָדוֹ' אם פעל בצדק אם אַיִן", וכן שד"ל: "לא היה משגיח ופוקח עינו על דבר ממה שהיה עושה", ומלבי"ם: "לא לקח מאיתו חשבון".

אולם אונקלוס תרגם כאן: "לֵית רַב בֵּית אֲסִירֵי חָזֵי יָת כָּל סוּרְחַן בִּידֵיהּ" – אין שר בית הסוהר רואה את כל סרחון בידו, כלומר פגם. לפי הבנתו, המילה "מְאוּמָה" מתפרשת כאן כ"מאומה רע",274 וכמו בדברי המלאך לאברהם: "אַל תִּשְׁלַח יָדְךָ אֶל הַנַּעַר וְאַל תַּעַשׂ לוֹ מְאוּמָה" (לעיל כב יב), ובדברי אכיש על דוד: "וְלֹא מָצָאתִי בוֹ מְאוּמָה מִיּוֹם נפְלוֹ עַד הַיּוֹם הַזֶּה" (שמואל א' כט ג); וכך פירש 'בכור שור': "שלא ראה בו מעולם דבר שלא כהוגן, ואין יוצא מידו דבר שאינו מתוקן".275

מהלשון "אֵין שַׂר בֵּית הַסֹּהַר רֹאֶה אֶת כָּל מְאוּמָה בְּיָדוֹ" משתמע יותר כפירוש הראשון, שכן המילה "אֶת" מורה על כך שמדובר כאן במושא קיים וידוע – היה מאומה, אך השר לא ראה אותו; כל מה שבידו של יוסף, דהיינו כל מה שנעשה תחתיו – אין שר בית הסוהר רואה אותו, ואף לא מאומה ממנו. אך לפי הבנת התרגום, היה מתבקש שייכתב: "אֵין שַׂר בֵּית הַסֹּהַר רֹאֶה כָּל מְאוּמָה בְּיָדוֹ", ללא המילה "אֶת", שכן הוא לא ראה מאומה מכיון שאכן לא היה מאומה.

"בַּאֲשֶׁר ה' אִתּוֹ, וַאֲשֶׁר הוּא עֹשֶׂה ה' מַצְלִיחַ"

מילים אלו מציינות את הסיבה לכך ש"אֵין שַׂר בֵּית הַסֹּהַר רֹאֶה אֶת כָּל מְאוּמָה בְּיָדוֹ": ה' היה עם יוסף, ודבר זה התבטא בכך שהצליח במעשיו – ולפיכך סמך עליו שר בית הסוהר וידע כי ינהל את את הדברים כראוי גם מבלי פיקוחו (לפי הפירוש הראשון), או: ולפיכך שר בית הסוהר לא מצא כל פגם בהתנהלותו (לפי הפירוש השני).

בתיאור המקביל בעניין פוטיפר, נאמר: "וַיַּרְא אֲדֹנָיו כִּי ה' אִתּוֹ, וְכֹל אֲשֶׁר הוּא עֹשֶׂה ה' מַצְלִיחַ בְּיָדוֹ" (פס' ג), ואילו כאן נאמר: "וַאֲשֶׁר הוּא עֹשֶׂה ה' מַצְלִיחַ", ללא המילה "בְּיָדוֹ"; וכתב על כך ה'רוקח': "למעלה נתן בידו כסף וזהב, דבר הניתן מיד ליד, כתיב 'בְּיָדוֹ', כאן לא היה בידו לא כסף ולא זהב, אלא בדיבורו עושה, 'במימריה הוה מתעביד'", וכך גם בעל הטורים: "לא כתיב 'מַצְלִיחַ בְּיָדוֹ' כמו למעלה, לפי שבכאן היה בבית האסורים ולא היה לו עסק בידו, כמו למעלה שהיה בבית אדוניו ומתעסק בשל אדוניו". בבית פוטיפר ההצלחה היתה במעשי ידיו של יוסף, אשר עסק בהם בעצמו; אך בבית הסוהר עיקר תפקידו היה לנהל את העבודה 'מלמעלה' ולא עבודה בידו ממש, ולפיכך אין ה'עשייה' מיוחסת כאן ל'יד'.

אלא שניתן להקשות על כך מתחילת הפסוק, שבה נאמר "אֵין שַׂר בֵּית הַסֹּהַר רֹאֶה אֶת כָּל מְאוּמָה בְּיָדוֹ" – ומשם משמע שמלאכתו של יוסף בבית הסוהר, הגם שלא היתה ב'ידו' ממש, מיוחסת ל'יד'. אך נראה שישנו הבדל בין שני ההיגדים הללו בעניין משמעות שילוב ה'יד' בהם; שכן מהביטוי "מַצְלִיחַ בְּיָדוֹ" משתמע שה' מצליח את היד העושה, ולפיכך לא נאמר כך כאן, שכן מלאכתו לא נעשתה בידיו; אך בהיגד "אֵין שַׂר בֵּית הַסֹּהַר רֹאֶה אֶת כָּל מְאוּמָה בְּיָדוֹ" אין ה'יד' מסמלת עשייה אלא תחום אחריות – השר לא פיקח על מאומה ממה שהיה בידו של יוסף, דהיינו ברשותו276 ותחת סמכותו.
